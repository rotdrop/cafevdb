29use OCA\CAFEVDB\Database\Legacy\PME\PHPMyEdit;

31/** Substitue some legacy PME placeholder. */
32trait SubstituteSQLFragmentTrait
33{
34  /** @var PHPMyEdit */
35  protected PHPMyEdit $pme;
36
37  /**
38   * Replace $join_table etc. replacements by their proper table and column
39   * aliases, base on the ordering defined in $fdd.
40   *
41   * @param array $fdd Field description data.
42   *
43   * @param string $fieldName The field name which defines the pivot for the
44   * replacements.
45   *
46   * @param string $sql SQL fragment with replacements.
47   *
48   * @param null|int $fieldIndex Field index if known, if not given array_search() is used.
49   *
50   * @return The result of the substitutions.
51   *
52   * @throws UnexpectedValueException Thrown if $fieldName is not found in $fdd.
53   */
54  protected function substituteSQLFragment(array $fdd, string $fieldName, string $sql, ?int $fieldIndex = null):string
55  {
56    $fieldFdd = $fdd[$fieldName] ?? null;
57    if (empty($fieldFdd)) {
58      throw new UnexpectedValueException($this->t->t('The field-description-data did not contain data for the field "%s".', $fieldName));
59    }
60    $joinTable = $fieldFdd['values']['join']['reference'] ?? null;
61    if (empty($joinTable)) {
62      $fieldIndex = $fieldIndex ?? array_search($fieldName, array_keys($fdd));
63      $joinTable = $fieldIndex;
64    }
65    if (!str_starts_with($joinTable, PHPMyEdit::JOIN_ALIAS) && is_numeric($joinTable)) {
66      $joinTable = PHPMyEdit::JOIN_ALIAS . $joinTable;
67    }
68    $joinColumn = $fieldFdd['values']['column'] ?? null;
69    $joinColFqn = $joinTable . '.' . $joinColumn;
70
71    $substitutions = [
72      'main_table' => PHPMyEdit::MAIN_ALIAS,
73      'field_name' => $fieldName,
74      'join_table' => $joinTable,
75      'join_col_fqn' => $joinColFqn,
76      'join_col_enc' => sprintf($fdd['values']['encode'] ?? '', $joinColFqn),
77      'join_column' => $joinColumn,
78      'table' => $joinTable,
79      'column' => $joinColumn,
80    ];
81    return $this->pme->substituteVars($sql, $substitutions);
82  }
83}
