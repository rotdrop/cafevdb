228    $maxDigits = max($minDigits, $maxDigits);
229    $value = round($value, $maxDigits);
230    $factor = pow(10.0, $maxDigits);
231    $digitsPart = abs((int)($value * $factor) % $factor);
232    $majorPart = (int)$value;
233    $majorPartWords = $this->numberTransformer->toWords($majorPart);
234    $digits = [];
235    while ($maxDigits-- > 0) {
236      $digits[] = $digitsPart % 10;//$this->numberTransformer->toWords($digitsPart % 10);
237      $digitsPart /= 10;
238    }
239    // strip trailing zeros
240    while (true) {
241      $value = reset($digits);
242      if ($value > 0 || count($digits) <= $minDigits) {
243        break;
244      }
245      array_shift($digits);
246    }
247    if ($this->fractionalPartAscending) {
248      $digits = array_pad(array_reverse($digits), $minDigits, 0);
249    } else {

251    }
252    $digits = array_map(fn(int $digit) => $this->numberTransformer->toWords($digit), $digits);
253    if ($this->fractionalPartLast) {

256        $parts[] = $this->fractionalSeparatorWord;

259    } else {

261        $parts = $digits;
262        $parts[] = $this->fractionalSeparatorWord;
263      }

265    }

328  protected function numberToWordsPhp(float $value, int $minDigits = 1, int $maxDigits = 2):string
329  {
330    $minDigits = min($minDigits, $maxDigits);
331    $maxDigits = max($minDigits, $maxDigits);
332    $value = round($value, $maxDigits);
333    $factor = pow(10.0, $maxDigits);
334    $value = ((float)(int)$value) + (((int)($value * $factor) % $factor) / $factor);
335
336    $optionalFactor = pow(10.0, $maxDigits - $minDigits + 1);
337    $fractionalPart = abs((int)($value * $factor) % $factor);
338    $forceDigits = $fractionalPart % $optionalFactor == 0;
339
340    if ($forceDigits) {
341      $excessDigit = pow(10.0, -($minDigits + 1));
342      $value = $value >= 0 ? $value + $excessDigit : $value - $excessDigit;
343    }
344    $words = explode(' ', str_replace("\xc2\xad", '', $this->numberSpeller->format($value)));
345    if ($forceDigits) {
346      $separatorPos = array_search($this->fractionalSeparatorWord, $words);
347      if ($this->fractionalPartLast) {
348        if ($this->fractionalPartAscending) {
349          $words = array_slice($words, 0, $separatorPos + $minDigits + 1);
350        } else {
351          $words = array_merge(
352            array_slice($words, 0, $separatorPos + 1),
353            array_slice($words, - $minDigits),
354          );
355        }
356      } else {
357        if ($this->fractionalPartAscending) {
358          $words = array_merge(
359            array_slice($words, 0, $minDigits),
360            array_slice($words, -$separatorPos - 1),
361          );
362        } else {
363          $words = array_merge(
364            array_slice($words, $separatorPos - $minDigits, $minDigits),
365            array_slice($words, -$separatorPos - 1),
366          );
367        }
368      }
369    }
370    return implode(' ', $words);
