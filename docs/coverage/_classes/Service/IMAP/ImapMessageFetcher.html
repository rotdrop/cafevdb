215  private function getPart(Horde_Mime_Part $p, string $partNo, bool $isFetched): void {
216    // iMIP messages
217    // Handle text/calendar parts first because they might be attachments at the same time.
218    // Otherwise, some of the following if-conditions might break the handling and treat iMIP
219    // data like regular attachments.
220    $allContentTypeParameters = $p->getAllContentTypeParameters();
221    if ($p->getType() === 'text/calendar') {
222      // Handle event data like a regular attachment
223      // Outlook doesn't set a content disposition
224      // We work around this by checking for the name only
225      if ($p->getName() !== null) {
226        $this->attachments[] = [
227          'id' => $p->getMimeId(),
228          'messageId' => $this->uid,
229          'fileName' => $p->getName(),
230          'mime' => $p->getType(),
231          'size' => $p->getBytes(),
232          'cid' => $p->getContentId(),
233          'disposition' => $p->getDisposition()
234        ];
235      }
236
237      // return if this is an event attachment only
238      // the method parameter determines if this is a iMIP message
239      if (!isset($allContentTypeParameters['method'])) {
240        return;
241      }
242
243      if (in_array(strtoupper($allContentTypeParameters['method']), ['REQUEST', 'REPLY', 'CANCEL'])) {
244        $this->scheduling[] = [
245          'id' => $p->getMimeId(),
246          'messageId' => $this->uid,
247          'method' => strtoupper($allContentTypeParameters['method']),
248          'contents' => $this->loadBodyData($p, $partNo, $isFetched),
249        ];
250        return;
251      }
252    }
253
254    // Regular attachments
255    if ($p->isAttachment() || $p->getType() === 'message/rfc822') {
256      $this->attachments[] = [
257        'id' => $p->getMimeId(),
258        'messageId' => $this->uid,
259        'fileName' => $p->getName(),
260        'mime' => $p->getType(),
261        'size' => $p->getBytes(),
262        'cid' => $p->getContentId(),
263        'disposition' => $p->getDisposition()
264      ];
265      return;
266    }
267
268    // Inline attachments
269    // Horde doesn't consider parts with content-disposition set to inline as
270    // attachment so we need to use another way to get them.
271    // We use these inline attachments to render a message's html body in $this->getHtmlBody()
272    $filename = $p->getName();
273    if ($p->getType() === 'message/rfc822' || isset($filename)) {
274      if (in_array($filename, $this->attachmentsToIgnore)) {
275        return;
276      }
277      $this->inlineAttachments[] = [
278        'id' => $p->getMimeId(),
279        'messageId' => $this->uid,
280        'fileName' => $filename,
281        'mime' => $p->getType(),
282        'size' => $p->getBytes(),
283        'cid' => $p->getContentId()
284      ];
285      return;
286    }
287
288    if ($p->getPrimaryType() === 'multipart') {
289      $this->handleMultiPartMessage($p, $partNo, $isFetched);
290      return;
291    }
292
293    if ($p->getType() === 'text/plain') {
294      $this->handleTextMessage($p, $partNo, $isFetched);
295      return;
296    }
297
298    if ($p->getType() === 'text/html') {
299      $this->handleHtmlMessage($p, $partNo, $isFetched);
300      return;
301    }
302
303    // EMBEDDED MESSAGE
304    // Many bounce notifications embed the original message as type 2,
305    // but AOL uses type 1 (multipart), which is not handled here.
306    // There are no PHP functions to parse embedded messages,
307    // so this just appends the raw source to the main message.
308    if ($p[0] === 'message') {
309      $data = $this->loadBodyData($p, $partNo, $isFetched);
310      $this->plainMessage .= trim($data) . "\n\n";
311    }

441  private function parseHeaders(Horde_Imap_Client_Data_Fetch $fetch): void {
442    /** @var resource $headersStream */
443    $headersStream = $fetch->getHeaderText('0', Horde_Imap_Client_Data_Fetch::HEADER_STREAM);
444    $this->rawHeaders = stream_get_contents($headersStream);
445    rewind($headersStream);
446    $parsedHeaders = Horde_Mime_Headers::parseHeaders($headersStream);
447    fclose($headersStream);
448
449    $references = $parsedHeaders->getHeader('references');
450    if ($references !== null) {
451      $this->rawReferences = $references->value_single;
452    }
453
454    $dispositionNotificationTo = $parsedHeaders->getHeader('disposition-notification-to');
455    if ($dispositionNotificationTo !== null) {
456      $this->dispositionNotificationTo = $dispositionNotificationTo->value_single;
457    }
458
459    $dkimSignatureHeader = $parsedHeaders->getHeader('dkim-signature');
460    $this->hasDkimSignature = $dkimSignatureHeader !== null;
461
462    $listUnsubscribeHeader = $parsedHeaders->getHeader('list-unsubscribe');
463    if ($listUnsubscribeHeader !== null) {
464      $listHeaders = new Horde_ListHeaders();
465      $headers = $listHeaders->parse($listUnsubscribeHeader->name, $listUnsubscribeHeader->value_single);
466      if (!$headers) {
467        // Unable to parse headers
468        return;
469      }
470      foreach ($headers as $header) {
471        if (str_starts_with($header->url, 'http')) {
472          $this->unsubscribeUrl = $header->url;
473
474          $unsubscribePostHeader = $parsedHeaders->getHeader('List-Unsubscribe-Post');
475          if ($unsubscribePostHeader !== null) {
476            $this->isOneClickUnsubscribe = strtolower($unsubscribePostHeader->value_single) === 'list-unsubscribe=one-click';
477          }
478          break;
479        }
480        if (str_starts_with($header->url, 'mailto')) {
481          $this->unsubscribeMailto = $header->url;
482          break;
483        }
484      }

488  private function isEncrypted(Horde_Imap_Client_Data_Fetch $message): bool {
489    $headers = $message->getHeaderText('0', Horde_Imap_Client_Data_Fetch::HEADER_PARSE);
490    if (!isset($headers['content-type'])) {
491      return false;
492    }
493
494    /** @var Horde_Mime_Headers_ContentParam_ContentType $contentType */
495    $contentType = $headers['content-type'];

497      && str_ends_with($contentType->stype, 'pkcs7-mime')
498      && isset($contentType['smime-type'])
499      && $contentType['smime-type'] === 'enveloped-data';
