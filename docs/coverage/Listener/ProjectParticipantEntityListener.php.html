94    $l = $this->appContainer->get(IL10N::class);
95    $field = 'participationStatus';
96    if ($event->hasChangedField($field)) {
97      $participationStatus = (string)$event->getNewValue($field);
98      switch ($participationStatus) {
99        case ParticipationStatus::PASSIVE:
100          // confirmed partiticipants must not be "passive"
101          if ($entity->getRegistration()) {
102            throw new Exceptions\EnduserNotificationException(
103              $l->t(
104                'Confirmed project participants cannot have a participation status of "%s".',
105                ParticipationStatus::getL10NValues($this->l)[$participationStatus],
106              ),
107            );
108          }
109          break;
110        case ParticipationStatus::ASSOCIATED:
111          // participants playing real instruments must not be associated
112          $realInstruments = $entity->getRealInstruments();
113          if (!$realInstruments->isEmpty()) {
114            throw new Exceptions\EnduserNotificationException(
115              $l->t(
116                'The participation-status cannot be set to "%1$s" as long as the participant is registered to perform with real instruments (%2$s)',
117                [
118                  ParticipationStatus::getL10NValues($this->l)[$participationStatus],
119                  implode(', ', array_map(fn(Entities\ProjectInstrument $instrument) => $instrument->getInstrument()->getName(), $realInstruments->toArray())),
120                ],
121              ),
122            );
123          }
124          break;
125      }
126    }
127    $field = 'registration';
128    if ($event->hasChangedField($field)) {
129      $oldValue = $event->getOldValue($field);
130      $this->entityManager->dispatchEvent(
131        new Events\PreChangeRegistrationConfirmation(
132          $entity,
133          !empty($oldValue),
134          !empty($event->getNewValue($field)),
135        )
136      );
137
138      $key = self::entityId($entity, $field);
139      $this->preUpdateValues[$key] = $oldValue;
140    }
141    $field = 'deleted';
142    if ($event->hasChangedField($field)) {
143      $oldValue = $event->getOldValue($field);
144      $key = self::entityId($entity, $field);
145      $this->preUpdateValues[$key] = $oldValue;
146    }

229  /**
230   * Register a pre-commit action with undo in order to add / remove the
231   * cloud-account of a newly registered executive board member to the
232   * respective cloud user groups. Also sync a linked address book entry if
233   * present.
234   *
235   * @param Entity $entity
236   *
237   * @param bool $remove IF true the this originated from a postRemove or
238   * soft-deletion event. Otherwise from postPersist or (soft-)undeletion.
239   *
240   * @return void
241   */
242  private function registerPreCommitAction(Entity $entity, bool $remove):void
243  {
244    $entityId = self::entityId($entity);
245    if (!empty($this->preCommitActions[$entityId])) {
246      // avoid recursion
247      return;
248    }
249    $this->preCommitActions[$entityId] = true;
250
251    /** @var ContactsService $contactsService */
252    $contactsService = $this->appContainer->get(ContactsService::class);
253    $musician = $entity->getMusician();
254    switch ($contactsService->registerContactSynchronization($musician)) {
255      case null:
256        $this->logInfo('Contacts synchronization is already scheduled for ' . $musician->getPublicName());
257        break;
258      case false:
259        $this->logError('Contacts-synchronization could not be registered for ' . $musician->getPublicName());
260        break;
261      default:
262        // be silent
263        break;
264    }
265
266    /** @var IConfig $cloudConfig */
267    $cloudConfig = $this->appContainer->get(IConfig::class);
268
269    $appName = $this->appContainer->get('appName');
270
271    $orchestraGroupId = $cloudConfig->getAppValue($appName, ConfigConstants::USER_GROUP_KEY);
272    if (empty($orchestraGroupId)) {
273      return;
274    }
275
276    /** @var EncryptionService $encryptionService */
277    $encryptionService = $this->appContainer->get(EncryptionService::class);
278    $executiveBoardProjectId = $encryptionService->getConfigValue(ConfigConstants::EXECUTIVE_BOARD_PROJECT_ID_KEY, -1);
279    if ($entity->getParticipationContext() == ParticipationContext::ASSOCIATES
280        || $entity->getProject()->getId() != $executiveBoardProjectId) {
281      // nothing to do
282      return;
283    }
284
285    /** @var IGroupManager $groupManager */
286    $groupManager = $this->appContainer->get(IGroupManager::class);
287
288    $managementGroupId = $orchestraGroupId . AuthorizationService::MANAGEMENT_GROUP_SUFFIX;
289    /** @var IGroup $managementGroup */
290    $managementGroup = $groupManager->get($managementGroupId);
291    /** @var IGroup $orchestraGroup */
292    $orchestraGroup = $groupManager->get($orchestraGroupId);
293
294    /** @var IUserManager $userManager */
295    $userManager = $this->appContainer->get(IUserManager::class);
296    $userId = $musician->getUserIdSlug();
297    $user = $userManager->get($userId);
298
299    $closureArguments = compact([
300      'groupManager',
301      'managementGroup',
302      'orchestraGroup',
303      'remove',
304      'user',
305      'userId',
306      'userManager',
307    ]);
308
309    $this->entityManager->registerPostCommitAction(
310      new GenericUndoable(
311        function() use ($closureArguments) {
312          extract($closureArguments);
313          $previousStatus = [
314            'orchestraGroup' => $orchestraGroup->inGroup($user),
315            'managementGroup' => $managementGroup->inGroup($user),
316          ];
317          if ($remove) {
318            $managementGroup->removeUser($user);
319          } else {
320            $managementGroup->addUser($user);
321            $orchestraGroup->addUser($user);
322          }
323          return $previousStatus;
324        },
325        function(array $previousStatus) use ($closureArguments) {
326          extract($closureArguments);
327          foreach ($previousStatus as $groupKey => $status) {
328            /** @var IGroup $group */
329            $group = $closureArguments[$groupKey];
330            if ($status && !$group->inGroup($user)) {
331              $group->addUser($user);
332            }
333            if (!$status && $group->inGroup($user)) {
334              $group->removeUser($user);
335            }
336          }
337        },
338      ),
339    );
340  }
