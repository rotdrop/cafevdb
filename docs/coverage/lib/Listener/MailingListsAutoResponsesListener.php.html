95  public function handle(Event $event):void
96  {
97    $nodes = [];
98    $eventClass = get_class($event);
99    switch ($eventClass) {
100      case NodeDeletedEvent::class:
101        /** @var NodeDeletedEvent $event */
102        $nodes[self::DEL_KEY] = $event->getNode();
103        break;
104      case NodeRenamedEvent::class:
105        /** @var NodeRenamedEvent $event */
106        $nodes[self::DEL_KEY] = $event->getSource();
107        $nodes[self::ADD_KEY] = $event->getTarget();
108        // rename gets another NodeWrittenEvent
109        break;
110      case NodeWrittenEvent::class:
111        /** @var NodeWrittenEvent $event */
112        $nodes[self::ADD_KEY] = $event->getNode();
113        break;
114      case NodeCopiedEvent::class:
115        /** @var NodeCopiedEvent $event */
116        $nodes[self::ADD_KEY] = $event->getTarget();
117        break;
118      case NodeCreatedEvent::class:
119      case NodeTouchedEvent::class:
120        /** @var NodeTouchedEvent $event */
121        $nodes[self::ADD_KEY] = $event->getNode();
122        break;
123      default:
124        return;
125    }
126
127    // initialize only now in order to keep the overhead for unhandled events small
128    $this->user = $this->appContainer->get(IUserSession::class)->getUser();
129    if (empty($this->user)) {
130      return;
131    }
132
133    $this->appName = $this->appContainer->get('appName');
134    $this->logger = $this->appContainer->get(ILogger::class);
135    $this->l = $this->appContainer->get(IL10N::class);
136
137    /** @var \OCP\Files\Node $node */
138    foreach ($nodes as $key => $node) {
139      if (false && $node instanceof \OC\Files\Node\NonExistingFile) {
140        unset($nodes[$key]);
141        continue;
142      }
143      $nodePath = $node->getPath();
144      if ($key == self::ADD_KEY) {
145        // Can ony use plain text files for the autoresponses.
146        try {
147          $eventMimeType = $node->getMimetype();
148        } catch (Throwable $t) {
149          // ignore
150          $eventMimeType = null;
151        }
152        if (!empty($eventMimeType) && $eventMimeType != 'text/plain' && $eventMimeType != 'text/markdown') {
153          unset($nodes[$key]);
154          continue;
155        }
156      }
157      $template = str_replace(
158        MailingListsService::TEMPLATE_FILE_SEPARATOR,
159        MailingListsService::TEMPLATE_MAILMAN_SEPARATOR,
160        pathinfo($nodePath, PATHINFO_FILENAME),
161      );
162      // first look at the base name, it must start with one of the known prefixes.
163      if (!str_starts_with($template, MailingListsService::TEMPLATE_MAILMAN_PREFIX)) {
164        unset($nodes[$key]);
165        continue;
166      }
167    }
168
169    if (empty($nodes)) {
170      return;
171    }
172
173    /** @var ConfigService $configService */
174    $configService = \OC::$server->query(ConfigService::class);
175    if (empty($configService)) {
176      return;
177    }
178
179    /** @var IRootFolder $rootFolder */
180    $rootFolder = $this->appContainer->get(IRootFolder::class);
181    if (empty($rootFolder)) {
182      return;
183    }

186    /** @var MailingListsService $listsService */
187    $listsService = $this->appContainer->get(MailingListsService::class);
188

190    $baseFolderShareUri = $listsService->ensureTemplateFolder($baseFolderPath);
191    $baseFolderPath = $userFolder . $baseFolderPath;

193    foreach ($nodes as $key => $node) {

195      $nodeBase = basename($nodePath);

197      if (!$this->matchPrefixDirectory($nodePath, $baseFolderPath)) {

228      }

230      if (empty($templateFolderPath)) {

239      );

241      foreach ($lists as $list) {
242        // try each list in turn but do not bail out on certain types of errors like "not found".
243        try {
244          if ($key == self::DEL_KEY) {
245            $listsService->setMessageTemplate($list, $template, null);
246            $this->logInfo('Removed ' . $template . ' from list ' . $list);
247          } else {

250            foreach ($lists as $list) {
251              $listsService->setMessageTemplate($list, $template, $templateUri);
252              $this->logInfo('Added ' . $template . ' to list ' . $list . ', URI ' . $templateUri);
253            }
254          }
255        } catch (Throwable $t) {
256          if (($t instanceof ClientException) && $t->getCode() == 404) {
257            $this->logException($t, 'Mailing-list "' . $list . '" probably does not exist on the server.');
258          } else {
259            $this->logException($t, 'Unable to modify template "' . $template . '" for list "' . $list . '".');
260          }
261        }
262      }
263    }
264  }

266  /**

268   *
269   * @param string $folderPrefix The folder-prefix to compare the

281  }
