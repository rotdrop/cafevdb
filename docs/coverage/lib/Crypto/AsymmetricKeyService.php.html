94  /**
95   * Initialize a private/public key-pair by either retreiving it from the
96   * config-space or generating a new one. If a new key-pair has to be
97   * generated the two events Events\BeforeEncryptionKeyPairChanged and
98   * Events\AfterEncryptionKeyPairChanged are fired. The old key-pair may be
99   * missing if the password used to secure the old private key is not
100   * available.
101   *
102   * @param null|string $ownerId The owner-id. If used for a group then it should
103   * be prefixed by '@'. If null then the currently logged in user is used.
104   *
105   * @param null|string $keyPassphrase The passphrase used to protect the
106   * private key. If null then the currently logged in user's password is used
107   * if the cloud's credentials store is able to provide the password.
108   *
109   * @param bool $forceNewKeyPair Generate a new key pair even if an
110   * old one is found.
111   *
112   * @return array<string, string>
113   * ```
114   * [
115   *   self::PRIVATE_ENCRYPTION_KEY_CONFIG => PRIV_KEY,
116   *   self::PUBLIC_ENCRYPTION_KEY_CONFIG => PUB_KEY,
117   * ]
118   * ```
119   *
120   * @throws Exceptions\EncryptionKeyException
121   */
122  public function initEncryptionKeyPair(?string $ownerId = null, ?string $keyPassphrase = null, bool $forceNewKeyPair = false)
123  {
124    if (empty($ownerId)) {
125      $ownerId = $this->getSessionUserId();
126    }
127
128    if (empty($ownerId)) {
129      throw new Exceptions\EncryptionException($this->l->t('The owner-id was not specified and could not be determined automatically.'));
130    }
131
132    if (empty($keyPassphrase)) {
133      $keyPassphrase = $this->getLoginPassword($ownerId);
134    }
135
136    if (empty($ownerId) || empty($keyPassphrase)) {
137      throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize SSL key-pair without user and password'));
138    }
139
140    if (!$forceNewKeyPair && !empty(self::$keyPairs[$ownerId])) {
141      return self::$keyPairs[$ownerId];
142    }
143
144    try {
145      $keyPair = $forceNewKeyPair ? null : $this->keyStorage->getKeyPair($ownerId, $keyPassphrase);
146    } catch (Throwable $t) {
147      $this->logException($t, $this->l->t('Unable to retrieve old key-pair for "%s".', [ $ownerId ]));
148      $keyPair = [];
149      // throw new Exceptions\EnduserNotificationException('DEBUG: IGNORE FAILED KEYPAIR REQUESTS');
150    }
151    if (empty($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG]) || empty($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG])) {
152
153      $oldKeyPair = self::$keyPairs[$ownerId] ?? null;
154      if (empty($oldKeyPair) && $ownerId == $this->getSessionUserId()) {
155        try {
156          $oldKeyPair = $this->keyStorage->getKeyPair($ownerId, $keyPassphrase);
157        } catch (Throwable $t) {
158          $this->logException($t, 'Unable to fetch old encryption key pair for "' . $ownerId . '".');
159        }
160      }
161
162      // make sure the old key-pair cache is set
163      self::$keyPairs[$ownerId] = $oldKeyPair;
164
165      $this->eventDispatcher->dispatchTyped(new Events\BeforeEncryptionKeyPairChanged($ownerId, $oldKeyPair));
166
167      $this->keyStorage->backupKeyPair($ownerId); // make a backup-copy if old key exists
168      $keyPair = $this->keyStorage->generateKeyPair($ownerId, $keyPassphrase);
169
170      $this->eventDispatcher->dispatchTyped(new Events\AfterEncryptionKeyPairChanged($ownerId, $oldKeyPair, $keyPair));
171    }
172
173    self::$keyPairs[$ownerId] = $keyPair;
174
175    // ensure that the cached cryptor has the correct key
176    /* $cryptor = */$this->getCryptor($ownerId)
177      ->setPrivateKey($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null)
178      ->setPublicKey($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? null);
179
180    return $keyPair;
181  }

259  {
260    /** @var AsymmetricCryptorInterface $cryptor */
261    $cryptor = self::$cryptors[$ownerId] ?? null;
262    if (empty($cryptor)) {
263      $keyPair = self::$keyPairs[$ownerId] ?? null;
264      if (empty($keyPair) && $ownerId == $this->getSessionUserId()) {
265        // effect of LazyGhosts: EncryptionService CTOR has not been run

268      }
269      $privKey = $keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null;
270      $pubKey = $keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? $this->keyStorage->getPublicKey($ownerId);
271      $cryptor = clone $this->cryptorPrototype;
272      $cryptor->setPrivateKey($privKey);
273      $cryptor->setPublicKey($pubKey);
274      self::$cryptors[$ownerId] = $cryptor;
275    }

457    $requestData = $this->appContainer->get(ITimeFactory::class)->getTime();
