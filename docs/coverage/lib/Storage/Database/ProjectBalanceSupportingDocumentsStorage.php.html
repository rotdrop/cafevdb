126  /**
127   * Return the sequence number if the given path refers to a document
128   * container directory, otherwise return null.
129   *
130   * @param string $path File-system path.
131   *
132   * @return null|int
133   */
134  private function isContainerDirectory(string $path):?int
135  {
136    switch ($this->project->getType()) {
137      case ProjectType::TEMPLATE:
138        break;
139      case ProjectType::PERMANENT:
140        // Folder have a year prefix and the supporting document has the year in its name
141        if (preg_match('|^/?' . '\d{4}' . '/' . $this->project->getName() . '-' . '\d{4}' . '-' . '(\d{3})/?$|', $path, $matches)) {
142          return $matches[1];
143        }
144        break;
145      case ProjectType::TEMPORARY:
146        // project-name contains already the year
147        if (preg_match('|^/?' . $this->project->getName() . '-' . '(\d{3})/?$|', $path, $matches)) {
148          return $matches[1];
149        }
150        break;
151    }
152    return null;
153  }

294  public function rename(string $path1, string $path2): bool
295  {

297

299    $path2 = $this->buildPath($path2);
300    list('dirname' => $dirName1, /* 'basename' => $baseName1 */) = self::pathinfo($path1);
301    list('dirname' => $dirName2, 'basename' => $baseName2) = self::pathinfo($path2);
302    $baseName2 = self::pathInfo($path2, PATHINFO_BASENAME);
303
304    $sequence1 = $this->isContainerDirectory($path1);
305    $sequence2 = $this->isContainerDirectory($path2);
306
307    if (($sequence1 === null) !== ($sequence2 === null)) {
308      // balance folder may only be renamed to balance folders
309      // $this->logInfo('SEQUENCES ' . $sequence1 . ' / ' . $sequence2);
310      return false;
311    }
312
313    $year1 = $this->isYearDirectory($path1);
314    $year2 = $this->isYearDirectory($path2);
315
316    if (($year1 === null) !== ($year2 === null)) {
317      // year directories may only be renamed to year directories
318      // $this->logInfo('YEARS ' . $year1 . ' / ' . $year2);
319      return false;
320    }
321
322    /** @var Entities\DatabaseStorageDirEntry $dirEntry */
323    $dirEntry = $this->fileFromFileName($path1);
324    if (empty($dirEntry)) {

326      return false;
327    }
328

330      $dirEntry = $this->persistInMemoryFileNode($dirEntry);
331    }
332
333    if ($dirName1 != $dirName2) {
334      $parent2 = $this->fileFromFileName($dirName2);
335      if (empty($parent2)) {
336        // $this->logInfo('NO PARENT2 for ' . $path2);
337        return false;
338      }
339    }
340

342    try {
343      $dirEntry->setName($baseName2);
344      if (!empty($parent2)) {
345        $dirEntry->setParent($parent2);
346      }
