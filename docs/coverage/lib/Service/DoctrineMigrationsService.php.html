167  public function apply(string $version, EnumMigrationDirection $direction = EnumMigrationDirection::UP): void
168  {
169    $dependencyFactory = $this->getDependencyFactory();
170    $dependencyFactory->getMetadataStorage()->ensureInitialized();
171    $planCalculator = $dependencyFactory->getMigrationPlanCalculator();
172    $migrationClassName = MigrationsNamespace::class . '\\Version' . $version;
173    try {
174      $plan = $planCalculator->getPlanForVersions(
175        array_map(static fn (string $version): Version => new Version($version), [$migrationClassName]),
176        $direction->value,
177      );
178    } catch (MigrationClassNotFound $t) {
179      throw new InvalidArgumentException(
180        $this->l->t('A migration with the version "%s" does not exist.', $version),
181        previous: $t,
182      );
183    }
184    $numTransactional = 0;
185    $numNonTransactional = 0;
186    $planItems = $plan->getItems();
187    foreach ($planItems as $planItem) {
188      $transactional = (int)$planItem->getMigration()->isTransactional();
189      $numTransactional += $transactional;
190      $numNonTransactional += 1 - $transactional;
191    }
192    if ($numTransactional == 0) {
193      $allOrNothing = false;
194    } elseif ($numNonTransactional == 0) {
195      $allOrNothing = true;
196    } else {
197      throw new InvalidArgumentException(
198        $this->l->t(
199          'The migration with the version "%s" involves other additional migrations,'
200          . ' however, this mixed structural and content changes which is not supported at the moment.', $version),
201      );
202    }
203
204    $this->logInfo('Executing {version} {direction}', [
205      'direction' => $plan->getDirection(),
206      'version' => $version,
207    ]);
208    $migrator = $dependencyFactory->getMigrator();
209    $migratorConfiguration = new MigratorConfiguration()
210      ->setDryRun(false)
211      ->setTimeAllQueries(true)
212      ->setAllOrNothing($allOrNothing);
213    /* $sql = */$migrator->migrate($plan, $migratorConfiguration);
214
215    foreach ($planItems as $planItem) {
216      $versionString = substr((string)$planItem->getVersion(), -14);
217      $description = $planItem->getMigration()->getDescription();
218      if ($direction === EnumMigrationDirection::UP) {
219        $this->appliedMigrations[$versionString] = $description;
220        unset($this->unappliedMigrations[$versionString]);
221        $this->executedMigrations[] = $versionString;
222      } else {
223        $this->unappliedMigrations[$versionString] = $description;
224        unset($this->appliedMigrations[$versionString]);
225        $index = array_search($versionString, $this->executedMigrations);
226        if ($index !== false) {
227          unset($this->executedMigrations[$versionString]);
228        }
229      }
230    }
231  }
