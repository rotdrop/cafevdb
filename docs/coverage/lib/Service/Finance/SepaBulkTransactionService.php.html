517  /**

520   * contain payments for different projects this is not implemented here.
521   *

534   * should only return more than one item if there are also deleted items.
535   */
536  public function generateProjectPayments(
537    Entities\ProjectParticipant $participant,
538    array $receivableOptions,
539    ?\DateTimeInterface $transactionDueDate = null,
540  ): Entities\CompositePayment {
541    $payments = new ArrayCollection();
542    $totalAmount = MonetaryNumberType::zero();
543    $project = $participant->getProject();
544    $musician = $participant->getMusician();
545
546    /** @var Entities\CompositePayment $compositePayment */
547    $compositePayment = (new Entities\CompositePayment)
548      ->setProjectParticipant($participant)
549      ->setProjectPayments($payments)
550      ->setAmount($totalAmount)
551      ->setSubject('');
552
553    if (empty($receivableOptions)) {
554      return $compositePayment;
555    }
556
557    /** @var Entities\ProjectParticipantFieldDataOption $receivableOption */
558    foreach ($receivableOptions as $receivableOption) {
559      if ($project != $receivableOption->getField()->getProject()) {
560        throw new RuntimeException(
561          $this->l->t('Refusing to generate payments for mismatching projects, current is "%s", but the receivable belongs to project "%s".', [
562            $project->getName(),
563            $receivableOption->getField()->getProject()->getName(),
564          ]));
565      }
566      $receivableDueDate = $receivableOption->getField()->getDueDate();
567      $depositDueDate = $receivableOption->getField()->getDepositDueDate();
568      /** @var Entities\ProjectParticipantFieldDatum $receivable */
569      foreach ($receivableOption->getMusicianFieldData($musician) as $receivable) {
570        $paidAmount = $receivable->amountPaid();
571        $payableAmount = $receivable->amountPayable();
572        $depositAmount = $receivable->depositAmount() ?? MonetaryNumberType::zero();
573        if ($payableAmount->sign() * $depositAmount->sign() < 0) {
574          throw new RuntimeException(
575            $this->l->t(
576              'Payable amount "%1$s" and deposit amount "%2$s" should have the compatible signs.', [
577                $payableAmount->toDecimal(2), $depositAmount->toDecimal(2),
578              ]));
579        }
580        if (!empty($transactionDueDate) && !empty($receivableDueDate)) {
581          if ($payableAmount->gt(0)) {
582            // debit note
583            if ($receivableDueDate <= $transactionDueDate) {
584              // past due-date, just keep billing the entire amount
585            } elseif ($depositDueDate <= $transactionDueDate) {
586              // past deposit-due date, charge the deposit
587              $payableAmount = $depositAmount;
588            } else {
589              // too early, just don't charge anything
590              $payableAmount->assign(0);
591              $this->logInfo('NOT YET DUE; DOING NOTHING');
592            }
593          } else {
594            // bank transfer
595            if ($transactionDueDate <= $depositDueDate) {
596              // early payment, only up to deposit
597              $payableAmount = $depositAmount;
598            } else {
599              // just transfer anything, i.e. keep the amount as is
600            }
601          }
602        }
603        $debitAmount = $payableAmount->sub($paidAmount)->round(2);
604        if ($debitAmount->eq(0)) {
605          // No need to debit empty amounts
606          // @todo Perhaps empty amounts should also be recorded.
607          continue;
608        }
609        $this->logInfo('RECEIVABLE FIELD ID ' . $receivable->getField()->getId());
610        $this->logInfo('RECEIVABLE FIELD MULTIPLICITY ' . $receivable->getField()->getMultiplicity()->value);
611        $this->refreshEntity($receivable->getField());
612        $this->logInfo('RECEIVABLE FIELD NAME ' . $receivable->getField()->getName());
613        /** @var Entities\ProjectPayment $payment */
614        $payment = (new Entities\ProjectPayment)
615                 ->setProject($project)
616                 ->setMusician($musician)
617                 ->setReceivable($receivable)
618                 ->setReceivableOption($receivableOption)
619                 ->setAmount($debitAmount)
620                 ->setCompositePayment($compositePayment)
621                 ->setSubject($receivable->paymentReference())
622                 ->setIsDonation(false)
623          ;
624
625        $payments->add($payment);
626        $totalAmount->addEq($debitAmount);
627      }
628    }
629
630    $compositePayment
631      ->setAmount($totalAmount)
632      ->updateSubject(
633        maxLength: FinanceService::SEPA_PURPOSE_LENGTH,
634        transliterate: fn($x) => $this->sepaTransliterator->translit($x),
635      );
636
637    return $compositePayment;
638  }

899      } elseif ($bulkTransaction instanceof Entities\SepaDebitNote) {

906      foreach ($bulkTransaction->getPayments() as $payment) {

910        } elseif ($project != $paymentProject) {
911          throw new UnexpectedValueException($this->l->t(
912            'Conflicting projects "%1$s" vs. "%2$s".', [
913              (string)$project,  (string)$paymentProject
914            ]));
915        }

919        $timeStamp,
920        $transactionType,
921        !empty($project) ? $project->getName() : null,
922        $format,
923      ])) . '.' . $exporter->fileExtension($bulkTransaction);
924

927      if (empty($exportDocument)) {
