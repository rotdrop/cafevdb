110        $this->l->t('Unsupported bulk-transaction class: "%s".', get_class($transaction)));
111    }
112
113    $namespace = $isDebit ? self::DEBIT_NAMESPACE : self::TRANSFER_NAMESPACE;

117    $dom->formatOutput = true;

119    $document = $dom->createElementNS($namespace, 'Document');
120    $dom->appendChild($document);
121    $root = $this->append($dom, $document, $rootName);

125    $messageId = $this->messageId($now);
126
127    // group header

129    $this->append($dom, $header, 'MsgId', $messageId);
130    $this->append($dom, $header, 'CreDtTm', $now->format('Y-m-d\TH:i:s'));
131    $this->append($dom, $header, 'NbOfTxs', (string)count($payments));
132    $this->append($dom, $header, 'CtrlSum', $this->decimal($this->controlSum($payments)));
133    $initiator = $this->append($dom, $header, 'InitgPty');

135
136    $dueDate = ($transaction->getDueDate() ?? $now)->format('Y-m-d');
137
138    if ($isDebit) {
139      // one payment-information block per sequence type
140      $groups = [];
141      foreach ($payments as $payment) {
142        $nonRecurring = $payment->getSepaDebitMandate()->getNonRecurring();
143        $sequence = $nonRecurring ? self::SEQUENCE_ONCE : self::SEQUENCE_FOLLOWING;
144        $groups[$sequence][] = $payment;
145      }
146      foreach ($groups as $sequence => $groupPayments) {
147        $this->appendDebitBlock($dom, $root, $messageId . '-' . $sequence, $sequence, $dueDate, $groupPayments);
148      }
149    } elseif (!empty($payments)) {
150      $this->appendTransferBlock($dom, $root, $messageId . '-1', $dueDate, $payments);
151    }
