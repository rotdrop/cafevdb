150  public function generateAccountsAutocompleteData(): ?array
151  {
152    $accountsExport = $this->encryptionService->getAppValue(AdminSettings::GNU_CASH_ACCOUNTS_TREE_DATA_KEY);
153    if (empty($accountsExport)) {
154      $this->logError('ACCOUNTS EXPORT FILE IS NOT SET');
155      return null;
156    }
157    $accountsExportFile = $this->userStorage->getFile($accountsExport);
158    if (empty($accountsExportFile)) {
159      $this->logError('UNABLE TO OPEN ACCOUNTS EXPORTS FILE ' . $accountsExport);
160      return null;
161    }
162
163    $accountsAutocompleteFile = $this->appStorage->getFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE, throw: false);
164    if ($accountsAutocompleteFile !== null && $accountsExportFile->getMTime() <= $accountsAutocompleteFile->getMTime()) {
165      return json_decode($accountsAutocompleteFile->getContent(), true);
166    }
167
168    $permanentProjects = $this->entityManager->getRepository(Entities\Project::class)->findNames(onlyType: ProjectType::PERMANENT);
169
170    $leafAccountRe = '/:([^0-9]+[0-9]{4}|'
171      . implode('|', array_map(fn(string $name) => preg_quote($name), $permanentProjects))
172      . ')$/';
173
174    $autocompleteData = [
175      self::GNU_CASH_EXPENSE_KEY => [],
176      self::GNU_CASH_INCOME_KEY => [],
177    ];
178
179    $exportData = explode("\n", $accountsExportFile->getContent());
180
181    foreach ($exportData as $dataLine) {
182      $lineData = str_getcsv($dataLine, ';', escape: '');
183      if (empty($lineData) || $lineData[0] === null) {
184        break;
185      }
186      $type = strtolower($lineData[0]);
187      if ($type != self::GNU_CASH_EXPENSE_KEY && $type != self::GNU_CASH_INCOME_KEY) {
188        continue;
189      }
190      $account = preg_replace($leafAccountRe, '', $lineData[1]);
191      $autocompleteData[$type][] = $account;
192    }
193    foreach ($autocompleteData as &$accounts) {
194      $accounts = array_unique($accounts);
195      sort($accounts);
196      $accounts = array_values($accounts);

198      $count = count($accounts);
199      for ($i = 0; $i < $count - 1; ++$i) {
200        if (str_starts_with($accounts[$i + 1], $accounts[$i] . ':')) {

202        }
203      }
204      $accounts = array_values($accounts);
205    }

208      $accountsAutocompleteFile = $this->appStorage->ensureFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE);
209    }
210    $accountsAutocompleteFile->putContent(json_encode($autocompleteData));
211
212    return $autocompleteData;

293
294    $receivableAccounts = [];
295    $musician = $compositePayment->getProjectParticipant()->getMusician();
296    $description = $compositePayment->getSubject() . '; ' . $musician->getPublicName(false);

298    // it need not be the case that a composite payment result in the same
