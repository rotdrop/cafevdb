147   * $calendarOffset is non-empty, then the resulting absolute
148   * distance from $fromDate will be the maxium of both offsets. The
149   * offsets must not have different signs.
150   *
151   * Business-days are relative to the SEPA TARGET2 calendar,
152   * {@see https://en.wikipedia.org/wiki/TARGET2#Holidays}
153   *
154   * @param int $businessOffset Offset in TARGET2 business days.
155   *
156   * @param int|null $calendarOffset Offset in calendar days.
157   *
158   * @param \DateTimeInterface|null $fromDate The pivot-date.
159   *
160   * @return DateTimeInterface The resulting date.
161   */
162  public function targetDeadline(
163    int $businessOffset,
164    ?int $calendarOffset = null,
165    ?\DateTimeInterface $fromDate = null,
166  ):DateTimeInterface {
167    /** @var CarbonImmutable $fromDate */
168    if (empty($fromDate)) {
169      $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
170    } else {
171      $fromDate = (new CarbonImmutable)
172                ->setTimezone($fromDate->getTimezone())
173                ->setTimestamp($fromDate->getTimestamp());
174    }
175
176    if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
177      throw new RuntimeException(
178        $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
179          $businessOffset, $calendarOffset
180        ]));
181    }
182
183    CarbonImmutable::setHolidaysRegion('target2');
184    $businessDeadline = $fromDate->addBusinessDays($businessOffset);
185    if (!empty($calendarOffset)) {
186      $calendarDeadline = $fromDate->addDays($calendarOffset);
187
188      $businessInterval = $fromDate->diff($businessDeadline, true);
189      $calendarInterval = $fromDate->diff($calendarDeadline, true);
190
191      if ($calendarInterval->days > $businessInterval->days) {
192        $businessDeadline = $calendarDeadline->nextBusinessDay();
193      }
194    }
195
196    return $businessDeadline;
