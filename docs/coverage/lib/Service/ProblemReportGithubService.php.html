139   * Submit a report: either create a new issue or comment on the existing
140   * issue for the same fingerprint.
141   *
142   * @param string $fingerprint
143   *
144   * @param string $title
145   *
146   * @param string $body Issue body for new issues.
147   *
148   * @param string $comment Comment for the issue (new and existing).
149   *
150   * @return array{url:string, created:bool}
151   *
152   * @throws Exceptions\EnduserNotificationException
153   */
154  public function report(string $fingerprint, string $title, string $body, string $comment): array
155  {
156    $repository = $this->getRepository();
157    if (!$this->isConfigured() || !preg_match(self::REPOSITORY_PATTERN, $repository)) {
158      throw new Exceptions\EnduserNotificationException($this->l->t('The GitHub problem-report channel is not configured.'));
159    }
160
161    $known = $this->findIssue($fingerprint, $repository);
162    if ($known !== null) {
163      try {
164        $this->request('POST', '/repos/' . $repository . '/issues/' . $known['issue_number'] . '/comments', [
165          'body' => $this->l->t('The problem has been reported again.') . "\n\n" . $comment,
166        ]);
167        $this->touchIssue($known['id']);
168        return [ 'url' => $known['issue_url'], 'created' => false ];
169      } catch (Exceptions\EnduserNotificationException $e) {
170        $code = $e->getPrevious()?->getCode();
171        if ($code !== 404 && $code !== 410) {
172          throw $e;
173        }
174        // the issue has been deleted or transferred, start afresh
175        $this->forgetIssue($known['id']);
176      }
177    }
178
179    $issue = $this->request('POST', '/repos/' . $repository . '/issues', [
180      'title' => $title,
181      'body' => $body,
182    ]);
183    $url = (string)($issue['html_url'] ?? '');
184    $number = (int)($issue['number'] ?? 0);
185    if (empty($url) || $number <= 0) {
186      throw new Exceptions\EnduserNotificationException($this->l->t('GitHub returned an unexpected response.'));
187    }
188    $this->rememberIssue($fingerprint, $repository, $number, $url);
189    try {
190      $this->request('POST', '/repos/' . $repository . '/issues/' . $number . '/comments', [ 'body' => $comment ]);
191    } catch (Throwable $t) {
192      // the issue exists, the attribution is nice to have
193      $this->logException($t, 'Unable to add attribution comment to GitHub issue ' . $url);
194    }
195    return [ 'url' => $url, 'created' => true ];
196  }

198  /**
199   * @param string $method
200   *
201   * @param string $path
202   *
203   * @param null|array $payload
204   *
205   * @param null|string $token Override the stored token.
206   *
207   * @return array Decoded response.
208   *
209   * @throws Exceptions\EnduserNotificationException
210   */
211  private function request(string $method, string $path, ?array $payload = null, ?string $token = null): array
212  {
213    if ($token === null) {
214      try {
215        $token = $this->crypto->decrypt((string)$this->encryptionService->getAppValue(AdminSettings::PROBLEM_REPORT_GITHUB_TOKEN_KEY, ''));
216      } catch (Throwable $t) {
217        throw new Exceptions\EnduserNotificationException($this->l->t('Unable to decrypt the GitHub access token.'), 0, $t);
218      }
219    }
220    $options = [
221      'headers' => [
222        'Accept' => 'application/vnd.github+json',
223        'Authorization' => 'Bearer ' . $token,
224        'X-GitHub-Api-Version' => self::API_VERSION,
225        'User-Agent' => $this->appName . '-problem-report',
226      ],
227      'timeout' => self::TIMEOUT,
228    ];
229    if ($payload !== null) {
230      $options['body'] = json_encode($payload, JSON_UNESCAPED_SLASHES|JSON_UNESCAPED_UNICODE|JSON_THROW_ON_ERROR);
231      $options['headers']['Content-Type'] = 'application/json';
232    }
233    try {
234      $client = $this->clientService->newClient();
235      $response = $client->{strtolower($method)}(self::API_BASE_URL . $path, $options);
236      $decoded = json_decode((string)$response->getBody(), true);
237      return is_array($decoded) ? $decoded : [];
238    } catch (Throwable $t) {
239      // never log the options, they contain the token
240      $this->logError('GitHub request ' . $method . ' ' . $path . ' failed: ' . get_class($t) . ' ' . $t->getCode());
241      throw new Exceptions\EnduserNotificationException(
242        $this->l->t('Communication with GitHub failed (code %d).', (int)$t->getCode()),
243        0,
244        $t,
245      );
246    }
247  }
