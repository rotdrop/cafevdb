267    array $instrumentValues = [],
268    ?string $only = null,

272
273    $this->logDebug($context . ' / ' . print_r($recordId, true) . ' / ' . print_r($instrumentValues, true));
274    if (empty($instrumentValues)) {
275      $instrumentValues = [];
276    }
277
278    if ($contextCase == EnumValidateInstrumentsContext::MUSICIAN && empty($recordId['projectId'])) {
279      if (empty($recordId['id'])) {
280        throw new InvalidArgumentException($this->l->t("Musician-id is missing"));
281      }
282      /** @var Entities\Musician $musician */
283      $musician = $this->getDatabaseRepository(Entities\Musician::class)->find([ 'id' => $recordId['id'] ]);

285        throw new UnexpectedValueException($this->l->t("Unable to fetch the musicican with given key %s from the database.", print_r($recordId, true)));
286      }
287    } else {
288      if (empty($recordId['projectId']) || empty($recordId['musicianId'])) {
289        throw new InvalidArgumentException(
290          $this->l->t(
291            "Project- or musician-id is missing (%s/%s)",
292            [ $recordId['projectId'], $recordId['musicianId'], ]
293          ),
294        );
295      }
296
297      /** @var Entities\ProjectParticipant $projectParticipant */
298      $projectParticipant = $this->getDatabaseRepository(Entities\ProjectParticipant::class)->find([

301      ]);
302      if (empty($projectParticipant)) {
303        throw new UnexpectedValueException(

306      }
307
308      $projectInstruments = [];
309      foreach ($projectParticipant['projectInstruments'] as $projectInstrument) {
310        $projectInstruments[$projectInstrument['instrument']['id']] = $projectInstrument;
311      }
312

314    }

316    $musicianInstruments = [];

327    }

336    });
