46/**
47 * Apply all unapplied migrations one by one. In contrast to the stock
48 * "migrations:migrate" command this does not use "all-or-nothing" for the
49 * complete set, as structural (non-transactional) and content (transactional)
50 * migrations must not be mixed in one run.
51 *
52 * Output depends on the verbosity:
53 * - quiet: nothing
54 * - normal: one line per applied migration, errors
55 * - verbose (-v): additionally a summary table after the run
56 * - very verbose (-vv): additionally a summary table before the run
57 */

116  /**
117   * Authenticate non-interactively if "--user" and "--password-file" are
118   * given, otherwise prompt on the console.
119   *

126  protected function authenticate(InputInterface $input, OutputInterface $output): int
127  {
128    $userId = $input->getOption('user');
129    $passwordFile = $input->getOption('password-file');
130    if (empty($userId) || empty($passwordFile)) {
131      return $this->interactiveAuthenticate($input, $output);
132    }
133
134    $password = is_readable($passwordFile) ? file_get_contents($passwordFile) : false;
135    $user = $this->userManager->get($userId);
136    if ($password === false || $password === '' || $user === null) {
137      // TRANSLATORS: Error message being unable to validate the given userId
138      // TRANSLATORS: or get the password for that person. The userId is
139      // TRANSLATORS: sustutued for %s.
140      $output->writeln('<error>' . $this->l->t('Unable to obtain the credentials for "%s".', $userId) . '</error>');
141      return self::ERROR;
142    }
143    $this->userSession->setUser($user);
144    // The login event-handler binds the encryption-service and the entity-manager.
145    if (!$this->userSession->login($userId, $password)) {
146      $output->writeln('<error>' . $this->l->t('Login failed.') . '</error>');
147      return self::ERROR;
148    }
149    $this->userId = $userId;
150    $this->userPassword = $password;
151    $this->entityManager = $this->appContainer->get(EntityManager::class);
152    return self::SUCCESS;
153  }

156  protected function execute(InputInterface $input, OutputInterface $output): int
157  {
158    $result = $this->authenticate($input, $output);
159    if ($result != 0) {
160      return self::ERROR;
161    }
162
163    $dryRun = false;
164    try {
165      /** @var MigrationsServiceInterface $migrationsService */
166      $migrationsService = $this->appContainer->get(MigrationsServiceInterface::class);
167
168      if ($input->getOption('list')) {
169        $this->renderSummary($output, $migrationsService);
170        return self::SUCCESS;
171      }
172
173      $unapplied = $migrationsService->getUnapplied();
174      if (empty($unapplied)) {
175        $output->writeln($this->l->t('No unapplied migrations.'));
176        return self::SUCCESS;
177      }
178
179      if ($output->isVeryVerbose()) {
180        $this->renderSummary($output, $migrationsService);
181      }
182
183      $dryRun = (bool)$input->getOption('dry-run');
184
185      foreach ($unapplied as $version => $description) {
186        if ($dryRun) {
187          $output->writeln($this->l->t('Would apply migration %1$s: %2$s', [ $version, $description ]));
188          continue;
189        }
190        // TRANSLATORS: %1$s is version number for a database migration, %2$
191        // TRANSLATORS: its description.
192        $output->writeln($this->l->t('Applying migration %1$s: %2$s', [ $version, $description ]));
193        $migrationsService->apply($version);
194      }
195    } catch (Throwable $t) {
196      $output->writeln('<error>' . $this->l->t('Migration failed: %s', $t->getMessage()) . '</error>');
197      if ($output->isVerbose()) {
198        $output->writeln($t->getTraceAsString());
199      }
200      return self::ERROR;
201    }
202
203    if ($dryRun) {
204      $output->writeln($this->l->t('Dry run: %d migration(s) would be applied.', count($unapplied)));
205      return self::SUCCESS;
206    }
207
208    $output->writeln($this->l->t('Applied %d migration(s).', count($unapplied)));
209
210    if ($output->isVerbose()) {
211      $migrationsService->clearCache();
212      $this->renderSummary($output, $migrationsService);
213    }
214
215    return self::SUCCESS;
216  }

218  /**
219   * Render a table of all known migrations with status and execution date. The table is fitted to the terminal width, long descriptions are
220   * wrapped into taller cells.
221   *
222   * @param OutputInterface $output
223   *
224   * @param MigrationsServiceInterface $migrationsService
225   *
226   * @return void
227   */
228  private function renderSummary(OutputInterface $output, MigrationsServiceInterface $migrationsService): void
229  {
230    $applied = $migrationsService->getApplied();
231    $unapplied = $migrationsService->getUnapplied();
232
233    $executed = [];
234    /** @var DoctrineMigrationsVersion $entry */
235    foreach ($this->entityManager->getRepository(DoctrineMigrationsVersion::class)->findAll() as $entry) {
236      $executed[substr($entry->getVersion(), -14)] = $entry;
237    }
238
239    // Symfony silently falls back to 80 columns, so only trust it if the
240    // width can really be determined.
241    $terminalWidth = self::DEFAULT_WIDTH;
242    if (getenv('COLUMNS') !== false || (defined('STDOUT') && stream_isatty(STDOUT))) {
243      $terminalWidth = (new Terminal)->getWidth() ?: self::DEFAULT_WIDTH;
244    }
245    $descriptionWidth = max(self::MIN_DESCRIPTION_WIDTH, $terminalWidth - self::FIXED_COLUMNS_WIDTH);
246
247    $migrations = $applied + $unapplied;
248    ksort($migrations);
249
250    $rows = [];
251    foreach ($migrations as $version => $description) {
252      $entry = $executed[$version] ?? null;
253      $rows[] = [
254        $version,
255        isset($applied[$version]) || $entry !== null ? 'migrated' : 'new',
256        $entry?->getExecutedAt()?->format('Y-m-d') ?? '',
257        wordwrap((string)$description, $descriptionWidth, "\n", true),
258      ];
259    }
260
261    $table = new Table($output);
262    $table
263      ->setHeaders([ 'Version', 'Status', 'Migrated At', 'Description' ])
264      ->setRows($rows);
265    $table->render();
266  }
