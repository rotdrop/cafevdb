425  /**
426   * Check for a valid database connection, try to establish a connection if
427   * not connected.
428   *
429   * @return bool
430   */
431  public function connected():bool
432  {
433    try {
434      $connection = $this->getConnection();
435    } catch (Exceptions\DatabaseNotConnectedException $e) {
436      if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
437        $this->logException($e);
438      }
439      return false;
440    }
441    $params = $connection->getParams();
442    $impossible = false;
443    foreach ([ 'host', 'user', 'password', ConfigConstants::APP_DB_NAME ] as $key) {
444      if (empty($params[$key])) {
445        $impossible = true;
446      }
447    }
448    if ($impossible) {
449      $this->logError('Unable to access database, connection parameters are unset');
450      return false;
451    }
452    try {
453      if (!$connection->isConnected()) {
454        // $connection->connect() is deprecated
455        $connection->getNativeConnection();
456        return $connection->isConnected();
457      }
458    } catch (Throwable $t) {
459      if (str_contains($this->request->getPathInfo() ?? '', 'apps/' . $this->appName)) {
460        $this->logException($t, 'Caught execption checking connection to database server ' . $params['user'] . '@' . $params['host'] . ':' . $params[ConfigConstants::APP_DB_NAME]);
461      }
462      return false;
463    }
464    return true;
465  }

837  /**
838   * Persist an entity after performing some special tweaks:
839   *
840   * - nesting primary foreign keys is not supported, but a
841   *   work-around can be established by additional associations. Fill
842   *   those to-one associations with references if the necessary keys
843   *   are available.
844   *
845   * Then just move the modified entity on to the ordinary persister.
846   *
847   * @param object $entity The entity instance to persist.
848   *
849   * @return void
850   *
851   * @todo This unfortunately does not hack similar problems with
852   * cascade="persist". There the "stock" persist operation seemingly
853   * still causes problems.
854   */
855  public function persist(object $entity):void
856  {
857    $meta = $this->getClassMetadata(get_class($entity));
858    if ($meta->containsForeignIdentifier) {
859      $columnValues = $meta->getIdentifierColumnValues($entity);
860
861      foreach ($meta->associationMappings as $property => $association) {
862
863        if (!$meta->isSingleValuedAssociation($property)) {
864          // only to-one makes sense
865          continue;
866        }
867
868        if (array_search($property, $meta->identifier) !== false) {
869          // all primary keys must already be there
870          continue;
871        }
872
873        if (!empty($meta->getFieldValue($entity, $property))) {
874          // don't override values already present
875          continue;
876        }
877
878        try {
879          $targetEntity = $association['targetEntity'];
880          $targetMeta = $this->getClassMetadata($targetEntity);
881
882          // the actual keys may need remapping through join columns
883          $targetColumnValues = $columnValues;
884          foreach ($association['joinColumns'] as $joinColumn) {
885            $sourceColumn = $joinColumn->name;
886            $targetColumn = $joinColumn->referencedColumnName;
887            if (isset($targetColumnValues[$sourceColumn])) {
888              $value = $targetColumnValues[$sourceColumn];
889              unset($targetColumnValues[$sourceColumn]);
890              $targetColumnValues[$targetColumn] = $value;
891            }
892          }
893
894          $targetEntityId = $targetMeta->extractKeyValues($targetColumnValues);
895          if (empty($targetEntityId)) {
896            $reference = null;
897          } else {
898            $reference = $this->getReference($targetEntity, $targetEntityId);
899          }
900
901          $meta->setFieldValue($entity, $property, $reference);
902
903        } catch (Throwable $t) {
904          if (\ROT_DROP_PHPUNIT) {
905            throw $t;
906          }
907          // $this->logException($t);
908        }
909      }
910
911    }
912    parent::persist($entity);
913  }

915  /**
916   * Register a pre-commit action and optionally an associated
917   * undo-action. The actions are run after the currently active --
918   * potentially nested -- transaction is committed. A commit will only
919   * execute the actions registered at the current transaction nesting level.
920   *
921   * If the action succeeds, then its $undoAction will be registered for the
922   * case that the commit throws an exception. In this case all
923   * undo-actions will be executed in reverse order.
924   *
925   * In case of an error $action must throw an \Exception, its return
926   * value is ignored.
927   *
928   * The callables need to run "stand-alone" without parameters.
929   *
930   * @param Closure|IUndoable $action Action to register.
931   *
932   * @param null|callable $undo The associated undo-action. If $actionm
933   * instanceof IUndoable then the $undo action is ignored. It should rather
934   * be specified while constructing the IUndoable instance. The $undo
935   * callable receives the return value of the $action callable as argument.
936   *
937   * @return UndoableRunQueue  Return the run-queue for  easy chaining
938   * via UndoableRunQueue::register().
939   */
940  public function registerPreCommitAction(Closure|IUndoable $action, ?Closure $undo = null):UndoableRunQueue
941  {
942    if (!$this->isOwnTransactionActive()) {
943      throw new Exceptions\DatabaseTransactionNotActiveException($this->l->t('There is no active database transaction, cannot register pre-commit actions.'));
944    }
945    $level = $this->getOwnTransactionNestingLevel() - 1;
946    $actions = $this->preCommitActions[$level];
947    if ($action instanceof Closure) {
948      $actions->register(new GenericUndoable($action, $undo));
949    } elseif ($action instanceof IUndoable) {
950      $actions->register($action);
951    } else {
952      throw new RuntimeException($this->l->t('$action must be callable or an instance of "%s".', IUndoable::class));
953    }
954    return $actions;
955  }

959   * execution matters. Only the pre-commit actions which were registered at
960   * the current or a higher level will be executed.
961   *
962   * @return void
963   *
964   * @throws Exceptions\UndoableRunQueueException TBD.
965   */
966  public function executePreCommitActions()
