79  {
80    $id = null;
81    $name = null;
82    if (is_numeric($projectIdentifier)) {

84        $id = $projectIdentifier;
85      }
86    } elseif (is_string($projectIdentifier)) {
87      $name = $projectIdentifier;
88    } elseif (is_array($projectIdentifier)) {

90      $name = $projectIdentifier['name'] ?? null;
91    }
92    if ($id === null && $name === null) {
93      throw new Exceptions\DatabaseMissingIdentifierException(
94        sprintf(self::t('The identifier is missing for a query to find an instance of "%1$s".'), $this->entityName),
95        entityClassName: $this->entityName,
96        incompleteIdentifier: $projectIdentifier,
97      );
98    }
99    $criteria = [];
100    if (!empty($id)) {
101      $criteria[] = ['id' => $id];
102    }
103    if (!empty($name)) {

105    }
