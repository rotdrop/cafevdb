464  /**
465   * Generate a data-item referring to the default value of the given
466   * field. If the field does not have a default return \null.
467   *
468   * @param ProjectParticipantField $field
469   *
470   * @param ProjectParticipant $participant
471   *
472   * @return ?self
473   */
474  public static function fromDefaultValue(ProjectParticipantField $field, ProjectParticipant $participant): ?self
475  {
476    $defaultValue = $field->getDefaultValue();
477    if ($defaultValue === null) {
478      return null;
479    }
480    $fieldContext = $field->getParticipationContext();
481    $participantContext = $participant->getParticipationContext();
482    if ($fieldContext != ParticipationContext::UNRESTRICTED
483        && $participantContext != ParticipationContext::UNRESTRICTED
484        && $fieldContext != $participantContext) {
485      return null;
486    }
487
488    $datum = (new ProjectParticipantFieldDatum)
489      ->setField($field)
490      ->setDataOption($defaultValue)
491      ->setProjectParticipant($participant);
492    switch ($field->getMultiplicity()) {
493      case FieldMultiplicity::RECURRING:
494      case FieldMultiplicity::SIMPLE:
495        // value is stored in the the data item
496        $datum->setOptionValue($defaultValue->getData());
497        $datum->setDeposit($defaultValue->getDeposit());
498        break;
499      default:
500        break;
501    }
502    // Perform some validation and check whether the resulting field value makes sense.
503    $effectiveValue = $datum->getEffectiveValue();
504    switch ($field->getDataType()) {
505      case FieldDataType::BOOLEAN:
506        $valid = null !== filter_var($effectiveValue, FILTER_VALIDATE_BOOLEAN, ['flags' => FILTER_NULL_ON_FAILURE]);
507        break;
508      case FieldDataType::INTEGER:
509        $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_INT);
510        break;
511      case FieldDataType::DB_FILE:
512        // database id must be >= 1
513        $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]]);
514        break;
515      case FieldDataType::DATE:
516        // fallthrough
517      case FieldDataType::DATETIME:
518        try {
519          new DateTimeImmutable($effectiveValue);
520          $valid = true;
521        } catch (Throwable) {
522          $valid = false;
523        }
524        break;
525      case FieldDataType::FLOAT:
526        $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_FLOAT);
527        break;
528      // string types
529      case FieldDataType::CLOUD_FILE:
530      case FieldDataType::CLOUD_FOLDER:
531      case FieldDataType::TEXT:
532      case FieldDataType::HTML:
533        $valid = is_string($effectiveValue);
534        break;
535      case FieldDataType::LIABILITIES:
536      case FieldDataType::RECEIVABLES:
537        $valid = null !== MonetaryNumberType::create($effectiveValue);
538        break;
539      default:
540        // not reached, but a safeguard for future changes
541        return null;
542    }
