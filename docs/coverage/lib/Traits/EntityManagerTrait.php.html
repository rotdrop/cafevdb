131  /**
132   * Removes an entity instance.
133   *
134   * A removed entity will be removed from the database at or before transaction commit
135   * or as a result of the flush operation.
136   *
137   * @param mixed $entity The entity instance to remove.
138   *
139   * @param bool $flush Initiate a flush if true.
140   *
141   * @param bool $hard Issue a double remove: if $entity is managed by
142   * Gedmo\SoftDeleteable then this will "hard" (read: finally) delete
143   * the entity unless it is tagged as "in use" by other means.
144   *
145   * @param bool $soft For entities managed by Gedmo\SoftDeleteable
146   * first check if this entity is already tagged as "soft deleted"
147   * and if so do nothing.
148   *
149   * @param bool $useTransaction Wrap the remove operation into a transaction
150   * or not. This parameter is ignored if $flush is false.
151   *
152   * @return void
153   *
154   * @throws ORMInvalidArgumentException
155   * @throws ORMException
156   * @throws Exceptions\DatabaseEntityNotFoundException
157   */
158  protected function remove(
159    mixed $entity,
160    bool $flush = false,
161    bool $hard = false,
162    bool $soft = false,
163    bool $useTransaction = false,
164  ):void {
165    if (filter_var($entity, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]])) {
166      $entity = [ 'id' => $entity ];
167    }
168    if (is_array($entity)) {
169      $key = $entity;
170      // getReference() will not always work, and in particular not with
171      // foreign keys, so just fetch the entity first.
172      //
173      // $entity = $this->entityManager->getReference($this->entityClassName, $key);
174      $entity = $this->entityManager->find($this->entityClassName, $key);
175      if (empty($entity)) {
176        throw new Exceptions\DatabaseEntityNotFoundException($this->entityClassName);
177      }
178    }
179    if ($soft && !$hard && method_exists($entity, 'isDeleted') && $entity->isDeleted()) {
180      return;
181    }
182    try {
183      if ($useTransaction && $flush) {
184        $this->entityManager->beginTransaction();
185      }
186      if ($hard && method_exists($entity, 'setDeleted')) {
187        $entity->setDeleted('now');
188      }
189      $this->entityManager->remove($entity);
190      if ($hard && !method_exists($entity, 'setDeleted')) {
191        $this->flush();
192        $this->entityManager->remove($entity);
193      }
194      if ($flush) {
195        $this->flush();
196        if ($useTransaction) {
197          $this->entityManager->commit();
198        }
199      }
200    } catch (Throwable $t) {
201      if ($useTransaction) {
202        $this->entityManager->rollback();
203      }
204      throw $t;
205    }
206  }

247   * Flushes all changes to objects that have been queued up to now to the database.
248   * This effectively synchronizes the in-memory state of managed objects with the

252   * flush is called.

254   * @param null|object|array $entity
255   *
256   * @param bool $useTransaction Wrap the flush operation into a transaction or not.
257   *
258   * @return void
259   *
260   * @throws \OCA\CAFEVDB\Wrapped\Doctrine\ORM\OptimisticLockException If a version check on an entity that
261   *         makes use of optimistic locking fails.
262   * @throws ORMException
263   */
264  protected function flush(mixed $entity = null, bool $useTransaction = false):void
265  {
266    if (!empty($entity)) {
267      $this->entityManager->persist($entity);
268    }
269    if ($useTransaction) {
270      $this->entityManager->beginTransaction();
271      try {
272        $this->entityManager->flush();
273        $this->entityManager->commit();
274      } catch (Throwable $t) {
275        if ($this->entityManager->isTransactionActive()) {
276          $this->entityManager->pushTransactionException($t);
277          $this->entityManager->rollback();
278        } else {
279          \OCP\Service::get(\Psr\Log\LoggerInterface::class)->error('Exception during flush()', [ 'exception' => $t ]);
280        }
281      }
282    } else {
283      $this->entityManager->flush();
284    }
285  }
