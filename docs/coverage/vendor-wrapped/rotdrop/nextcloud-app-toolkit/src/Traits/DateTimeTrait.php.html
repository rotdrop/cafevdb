47    /**
48     * Convert the given argument to a DateTimeImmutable object. allowed are
49     * timestamps, other DateTimeInterface instances and strings which are
50     * understood by the constructor or DateTimeImmutable (see PHP manual).
51     *
52     * @param null|string|int|float|DateTimeInterface $dateTime
53     *
54     * @return ?DateTimeImmutable
55     */
56    protected static function convertToDateTime(null|string|int|float|DateTimeInterface $dateTime): ?DateTimeImmutable
57    {
58        if ($dateTime === null || $dateTime === '') {
59            return null;
60        } elseif (!$dateTime instanceof DateTimeInterface) {
61            $timeStamp = filter_var($dateTime, \FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]]);
62            if ($timeStamp === \false) {
63                $timeStamp = filter_var($dateTime, \FILTER_VALIDATE_FLOAT, ['options' => ['min_range' => 1]]);
64            }
65            if ($timeStamp !== \false) {
66                // NOTE: setTimestamp() cannot be used for sub-second resolution.
67                return (new DateTimeImmutable())->modify('@' . $timeStamp);
68            } elseif (is_string($dateTime)) {
69                return new DateTimeImmutable($dateTime);
70            } else {
71                throw new InvalidArgumentException('Cannot convert input to DateTime.');
72            }
73        } elseif ($dateTime instanceof DateTime) {
74            return DateTimeImmutable::createFromMutable($dateTime);
75        } else {
76            return $dateTime;
77        }
78        // Not reached.
79    }
80    /**
81     * Reinterprete the date portion of a \DateTimeInterface object at time 00:00:00 in another time-zone.
82     *
83     * @param DateTimeInterface $date
84     *
