25/**
26 * Helper trait for converting between camelcase and "dashed" variants.
27 */
28trait CamelCaseToDashesTrait
29{
30    /**
31     * Take any dashed or "underscored" lower-case string and convert to
32     * camel-case.
33     *
34     * @param string $string the string to convert.
35     *
36     * @param bool $capitalizeFirstCharacter self explaining.
37     *
38     * @param string $dashes Characters to replace, defaults to '-_',
39     * i.e. dashes and underscores are both considered as separators.
40     *
41     * @return string
42     */
43    protected static function dashesToCamelCase(string $string, bool $capitalizeFirstCharacter = \false, string $dashes = '_-'): string
44    {
45        if ($string === null) {
46            return null;
47        }
48        $str = str_replace(str_split($dashes), '', ucwords($string, $dashes));
49        if (!$capitalizeFirstCharacter) {
50            $str[0] = strtolower($str[0]);
51        }
52        return $str;
53    }
54    /**
55     * Take an camel-case string and convert to lower-case with dashes or
56     * underscores between the words. First letter may or may not be upper
57     * case. Sequences of numbers and punctutation characters (different from
58     * what is passed in the $dashes argument) are considered as words,
59     * e.g. 'Example99!' will be converted to 'Example_99_!'
60     *
61     * @param string $string String to work on.
62     *
63     * @param string $separator Separator to use, defaults to '-'.
64     *
65     * @return string
66     */
67    protected static function camelCaseToDashes(string $string, string $separator = '-'): string
68    {
69        return strtolower(preg_replace('/([A-Z]|[0-9]+|[[:punct:]]+)/', $separator . '$1', lcfirst($string)));
70    }
71}
