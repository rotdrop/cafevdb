155    public function toDecimal(int $scale = -1, int $precision = 0): string
156    {
157        $sign = $this->sign();
158        $abs = $this->abs();
159        if ($scale === -1) {
160            $result = $abs->whole;
161            $fractionalLimit = static::DECIMAL_DIGITS_MAX - max($abs->numerator > 1 ? strlen($abs->numerator) : 0, $result == 0 ? 0 : strlen($result));
162            $abs = $abs->round($fractionalLimit);
163            $abs->subEq($result);
164            $fractionalPart = '';
165            while ($abs->numerator != 0 && strlen($fractionalPart) < $fractionalLimit) {
166                $abs->mulEq(10);
167                $digit = $abs->whole;
168                $abs->subEq($digit);
169                $fractionalPart .= $digit;
170            }
171            $result .= '.' . $fractionalPart;
172            return $sign < 0 ? '-' . $result : $result;
173        }
174        $rollIn = pow(10, $scale + 1);
175        $fixedPoint = str_pad(intdiv($abs->mul($rollIn)->whole + 5, 10), $scale + 1, '0', \STR_PAD_LEFT);
176        $integralPart = substr($fixedPoint, 0, -$scale);
177        $fractionalPart = substr($fixedPoint, -$scale);
178        $result = $integralPart . '.' . $fractionalPart;
179        if ($precision > 0 && strlen(ltrim($integralPart, '0')) + strlen($fractionalPart) > $precision) {
180            $bound = str_pad('', $precision - $scale, '9') . '.' . str_pad('', $scale, '9');
181            throw new OutOfBoundsException('The rational number ' . (string) $this . ' (' . $this->toFloat() . ') does not fit into the range [-' . $bound . ', ' . $bound . '].');
182        }
183        return $sign < 0 ? '-' . $result : $result;
184    }

200    {
201        if (is_int($decimal)) {
202            return new static($decimal, 0, 1);
203        }
204        $matches = [];
205        if (!preg_match_all('/^(-)?([1-9]\d*|0?)(?:\.(\d*))?$/', $decimal, $matches)) {
206            throw new InvalidArgumentException('Unable to parse input string "' . $decimal . '".');
207        }
208        $sign = $matches[1][0] == '-' ? -1 : 1;
209        $integralPart = empty($matches[2]) ? 0 : (int) $matches[2][0];
210        $numerator = empty($matches[3]) ? 0 : (int) $matches[3][0];
211        $denominator = pow(10, strlen($matches[3][0] ?? ''));
212        return new static($sign * $integralPart, $sign * $numerator, $denominator);
213    }

218     * self::jsonSerialize() back to a number.

224    public static function fromFractionString(string $input): static
225    {
226        $originalInput = $input;
227        $input = trim($input);
228        $negate = \false;
229        if (str_starts_with($input, '-')) {
230            $negate = \true;
231            $input = substr($input, 1);
232        }
233        $isInteger = filter_var($input, \FILTER_VALIDATE_INT, ['options' => ['min_range' => 0]]);
234        if (!$isInteger && !str_contains($input, '/')) {
235            throw new InvalidArgumentException("The given string \"{$originalInput}\" does not appear to be a valid fraction.");
236        }
237        // replace super- and subscripts back to ordinary numbers.
238        $from = array_merge(self::SUPER_SCRIPTS, self::SUB_SCRIPTS);
239        $from[] = ' ';
240        $to = array_merge(array_keys(self::SUPER_SCRIPTS), array_keys(self::SUB_SCRIPTS));
241        $to[] = '+';
242        $input = str_replace($from, $to, $input);
243        // now we have at any rate the jsonSerialize() representation.
244        $parts = array_filter(explode('+', $input));
245        $numParts = count($parts);
246        switch ($numParts) {
247            case 2:
248                [$whole, $fraction] = $parts;
249                break;
250            case 1:
251                if (str_contains($parts[0], '/')) {
252                    $whole = 0;
253                    $fraction = $parts[0];
254                } else {
255                    $whole = $parts[0];
256                    $fraction = '0/1';
257                }
258                break;
259            default:
260            case 0:
261                throw new InvalidArgumentException("The given string \"{$originalInput}\" does not appear to be a valid fraction.");
262        }
263        [$numerator, $denominator] = explode('/', $fraction);
264        $number = new static((int) $whole, (int) $numerator, (int) $denominator);
265        return $negate ? $number->negate() : $number;
266    }
