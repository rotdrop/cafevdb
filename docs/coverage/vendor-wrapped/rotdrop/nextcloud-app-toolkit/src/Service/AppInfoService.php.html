28/**
29 * Access info from the appinfo/info.xml.
30 */
31class AppInfoService
32{
33    private static string $appInfoPath;

37    /**
38     * Determine the path to the app's info.xml file assuming that the app PHP
39     * files start at lib/ from the app directory.
40     *
41     * @return string
42     */
43    public static function getAppInfoPath(): string
44    {
45        if (self::$appInfoPath ?? null) {
46            return self::$appInfoPath;
47        }
48        self::$appInfoPath = self::getAppFolderPath() . Constants::PATH_SEP . Constants::INFO_FILE;
49        return self::$appInfoPath;
50    }
51    /**
52     * Determine the path of the application folder.
53     *
54     * @return string
55     */
56    public static function getAppFolderPath(): string
57    {
58        if (self::$appFolderPath ?? null) {
59            return self::$appFolderPath;
60        }
61        // When installed as composer package (possibly namespace-wrapped) the
62        // nesting of the class name does not reflect the position inside the app
63        // folder, so first look for the info file in the parent directories.
64        for ($dir = __DIR__, $parent = dirname($dir); $parent !== $dir; $dir = $parent, $parent = dirname($dir)) {
65            if (is_file($dir . Constants::PATH_SEP . Constants::INFO_FILE)) {
66                self::$appFolderPath = $dir;
67                return self::$appFolderPath;
68            }
69        }
70        // Fallback: extract the directory nesting level from the class-name, so this counts
71        // the part after OCA\APP_NAME_SPACE. Thus OCA\APP_NAME_SPACE\AppInfo\Application.php
72        // yields a nesting-level of 2 and yes, the info file is
73        // lib/AppInfo/../../appinfo/info.xml
74        $nestingLevel = count(explode('\\', __CLASS__)) - 2;
75        $pathPrefix = str_repeat(Constants::PATH_SEP . '..', $nestingLevel);
76        self::$appFolderPath = __DIR__ . $pathPrefix;
77        return self::$appFolderPath;
78    }
79    /**
80     * @return null|string The app-name from the info.xml file or null if that
81     * cannot be found.
82     */
83    public static function getAppInfoAppName(): ?string
84    {
85        $infoXml = self::getAppInfoXML();
86        return !empty($infoXml->id) ? (string) $infoXml->id : null;
87    }

89     * @return SimpleXMLElement The parse app-info file.
90     */

92    {
93        if (self::$appInfoXML ?? null) {
94            return self::$appInfoXML;

96        // we do not try-catch here as this file MUST be there and parseable.
97        self::$appInfoXML = new SimpleXMLElement(file_get_contents(self::getAppInfoPath()));
98        return self::$appInfoXML;
99    }
100    /**
101     * Ask IAppManager about the full app-info file.
102     *

104     */
105    public static function getAppInfo(): array

112        self::$appInfo = $appManager->getAppInfoByPath(self::getAppInfoPath());
113        return self::$appInfo;
114    }
