124    /**
125     * Add some "missing" convenience stuff to findBy()
126     *
127     * - allow filtering by association fields. Association-fields are
128     *   simply accessed by dot-notation, e.g. 'musician.name' where
129     *   'musician' in this example is the name of the property in the
130     *   master entity.
131     * - allow sorting by association fields
132     * - allow indexing by adding an EnumOrderByOptions::INDEX option to $orderBy.
133     * - allow wild-cards, uses "LIKE" or "REGEXP" in comparison if specified by
134     *   query-options (see below $criteria). Wildcards unescaped '%' and '_' as
135     *   in vanilla LIKE. When REGEXP is used then '%' maps to '.*' and '_' maps
136     *   to '.?'.
137     * - allow the basic comparators =, <, >, ! in prefix notation,
138     *   e.g. ```!>FIELD => SOMETHING``` will be translated to the
139     *   expression ```!(FIELD > SOMETHING)```
140     * - in particulular ```'!FIELD' => SOMETHING``` will just negate
141     *   the criterion
142     * - logical junctors & and | and parenthesis are also supported: criteria can be
143     *   prefixed by '(|' and '(&' to indicate the boolean junctor with the other
144     *   criteria. Grouping is required by using '(' and ')'. The final closing
145     *   paren may be omitted.
146     * - criteria can be passed as FIELD_EXPRESSION => VALUE or if this would
147     *   yield non-unique array keys also as pairs [ FIELD_EXPRESSION => VALUE ]
148     *
149     *   Example:
150     *   ```
151     *   [
152     *     __OPTIONS__ => [ 'LIKE' => false ], // default
153     *     ... STUFF
154     *     '!fieldId' => 13,       // search for field_id != 13
155     *     '(|optionValue' => ''   // and (option_value the empty string ...
156     *       'optionValue' => null //      or option_value is NULL ...
157     *       '(&blah' => 2',       //      or (blah is 2 ...
158     *        'blub' => 3,         //          and blub is 3 ...
159     *       ')(&foo' => 5         //         ) or (foo is 5 ...
160     *          'bar' => 6         //               and bar is 6
161     *       ')' => ANYTHING       //              ) close the AND-group
162     *                             // implicitly close the first or group
163     *   ],
164     *   ```
165     *
166     * - supports Collections\Criteria, these are applied at the end.
167     *
168     * - support group-functions in order to collect grouped data, e.g.
169     *   ```
170     *   [
171     *     __OPTIONS__ => [ 'LIKE' => true ],
172     *     [ foo.bar@GROUP_CONCAT(%s) => '%SEARCH%' ],
173     *   ]
174     *   ```
175     *   Such search criteria will end up in the having clause.
176     *
177     * - support ordinary-functions in the where part with the syntax
178     *   ```
179     *   [
180     *     __OPTIONS__ => [ 'LIKE' => true ],
181     *     [ foo.bar#BIN_TO_UUID(%s) => '%SEARCH%' ],
182     *   ]
183     *   ```
184     *   The difference to the '@' syntax is that these criteria will end up in
185     *   the WHERE clause.
186     *
187     * - support explicit type specification for "complicated" cases where a
188     *   mere string conversion would leads to wrong results, e.g.
189     *   ```[ foo.bar:uuid_binary => Uuid::NIL ]```
190     *   Automatic type-deduction is used only for fields of the main-table.
191     *
192     * In order to filter by empty collections a left-join with the
193     * main-table is performed.
194     *
195     * @param array $criteria Search criteria.
196     *
197     * @param null|array $orderBy Order-by criteria. In addition to ASC and DESC
198     * an ordering of INDEX is also supported. INDEX will add a corresponding
199     * "INDEX BY" directive to the generated DQL query which will generate a
200     * result array indexed by the specified column.

366                            }

368                                throw new DatabaseException('Comparison with null is undefined, only equality is allowed.');
369                            }
370                            if (is_array($value)) {
371                                throw new DatabaseException('Array-valued comparisons are not allowed.');
372                            }
373                            $operators = substr($operators, strlen($abbr));
374                            $criterion['comparator'] = $comparator;
375                        }
376                        if (empty($operators)) {
377                            break 2;
378                        }
379                    }
380                }
