157  /**
158   * Pre-process the given $key. If a key starts with a known prefix, but has
159   * a '-' as separator, then replace all dashes by our standard separator.
160   *
161   * @param string $key The key to process.
162   *
163   * @return string
164   */
165  private function preProcessKey(string $key):string
166  {
167    foreach (self::SUBKEY_PREFIXES as $prefix) {
168      if (str_starts_with($key, $prefix . '-')) {
169        $key = str_replace(['--', '+-'], ['-minus-', '-plus-'], $key);
170        $lastChar = substr($key, -1);
171        switch ($lastChar) {
172          case '-':
173            $key = substr($key, 0, -1) . '-minus';
174            break;
175          case '+':
176            $key = substr($key, 0, -1) . '-plus';
177            break;
178        }
179        return str_replace('-', self::SUB_KEY_SEP, $key);
180      }
181    }
182    return $key;
183  }

185  /**
186   * Return a translated tool-tip for the given key.
187   *
188   * @param string $key The tool-tip key to look up.
189   *
190   * @param bool $escape Escape HTML entities.
191   *
192   * @return null|string
193   */
194  public function fetch(string $key, bool $escape = true):?string
195  {
196    if ($key == $this->lastKey && $escape) {
197      return $this->lastToolTip;
198    }
199    $this->lastKey = $key;
200
201    $toolTipsData = ToolTipsDataService::get();
202
203    $key = $this->preprocessKey($key);
204
205    $keys = explode(self::SUB_KEY_SEP, $key);
206    while (\count($keys) > 0) {
207      $key = array_shift($keys);
208      $toolTipsData = $toolTipsData[$key] ?? ($toolTipsData[self::DEFAULT_KEY] ?? null);
209    }
210    $tip = $toolTipsData[self::DEFAULT_KEY] ?? $toolTipsData;
211
212    if (!isset($tip['text'])) {
213      $tip = null;
214    }
215
216    $options = $tip['options'] ?? self::DEFAULT_OPTIONS;
217
218    if (empty($tip)) {
219      $this->failedKeys[] = $this->lastKey;
220      if ($this->debug) {
221        $tip = $this->l->t('Unknown Tooltip for key "%s" requested.', $this->lastKey);
222      }
223    } else {
224      $this->failedKeys = [];
225      $parameters = array_map(
226        fn(mixed $value) => $value instanceof Closure ? $value($this->l, $this->appContainer) : $value,
227        $tip['parameters'] ?? [],
228      );
229      $tip = $this->l->t($tip['text'], $parameters);
230      if ($this->debug) {
231        $tip .= ' (' . $this->l->t('ToolTip-Key "%s"', $this->lastKey) . ')';
232      }
233    }
234
235    if (!empty($tip)) {
236      if ($escape && !$options[self::OPTION_HTML]) {
237        // all tooltips are rendered as HTML, this implies that non-HTML
238        // tips have to be encoded twice.
239        $tip = htmlspecialchars($tip);
240      }
241      // multiple lines consisting only of spaces and linebreaks are replaced by paragraphs
242      $tip = preg_replace('/(^\s*[\n])+/m', self::PARAGRAPH . "\n", $tip);
243      if ($escape) {
244        $tip = htmlspecialchars($tip);
245      }
246      $this->lastToolTip = $tip;
247    } else {
248      $this->lastToolTip = null;
249    }
250
251    return $this->lastToolTip;
252  }
