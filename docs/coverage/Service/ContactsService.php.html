180    $musicianCardBackend = $this->appContainer->get(MusicianCardBackend::class);

182    $result = [];
183    $addressBooks = $this->contactsManager->getUserAddressBooks();
184    /** @var IAddressBook $addressBook */
185    foreach ($addressBooks as $addressBook) {
186      if ($addressBook->getKey() == $musiciansKey) {
187        continue;
188      }
189      $bookName = $addressBook->getDisplayName();

196          continue;
197        }
198        if (!is_array($emails)) {
199          $emails = [ $emails ];
200        }

203          if (empty($email)) {

212        }

216          return strcmp($aName, $bName);
217        });

220    }

286        if ($withTypes) {

289            'value' => $property->getValue()

293        }
