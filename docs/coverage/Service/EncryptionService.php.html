395    if (!$this->verifyHash($userDatabaseKey, $sysDatabaseKeyHash)) {
396      $failCount = 0;

398      for ($i = 0; $i < $numberOfTests; ++$i) {

400      }
401      if ($failCount < 10) {

403        $this->logError(

526  public function getConfigValue(string $key, mixed $default = null, bool $ignoreLock = false)
527  {
528    if (!$ignoreLock && !empty($this->getAppValue(ConfigConstants::CONFIG_LOCK_KEY))) {
529      throw new Exceptions\ConfigLockedException('Configuration locked, not retrieving value for ' . $key);
530    }
531    $value = $this->getAppValue($key, $default);
532
533    if (in_array($key, self::NEVER_ENCRYPT) && $this->appCryptor->isEncrypted($value) === false) {
534      // short-cut, no need to bail out or complain. The "not enecrypted"
535      // check is necessary for the case that items are added to the
536      // NEVER_ENCRYPT list.
537      return $value;
538    }
539
540    if (!empty($value) && ($value !== $default)) {
541      // null is error or uninitialized, string '' means no encryption
542      if ($this->appCryptor->getEncryptionKey() === null) {
543        if ($this->appCryptor->isEncrypted($value) === false) {
544          return $value;
545        }
546        if (!empty($this->userId)) {
547          $message = $this->l->t('Decryption requested for user "%s", but not configured, empty encryption key.', $this->userId);

549          // $this->logError($message);
550        }
551        return false;
552      }

555      } catch (Throwable $t) {
556        throw new Exceptions\DecryptionFailedException($this->l->t('Unable to decrypt value "%s" for "%s"', [$value, $key]), $t->getCode(), $t);
557      }
558      //$this->logInfo("Decrypted value for $key: ".$value);
559    }
560
561    return $value;
