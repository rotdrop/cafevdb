38  protected ContainerInterface $appContainer;

42  /**
43   * Remove "forbidden" characters as configured in order to achieve a
44   * filename which is also valid on certain strange operating systems.
45   *
46   * @param string $name
47   *
48   * @param null|string $mimeType
49   *
50   * @return string
51   *
52   * @throws Exceptions\EnduserNotificationException
53   */
54  protected function sanitizeFilename(string $name, ?string $mimeType = null): string
55  {
56    if ($name === '') {
57      return $name; // root folder
58    }
59    $oldName = $name;
60    try {
61      /** @var FilenameValidator $filenameValidator */
62      $filenameValidator = $this->appContainer->get(FilenameValidator::class);
63
64      $forbiddenCharacters = $filenameValidator->getForbiddenCharacters();
65      $charReplacement = array_diff(['_', ' ', '-'], $forbiddenCharacters);
66      $charReplacement = reset($charReplacement) ?: '';
67
68      $forbiddenExtensions = $filenameValidator->getForbiddenExtensions();
69      foreach ($forbiddenExtensions as $extension) {
70        if (str_ends_with($name, $extension)) {
71          $name = substr($name, 0, strlen($name) - strlen($extension));
72          $extension = $this->fileExtensionFromMimeType($mimeType);
73          if ($extension && !in_array($extension, $forbiddenExtensions)) {
74            $name .= '.' . $extension;
75          }
76        }
77      }
78
79      $basename = substr($name, 0, strpos($name, '.', 1) ?: null);
80      if (in_array($basename, $filenameValidator->getForbiddenBasenames())) {
81        $name = str_replace($basename, $this->l->t('%1$s (renamed)', [$basename]), $name);
82      }
83
84      if ($name === '') {
85        $name = $this->l->t('renamed file');
86      }
87
88      $forbiddenCharacter = $filenameValidator->getForbiddenCharacters();
89      $name = str_replace($forbiddenCharacter, $charReplacement, $name);
90    } catch (Throwable $t) {
91      throw new Exceptions\EnduserNotificationException(
92        $this->l->t('Unable to sanitize filename "%s".', $oldName),
93        previous: $t,
94      );
95    }
96    return $name;
97  }

100    'image/png' => 'png',
101    'image/jpeg' => 'jpg',
102    'image/gif' => 'gif',
103    'image/vnd.microsoft.icon' => 'ico',

108  /**
109   * @param null|string $mimeType
110   *
111   * @return null|string
112   */
113  protected function fileExtensionFromMimeType(?string $mimeType):?string
114  {
115    if (!$mimeType) {
116      return null;
117    }
118    if (!empty(self::FILE_EXTENSIONS_BY_MIME_TYPE[$mimeType])) {
119      return self::FILE_EXTENSIONS_BY_MIME_TYPE[$mimeType];
120    }
121    // as a wild guess we return anything after the slash if it is at
122    // most 4 characters.
123    list(/* $first*/, $second) = explode('/', $mimeType);
124    if (strlen($second) <= 4) {
125      return $second;
126    }
127    return null;
128  }
