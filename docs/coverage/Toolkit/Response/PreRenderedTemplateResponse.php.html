27/**
28 * A template response which optionally pre-renders its content.
29 *
30 * This can be used to catch ecxeptions thrown during render. The stock NC
31 * TemplateResponse::render() method is called outside the middle-ware loop
32 * and directly triggers the exceptions handler which then renders the core
33 * exception template.
34 */
35class PreRenderedTemplateResponse extends TemplateResponse
36{
37  protected ?string $content = null;

49
50  /**
51   * Call parent::render() and cache its output.
52   *
53   * @return string
54   */
55  public function preRender()
56  {
