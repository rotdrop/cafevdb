81   * Register the needed extension to MIME-type mappings with the Nextcloud server.
82   *
83   * @return void
84   *

86   */
87  public function registerMimeTypeMappings():void
88  {
89    if (!($this->mimeTypeDetector instanceof MimeTypeDetector)) {
90      return;
91    }
92
93    $missingMimeMappings = $this->getMissingMimeTypeMappings();

97    $this->allMimeTypeMappings = null;

120  /**
121   * @return array<string, string> An array EXT => MIME of the supported
122   * archive MIME-types. That is, the cloud must know the mime type and the
123   * archive backend must support it in order to have an extension and
124   * mime-type added to the list.
125   */
126  public function getSupportedArchiveMimeTypes():array
127  {
128    if ($this->supportedMimeTypes !== null) {
129      return $this->supportedMimeTypes;
130    }
131    $this->registerMimeTypeMappings();
132    $mimeTypeMapping = $this->getAllMimeTypeMappings();
133    $supportedFormats = ArchiveFormats::getSupportedDriverFormats();
134    $this->supportedMimeTypes = [];
135    $unsupportedMimeTypes = [];
136    foreach ($mimeTypeMapping as $extension => $mimeTypes) {
137      if (count($mimeTypes) == 0) {
138        $this->logError('Buggy config file, no mime-types for extension ' . $extension);
139      } elseif (count($mimeTypes) > 1) {
140        $this->logDebug('More than one mime-type for extension "' . $extension . '": ' . print_r($mimeTypes, true));
141      }
142      $mimeType = $mimeTypes[0];
143      if ($mimeType == 'application/x-gtar') {
144        $this->logInfo('MIME TYPE ' . $mimeType);
145      }
146      $format = ArchiveFormats::detectArchiveFormat('FOO.' . $extension, contentCheck: false);
147      if (!empty($supportedFormats[$format])) {
148        $this->supportedMimeTypes[$extension] = $mimeType;
149      } else {
150        $unsupportedMimeTypes[$extension] = $mimeType;
151      }
152    }
153    // $this->logInfo('SUPPORTED MIME TYPES ' . print_r($this->supportedMimeTypes, true));
154    // $this->logInfo('UNSUPPORTED MIME TYPES ' . print_r($unsupportedMimeTypes, true));
155
156    return $this->supportedMimeTypes;
157  }

174  /** @return array Slurp in and cache the extension to mime-type mapping. */
175  private function getAppMimeTypeMappings():array
176  {
177    if ($this->appMimeTypeMappings !== null) {
178      return $this->appMimeTypeMappings;
179    }
180
181    $baseDirs = [ $this->appPath, __DIR__ . '/../../' ];
182    foreach ($baseDirs as $prefixDir) {
183      $dataFile = $prefixDir . self::MIME_TYPE_MAPPING_DATA_FILE;
184      $jsonData = file_get_contents($dataFile);
185      if (empty($jsonData)) {
186        $this->logInfo('Unable to read "' . $dataFile . '".');
187        continue;
188      }
189      try {
190        $arrayData = json_decode($jsonData, true);
191        break;
192      } catch (Throwable $t) {
193        $this->logException($t, 'Unable to decode mime-type mapping. "' . self::MIME_TYPE_MAPPING_DATA_FILE . '".');
194        $arrayData = null;
195        continue;
196      }
197    }
198
199    if (empty($arrayData)) {
200      $this->appMimeTypeMappings = [];
201      return $this->appMimeTypeMappings;
202    }
203
204    $this->appMimeTypeMappings = array_filter($arrayData, fn(string $key) => !str_starts_with($key, '_'), ARRAY_FILTER_USE_KEY);
205    // $this->logInfo('MIME MAPPINGS ' . print_r($this->appMimeTypeMappings, true));
206
207    return $this->appMimeTypeMappings;
208  }
