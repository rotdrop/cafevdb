232    $musician = $this->getMusician();
233    if ($musician) {
234      $indexByValue = $instrument?->getId();
235      $musicianInstruments = $musician->getInstruments();
236      if (!$musicianInstruments->contains($this)) {
237        if ($indexByValue) {
238          $musicianInstruments->set($indexByValue, $this);
239        } else {
240          $musicianInstruments->add($this);
241        }
242      } elseif ($indexByValue && !$musicianInstruments->get($indexByValue)) {
243        $musicianInstruments->removeElement($this);
244        $musicianInstruments->set($indexByValue, $this);
245      }
246    }
247    if ($instrument) {
248      $indexByValue = $musician?->getId();
249      $musicianInstruments = $instrument->getMusicianInstruments();
250      if (!$musicianInstruments->contains($this)) {

252          $musicianInstruments->set($indexByValue, $this);
253        } else {

257        $musicianInstruments->removeElement($this);
