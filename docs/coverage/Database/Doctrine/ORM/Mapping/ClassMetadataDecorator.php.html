260  /**
261   * Update the inverse side of an association after auto-installing
262   * $targetEntity into $entity (the owning side). This is only done if it
263   * does not produce reloads from the data-base, so only "real" entities,
264   * initialized proxies and initialized lazy collections are updated.
265   *
266   * @param object $entity The current $entity which must be the owning side
267   * of the association.
268   *
269   * @param AssociationMapping $association The association mapping of the current field
270   * (the field itself is not needed here).
271   *
272   * @param object $targetEntity The target-entity which must be the inverse
273   * side of the association.
274   *
275   * @param ClassMetadata|ClassMetadataDecorator $targetMeta The class meta-data of the target-entity.
276   *
277   * @return void
278   */
279  private function updateInverseSide(
280    object $entity,
281    AssociationMapping $association,
282    object $targetEntity,
283    ClassMetadataInterface $targetMeta,
284  ):void {
285    if (!empty($association['mappedBy'])) {
286      $this->debug('WE ARE THE INVERSE SIDE ' . print_r($association, true));
287    }
288    // try to maintain connectivity if it does not cause additional direct database access
289    if (!empty($association['inversedBy'])
290        && (!($targetEntity instanceof Proxy) || $targetEntity->__isInitialized())) {
291      $inversedBy = $association['inversedBy'];
292      $associationType = $association['type'];
293      switch ($associationType) {
294        case ClassMetadata::ONE_TO_ONE:
295          $targetMeta->setFieldValue($targetEntity, $inversedBy, $entity);
296          break;
297        case ClassMetadata::MANY_TO_ONE:
298          $inversedByValue = $targetMeta->getFieldValue($targetEntity, $inversedBy);
299          if (!($inversedByValue instanceof Collection)
300              || ($inversedByValue instanceof PersistentCollection) && !$inversedByValue->isInitialized()) {
301            // skip if the collection would have to be fetches or if
302            // it is not there already (the latter should not happen).
303            break;
304          }
305          $targetAssociation = $targetMeta->associationMappings[$inversedBy];
306          $indexBy = $targetAssociation['indexBy'] ?? null;
307          // $orderBy = $targetAssociation['orderBy']; complicated
308          if (!empty($indexBy)) {
309            // $indexByValue = (string)$this->metaData->getFieldValue($entity, $indexBy);
310            $indexByValue = (string)$this->getColumnValue($entity, $indexBy);
311            if ($inversedByValue->contains($entity)) {
312              $inversedByValue->removeElement($entity);
313            }
314            $inversedByValue->set($indexByValue, $entity);
315          } elseif (!$inversedByValue->contains($entity)) {
316            $inversedByValue->add($entity);
317          }
318          break;
319      }
320    }
321  }

323  /**
324   * The related MetaData::getIdentifierValues() function does not
325   * handle recursion into associations. Extract the column values of
326   * primary foreign keys by recursing into the meta-data.
327   *
328   * @param mixed $entity The entity to extract the values from.
329   *
330   * @return array
331   * ```
332   * [ COLUMN1 => VALUE1, ... ]
333   * ```
334   * The array is indexed by the database column-names.
335   *
336   * @note As a side-effect, $entity is modified if a given foreign
337   * key is just a simple identifier value (like an int) and not an
338   * entity instance or reference.
339   */
340  public function getIdentifierColumnValues(mixed $entity)
341  {
342    $columnValues = [];
343    foreach ($this->metaData->getIdentifierValues($entity) as $field => $value) {
344      if (isset($this->metaData->associationMappings[$field])) {
345        $association = $this->metaData->associationMappings[$field];
346        $targetEntity = $association->targetEntity;
347        /** @var ClassMetadata $targetMeta */
348        $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
349        if (count($association->joinColumns) != 1) {
350          throw new Exceptions\DatabaseException($this->l->t('Foreign keys as principle keys cannot be composite'));
351        }
352        $joinInfo = $association->joinColumns[0];
353        $columnName = $joinInfo->name;
354        $targetColumn = $joinInfo->referencedColumnName;
355        $targetField = $targetMeta->fieldNames[$targetColumn];
356        if ($value instanceof $targetEntity) {
357          $columnValues[$columnName] = $targetMeta->getFieldValue($value, $targetField);
358          // try to maintain connectivity if it does not cause additional direct database access
359          $this->updateInverseSide($entity, $association, $value, $targetMeta);
360        } else {
361          // avoid references with empty identifiers
362          if (empty($value)) {
363            $value = null;
364            $reference = null;
365          } else {
366            // replace the value by a reference
367            $reference = $this->entityManager->getReference($targetEntity, [ $targetField => $value ]);
368
369            // try to maintain connectivity if it does not cause additional direct database access
370            $this->updateInverseSide($entity, $association, $reference, $targetMeta);
371          }
372          $this->doSetFieldValue($this->metaData, $entity, $field, $reference);
373          // assume this is the column value, not the entity of the foreign key
374          $columnValues[$columnName] = $value;
375        }
376      } else {
377        $columnName = $this->metaData->fieldMappings[$field]->columnName;
378        $columnValues[$columnName] = $value;
379      }
380    }
381    return $columnValues;
382  }

384  /**
385   * Generate ids for use with find and self::persist() from database
386   * column values. $columnValues is allowed to contain excess data
387   * which comes in handy when recursing into associations.
388   *
389   * @param array $columnValues The actual identifier values indexed by
390   * the database column names (read: not the entity-class-names, but
391   * the raw column names in the database).
392   *
393   * @param bool $ignoreMissing Ignore missing key values, just continue and
394   * return an incomplete set of keys.
395   *
396   * @return array
397   * ```
398   * [
399   *   PROPERTY => ELEMENTARY_FIELD_VALUE,
400   *   ...
401   * ]
402   * ```
403   *
404   * @todo $columnValues sometimes contains raw data-base values as it
405   * is passed down here from code using the legacy phpMyEdit
406   * stuff. ATM we hack around by converting string values to their
407   * proper PHP values, but this is an ugly hack.
408   */
409  public function extractKeyValues(array $columnValues, bool $ignoreMissing = false):array
410  {
411    $entityId = [];
412    foreach ($this->metaData->identifier as $field) {
413      $dbalType = null;
414      if (isset($this->metaData->associationMappings[$field])) {
415        $association = $this->metaData->associationMappings[$field];
416        $targetEntity = $association->targetEntity;
417        /** @var ClassMetadata $targetMeta */
418        $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
419        if (count($association->joinColumns) != 1) {
420          throw new Exceptions\DatabaseException($this->l->t('Foreign keys as principle keys cannot be composite'));
421        }
422        $joinInfo = $association->joinColumns[0];
423        $columnName = $joinInfo->name;
424        $value = $columnValues[$columnName];
425        if (!($value instanceof $targetEntity)) {
426          $targetColumn = $joinInfo->referencedColumnName;
427          $targetField = $targetMeta->fieldNames[$targetColumn];
428          $value = $this->entityManager->getReference($association->targetEntity, [ $targetField => $value ]);
429        }
430      } else {
431        $columnName = $this->metaData->fieldMappings[$field]->columnName;
432        if (!isset($columnValues[$columnName])) {
433          // possibly an attempt to extract from non-existing field.
434          if ($ignoreMissing || $this->metaData->usesIdGenerator()) {
435            continue;
436          }
437          throw new Exceptions\DatabaseException(
438            $this->l->t('Missing value and no generator for identifier field: %s::%s', [ $this->getName(), $field ]));
439        }
440        $dbalType = Type::getType($this->metaData->fieldMappings[$field]->type);
441        $value = $columnValues[$columnName];
442      }
443      if (!empty($dbalType) && is_string($value)) {
444        $value = $dbalType->convertToPHPValue($value, $this->entityManager->getPlatform());
445      }
446      $entityId[$field] = $value;
447    }
448    return $entityId;
449  }

451  /**

453   * introduced through associations. The mapped field is then the
454   * first found association field which references the column.

457   */
458  private function columnAssociations():array
459  {
460    if (empty($this->columnAssociationsCache)) {
461      $meta = $this->metaData;
462      $fields = [];

464        foreach (($mapping['joinColumns'] ?? []) as $joinColumn) {
465          $fields[$joinColumn->name][$joinColumn->referencedColumnName] = $field;
466        }
467      }

469    }

523  {

525
526    $meta = $this->metaData;
527    $numSetters = 0;
528    if (isset($meta->fieldNames[$column])) {
529      // simple case, just set the entity property
530      $this->doSetFieldValue($meta, $entity, $meta->fieldNames[$column], $value);

532    }
533    $columnAssociations = $this->columnAssociations();
534    foreach (($columnAssociations[$column]??[]) as $field) {
535      // If the column is a join-column then try to create references to the
536      // respective entity. Multi-column associations are supported if the
537      // column is the only one which is not known.
538      if (empty($value)) {
539        // an empty join-column value must void the association as the
540        // join-columns are the identifier of the referenced entity and hence
541        // cannot be null unless the referenced entity is null
542        $this->debug('DO CLEAR FIELD VALUE FOR ' . $field);
543        $this->doSetFieldValue($meta, $entity, $field, null);
544        continue;
545      }
546      $associationMapping = $meta->associationMappings[$field];
547      $targetEntity = $associationMapping['targetEntity'];
548      $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
549      $referencedColumnId = [];
550      foreach ($associationMapping['joinColumns'] as $joinInfo) {

552        $targetColumn = $joinInfo->referencedColumnName;

555          if (!empty($fieldValue) &&
556              $value == $targetMeta->getColumnValue($fieldValue, $targetColumn)) {
557            // do not destroy already installed associated entities
558            continue 2;

560          $targetValue = $value;
561        } else {
562          $targetValue = $this->getColumnValue($entity, $joinColumnName);
563        }
564        if (empty($targetValue)) {
565          continue 2;
566        }
567        $referencedColumnId[$targetColumn] = $targetValue;
568      }
569      $referencedId = $targetMeta->extractKeyValues($referencedColumnId);
570      $fieldValue = $this->entityManager->getReference($targetEntity, $referencedId);
571      $this->doSetFieldValue($meta, $entity, $field, $fieldValue);
572      $this->updateInverseSide($entity, $associationMapping, $fieldValue, $targetMeta);
573      ++$numSetters;

598      return $meta->getFieldValue($entity, $meta->fieldNames[$column]);

601    $columnAssociations = $this->columnAssociations();

617              'Association field must eiter be an entity or the value of the single join column: %s',

619            ),
620          );
621        }
622        return $fieldValue;
623      }
624    }
625    if (!empty($this->temporaryColumnStorage[$column])) {
