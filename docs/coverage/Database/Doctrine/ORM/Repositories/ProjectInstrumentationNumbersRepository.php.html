53   * Fetch the instrumentation balance for the given project-id, that

55   * instruments vs. pro-forma registered instrument vs. confirmed

60   *

62   * ```

69   *   ],
70   *   ...

74  public function fetchInstrumentationBalance(int $projectId):array
75  {
76    $qb = $this->createQueryBuilder('pin');
77    $qb->select([
78      'i.name AS instrument',
79      'pin.voice AS voice',
80      'pin.quantity AS required',
81      'count(pi.instrument) AS registered',
82      'count(pp.registration) AS confirmed' ])
83      ->leftJoin('pin.instrument', 'i')
84      ->leftJoin('pin.projectInstruments', 'pi')
85      ->leftJoin(
86        Entities\ProjectParticipant::class, 'pp',
87        Query\Expr\Join::WITH,

89        .' AND '
90                  .'pi.musician = pp.musician'
91                  .' AND '

93       ->where($qb->expr()->eq('identity(pin.project)', ':projectId'))
94       ->setParameter('projectId', $projectId)
95       ->addGroupBy('pin.project')
96       ->addGroupBy('pin.instrument')
97       ->addGroupBy('pin.voice')
