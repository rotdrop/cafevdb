103 $historyPostData = [];
104
105 $queryHash = $this->request->getParam('hash');
106 if ($queryHash) {
107 $initialPostData = $this->historyService->get($queryHash);
108 $this->logInfo('HASH VALUE ' . $queryHash . ' DATA ' . print_r($initialPostData ?? [], true));
109 if (!empty($initialPostData)) {
110 $historyPostData = CommonUtil::arrayMergeRecursive(

121 $lastUrlPath = $this->historyService->getLastUrlPath();
122 if (!empty($lastUrlPath)) {
123 $queryData = [];
124 parse_str(parse_url($lastUrlPath, PHP_URL_QUERY), $queryData);
125 $hash = $queryData['hash'] ?? null;
126 if ($hash) {
127 $postData = $this->historyService->get($hash);
128 if (!empty($postData)) {
129 $historyPostData = CommonUtil::arrayMergeRecursive(
130 $historyPostData, [
131 'post' => [
132 $hash => $postData,
133 ],
134 'lastUrlPath' => $lastUrlPath,
135 'lastUrlHash' => $hash,
136 ]
137 );
138 }
139 }
140 }
141
142 if (!empty($historyPostData)) {
143 $this->initialStateService->provideInitialState($this->appName, 'historyPostData', $historyPostData);
144 }

163 public function navigation(string $template, ?int $projectId = null, ?string $projectName = null)
164 {
165 $template = urldecode($template);
166 if ($template == 'home') {
167 $navigationItems = [
168 PageRenderer\Projects::navigationItem(),
169 PageRenderer\AllMusicians::navigationItem(),
170 PageRenderer\Blog::navigationItem(),
171 ];
172 } else {
173 try {
175 $renderer = $this->appContainer->query(PageRenderer\Registration::TEMPLATE_PREFIX . $template);
176 } catch (Throwable $t) {
177 throw new Exceptions\EnduserNotificationException(
178 $this->l->t('Unable to generate the navigation menu for page "%s".', $template),
179 previous: $t,
180 );
181 }
182 $navigationItems = $renderer->navigationItems();
183 }
184 $userPermissions = $this->authorizationService->getUserPermissions();
185 $navigationItems = array_filter($navigationItems, fn($item) => ($item['permissions'] === ($item['permissions'] & $userPermissions)));
186 foreach ($navigationItems as &$item) {
187 $item['nameKey'] = $item['name'];
188 $item['name'] = $this->toolTipsService[$item['name']] ?: $item['name'];
189 $item['tooltipKey'] = $item['tooltip'];
190 $item['tooltip'] = $this->toolTipsService[$item['tooltip']] ?: $item['tooltip'];
191 }
192 return self::dataResponse([
193 'navigation' => $navigationItems,
194 ]);
195 }
