107 protected function afterThrowable($controller, $methodName, Throwable $exception)
108 {
109 $reflectionMethod = new ReflectionMethod($controller, $methodName);
110 if ($this->hasAnnotationOrAttribute($reflectionMethod, Attributes\DoNotCatchExceptions::class)) {
111 throw $exception;
112 }
113 if (!($exception instanceof Exceptions\EnduserNotificationException)) {
114 $originalException = $exception;
115 $exceptionMessage = $this->l->t(
116 'Unable to serve request to "%1$s": %2$s',
117 [ $this->request->getPathInfo(), $originalException->getMessage() ],
118 );
119 $appRootFolder = $this->appContainer->get(App::APP_ROOT_FOLDER);
120 $exceptionMessage = str_replace($appRootFolder, '...', $exceptionMessage);
121
122 $context = [];
123 switch (get_class($originalException)) {
124 case QueryNotFoundException::class:
125 $httpStatusCode = Http::STATUS_NOT_FOUND;
126 break;
127 case Exceptions\NotAuthorizedException::class:
129 $httpStatusCode = Http::STATUS_UNAUTHORIZED;
130 // @todo: the Vue error-page should compose a useful hint out of
131 // this information.
132 $context['authorization'] = [
133 'userId' => $originalException->getUserId(),
134 'actualPermissions' => $originalException->getActualPermissions(),
135 'requiredPermissions' => $originalException->getRequiredPermissions(),
136 ];
137 break;
138 case InvalidArgumentException::class:
139 $httpStatusCode = Http::STATUS_BAD_REQUEST;
140 break;
141 default:
142 $httpStatusCode = Http::STATUS_INTERNAL_SERVER_ERROR;
143 break;
144 }
145
146 $exception = new Exceptions\EnduserNotificationException(
147 $exceptionMessage, 0, $originalException,
148 httpStatusCode: $httpStatusCode,
149 context: $context,
150 );
151 }
152 $httpStatusCode = $exception->getHttpStatusCode();
153 $context = Util::arrayMergeRecursive(
154 [
155 'httpStatusCode' => $httpStatusCode,
156 ],
157 $exception->getContext() ?? [],
158 );
159 $logEntry = $this->logException(
160 $exception,
161 message: $exception->getMessage(),
162 context: $context,
163 returnLogEntry: true,
164 shift: PHP_INT_MIN, // do not decorate with prefix
165 );
166 // $this->logInfo('LOG_ENTRY ' . print_r($logEntry, true));
167 // $this->logInfo('LOG_ENTRY ' . \OCA\CAFEVDB\Common\Functions\dump($logEntry));
168 if (is_array($logEntry)) {
169 array_walk_recursive($logEntry, fn(&$value) => $value = str_replace(\OC::$SERVERROOT, '', $value));
170 } else {
171 $this->logError('Log entry is null');
172 }
173 $this->logDebug('LOG_ENTRY ' . print_r($logEntry, true));
174 return new JSONResponse($logEntry ?? [], $httpStatusCode);
175 }
