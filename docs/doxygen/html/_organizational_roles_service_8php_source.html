91 public function dedicatedBoardMemberContact(string $role, int $musicianId = 0):?array
92 {
93 $participant = $this->dedicatedBoardMemberParticipant($role, $musicianId);
94 if (empty($participant)) {
95 return null;
96 }
97 $musician = $participant->getMusician();
98 $data = [
99 'email' => $musician->getEmail(),
100 'name' => $musician->getPublicName(firstNameFirst: true),
101 'firstName' => $musician->getFirstName(),
102 'surName' => $musician->getSurName(),
103 'street' => $musician->getStreet(),
104 'streetNumber' => $musician->getStreetNumber(),
105 'streetAndNumber' => $musician->getStreet() . ' ' . $musician->getStreetNumber(),
106 'postalCode' => $musician->getPostalCode(),
107 'city' => $musician->getCity(),
108 'phone' => $musician->getFixedLinePhone(),
109 'mobile' => $musician->getMobilePhone(),
110 ];
111
112 // Override with configured functional address
113 $roleEmail = $this->getConfigValue($role . 'Email', null);
114 if (!empty($roleEmail)) {
115 $data['email'] = $roleEmail;
116 }
117
118 $missingFields = array_keys(array_filter($data, fn($value) => empty($value)));
119
120 if (!empty($missingFields)) {
121 $this->logInfo('Missing fields for ' . $role . ' ' . print_r($missingFields, true) . ' ' . print_r($data, true));
122 }
123
124 // if some field are missing try to fill in from the cloud data
125 if (!empty($missingFields)) {
126
127 $roleUid = $this->getConfigValue($role.'UserId', null);
128 if (!empty($roleUid)) {
129 $user = $this->user($roleUid);
131 $accountManager = $this->di(IAccountManager::class);
133 $account = $accountManager->getAccount($user);
134
135 foreach ($missingFields as $key) {
136 switch ($key) {
137 case 'email':
138 $item = $user->getEMailAddress();
139 break;
140 case 'name':
141 $item = $user->getDisplayName();
142 break;
143 case 'streetAndNumber':
144 $item = $account->getProperty(IAccountManager::PROPERTY_ADDRESS)->getValue();
145 break;
146 case 'phone':
147 $item = $account->getProperty(IAccountManager::PROPERTY_PHONE)->getValue();
148 break;
149 case 'postalCode':
150 case 'city':
151 case 'mobile':
152 $item = null;
153 break;
154 }
155 if (!empty($item)) {
156 $data[$key] = $item;
157 }
158 }
159 }
160 }
161
162 return $data;
163 }

323 public function isDedicatedBoardMember(string $role, ?string $uid = null, bool $allowGroupAccess = false):bool
324 {
325 empty($uid) && $uid = $this->userId();
