94 public function dedicatedBoardMemberContact(string $role, int $musicianId = 0):?array
95 {
96 $participant = $this->dedicatedBoardMemberParticipant($role, $musicianId);
97 if (empty($participant)) {
98 return null;
99 }
100 $musician = $participant->getMusician();
101 $data = [
102 'email' => $musician->getEmail(),
103 'name' => $musician->getPublicName(firstNameFirst: true),
104 'firstName' => $musician->getFirstName(),
105 'surName' => $musician->getSurName(),
106 'street' => $musician->getStreet(),
107 'streetNumber' => $musician->getStreetNumber(),
108 'streetAndNumber' => $musician->getStreet() . ' ' . $musician->getStreetNumber(),
109 'postalCode' => $musician->getPostalCode(),
110 'city' => $musician->getCity(),
111 'phone' => $musician->getFixedLinePhone(),
112 'mobile' => $musician->getMobilePhone(),
113 ];
114
115 // Override with configured functional address
116 $roleEmail = $this->getConfigValue($role . 'Email', null);
117 if (!empty($roleEmail)) {
118 $data['email'] = $roleEmail;
119 }
120
121 $missingFields = array_keys(array_filter($data, fn($value) => empty($value)));
122
123 if (!empty($missingFields)) {
124 $this->logInfo('Missing fields for ' . $role . ' ' . print_r($missingFields, true) . ' ' . print_r($data, true));
125 }
126
127 // if some field are missing try to fill in from the cloud data
128 if (!empty($missingFields)) {
129
130 $roleUid = $this->getConfigValue($role.'UserId', null);
131 if (!empty($roleUid)) {
132 $user = $this->user($roleUid);
134 $accountManager = $this->di(IAccountManager::class);
136 $account = $accountManager->getAccount($user);
137
138 foreach ($missingFields as $key) {
139 switch ($key) {
140 case 'email':
141 $item = $user->getEMailAddress();
142 break;
143 case 'name':
144 $item = $user->getDisplayName();
145 break;
146 case 'streetAndNumber':
147 $item = $account->getProperty(IAccountManager::PROPERTY_ADDRESS)->getValue();
148 break;
149 case 'phone':
150 $item = $account->getProperty(IAccountManager::PROPERTY_PHONE)->getValue();
151 break;
152 case 'postalCode':
153 case 'city':
154 case 'mobile':
155 $item = null;
156 break;
157 }
158 if (!empty($item)) {
159 $data[$key] = $item;
160 }
161 }
162 }
163 }
164
165 return $data;
166 }

326 public function isDedicatedBoardMember(string $role, ?string $uid = null, bool $allowGroupAccess = false):bool
327 {
