119 public function executeActions(bool $gracefully = false):bool
120 {
121 if ($this->executing) {

123 }
124 $this->executing = true;
125 $this->runQueueExceptions = [];
126 $this->undoExceptions = [];
127 $this->undoStack = [];
128 usort($this->actionQueue, fn(IUndoable $a, IUndoable $b) => $a->getSortOrder() <=> $b->getSortOrder());
129 while (!empty($this->actionQueue)) {
130 try {
131 $action = array_shift($this->actionQueue); // from front
132 $action->do();
133 array_unshift($this->undoStack, $action); // at front
134 } catch (Throwable $t) {
135 $message = $this->l->t(
136 'Exception during execution of run-queue: %1$s. Number of successful actions: %2$d.', [
137 $t->getMessage(),
138 count($this->undoStack),
139 ]);
140 $this->runQueueExceptions[] = $t;
141 if ($gracefully) {
142 $this->logException($t, $message);
143 } else {
144 $this->executing = false;
145 throw new UndoableRunQueueException($this, $message, 0, $t);
146 }
147 }
148 }
149 $this->executing = false;
