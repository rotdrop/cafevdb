133 empty($projectName) ? '?' : $projectName,
134 ],
135 ),
136 );
137 }
138
139 $selectedEvents = array_unique($this->request->getParam('eventSelect', []));
140 $calendarIds = array_unique($this->request->getParam('calendarId', []));
141
142 $exports = []; // array marking selected events
143 foreach ($selectedEvents as $eventIdentifier) {
144 $eventIdentifier = json_decode($eventIdentifier, true);
145 $flatIdentifier = EventsService::makeFlatIdentifier($eventIdentifier);
146 $exports[$flatIdentifier] = $eventIdentifier;
147 }
148
149 if (empty($exports)) {
150 foreach ($calendarIds as $eventUri => $calendarId) {
151 $exports[] = [ 'calendarId' => $calendarId, 'uri' => $eventUri ];
152 }
153 }
154
155 $fileName = $projectName.'-'.$this->timeStamp().'.ics';
156
157 return $this->dataDownloadResponse(
158 $this->eventsService->exportEvents($exports, $projectName),
159 $fileName,
160 'text/calendar');
