144 $selectedEvents = array_unique($this->request->getParam('eventSelect', []));
145 $calendarIds = array_unique($this->request->getParam('calendarId', []));
146
147 $exports = []; // array marking selected events
148 foreach ($selectedEvents as $eventIdentifier) {
149 $eventIdentifier = json_decode($eventIdentifier, true);
150 $flatIdentifier = EventsService::makeFlatIdentifier($eventIdentifier);
151 $exports[$flatIdentifier] = $eventIdentifier;
152 }
153
154 if (empty($exports)) {
155 foreach ($calendarIds as $eventUri => $calendarId) {
156 $exports[] = [ 'calendarId' => $calendarId, 'uri' => $eventUri ];
157 }
158 }
159
160 $fileName = $projectName.'-'.$this->timeStamp().'.ics';
161
162 return $this->dataDownloadResponse(
163 $this->eventsService->exportEvents($exports, $projectName),
