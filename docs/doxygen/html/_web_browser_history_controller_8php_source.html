116 private function exportHistoryState(
117 Entities\WebBrowserHistoryState $historyState,
118 string $modeOrKey,
119 ):array {
120 switch ($modeOrKey) {
121 case self::GET_MODE_DEEP:
122 case self::GET_MODE_SHALLOW:
123 $dataItem = [
124 'modificationTime' => $historyState->getCreated()->format('U.v'),
125 'position' => $historyState->getPos()->getKey(),
126 'history' => [],
127 ];
128 if ($modeOrKey === self::GET_MODE_DEEP) {
129 $dataItem['requestData'] = [];
130 }
132 foreach ($historyState->getStack() as $key => $entry) {
133 $hash = $entry->getData()->getHash();
134 $dataItem['history'][$key] = $this->exportHistoryEntry($entry, $modeOrKey);
135 if ($modeOrKey === self::GET_MODE_DEEP) {
136 $dataItem['requestData'][$hash] = $entry->getData()->getData();
137 }
138 }
139 break;
140 default:
141 // just the data form a single item, this is always "deep" including the post data.
142 $entry = $historyState->getEntry($modeOrKey);
143 if ($entry === null) {
144 throw new Exceptions\EnduserNotificationException(
145 $this->l->t('History record with key "%1$s" at time "%2$s" for user "%3$s" could not be found.', [
146 $modeOrKey,
147 $this->dateTimeFormatter->formatDateTime($historyState->getCreated()),
148 $this->userId,
149 ]),
150 0,
151 httpStatusCode: Http::STATUS_NOT_FOUND,
152 );
153 }
154 $dataItem = $this->exportHistoryEntry($entry, self::GET_MODE_DEEP);
155 break;
156 }
157 return $dataItem;
158 }

167 #[CoreAttributes\NoAdminRequired]
168 public function get(string|float $timestamp, string $modeOrKey = self::GET_MODE_SHALLOW)
169 {
170 $repository = $this->getDatabaseRepository(Entities\WebBrowserHistoryState::class);
171 switch ($timestamp) {
172 case self::GET_REQUEST_ALL:
173 $historyStates = $repository->findBy(
174 criteria: [ 'userId' => $this->userId ],
175 orderBy: [ 'created' => 'DESC' ],
176 );
177 $data = [];
179 foreach ($historyStates as $historyState) {
180 $item = $this->exportHistoryState($historyState, $modeOrKey);
181 $data[$item['modificationTime']] = $item;
182 }
183 break;
184 case self::GET_REQUEST_TIMESTAMPS:
185 $historyStates = $repository->findBy(
186 criteria: [ 'userId' => $this->userId ],
187 orderBy: [ 'created' => 'DESC' ],
188 );
189 $data = [];
190 foreach ($historyStates as $historyState) {
191 // return fractional milliseconds as this is what the frontend uses.
192 $data[] = $historyState->getCreated()->format('U.v');
193 }
194 break;
195 default:
196 $date = self::convertToDateTime($timestamp);
197 $historyState = $repository->findOneBy([ 'userId' => $this->userId, 'created' => $date ]);
198 if (empty($historyState)) {
199 throw new Exceptions\EnduserNotificationException(
200 $this->l->t('History record at time "%1$s" for user "%2$s" could not be found.', [
201 $this->dateTimeFormatter->formatDateTime($date),
202 $this->userId,
203 ]),
204 0,
205 httpStatusCode: Http::STATUS_NOT_FOUND,
206 );
207 }
208 $data = $this->exportHistoryState($historyState, $modeOrKey);
209 break;
210 }
211 return self::dataResponse($data);
212 }

229 {
230 $historyState = new Entities\WebBrowserHistoryState($timestamp, $this->userId);
231 $this->entityManager->beginTransaction();
232 try {
233 $this->persist($historyState);
234 $this->flush(); // we need the autoincrement value
235
236 $dataRepository = $this->getDatabaseRepository(Entities\WebBrowserHistoryData::class);
237 $data = [];
238 foreach ($requestData as $hash => $postData) {
239 $dataItem = $dataRepository->find($hash);
240 if (!$dataItem) {
241 $dataItem = new Entities\WebBrowserHistoryData($hash, $postData);
242 $this->persist($dataItem);
243 }
244 $data[$hash] = $dataItem;
245 }
246
247 foreach ($history as $key => $entryData) {
248 $entry = new Entities\WebBrowserHistoryEntry(
249 state: $historyState,
250 key: $key,
251 path: $entryData['path'],
252 data: $data[$entryData['hash']],
253 );
254 $this->persist($entry);
255 $historyState->addEntry($entry);
256 }
257
258 $historyState->setPos($historyState->getEntry($position));
259
260 $this->entityManager->flush();
261 $this->entityManager->commit();
262 } catch (Throwable $t) {
263 $this->entityManager->rollback();
264
265 if ($t instanceof DBALException\UniqueConstraintViolationException) {
266 throw new Exceptions\EnduserNotificationException(
267 $this->l->t('History record at time "%1$s" for user "%2$s" already has been saved previously.', [
268 $this->dateTimeFormatter->formatDateTime($historyState->getCreated()),
269 $this->userId,
270 ]),
271 0,
272 $t,
273 httpStatusCode: Http::STATUS_CONFLICT,
274 );
275 }
276
277 throw new Exceptions\DatabaseException(
278 $this->l->t('Unable to store history data.'),
279 0,
280 $t
281 );
282 }
283 return self::dataResponse([
284 'message' => $this->l->t('History record at time "%1$s" for user "%2$s" has been saved into the database.', [
285 $this->dateTimeFormatter->formatDateTime($historyState->getCreated()),
286 $this->userId,
287 ]),
288 ]);
