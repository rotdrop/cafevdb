74 public function findLike(
75 array $criteria,
76 ?array $orderBy = null,
77 ?int $limit = null,
78 ?int $offset = null,
79 ): array {
80 $qb = $this->createQueryBuilder('table');
81 $andX = $qb->expr()->andX();
82 foreach ($criteria as $key => &$value) {
83 $value = str_replace('*', '%', $value);
84 if (strpos($value, '%') !== false) {
85 $andX->add($qb->expr()->like('table'.'.'.$key, ':'.$key));
86 } else {
87 $andX->add($qb->expr()->eq('table'.'.'.$key, ':'.$key));
88 }
89 }
90 foreach ($criteria as $key => $value) {
91 $qb->setParameter($key, $value);
92 }
93 $qb->where($andX);
94
95 self::addOrderBy($qb, $orderBy, $limit, $offset, 'table');
96
97 return $qb->getQuery()->execute();
98 }

242 public function findBy(
243 array $criteria,
244 ?array $orderBy = null,
245 ?int $limit = null,
246 ?int $offset = null,
247 ):array {
248 $queryParts = $this->prepareFindBy($criteria, $orderBy);
249
250 // The stock findBy() / findOneBy() functions do not use query-hints, see
251 // https://github.com/doctrine/orm/issues/6751
252 if (empty($this->getEntityManager()->getConfiguration()->getDefaultQueryHints())
253 && empty($queryParts['joinEntities'])
254 && empty($queryParts['indexBy'])
255 && empty($queryParts['modifiers'])
256 && empty($queryParts['collectionCriteria'])) {
257 // vanilla
258 return parent::findBy($criteria, $orderBy, $limit, $offset);
259 }
260
261 $qb = $this->generateFindBySelect($queryParts);
262
263 if (count($queryParts['joinEntities']) > 0) {
264 // joining may produce excessive extra columns, try to group by the primary keys.
265 foreach ($this->getClassMetadata()->identifier as $field) {
266 $qb->addGroupBy('mainTable.'.$field);
267 }
268 }
269
270 $this->generateFindByWhere($qb, $queryParts, $limit, $offset);
271
272 // $this->log('SQL '.$qb->getQuery()->getSql());
273 // $this->log('PARAM '.print_r($qb->getQuery()->getParameters()->toArray(), true));
274
275 return $qb->getQuery()->getResult();

336 protected function prepareFindBy(array $criteria, ?array $orderBy = null): array
337 {
338 $options = [
339 Constants::QUERY_OPTION_WILDCARDS => false,
340 ];
341 if (!empty($criteria)
342 && array_keys($criteria)[0] === self::OPTIONS_KEY
343 && is_array($criteria[self::OPTIONS_KEY])) {
344 $options = array_merge($options, $criteria[self::OPTIONS_KEY]);
345 unset($criteria[self::OPTIONS_KEYS]);
346 }
347
348 $orderBy = $orderBy?:[];
349
350 // filter out instances of criteria
351 $collectionCriteria = [];
352 foreach ($criteria as $key => $value) {
353 if ($value instanceof Collections\Criteria) {
354 $collectionCriteria[] = $value;
355 unset($criteria[$key]);
356 }
357 }
358
359 $whereCriteria = [];
360 $havingCriteria = [];
361 foreach ($criteria as $key => $value) {
362 // if key is numeric we assume $value is a pair [ EXPR => VALUE ]
363 if (is_numeric($key)) {
364 $key = array_key_first($value);
365 $value = $value[$key];
366 }
367 $groupFunction = null;
368 $fctPos = strpos($key, '@');
369 if ($fctPos !== false) {
370 $groupFunction = substr($key, $fctPos + 1);
371 $key = substr($key, 0, $fctPos);
372 }
373 $sqlFunction = null;
374 $fctPos = strpos($key, '#');
375 if ($fctPos !== false) {
376 $sqlFunction = substr($key, $fctPos +1);
377 $key = substr($key, 0, $fctPos);
378 }
379 $fieldType = null;
380 $typePos = strpos($key, ':');
381 if ($typePos !== false) {
382 $fieldType = substr($key, $typePos + 1);
383 $key = substr($key, 0, $typePos);
384 }
385
386 $criterion = [
387 'field' => $key,
388 'fieldType' => $fieldType,
389 'value' => $value,
390 'modifiers' => [],
391 'junctors' => [],
392 'comparator' => null,
393 'literal' => false,
394 'index' => count($whereCriteria) + count($havingCriteria),
395 'groupFunction' => $groupFunction,
396 'sqlFunction' => $sqlFunction,
397 ];
398
399 if (preg_match('/^[!=<>&|()]+/', $key, $matches)) {
400
401 $operators = $matches[0];
402 $key = substr($key, strlen($operators));
403
404 $criterion['field'] = $key;
405
406 $count = 0;
407 while (!empty($operators)) {
408 ++$count;
409 if ($count > 100) {
410 throw new DatabaseException('INFINITE LOOP "' . $operators . '" ' . print_r($criterion, true) . ' ' . print_r($criteria, true));
411 }
412 // reduce multiple ! signs
413 while (strpos($operators, '!!') !== false) {
414 $operators = str_replace('!!', '', $operators);
415 }
416 // (&, (|, !(|, !(&
417 foreach (self::JUNCTORS as $abbr => $junctor) {
418 $pos = strpos($operators, $abbr);
419 if ($pos === 0) {
420 $operators = substr($operators, strlen($abbr));
421 $criterion['junctors'] = array_merge($criterion['junctors'], $junctor);
422 }
423 if (empty($operators)) {
424 break 2;
425 }
426 }
427 // this is only !
428 foreach (self::MODIFIERS as $abbr => $modifier) {
429 $pos = strpos($operators, $abbr);
430 if ($pos === 0) {
431 $operators = substr($operators, strlen($abbr));
432 $criterion['modifiers'][] = $modifier;
433 }
434 if (empty($operators)) {
435 break 2;
436 }
437 }
438 foreach (self::COMPARISONS as $abbr => $comparator) {
439 $pos = strpos($operators, $abbr);
440 if ($pos === 0) {
441 if (!empty($criterion['comparator'])) {
442 throw new DatabaseException('Comparison for key "' . $key . '" already set to "' . $criterion['comparator'] . '".');
443 }
444 if ($abbr == '=' && $value === null) {
445 throw new DatabaseException('Comparison with null is undefined, only equality is allowed.');
446 }
447 if (is_array($value)) {
448 throw new DatabaseException('Array-valued comparisons are not allowed.');
449 }
450 $operators = substr($operators, strlen($abbr));
451 $criterion['comparator'] = $comparator;
452 }
453 if (empty($operators)) {
454 break 2;
455 }
456 }
457 }
458 } // !empty($operators)
459
460 if ($groupFunction === null) {
461 $whereCriteria[] = $criterion;
462 } else {
463 $havingCriteria[] = $criterion;
464 }
465 }
466
467 // walk through criteria and find associations ASSOCIATION.FIELD
468 $joinEntities = [];
469 foreach (array_merge($whereCriteria, $havingCriteria) as $criterion) {
470 $field = $criterion['field'];
471 $dotPos = strpos($field, '.');
472 $joinParent = 'mainTable';
473 $joinAlias = [];
474 while ($dotPos !== false) {
475 $joinTable = substr($field, 0, $dotPos);
476 $joinAlias[] = $joinTable;
477 $aliasKey = implode('_', $joinAlias);
478 $joinEntities[$aliasKey] = [
479 'parent' => $joinParent,
480 'association' => $joinTable,
481 'alias' => $aliasKey,
482 ];
483 $field = substr($field, $dotPos + 1);
484 $dotPos = strpos($field, '.');
485 $joinParent = $joinTable;
486 }
487 }
488 $indexBy = [];
489 foreach ($orderBy as $key => $ordering) {
490 $dotPos = strpos($key, '.');
491 if ($dotPos !== false) {
492 $joinParent = 'mainTable';
493 $joinAlias = [];
494 while ($dotPos !== false) {
495 $joinTable = substr($field, 0, $dotPos);
496 $joinAlias[] = $joinTable;
497 $aliasKey = implode('_', $joinAlias);
498 $joinEntities[] = [
499 'parent' => $joinParent,
500 'association' => $joinTable,
501 'alias' => $aliasKey,
502 ];
503 $joinAlias[] = $joinTable;
504 $field = substr($field, $dotPos + 1);
505 $dotPos = strpos($field, '.');
506 $joinParent = $joinTable;
507 }
508 } else {
509 $tableAlias = 'mainTable';
510 $field = $tableAlias.'.'.$key;
511 }
512 if (strtoupper($ordering) == 'INDEX') {
513 $indexBy[$tableAlias] = $field;
514 unset($orderBy[$key]);
515 }
516 }
517
518 // modify and also return
519 $queryParts = [
520 'options' => $options,
521 'criteria' => [
522 'where' => $whereCriteria,
523 'having' => $havingCriteria,
524 ],
525 'orderBy' => $orderBy,
526 'joinEntities' => $joinEntities,
527 'indexBy' => $indexBy,
528 'collectionCriteria' => $collectionCriteria,
529 ];
530
531 return $queryParts;
532 }

544 protected function generateFindBySelect(array $queryParts, ?array $select = null)
545 {
546 $indexBy = $queryParts['indexBy']?:['mainTable' => null];
547 $qb = $this->createQueryBuilder('mainTable', $indexBy['mainTable']);
548 foreach ($queryParts['joinEntities'] as $joinInfo) {
549 $association = $joinInfo['association'];
550 $alias = $joinInfo['alias'];
551 $qb->leftJoin($joinInfo['parent'] . '.' . $association, $alias, null, null, $indexBy[$alias] ?? null);
552 }
553 if (!empty($select)) {
554 $qb->select($select);
555 }
556 return $qb;

616 protected function generateFindByWhere(
617 ORM\QueryBuilder $qb,
618 array $queryParts,
619 ?int $limit = null,
620 ?int $offset = null,
621 ):ORM\QueryBuilder {
622
624 $class = $this->getClassMetadata();
625
627 $entityManager = $this->getEntityManager();
628
629 // unpack parameter array
630 // foreach ($queryParts as $key => $part) {
631 // ${$key} = $part;
632 // }
633 extract($queryParts);
634
635 foreach (['where', 'having'] as $conditionType) {
636
637 if (!empty($criteria[$conditionType])) {
638 $groupExpression = [
639 0 => [
640 'junctor' => 'andX',
641 'components' => [],
642 ]
643 ];
644 $groupLevel = 0;
645
646 foreach ($criteria[$conditionType] as &$criterion) {
647
648 $field = $criterion['field'];
649 $joinParts = explode('.', $field);
650 if (count($joinParts) > 1) {
651 $col = array_pop($joinParts);
652 $field = implode('_', $joinParts) . '.' . $col;
653 }
654
655 $value = $criterion['value'];
656
657 // $this->log('FIELD ' . $conditionType . ': ' . $field);
658
659 foreach ($criterion['junctors'] as $junctor) {
660 if ($junctor !== ')') {
661 $expression = [
662 'junctor' => $junctor,
663 'components' => [],
664 ];
665 $groupExpression[++$groupLevel] = $expression;
666 } else {
667 $expression = array_pop($groupExpression);
668 $groupLevel--;
669 $junctor = $expression['junctor'];
670 $components = $expression['components'];
671 $compositeExpr = call_user_func_array([ $qb->expr(), $junctor ], $components);
672 $groupExpression[$groupLevel]['components'][] = $compositeExpr;
673 }
674 }
675 if (empty($field)) {
676 continue;
677 }
678 // $this->log('FIELD ' . $field);
679 $fieldIsCollection = false;
680 $dotPos = strpos($field, '.');
681 if ($dotPos !== false) {
682 $tableAlias = substr($field, 0, $dotPos);
683 $column = substr($field, $dotPos + 1);
684 } else {
685 $tableAlias = 'mainTable';
686 $column = $field;
687 $field = $tableAlias . '.' . $field;
688 }
689 if ($tableAlias == 'mainTable') {
690 $fieldIsCollection = $class->isCollectionValuedAssociation($column);
691 } elseif ($class->hasAssociation($tableAlias)) {
692 $targetClass = $entityManager->getClassMetadata($class->getAssociationTargetClass($tableAlias));
693 $fieldIsCollection = $targetClass->isCollectionValuedAssociation($column);
694 }
695 $param = str_replace('.', '_', $field) . '_' . $criterion['index'];
696 if (!empty($criterion['groupFunction'])) {
697 $field = sprintf($criterion['groupFunction'], $field);
698 } elseif (!empty($criterion['sqlFunction'])) {
699 $field = sprintf($criterion['sqlFunction'], $field);
700 }
701 $comparator = $criterion['comparator'] ?? 'eq';
702 if ($value === null) {
703 $criterion['literal'] = true;
704 $expr = $qb->expr()->isNull($field);
705 } elseif (is_array($value)) {
706 // special case empty array:
707 // - in principle always FALSE (nothing is in an empty set)
708 // - FIELD == NULL ? NULL in any given set would be FALSE, we
709 // keep it that way, even if the set is empty
710 if (empty($value)) {
711 $criterion['literal'] = true;
712 // unfortunately, a literal 0 just cannot be modelled with the query builder
713 $expr = $qb->expr()->eq(1, 0);
714 } else {
715 // array values could contain wildcards
716 if (($options[Constants::QUERY_OPTION_WILDCARDS] ?? false) === true
717 && !empty(array_filter($value, fn($x) => self::containsWildcards($x)))) {
718 $value = implode('|', array_map(fn($x) => self::wildcardsToRegexp($x), $value));
719 $value = '^' . $value . '$';
720 $expr = $qb->expr()->eq(new Expr\Func('REGEXP', [ $field, ':' . $param ]), 1);
721 $criterion['value'] = $value;
722 } else {
723 $expr = $qb->expr()->in($field, ':' . $param);
724 }
725 }
726 } elseif (is_string($value)) {
727 if (($options[Constants::QUERY_OPTION_WILDCARDS] ?? false) === true && self::containsWildcards($value)) {
728 $expr = $qb->expr()->like($field, ':' . $param);
729 $criterion['value'] = $value;
730 } else {
731 if ($fieldIsCollection && $comparator == 'eq') {
732 $expr = $qb->expr()->isMemberOf(':' . $param, $field);
733 } else {
734 $expr = $qb->expr()->$comparator($field, ':' . $param);
735 }
736 }
737 } elseif ($fieldIsCollection && $comparator == 'eq') {
738 $expr = $qb->expr()->isMemberOf(':' . $param, $field);
739 } else {
740 $expr = $qb->expr()->$comparator($field, ':' . $param);
741 }
742 foreach ($criterion['modifiers'] as $modifier) {
743 $expr = $qb->expr()->$modifier($expr);
744 }
745 $groupExpression[$groupLevel]['components'][] = $expr;
746 }
747
748 // closing parenthesis maybe omitted at end, can be arbitrary many
749 while ($groupLevel >= 0) {
750 $expression = array_pop($groupExpression);
751 $groupLevel--;
752 $junctor = $expression['junctor'];
753 $components = $expression['components'];
754 $compositeExpr = call_user_func_array([ $qb->expr(), $junctor ], $components);
755 if ($groupLevel >= 0) {
756 $groupExpression[$groupLevel]['components'][] = $compositeExpr;
757 } else {
758 $qb->{$conditionType}($compositeExpr);
759 }
760 }
761 unset($criterion); // break reference
762 }
763
764 foreach ($criteria[$conditionType] as $criterion) {
765 if ($criterion['literal']) {
766 continue;
767 }
768 $field = $criterion['field'];
769 if (empty($field)) {
770 continue;
771 }
772 // $this->log('PARAMETER ' . $field);
773 $dotPos = strpos($field, '.');
774 if ($dotPos !== false) {
775 $tableAlias = substr($field, 0, $dotPos);
776 $column = substr($field, $dotPos+1);
777 } else {
778 $tableAlias = 'mainTable';
779 $column = $field;
780 $field = $tableAlias . '.' . $column;
781 }
782 $param = str_replace('.', '_', $field) . '_' . $criterion['index'];
783
784 $value = $criterion['value'];
785 $fieldType = null;
786 if ($criterion['fieldType'] !== null) {
787 $fieldType = $criterion['fieldType'];
788 } elseif ($tableAlias == 'mainTable') {
789 if (!is_array($value)
790 && empty($criterion['groupFunction'])
791 && empty($criterion['sqlFunction'])) {
792 // try to deduce the type as this is NOT done by ORM here
793 $fieldType = $this->getClassMetadata()->getTypeOfField($column);
794 }
795 }
796 $qb->setParameter($param, $value, $fieldType);
797 }
798 }
799
800 self::addOrderBy($qb, $orderBy, $limit, $offset, 'mainTable');
801
802 foreach ($collectionCriteria as $selectableCriteria) {
803 $qb->addCriteria($selectableCriteria);
804 }
805
806 return $qb;
807 }
