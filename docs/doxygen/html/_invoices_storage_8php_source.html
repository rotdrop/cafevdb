90 public function addDocument(
91 Entity $entity,
92 Entities\EncryptedFile $file,
93 bool $flush = true,
94 string $conflict = DatabaseStorageFolder::ADD_DOCUMENT_CONFLICT_FAIL,
95 ):Entities\DatabaseStorageFile {
96 $mimeType = $file->getMimeType();
97 $extension = Util::fileExtensionFromMimeType($mimeType);
98 if (empty($extension) && !empty($file->getFileName())) {
99 $extension = strtolower(pathinfo($file->getFileName(), PATHINFO_EXTENSION));
100 }
101 $this->logInfo('MIME AND EXTENSION ' . $mimeType . ' "' . $extension . '"');
102 $folderName = $this->getInvoiceFileName($entity);
103 $fileName = $folderName . ($extension ? '.' . $extension : '');
104 $folderName = substr($folderName, strlen($this->getAppL10n()->t('invoice')) + 1);
105
106 if ($flush) {
107 $this->entityManager->beginTransaction();
108 }
109 try {
110 // search for the folder
111 $rootFolder = $this->getRootFolder(create: true);
112 if (empty($rootFolder)) {
113 throw new UnexpectedValueException($this->l->t('Root-folder does not exist.'));
114 }
115 $year = $this->getBirthTimeFromEntity($entity)->format('Y');
116 $yearFolder = $this->rootFolder->getFolderByName($year);
117 if (empty($yearFolder)) {
118 $yearFolder = $this->rootFolder->addSubFolder($year)
119 ->setUpdated($file->getUpdated())
120 ->setCreated($file->getCreated());
121 $this->persist($yearFolder);
122 }
123 $invoiceFolder = $yearFolder->getFolderByName($folderName);
124 if (empty($invoiceFolder)) {
125 $invoiceFolder = $yearFolder->addSubFolder($folderName)
126 ->setUpdated($file->getUpdated())
127 ->setCreated($file->getCreated());
128 $this->persist($invoiceFolder);
129 }
130
131 $document = $invoiceFolder->addDocument($file, $fileName, conflict: $conflict)
132 ->setCreated($file->getCreated())
133 ->setUpdated($file->getUpdated());
134 $this->persist($document);
135 $invoiceFolder
136 ->setCreated(min($file->getCreated(), $yearFolder->getCreated()))
137 ->setUpdated(max($file->getUpdated(), $yearFolder->getUpdated()));
138 $yearFolder
139 ->setCreated(min($invoiceFolder->getCreated(), $invoiceFolder->getCreated()))
140 ->setUpdated(max($invoiceFolder->getUpdated(), $invoiceFolder->getUpdated()));
141
142 if ($flush) {
143 $this->flush();
144 $this->entityManager->commit();
145 }
146
147 } catch (Throwable $t) {
148 if ($this->entityManager->isTransactionActive()) {
149 $this->entityManager->rollback();
150 }
151 throw new Exceptions\Exception($this->l->t('Unable to add new document "%s".', $file->getFileName()), previous: $t);
152 }
153
154 return $document;
155 }

254 public function touch($path, $mtime = null)
255 {
256 if ($this->is_dir($path)) {
257 // $this->logInfo('IS DIR ' . $path);
258 return false;
259 }
260 list('basename' => $baseName, 'dirname' => $dirName) = self::pathinfo($path);
261
262 $this->logInfo('EXPLODED PATH ' . print_r(explode(self::PATH_SEPARATOR, $path), true));
263
264 // Do not allow files with the same name as the year directories
265 if (preg_match('|^/?(\d{4})/?$|', $baseName)) {
266 return false;
267 }
268
270 $invoice = $this->entityRepository->findInvoiceByFileName(self::pathInfo($dirName, PATHINFO_BASENAME));
271 if ($invoice && $invoice->getNotificationEmail() && !str_contains($baseName, 'ocTransferId')) {
272 // Do not allow adding further documents except adding a folder
273 // documentation. The final rename will fail in addition if this is
274 // something else but a readme.
275 return false;
276 }
277
278 $this->entityManager->beginTransaction();
279 try {
280 $this->getRootFolder();
281
283 $dirEntry = $this->fileFromFileName($path);
284 if (empty($dirEntry)) {
285 $parent = $this->fileFromFileName($dirName);
286 if (empty($parent)) {
287 return false;
288 }
289 $file = new Entities\EncryptedFile($baseName, '', '');
290 if ($mtime !== null) {
291 $file->setCreated($mtime);
292 }
293 $this->persist($file);
294 $this->flush();
295 $dirEntry = $parent->addDocument($file, $baseName);
296 $this->persist($dirEntry);
297 if ($mtime !== null) {
298 $dirEntry->setCreated($mtime);
299 }
300 }
301 if ($dirEntry instanceof InMemoryFileNode) {
302 $dirEntry = $this->persistInMemoryFileNode($dirEntry);
303 }
304 if ($mtime !== null) {
305 $dirEntry->setUpdated($mtime);
306 $dirEntry->getFile()->setUpdated($mtime);
307 }
308 $this->flush();
309
310 $this->entityManager->commit();
311
312 $this->setFileNameCache($path, $dirEntry);
313
314 } catch (Throwable $t) {
315 if ($this->entityManager->isTransactionActive()) {
316 $this->entityManager->pushTransactionException($t);
317 $this->entityManager->rollback();
318 } else {
319 $this->logException($t);
320 }
321 return false;
322 }

345 public function rename(string $path1, string $path2): bool
346 {
347 $this->logInfo('RENAME ARGS ' . $path1 . ' || ' . $path2);
348
349 $path1 = $this->buildPath($path1);
350 $path2 = $this->buildPath($path2);
351 list('dirname' => $dirName1, 'basename' => $baseName1) = self::pathinfo($path1);
352 list('dirname' => $dirName2, 'basename' => $baseName2) = self::pathinfo($path2);
353
355 $dirEntry = $this->fileFromFileName($path1);
356 if (empty($dirEntry)) {
357 // $this->logInfo('NO DIR ENTRY FOR ' . $path1);
358 return false;
359 } elseif ($dirEntry instanceof Entities\DatabaseStorageFolder) {
360 // nope, we do not allow renaming folders
361 return false;
362 }
363
365 $invoice = $this->entityRepository->findInvoiceByFileName($baseName1);
366 if ($invoice) {
367 // keep it
368 return false;
369 }
370
371 $readMeFileNames = $this->getReadMeFactory()->getReadMeFileNames();
372
374 $invoice2 = $this->entityRepository->findInvoiceByFileName(self::pathInfo($dirName2, PATHINFO_BASENAME));
375 if ($invoice2 && $invoice2->getNotificationEmail() && !in_array($baseName2, $readMeFileNames)) {
376 $this->logError('Destination directory "' . $dirName2 . '" refers to a sent-out invoice and "' . $baseName2 . '" is not a README filename, denying changes.');
377 // @todo cleanup path1 if it is a temporary file
378 return false;
379 }
380
382 $invoice1 = $this->entityRepository->findInvoiceByFileName(self::pathInfo($dirName1, PATHINFO_BASENAME));
383 if ($invoice1 && $invoice1->getNotificationEmail() && !str_contains($baseName1, 'ocTransferId')) {
384 $this->logError('Source directory "' . $dirName1 . '" refers to a sent-out invoice and "' . $baseName1 . '" is not a README upload file, , denying changes.');
385 // @todo cleanup path1 if it is a temporary file
386 return false;
387 }
388
389 if ($dirEntry instanceof InMemoryFileNode) {
390 $dirEntry = $this->persistInMemoryFileNode($dirEntry);
391 }
392
393 if ($dirName1 != $dirName2) {
394 $parent2 = $this->fileFromFileName($dirName2);
395 if (empty($parent2)) {
396 // $this->logInfo('NO PARENT2 for ' . $path2);
397 return false;
398 }
399 }
400
401 $this->entityManager->beginTransaction();
402 try {
403 $dirEntry->setName($baseName2);
404 if (!empty($parent2)) {
405 $dirEntry->setParent($parent2);
406 }
407
408 $this->flush();
409
410 $this->entityManager->commit();
411
412 // update our local files cache
413 $this->setFileNameCache($path2, $dirEntry);
414 $this->unsetFileNameCache($path1);
