193 }

195 return $this->overriddenData[$offset];
196 }
197 $value = $flatEntity[$offset];
198 if ($this->depth > 0 && ($value instanceof EntityReference)) {

200 // This cannot happen:
201 //
202 // if (!isset($data->repositories[$value->entityClassName][$value->flatIdentifier])) {
203 // throw new UnexpectedValueException('Offset "' . $offset . '" has not been fetched from the database at depth "' . $this->depth . '".');
204 // }

213 return $result;
214 } elseif ($value instanceof EntityReferenceCollection) {
215 if ($this->depth === 0) {
216 // even on depth 0 return an empty array if applicable as it simplifies the exported object.

228 // },

233 // }

240 root: $this->root ?? $this,
