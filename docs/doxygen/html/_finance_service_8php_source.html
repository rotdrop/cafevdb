138 public function targetDeadline(
139 int $businessOffset,
140 ?int $calendarOffset = null,
141 ?\DateTimeInterface $fromDate = null,
142 ):DateTimeInterface {
144 if (empty($fromDate)) {
145 $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
146 } else {
147 $fromDate = (new CarbonImmutable)
148 ->setTimezone($fromDate->getTimezone())
149 ->setTimestamp($fromDate->getTimestamp());
150 }
151
152 if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
153 throw new RuntimeException(
154 $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
155 $businessOffset, $calendarOffset
156 ]));
157 }
158
159 CarbonImmutable::setHolidaysRegion('target2');
160 $businessDeadline = $fromDate->addBusinessDays($businessOffset);
161 if (!empty($calendarOffset)) {
162 $calendarDeadline = $fromDate->addDays($calendarOffset);
163
164 $businessInterval = $fromDate->diff($businessDeadline, true);
165 $calendarInterval = $fromDate->diff($calendarDeadline, true);
166
167 if ($calendarInterval->days > $businessInterval->days) {
168 $businessDeadline = $calendarDeadline->nextBusinessDay();
169 }
170 }
171
172 return $businessDeadline;
173 }

1352 * Das Ganze ist deswegen etwas spannend, weil eine Modulo-Rechnung, also eine Ganzzahl-Division mit einer
1353 * bis zu 38-stelligen Ganzzahl durchgefuehrt werden muss. Wegen der meist nur zur Verfuegung stehenden
1354 * 32-Bit-CPUs koennen mit PHP aber nur maximal 9 Stellen mit allen Ziffern genutzt werden.

1359 *
1360 * @return bool

1363 {

1365 $iban1 = substr($iban, 4)
1366 . strval(ord($iban[0]) - 55)
1367 . strval(ord($iban[1]) - 55)
1368 . substr($iban, 2, 2);
1369
1370 for ($i = 0; $i < strlen($iban1); $i++) {
1371 if (ord($iban1[$i]) > 64 && ord($iban1[$i]) < 91) {
1372 $iban1 = substr($iban1, 0, $i) . strval(ord($iban1[$i]) - 55) . substr($iban1, $i+1);
1373 }

1375 $rest = 0;
1376 for ($pos = 0; $pos < strlen($iban1); $pos += 7) {
1377 $part = strval($rest) . substr($iban1, $pos, 7);
1378 $rest = intval($part) % 97;

1380 $checkSum = sprintf("%02d", 98 - $rest);
1381

1383 return substr_replace($iban, $checkSum, 2, 2);
