173 public function targetDeadline(
174 int $businessOffset,
175 ?int $calendarOffset = null,
176 ?\DateTimeInterface $fromDate = null,
177 ):DateTimeInterface {
179 if (empty($fromDate)) {
180 $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
181 } else {
182 $fromDate = (new CarbonImmutable)
183 ->setTimezone($fromDate->getTimezone())
184 ->setTimestamp($fromDate->getTimestamp());
185 }
186
187 if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
188 throw new RuntimeException(
189 $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
190 $businessOffset, $calendarOffset
191 ]));
192 }
193
194 CarbonImmutable::setHolidaysRegion('target2');
195 $businessDeadline = $fromDate->addBusinessDays($businessOffset);
196 if (!empty($calendarOffset)) {
197 $calendarDeadline = $fromDate->addDays($calendarOffset);
198
199 $businessInterval = $fromDate->diff($businessDeadline, true);
200 $calendarInterval = $fromDate->diff($calendarDeadline, true);
201
202 if ($calendarInterval->days > $businessInterval->days) {
203 $businessDeadline = $calendarDeadline->nextBusinessDay();
204 }
205 }
206
207 return $businessDeadline;
208 }

324 public function generateInvoiceNumber(Entities\Musician $originator, null|int|Entities\Project $project, ?DateTimeInterface $date = null):string
325 {
326 $date = $date ?? (new DateTime);
327 $parts = [
328 $originator->getInitials(),
329 $date->format('Y-m'),
330 ];
331 if ($project !== null) {
333 $project = $this->ensureProject($project);
334 array_unshift($parts, $project->getName());
335 }
336
337 // Simplest thing would be to just use the number of seconds since the
338 // start of the month ... but it is much cooler to do this in a different
339 // way.
340
341 $idPrefix = implode(self::INVOICE_ID_GLUE, $parts);
342 $invoices = $this->getDatabaseRepository(Entities\Invoice::class)->findLike([ 'id' => $idPrefix . self::INVOICE_ID_GLUE . '%', ]);
343 $sequence = array_reduce(
344 $invoices,
345 fn(int $sequence, Entities\Invoice $invoice) => max($sequence, (int)substr($invoice->getId(), strlen($idPrefix))),
346 initial: 1,
347 );
348
349 return $idPrefix . self::INVOICE_ID_GLUE . $sequence;

498 public function generateReceiptOfDonationMailMergeData(
499 Entities\CompositePayment $compositePayment,
500 ):array {
501 // perform sanity checks
502
503 $numberOfDonations = $compositePayment->countDonations();
504 if ($numberOfDonations == 0) {
505 throw new InvalidArgumentException($this->l->t('The composite payment "%s" does not refer to a donation.', $compositePayment));
506 } elseif ($numberOfDonations > 1) {
507 throw new InvalidArgumentException($this->l->t('The composite pamynet "%s" contains more than one donation. This is not yet supported.', $compositePayment));
508 }
509
510 $donationAmount = $compositePayment->getDonationAmount();
511 $nonDonationAmount = $compositePayment->getNonDonationAmount();
512
513 if ($donationAmount <= 0) {
514 throw new InvalidArgumentException(
515 $this->l->t(
516 'The donation in the composite payment "%s" sums up to less or equal zero: %s',
517 [
518 $compositePayment,
519 $this->moneyValue($donationAmount),
520 ],
521 ));
522 }
523 if ($nonDonationAmount > 0) {
524 throw new InvalidArgumentException(
525 $this->l->t(
526 'The non-donation parts of the composite payment "%s" sum up to an income of %s. This is unsupported.',
527 [
528 $compositePayment,
529 $this->moneyValue($nonDonationAmount),
530 ],
531 ));
532 }
533
535 $donationPayment = $compositePayment->getDonations()->first();
536
537 $locale = $this->appLocale();
538
540 $numberFormatter = new NumberFormatter($locale);
541
542 // now compose the dataset for the document template engine
543 $templateData = [
544 'recipient' => EntityArrayAdapter::create($donationPayment->getMusician(), depth: 1),
545 'donation' => [
546 'amount' => $donationAmount,
547 'dateOfReceipt' => $compositePayment->getDateOfReceipt(),
548 'isWaivingOfReimbursement' => !($donationAmount + $nonDonationAmount),
549 'l10n' => [
550 'locale' => $locale,
551 'amount' => $numberFormatter->formatCurrency($donationAmount),
552 'amountText' => $numberFormatter->currencyToWords($donationAmount),
553 'absAmount' => $numberFormatter->formatCurrency($donationAmount),
554 'absAmountText' => $numberFormatter->currencyToWords($donationAmount),
555 ],
556 ],
557 ];
558
559 $paymentData = $this->generatePaymentMailMergeData($compositePayment);
560 foreach ($paymentData['payment']['payments'] as $paymentData) {
561 if ($paymentData['amount'] < 0) {
562 $templateData['donation']['waivedPayment'] = $paymentData;
563 }
564 }
565
566 $templateData = array_merge($paymentData, $templateData);

568 return $templateData;

1481 . strval(ord($iban[1]) - 55)
1482 . substr($iban, 2, 2);
1483

1485 if (ord($iban1[$i]) > 64 && ord($iban1[$i]) < 91) {
1486 $iban1 = substr($iban1, 0, $i) . strval(ord($iban1[$i]) - 55) . substr($iban1, $i+1);
