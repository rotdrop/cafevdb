168 public function targetDeadline(
169 int $businessOffset,
170 ?int $calendarOffset = null,
171 ?\DateTimeInterface $fromDate = null,
172 ):DateTimeInterface {
174 if (empty($fromDate)) {
175 $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
176 } else {
177 $fromDate = (new CarbonImmutable)
178 ->setTimezone($fromDate->getTimezone())
179 ->setTimestamp($fromDate->getTimestamp());
180 }
181
182 if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
183 throw new RuntimeException(
184 $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
185 $businessOffset, $calendarOffset
186 ]));
187 }
188
189 CarbonImmutable::setHolidaysRegion('target2');
190 $businessDeadline = $fromDate->addBusinessDays($businessOffset);
191 if (!empty($calendarOffset)) {
192 $calendarDeadline = $fromDate->addDays($calendarOffset);
193
194 $businessInterval = $fromDate->diff($businessDeadline, true);
195 $calendarInterval = $fromDate->diff($calendarDeadline, true);
196
197 if ($calendarInterval->days > $businessInterval->days) {
198 $businessDeadline = $calendarDeadline->nextBusinessDay();
199 }
200 }
201
202 return $businessDeadline;
203 }

319 public function generateInvoiceNumber(Entities\Musician $originator, null|int|Entities\Project $project, ?DateTimeImmutable $date = null):string
320 {
321 $date = $date ?? (new DateTime);
322 $parts = [
323 $originator->getInitials(),
324 $date->format('Y-m'),
325 ];
326 if ($project !== null) {
328 $project = $this->ensureProject($project);
329 array_unshift($parts, $project->getName());
330 }
331
332 // Simplest thing would be to just use the number of seconds since the
333 // start of the month ... but it is much cooler to do this in a different
334 // way.
335
336 $idPrefix = implode(self::INVOICE_ID_GLUE, $parts);
337 $invoices = $this->getDatabaseRepository(Entities\Invoice::class)->findLike([ 'id' => $idPrefix . self::INVOICE_ID_GLUE . '%', ]);
338 $sequence = array_reduce(
339 $invoices,
340 fn(int $sequence, Entities\Invoice $invoice) => max($sequence, (int)substr($invoice->getId(), strlen($idPrefix))),
341 initial: 1,
342 );
343
344 return $idPrefix . self::INVOICE_ID_GLUE . $sequence;

493 public function generateReceiptOfDonationMailMergeData(
494 Entities\CompositePayment $compositePayment,
495 ):array {
496 // perform sanity checks
497
498 $numberOfDonations = $compositePayment->countDonations();
499 if ($numberOfDonations == 0) {
500 throw new InvalidArgumentException($this->l->t('The composite payment "%s" does not refer to a donation.', $compositePayment));
501 } elseif ($numberOfDonations > 1) {
502 throw new InvalidArgumentException($this->l->t('The composite pamynet "%s" contains more than one donation. This is not yet supported.', $compositePayment));
503 }
504
505 $donationAmount = $compositePayment->getDonationAmount();
506 $nonDonationAmount = $compositePayment->getNonDonationAmount();
507
508 if ($donationAmount <= 0) {
509 throw new InvalidArgumentException(
510 $this->l->t(
511 'The donation in the composite payment "%s" sums up to less or equal zero: %s',
512 [
513 $compositePayment,
514 $this->moneyValue($donationAmount),
515 ],
516 ));
517 }
518 if ($nonDonationAmount > 0) {
519 throw new InvalidArgumentException(
520 $this->l->t(
521 'The non-donation parts of the composite payment "%s" sum up to an income of %s. This is unsupported.',
522 [
523 $compositePayment,
524 $this->moneyValue($nonDonationAmount),
525 ],
526 ));
527 }
528
530 $donationPayment = $compositePayment->getDonations()->first();
531
532 $locale = $this->appLocale();
533
535 $numberFormatter = new NumberFormatter($locale);
536
537 // now compose the dataset for the document template engine
538 $templateData = [
539 'recipient' => $this->flattenMusician($donationPayment->getMusician()),
540 'donation' => [
541 'amount' => $donationAmount,
542 'dateOfReceipt' => $compositePayment->getDateOfReceipt(),
543 'isWaivingOfReimbursement' => !($donationAmount + $nonDonationAmount),
544 'l10n' => [
545 'locale' => $locale,
546 'amount' => $numberFormatter->formatCurrency($donationAmount),
547 'amountText' => $numberFormatter->currencyToWords($donationAmount),
548 'absAmount' => $numberFormatter->formatCurrency($donationAmount),
549 'absAmountText' => $numberFormatter->currencyToWords($donationAmount),
550 ],
551 ],
552 ];
553
554 $paymentData = $this->generatePaymentMailMergeData($compositePayment);
555 foreach ($paymentData['payment']['payments'] as $paymentData) {
556 if ($paymentData['amount'] < 0) {
557 $templateData['donation']['waivedPayment'] = $paymentData;
558 }
559 }
560
561 $templateData = array_merge($paymentData, $templateData);
562
563 return $templateData;
564 }

1480 if (ord($iban1[$i]) > 64 && ord($iban1[$i]) < 91) {
