139 public function targetDeadline(
140 int $businessOffset,
141 ?int $calendarOffset = null,
142 ?\DateTimeInterface $fromDate = null,
143 ):DateTimeInterface {
145 if (empty($fromDate)) {
146 $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
147 } else {
148 $fromDate = (new CarbonImmutable)
149 ->setTimezone($fromDate->getTimezone())
150 ->setTimestamp($fromDate->getTimestamp());
151 }
152
153 if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
154 throw new RuntimeException(
155 $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
156 $businessOffset, $calendarOffset
157 ]));
158 }
159
160 CarbonImmutable::setHolidaysRegion('target2');
161 $businessDeadline = $fromDate->addBusinessDays($businessOffset);
162 if (!empty($calendarOffset)) {
163 $calendarDeadline = $fromDate->addDays($calendarOffset);
164
165 $businessInterval = $fromDate->diff($businessDeadline, true);
166 $calendarInterval = $fromDate->diff($calendarDeadline, true);
167
168 if ($calendarInterval->days > $businessInterval->days) {
169 $businessDeadline = $calendarDeadline->nextBusinessDay();
170 }
171 }
172
173 return $businessDeadline;

291 {

293 $parts = [
294 $originator->getInitials(),
295 $date->format('Y-m'),

297 if ($project !== null) {
299 $project = $this->ensureProject($project);

447 // perform sanity checks
448
449 $numberOfDonations = $compositePayment->countDonations();
450 if ($numberOfDonations == 0) {

452 } elseif ($numberOfDonations > 1) {

466 ],
467 ));
468 }
469 if ($nonDonationAmount > 0) {
470 throw new InvalidArgumentException(
471 $this->l->t(
472 'The non-donation parts of the composite payment "%s" sum up to an income of %s. This is unsupported.',
473 [
474 $compositePayment,
475 $this->moneyValue($nonDonationAmount),
476 ],
477 ));
478 }
479
481 $donationPayment = $compositePayment->getDonations()->first();
482
483 $locale = $this->appLocale();
484
486 $numberFormatter = new NumberFormatter($locale);
487
488 // now compose the dataset for the document template engine

491 'donation' => [
492 'amount' => $donationAmount,
493 'dateOfReceipt' => $compositePayment->getDateOfReceipt(),
494 'isWaivingOfReimbursement' => !($donationAmount + $nonDonationAmount),
495 'l10n' => [

498 'amountText' => $numberFormatter->currencyToWords($donationAmount),

501 ],
502 ],
503 ];

506 foreach ($paymentData['payment']['payments'] as $paymentData) {
507 if ($paymentData['amount'] < 0) {
508 $templateData['donation']['waivedPayment'] = $paymentData;
509 }
