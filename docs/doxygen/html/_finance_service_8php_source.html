163 public function targetDeadline(
164 int $businessOffset,
165 ?int $calendarOffset = null,
166 ?\DateTimeInterface $fromDate = null,
167 ):DateTimeInterface {
169 if (empty($fromDate)) {
170 $fromDate = new CarbonImmutable('now', $this->getDateTimeZone());
171 } else {
172 $fromDate = (new CarbonImmutable)
173 ->setTimezone($fromDate->getTimezone())
174 ->setTimestamp($fromDate->getTimestamp());
175 }
176
177 if (!empty($calendarOffset) && $businessOffset*$calendarOffset < 0) {
178 throw new RuntimeException(
179 $this->l->t('The business-day and calendar-day offset must have the same sign (%d / %d)', [
180 $businessOffset, $calendarOffset
181 ]));
182 }
183
184 CarbonImmutable::setHolidaysRegion('target2');
185 $businessDeadline = $fromDate->addBusinessDays($businessOffset);
186 if (!empty($calendarOffset)) {
187 $calendarDeadline = $fromDate->addDays($calendarOffset);
188
189 $businessInterval = $fromDate->diff($businessDeadline, true);
190 $calendarInterval = $fromDate->diff($calendarDeadline, true);
191
192 if ($calendarInterval->days > $businessInterval->days) {
193 $businessDeadline = $calendarDeadline->nextBusinessDay();
194 }
195 }
196
197 return $businessDeadline;
198 }

314 public function generateInvoiceNumber(Entities\Musician $originator, null|int|Entities\Project $project, ?DateTimeInterface $date = null):string
315 {
316 $date = $date ?? (new DateTime);
317 $parts = [
318 $originator->getInitials(),
319 $date->format('Y-m'),
320 ];
321 if ($project !== null) {
323 $project = $this->ensureProject($project);
324 array_unshift($parts, $project->getName());
325 }
326
327 // Simplest thing would be to just use the number of seconds since the
328 // start of the month ... but it is much cooler to do this in a different
329 // way.
330
331 $idPrefix = implode(self::INVOICE_ID_GLUE, $parts);
332 $invoices = $this->getDatabaseRepository(Entities\Invoice::class)->findLike([ 'id' => $idPrefix . self::INVOICE_ID_GLUE . '%', ]);
333 $sequence = array_reduce(
334 $invoices,
335 fn(int $sequence, Entities\Invoice $invoice) => max($sequence, (int)substr($invoice->getId(), strlen($idPrefix))),
336 initial: 1,
337 );
338
339 return $idPrefix . self::INVOICE_ID_GLUE . $sequence;

488 public function generateReceiptOfDonationMailMergeData(
489 Entities\CompositePayment $compositePayment,
490 ):array {
491 // perform sanity checks
492
493 $numberOfDonations = $compositePayment->countDonations();
494 if ($numberOfDonations == 0) {
495 throw new InvalidArgumentException($this->l->t('The composite payment "%s" does not refer to a donation.', $compositePayment));
496 } elseif ($numberOfDonations > 1) {
497 throw new InvalidArgumentException($this->l->t('The composite pamynet "%s" contains more than one donation. This is not yet supported.', $compositePayment));
498 }
499
500 $donationAmount = $compositePayment->getDonationAmount();
501 $nonDonationAmount = $compositePayment->getNonDonationAmount();
502
503 if ($donationAmount <= 0) {
504 throw new InvalidArgumentException(
505 $this->l->t(
506 'The donation in the composite payment "%s" sums up to less or equal zero: %s',
507 [
508 $compositePayment,
509 $this->moneyValue($donationAmount),
510 ],
511 ));
512 }
513 if ($nonDonationAmount > 0) {
514 throw new InvalidArgumentException(
515 $this->l->t(
516 'The non-donation parts of the composite payment "%s" sum up to an income of %s. This is unsupported.',
517 [
518 $compositePayment,
519 $this->moneyValue($nonDonationAmount),
520 ],
521 ));
522 }
523
525 $donationPayment = $compositePayment->getDonations()->first();
526
527 $locale = $this->appLocale();
528
530 $numberFormatter = new NumberFormatter($locale);
531
532 // now compose the dataset for the document template engine
533 $templateData = [
534 'recipient' => EntityArrayAdapter::create($donationPayment->getMusician(), depth: 1),
535 'donation' => [
536 'amount' => $donationAmount,
537 'dateOfReceipt' => $compositePayment->getDateOfReceipt(),
538 'isWaivingOfReimbursement' => !($donationAmount + $nonDonationAmount),
539 'l10n' => [
540 'locale' => $locale,
541 'amount' => $numberFormatter->formatCurrency($donationAmount),
542 'amountText' => $numberFormatter->currencyToWords($donationAmount),
543 'absAmount' => $numberFormatter->formatCurrency($donationAmount),
544 'absAmountText' => $numberFormatter->currencyToWords($donationAmount),
545 ],
546 ],
547 ];
548
549 $paymentData = $this->generatePaymentMailMergeData($compositePayment);
550 foreach ($paymentData['payment']['payments'] as $paymentData) {
551 if ($paymentData['amount'] < 0) {
552 $templateData['donation']['waivedPayment'] = $paymentData;
553 }
554 }
555
556 $templateData = array_merge($paymentData, $templateData);
557
558 return $templateData;
559 }
