283 public function checkRequirements(?string $dataBaseName):array
284 {
285 $status = self::REQUIREMENTS_OK;
286
287 $hints = [];
288 $userBackendEnabled = $this->appManager->isInstalled(self::CLOUD_USER_BACKEND);
289 $userBackendRestrictions = $this->appManager->getAppRestriction(self::CLOUD_USER_BACKEND);
290 if (!$userBackendEnabled) {
291 $status = self::REQUIREMENTS_MISSING;
292 $hints[] = $this->l->t('In order to be able to import the orchestra club-members as cloud-users the
293"%1$s"-app needs to be enabled. Please ask the administrator of this cloud-instance to install and unconditionally enable this app.', self::CLOUD_USER_BACKEND);
294 } elseif (!empty($userBackendRestrictions)) {
295 $status = self::REQUIREMENTS_MISSING;
296 $hints[] = $this->l->t(
297 'The requird user-backend "%1$s" seems to be installed and enabled,'
298 . ' however, the following app-restriction have been imposed on the app: "%2$s".', [
299 self::CLOUD_USER_BACKEND, implode(', ', $userBackendRestrictions),
300 ]);
301 }
302
303 if (!empty($dataBaseName) && $dataBaseName != $this->appDbName) {
304 $hints[] = $this->l->t(
305 'Please make sure that the user "%1$s@%2$s" has all -- and in particular: GRANT -- privileges on the database "%3$s".', [
306 $this->appDbUser, $this->appDbHost, $dataBaseName
307 ]);
308 }
309
310 return [
311 'status' => $status,
312 'hints' => $hints,

341 public function updateUserSqlViews(?string $dataBaseName):void
342 {
343 $cloudDbUser = $this->checkAndGetCloudDbUser();
344 $currentStatement = null;
345 try {
346 foreach (self::USER_SQL_VIEWS as $name => $sql) {
347 $viewName = $this->viewName($dataBaseName, self::USER_SQL_PREFIX, $name);
348 $statements = [
349 sprintf($sql, $viewName, $this->appName),
350 sprintf(self::GRANT_SELECT, $viewName, $cloudDbUser),
351 ];
352 if ($name === 'User') {
353 // allow changing the password from the cloud
354 $statements[] = sprintf(self::GRANT_FIELD_UPDATE, $viewName, $cloudDbUser, 'password');
355 // allow deactivation of users from the cloud
356 $statements[] = sprintf(self::GRANT_FIELD_UPDATE, $viewName, $cloudDbUser, 'inactive');
357 }
358 foreach ($statements as $sql) {
359 $currentStatement = $sql;
360 $this->logDebug('SQL ' . $currentStatement);
361 if (str_starts_with($sql, 'CREATE')) {
362 try {
363 $this->connection->prepare($sql . ' WITH CHECK OPTION')->execute();
364 } catch (DBALDriverException $e) {
365 if ($e->getCode() != self::CHECK_OPTION_ON_NON_UPDATABLE_VIEW_ERROR) {
366 throw $e;
367 }
368 $this->connection->prepare($sql)->execute();
369 }
370 } else {
371 $this->connection->prepare($sql)->execute();
372 }
373 }
374 }
375 } catch (Throwable $t) {
376 throw new Exceptions\DatabaseCloudConnectorViewException(
377 $this->l->t('Unable to create or update the user-sql cloud-connector views: %s.', $currentStatement),
378 $t->getCode(),
379 $t
380 );
381 }

526 public function configureCloudUserBackend(bool $erase = false):array
527 {
529 $requestService = $this->appContainer->get(RequestService::class);
530
531 $configKeys = $this->cloudConfig->getAppKeys($this->appName);
532 $prefix = self::CLOUD_USER_BACKEND . ':';
533 $prefixLen = strlen($prefix);
534 $cloudUserBackendKeys = array_map(function($key) use ($prefixLen) {
535 return substr($key, $prefixLen);
536 }, array_filter($configKeys, function($key) use ($prefix) {
537 return str_starts_with($key, $prefix);
538 }));
539
540 $this->logDebug('USER SQL KEYS ' . print_r($cloudUserBackendKeys, true));
541
542 $cloudUserBackendParams = [];
543 foreach ($cloudUserBackendKeys as $cloudUserBackendKey) {
544 $cloudUserBackendValue = $erase ? '' : $this->cloudConfig->getAppValue($this->appName, $prefix . $cloudUserBackendKey);
545 if (preg_match('/%system:(\w+)%/', $cloudUserBackendValue, $matches)) {
546 $cloudUserBackendValue = $this->cloudConfig->getSystemValue($matches[1]);
547 }
548 // $this->cloudConfig->setAppValue(self::CLOUD_USER_BACKEND, $cloudUserBackendKey, $cloudUserBackendValue);
549 $cloudUserBackendParams[str_replace('.', '-', $cloudUserBackendKey)] = $cloudUserBackendValue;
550 }
551 // $this->logInfo('USER SQL POST PARAMS ' . print_r($cloudUserBackendParams, true));
552
553 // try also to clear the cache after and before changing the configuration
554 $this->clearUserBackendCache();
555
556 $messages = [];
557
559 $requestService = $this->appContainer->get(RequestService::class);
560
561 $route = implode('.', [
562 self::CLOUD_USER_BACKEND,
563 'settings',
564 'saveProperties',
565 ]);
566 $result = $requestService->postToRoute($route, requestData: $cloudUserBackendParams, postType: RequestService::URL_ENCODED);
567 $messages[] = $result['message'] ?? $this->l->t('"%s" configuration may have succeeded.', self::CLOUD_USER_BACKEND);
568
569 // try also to clear the cache after and before changing the configuration
570 $this->clearUserBackendCache();
571
572 return $messages;
573 }

607 public function setCloudUserSubAdmins(bool $delete = false):void
608 {
610 $configService = $this->appContainer->get(ConfigService::class);
611
612 // finally add all sub-admins of the orchestra group to the catch-all-group of the backend
613 $subAdmins = $configService->getGroupSubAdmins();
614 $catchAllGroup = $configService->getGroup(self::CLOUD_USER_BACKEND); // same name as backend
615 if (!empty($catchAllGroup)) {
616 $subAdminManager = $configService->getSubAdminManager();
617 foreach ($subAdmins as $subAdmin) {
618 $isSubAdmin = $subAdminManager->isSubAdminOfGroup($subAdmin, $catchAllGroup);
619 if ($delete && $isSubAdmin) {
620 $configService->getSubAdminManager()->deleteSubAdmin($subAdmin, $catchAllGroup);
621 } elseif (!($delete || $isSubAdmin)) {
622 $configService->getSubAdminManager()->createSubAdmin($subAdmin, $catchAllGroup);
623 }
624 }

1057 } catch (Throwable $t) {
1058 throw new Exceptions\DatabaseCloudConnectorViewException(
1059 $this->l->t('Unable to create or update the personalized view: %s.', $currentStatement),
