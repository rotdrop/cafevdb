168 protected function generateInstruments(
169 bool $persist = false,
170 array $instrumentNames = ['violin', 'double bass'],
171 int $voices = 2,
172 array $familyNames = ['string', 'strings'],
173 ): void {
174 $l10n = $this->mockProvider->getL10N();
175 if ($persist) {
176 // The real database has the standard instruments already available by
177 // means of a setup migration.
178 // $this->entityManager->enableLogging();
179 $biL10n = $this->appContainer->get(BiDirectionalL10N::class);
180 // be a little bit sloppy here and allow translated as well as untranslated names.
181 $criteriaNames = array_merge(
182 $instrumentNames,
183 array_map(fn(string $name) => $biL10n->t($name), $instrumentNames),
184 );
185 $instruments = $this->entityManager->getRepository(Entities\Instrument::class)->findBy(['name' => $criteriaNames], ['sortOrder' => 'ASC']);
186 $untranslatedNames = array_map(fn($arg) => $arg->getUntranslatedName(), $instruments);
187 $this->assertEquals($instrumentNames, $untranslatedNames);
188 } else {
189 $families = [];
190 $idOffset = 0;
191 foreach ($familyNames as $familyName) {
192 $family = new Entities\InstrumentFamily()
193 ->setId(self::FAKED_ENTITY_ID + $idOffset++)
194 ->setFamily($l10n->t($familyName));
195 if ($persist) {
196 $this->entityManager->persist($family);
197 } else {
198 new ReflectionProperty($family, 'untranslatedFamily')->setValue($family, $familyName);
199 }
200 $families[] = $family;
201 }
202 $instruments = [];
203 $idOffset = 0;
204 foreach ($instrumentNames as $instrumentName) {
205 $instrument = new Entities\Instrument()
206 ->setId(self::FAKED_ENTITY_ID + $idOffset++)
207 ->setName($l10n->t($instrumentName));
208 new ReflectionProperty($instrument, 'untranslatedName')->setValue($instrument, $instrumentName);
209 foreach ($families as $family) {
210 $instrument->addFamily($family);
211 }
212 if ($persist) {
213 $this->entityManager->persist($instrument);
214 }
215 $instruments[] = $instrument;
216 }
217 }
218 $musicianInstruments = [];
220 foreach ($instruments as $instrument) {
221 $musicianInstrument = $this->musician->getInstruments()->get($instrument->getId());
222 if (empty($musicianInstrument)) {
223 $musicianInstrument = new Entities\MusicianInstrument()
224 ->setInstrument($instrument)
225 ->setMusician($this->musician);
226 if ($persist) {
227 $this->entityManager->persist($musicianInstrument);
228 }
229 }
230 $this->assertEquals($instrument, $musicianInstrument->getInstrument());
231 $this->assertEquals($this->musician, $musicianInstrument->getMusician());
232 $musicianInstruments[] = $musicianInstrument;
233 }
234
236 $musicianInstrument = $musicianInstruments[0];
237 $voicedInstrument = $musicianInstrument->getInstrument();
238 $instrumentationNumbers = [];
239 $numbersCollection = $this->project->getInstrumentationNumbers();
240 for ($voice = 0; $voice <= $voices; ++$voice) {
241 $instrumentationNumber = $numbersCollection->findFirst(
242 fn($key, Entities\ProjectInstrumentationNumber $number)
243 =>
244 $number->getVoice() == $voice && $number->getInstrument()->getId() == $voicedInstrument->getId(),
245 );
246 if (empty($instrumentationNumber)) {
247 $instrumentationNumber = new Entities\ProjectInstrumentationNumber(
248 $this->project,
249 $voicedInstrument,
250 $voice,
251 );
252 $this->project->getInstrumentationNumbers()->add($instrumentationNumber);
253 if ($persist) {
254 $this->entityManager->persist($instrumentationNumber);
255 }
256 }
257 $instrumentationNumber->setQuantity($voice);
258 $instrumentationNumbers[] = $instrumentationNumber;
259 }
260
261 $projectInstrument = new Entities\ProjectInstrument()
262 ->setProjectParticipant($this->participant)
263 ->setMusicianInstrument($musicianInstrument);
264 $this->assertEquals($musicianInstrument->getInstrument(), $projectInstrument->getInstrument());
265
266 $instrumentationNumber = $instrumentationNumbers[$voices > 0 ? 1 : 0];
267 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
268 $projectInstrument->getInstrument()->getProjectInstrumentationNumbers()->add($instrumentationNumber);
269 $instrumentationNumber->getProjectInstruments()->add($projectInstrument);
270
271 if ($persist) {
272 $this->entityManager->persist($projectInstrument);
273 }
274
275 if ($persist) {
276 $this->entityManager->beginTransaction();
277 try {
278 $this->entityManager->flush();
279 $this->entityManager->commit();
280 } catch (Throwable $t) {
281 if ($this->entityManager->isTransactionActive()) {
282 $this->entityManager->rollBack();
283 }
284 throw $t;
285 }
286 }
