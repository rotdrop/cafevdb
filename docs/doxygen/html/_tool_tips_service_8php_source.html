163 private function preProcessKey(string $key):string
164 {
165 foreach (self::SUBKEY_PREFIXES as $prefix) {
166 if (str_starts_with($key, $prefix . '-')) {
167 $key = str_replace(['--', '+-'], ['-minus-', '-plus-'], $key);
168 $lastChar = substr($key, -1);
169 switch ($lastChar) {
170 case '-':
171 $key = substr($key, 0, -1) . '-minus';
172 break;
173 case '+':
174 $key = substr($key, 0, -1) . '-plus';
175 break;
176 }
177 return str_replace('-', self::SUB_KEY_SEP, $key);
178 }
179 }
180 return $key;
181 }

192 public function fetch(string $key, bool $escape = true):?string
193 {
194 if ($key == $this->lastKey && $escape) {
195 return $this->lastToolTip;
196 }
197 $this->lastKey = $key;
198
199 $toolTipsData = ToolTipsDataService::get();
200
201 $key = $this->preprocessKey($key);
202
203 $keys = explode(self::SUB_KEY_SEP, $key);
204 while (\count($keys) > 0) {
205 $key = array_shift($keys);
206 $toolTipsData = $toolTipsData[$key] ?? ($toolTipsData['default'] ?? null);
207 }
208 $tip = $toolTipsData['default'] ?? $toolTipsData;
209
210 if (!isset($tip['text'])) {
211 $tip = null;
212 }
213
214 $options = $tip['options'] ?? self::DEFAULT_OPTIONS;
215
216 if (empty($tip)) {
217 $this->failedKeys[] = $this->lastKey;
218 if ($this->debug) {
219 $tip = $this->l->t('Unknown Tooltip for key "%s" requested.', $this->lastKey);
220 }
221 } else {
222 $this->failedKeys = [];
223 $parameters = array_map(
224 fn(mixed $value) => $value instanceof Closure ? $value($this->l, $this->appContainer) : $value,
225 $tip['parameters'] ?? [],
226 );
227 $tip = $this->l->t($tip['text'], $parameters);
228 if ($this->debug) {
229 $tip .= ' (' . $this->l->t('ToolTip-Key "%s"', $this->lastKey) . ')';
230 }
231 }
232
233 if (!empty($tip)) {
234 if ($escape && !$options[self::OPTION_HTML]) {
235 // all tooltips are rendered as HTML, this implies that non-HTML
236 // tips have to be encoded twice.
237 $tip = htmlspecialchars($tip);
238 }
239 // multiple lines consisting only of spaces and linebreaks are replaced by paragraphs
240 $tip = preg_replace('/(^\s*[\n])+/m', self::PARAGRAPH . "\n", $tip);
241 if ($escape) {
242 $tip = htmlspecialchars($tip);
243 }
244 $this->lastToolTip = $tip;
245 } else {
246 $this->lastToolTip = null;
247 }
248
249 return $this->lastToolTip;
250 }
