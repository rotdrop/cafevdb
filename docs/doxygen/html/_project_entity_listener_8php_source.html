103 if (!empty($this->preCommitActions[$project->getId()])) {

109 $this->preCommitActions[$project->getId()] = new GenericUndoable(
110 function() use ($project) {
111 $oldRegistrationEvent = $this->eventsService->findProjectRegistrationEvent($project);
112 if (!empty($oldRegistrationEvent)) {
113 $oldRegistrationEvent = Util::cloneArray($oldRegistrationEvent);
114 }
116 $this->eventsService->ensureProjectRegistrationEvent($project);
117 return $oldRegistrationEvent;
118 },
119 function(?array $oldRegistrationEvent) {
121 $calDavService = $this->appContainer->get(CalDavService::class);
122 if (!empty($oldRegistrationEvent)) {
123 try {
124 $calDavService->restoreCalendarObject(object: $oldRegistrationEvent);
125 } catch (Throwable $t) {
126 $this->logException($t, 'Cannot restore ' . $oldRegistrationEvent['uri'] . '.');
127 try {
128 $calendarId = $oldRegistrationEvent['calendarid'];
129 $localUri = $oldRegistrationEvent['uri'];
130 $data = $oldRegistrationEvent['calendardata'];
131 $calDavService->createCalendarObject($calendarId, $localUri, $data);
132 } catch (Throwable $t) {
133 $this->logException($t, 'Cannot recreate ' . $oldRegistrationEvent['uri'] . '.');
134 }
135 }
136 } else {

138 if (!empty($registrationEvent)) {
139 try {
140 $calDavService->deleteCalendarObject(
