152 ]);
153 }
154
155 // the what part ...
156 $qb = $this->createQueryBuilder('m')
157 ->select(implode(',', $selects))
158 ->leftJoin('m.payments', 'p')
159 ->leftJoin('p.sepaTransaction', 't')
160 ->groupBy('m.mandateReference');
161
162 // the where part ...
163 if (is_string($identifier)) { // assume it is the mandate-reference
164 $qb->where('m.mandateReference = :reference')
165 ->setParameter('reference', $identifier);
166 } elseif (is_array($identifier) || ($identifier instanceof Entities\SepaDebitMandate)) {
167 $qb->where('m.project = :project')
168 ->andWhere('m.musician = :musician')
169 ->andWhere('m.sequence = :sequence')
170 ->setParameter('project', $identifier['project'])
171 ->setParameter('musician', $identifier['musician'])
172 ->setParameter('sequence', $identifier['sequence']);
173 } else {
