153 public function searchFolders(string $regexp, string $topic = self::SEARCH_TOPIC_MOUNT):array
154 {
155 if ($this->sharedFolders === null) {
156 $this->fetchFolders();
157 }
158 if ($regexp[0] != $regexp[-1]) {
159 $regexp = '|^' . $regexp . '$|';
160 }
161 $this->logDebug('REGEXP ' . $regexp . ' / ' . $topic);
162 switch ($topic) {
163 case self::SEARCH_TOPIC_MOUNT:
164 return array_filter($this->sharedFolders, function($folderInfo) use ($regexp) {
165 return preg_match($regexp, $folderInfo['mount_point']);
166 });
167 case self::SEARCH_TOPIC_GROUP:
168 return array_filter($this->sharedFolders, function($folderInfo) use ($regexp) {
169 foreach (array_keys($folderInfo['groups']) as $group) {
170 if (preg_match($regexp, $group)) {
171 return true;
172 }
173 return false;
174 }
175 });
176 default:
177 throw new RuntimeException($this->l->t('Unknown search topic "%1$s"', $topic));
178 }
179 }

327 private function changeFolderManager(string $mountPoint, string $managerId, string $type, bool $canManage):void
328 {
329 // POST BASEURL/groupfolders/folders/8/acl
330 // [ acl => 0 / 1 ]
331 //
332 // POST BASEURL/groupfolders/folders/4/manageACL
333 // [ mappingType => MANAGER_TYPE, mappingId => MANAGER_ID, manageAcl=> 1 ]
334 //
335 $folderInfo = $this->getFolder($mountPoint);
336 if (empty($folderInfo)) {
337 throw new RuntimeException($this->l->t('Shared folder for mount-point "%1$s" does not exist, cannot modify manager "%2$s".', [ $mountPoint, $managerId ]));
338 }
339
340 $this->logDebug('FOLDER INFO ' . print_r($folderInfo, true));
341
342 // first check if anything needs to be done
343 $aclEnabled = !!($folderInfo['acl'] ?? false);
344 $isManager = 0 < count(array_filter($folderInfo['manage'] ?? [], function($manager) use ($managerId, $type) {
345 return $manager['type'] == $type && $manager['id'] == $managerId;
346 }));
347
348 if ($isManager == $canManage && ($canManage || $isManager == $canManage)) {
349 return;
350 }
351
352 $folderId = $folderInfo['id'];
353 if (!$aclEnabled) {
354 $this->folderManager->setFolderACL($folderId, true);
355 }
356
357 if ($isManager != $canManage) {
358 $this->folderManager->setManageACL($folderId, $type, $managerId, $canManage);
359 }
360 if (!$aclEnabled || $isManager != $canManage) {
361 $this->getFolderById($folderId);
362 }
363 }

434 public function changeMountPoint(string $mountPoint, string $targetMountPoint, bool $moveChildren = true):void
435 {
436 if ($mountPoint == $targetMountPoint) {
437 return;
438 }
439 $folderInfo = $this->getFolder($mountPoint);
440 if (empty($folderInfo)) {
441 throw new RuntimeException($this->l->t('Shared folder for mount-point "%1$s" does not exist, cannot change its mount-point to "%2$s".', [ $mountPoint, ]));
442 }
443 // POST BASE_URL/groupfolders/folders/FOLDER_ID/mountpoint
444 // DATA [ mountpoint => NEW_MOUNT_POINT ]
445
446 $this->folderManager->renameFolder($folderInfo['id'], $targetMountPoint);
447 $this->sharedFolders[$targetMountPoint] = $this->sharedFolders[$mountPoint];
448 unset($this->sharedFolders[$mountPoint]);
449 $this->sharedFolders[$targetMountPoint]['mount_point'] = $targetMountPoint;
450
451 if ($moveChildren) {
452 $children = $this->searchFolders('|^' . $mountPoint . '/.*$|');
453 foreach ($children as $childInfo) {
454 $oldChildMount = $childInfo['mount_point'];
455 $newChildMount = $targetMountPoint . substr($oldChildMount, strlen($mountPoint));
456 try {
457 $this->changeMountPoint($oldChildMount, $newChildMount, moveChildren: false);
458 } catch (\Throwable $t) {
459 $this->logException($t, 'Failed to move child mount from ' . $oldChildMount . ' to ' . $newChildMount);
460 }
461 }
462 }
