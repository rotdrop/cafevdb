236 public function tableTabs(?array $participantFields = null, bool $useFinanceTab = false)
237 {
238 $dfltTabs = $this->defaultTableTabs($useFinanceTab);
239
240 if (!is_array($participantFields)) {
241 return $dfltTabs;
242 }
243
244 $extraTabs = array();
245 foreach ($participantFields as $field) {
246 if (empty($field['Tab'])) {
247 continue;
248 }
249
250 $extraTab = $field['Tab'];
251 foreach ($dfltTabs as $tab) {
252 if ($extraTab === $tab['id'] ||
253 $extraTab === (string)$tab['name']) {
254 $extraTab = false;
255 break;
256 }
257 }
258 if ($extraTab !== false) {
259 $extraTabs[] = [
260 'id' => $extraTab,
261 'name' => $this->l->t($extraTab),
262 'tooltip' => $this->toolTipsService['participant-fields-extra-tab'],
263 ];
264 }
265 }
266
267 return array_merge($dfltTabs, $extraTabs);
