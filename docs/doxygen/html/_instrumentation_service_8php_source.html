270 public function tableTabs(?array $participantFields = null, bool $useFinanceTab = false)
271 {
272 $dfltTabs = $this->defaultTableTabs($useFinanceTab);
273
274 if (!is_array($participantFields)) {
275 return $dfltTabs;
276 }
277
278 $extraTabs = array();
279 foreach ($participantFields as $field) {
280 if (empty($field['Tab'])) {
281 continue;
282 }
283
284 $extraTab = $field['Tab'];
285 foreach ($dfltTabs as $tab) {
286 if ($extraTab === $tab['id'] ||
287 $extraTab === (string)$tab['name']) {
288 $extraTab = false;
289 break;
290 }
291 }
292 if ($extraTab !== false) {
293 $extraTabs[] = [
294 'id' => $extraTab,
295 'name' => $this->l->t($extraTab),
296 'tooltip' => $this->toolTipsService['participant-fields-extra-tab'],
297 ];
298 }
299 }
300
301 return array_merge($dfltTabs, $extraTabs);
302 }
