280 public function getFolderArchive($pathOrFolder, int $parentsToStrip = 0, string $format = 'zip'): ?string
281 {
283 if (!($pathOrFolder instanceof Folder)) {
284 $folder = $this->getFolder($pathOrFolder);
285 } else {
286 $folder = $pathOrFolder;
287 }
288
289 if (empty($folder)) {

293 // don't try to strip more components than the initial path-depth
294 $parentsToStrip = min($parentsToStrip, count(Util::explode(self::PATH_SEP, $folder->getPath())));

296 $dataStream = fopen("php://memory", 'w');
297 $zipStream = new ZipStream(outputStream: $dataStream);

353 public function copyTree(mixed $src, mixed $dst, ?string $excludeRegexp = null, int $flags = self::KEEP_DEST): ?Folder
354 {
355 // $this->logInfo('SRC // DST ' . $src . ' || ' . $dst);
356
358 if (!($src instanceof Folder)) {
359 $sourceFolder = $this->getFolder($src);
360 } else {
361 $sourceFolder = $src;
362 }
363
364 if (empty($sourceFolder)) {
365 $this->logInfo('NO SOURCE FOLDER ' . $src . ' || ' . $dst);
366 return null;
367 }
368
369 $sourceRootPath = $sourceFolder->getPath();
370 $sourceRootLen = strlen($sourceRootPath);
371
372 $userFolderPrefix = $this->userFolder->getPath();
373 $userFolderLen = strlen($userFolderPrefix);
374
375 if ($dst instanceof Folder) {
376 $destFolder = $dst;
377 $destRootPath = $destFolder->getPath();
378 $destUserPath = trim(substr($destRootPath, $userFolderLen), self::PATH_SEP);
379 } else {
380 $destUserPath = $dst;
381 try {
382 $destFolder = $this->userFolder->get($destUserPath);
383 $destRootPath = $destFolder->getPath();
384 } catch (FileNotFoundException $e) {
385 $destFolder = null;
386 }
387 }
388 if ($destFolder
389 && (($flags & self::CLEAN_DEST) || $destFolder->getType() != FileInfo::TYPE_FOLDER)) {
390 $destFolder->delete();
391 $destFolder = null;
392 }
393 if (empty($destFolder)) {
394 $destFolder = $this->ensureFolder($destUserPath);
395 $destRootPath = $destFolder->getPath();
396 }

398 $this->folderWalk($sourceFolder, function(
399 Node $sourceNode,
400 int $depth,
401 ) use (
402 $sourceFolder,
403 $sourceRootPath,
404 $sourceRootLen,
405 $destFolder,
406 $destRootPath,
407 $excludeRegexp,
408 $flags,
409 ) {
410 if (!empty($excludeRegexp) && preg_match($excludeRegexp, $sourceNode->getName())) {
411 $this->logDebug('EXCLUDE HIT FOR PATTERN "' . $excludeRegexp . '" given filename "' . $sourceNode->getName() . '".');
412 return false;
413 }
414
415 $sourcePath = $sourceNode->getPath();
416 $relativePath = trim(substr($sourcePath, $sourceRootLen), self::PATH_SEP);
417 $destPath = $destRootPath . self::PATH_SEP . $relativePath;
418 switch ($sourceNode->getType()) {
419 case FileInfo::TYPE_FOLDER:
420 try {
421 $destNode = $destFolder->get($relativePath);
422 if ($destNode->getType() != FileInfo::TYPE_FOLDER && ($flags & self::OVWR_DEST)) {
423 $destNode->delete();
424 throw new FileNotFoundException($this->l->t("Deleted non-directory %s", [$destPath]));
425 }
426 // $this->logInfo('TARGET FOLDER ALREADY EXISTS: ' . $destPath);
427 } catch (FileNotFoundException $e) {
428 $destNode = $destFolder->newFolder($relativePath);
429 }
430 break;
431 case FileInfo::TYPE_FILE:
432 try {
433 $destNode = $destFolder->get($relativePath);
434 if ($flags & self::OVWR_DEST) {
435 $destNode->delete();
436 throw new FileNotFoundException($this->l->t("Overwrite requested, deleted %s", [$destPath]));
437 }
438 } catch (FileNotFoundException $e) {
439 $sourceNode->copy($destPath);
440 }
441 break;
442 }
443 return true;
444 });
445
446 return null;

532 public function restore(string $path, ?array $timeInterval = null, bool $overwriteExisting = false): bool
533 {
535 $trashManager = $this->appContainer->get(ITrashManager::class);
536
537 if (empty($trashManager)) {
538 throw new RuntimeException($this->l->t('Unable to restore "%s": TrashManager cannot be loaded', $path));
539 }
540
541 $path = trim($path, self::PATH_SEP);
542
543 $candidate = [ 'time' => 0, 'trashItem' => null ];
544
546 foreach ($trashManager->listTrashRoot($this->user()) as $trashItem) {
547 $originalLocation = trim($trashItem->getOriginalLocation(), self::PATH_SEP);
548 $deletedTime = $trashItem->getDeletedTime();
549
550 if ($originalLocation == $path) {
551 if (empty($timeInterval)
552 || ($deletedTime >= $timeInterval[0] && $deletedTime <= $timeInterval[1])) {
553 if ($candidate['time'] < $deletedTime) {
554 $candidate['time'] = $deletedTime;
555 $candidate['trashItem'] = $trashItem;
556 }
557 }
558 }
559 }
560
561 if (empty($candidate['trashItem'])) {
562 $this->logInfo('Unable to find trash-bin item to undelete');
563 return false;
564 }
565
566 if ($overwriteExisting) {
567 $this->logInfo('Try delete ' . $path);
568 $this->delete($path);
569 }
570
571 $this->logInfo('Try restore ' . $candidate['trashItem']->getOriginalLocation());
572 $trashManager->restoreItem($candidate['trashItem']);
573
574 return true;
575 }
