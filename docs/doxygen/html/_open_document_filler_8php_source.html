168 private function fillInternal(File $templateFile, string $templateFileName, array $templateData, array $blocks, bool $asPdf)
169 {
170 ob_start();
171
172 $fillData = $this->fillData($templateData);
173
174 $this->backend->ResetVarRef(false);
175 $this->backend->VarRef = $fillData;
176
177 unset($fillData['now']); // the cache is just for this request, so "now" from the first call is good enough
178 unset($fillData['date']); // the cache is just for this request, so "now" from the first call is good enough
179 $templateDataHash = md5(json_encode($fillData, JSON_PARTIAL_OUTPUT_ON_ERROR));
180
181 $this->backend->LoadTemplate($templateFile->fopen('r'), OPENTBS_ALREADY_UTF8);
182
183 // OpenTBS does not support DateTimeInterface or time-zones, so
184 // convert everything to a time-stamp and add the timezone-offset
185 // to get correct dates and times.
186 array_walk_recursive($this->backend->VarRef, function(&$value, $key) {
187 if ($value instanceof DateTimeInterface) {
188 $stamp = $value->getTimestamp();
189 $stamp += $value->getOffset();
190 $this->logDebug('REPLACE DATE BY TIMESTAMP ' . $key . ': ' . print_r($value, true) . ' -> ' . $stamp);
191 $value = $stamp;
192 } elseif ($value === true) {
193 $value = 1;
194 } elseif ($value === false) {
195 $value = 0;
196 } elseif ($value instanceof RationalNumber) {
197 $value = $value->toDecimal();
198 } elseif ($value === null) {
199 $value = '';
200 }
201 });
202
203 $fileHash = $templateFile->getEtag();
204
205 if (empty($this->cache[$fileHash][$templateDataHash])) {
206
207 // do a serialize - unserialize
208 $this->backend->VarRef = json_decode(json_encode($this->backend->VarRef, JSON_PARTIAL_OUTPUT_ON_ERROR), true);
209
210 // Do an opportunistic block-merge for every key with is an array
211 foreach ($this->backend->VarRef as $key => $value) {
212 if (is_array($value)) {
213 // try to do a block-merge primarily to have shorter names in
214 // the tempalte. If the array is not a numeric array, then
215 // wrap it into a single element numeric array in order to
216 // please TBS.
217 $keys = array_keys($value);
218 if ($keys != array_filter($keys, 'is_int')) {
219 $value = [ $value ];
220 }
221 $this->backend->MergeBlock($key, $value);
222 }
223 }
224
225 // Do a block-merge for every explicitly defined block
226 foreach ($blocks as $key => $reference) {
227 $indices = explode('.', $reference);
228 $value = $this->backend->VarRef;
229 while (!empty($indices) && !empty($value)) {
230 $index = array_shift($indices);
231 $value = $value[$index]??null;
232 }
233 if (empty($value)) {
234 // throw new RuntimeException(
235 // $this->l->t(
236 // 'Data for block "%1$s" using the path "%2$s" could not be found in the substitution data.',
237 // [ $key, $reference ],
238 // ),
239 // );
240 $this->logWarn('Data for block "' . $key . '" using the path "' . $reference . '" could not be found in the substitution data.');
241 continue;
242 }
243 $keys = array_keys($value);
244 if ($keys != array_filter($keys, 'is_int')) {
245 $value = [ $value ];
246 }
247 $this->backend->MergeBlock($key, $value);
248 }
249
250 $this->backend->show(OPENTBS_STRING);
251
252 $output = ob_get_contents();
253 ob_end_clean();
254
255 if (!empty($output)) {
256 throw new Exceptions\Exception($output);
257 }
258
259 $fileData = $this->backend->Source;
260 $mimeType = $templateFile->getMimeType();
261
262 $this->cache[$fileHash][$templateDataHash] = [
263 'fileData' => $fileData,
264 'mimeType' => $mimeType,
265 ];
266 } else {
267 list('fileData' => $fileData, 'mimeType' => $mimeType) = $this->cache[$fileHash][$templateDataHash];
268 }
269
270 $fileName = basename($templateFileName);
271
272 if ($asPdf) {
273
274 if (empty($this->cache[$fileHash][$templateDataHash]['pdfData'])) {
275 $fileData = $this->anyToPdf->convertData($fileData);
276 $this->cache[$fileHash][$templateDataHash]['pdfData'] = $fileData;
277 } else {
278 $fileData = $this->cache[$fileHash][$templateDataHash]['pdfData'];
279 }
280
281 $mimeType = 'application/pdf';
282 $pathInfo = pathinfo($fileName);
283 $fileName = $pathInfo['filename'] . '.pdf';
284 }
285
286 return [
287 $fileData,
288 $mimeType,
289 $fileName,
290 ];
291 }

395 $substitutions['org']['taxAuthorities'] = [
396 'exemptionNotices' => [],
397 ];
398
399 $taxNames = TaxType::getL10NValues($this->appL10n());
400 $taxKeys = array_combine(
401 array_keys($taxNames),
402 array_map(fn(string $type) => Util::dashesToCamelCase($type, dashes: ' -_'), array_keys($taxNames)),
403 );
404
406 $repository = $this->getDatabaseRepository(Entities\TaxExemptionNotice::class);
407 foreach (TaxType::values() as $taxType) {
408 $notice = $repository->getLatestValid($taxType);
409 if ($notice === null) {
410 continue;
411 }
412 $noticeData = $notice->jsonSerialize();
413 if ($notice->getWrittenNotice()) {
414 $noticeData['writtenNotice'] = $notice->getWrittenNotice()->getData(Entities\FileData::DATA_FORMAT_URI);
415 }
416 $noticeData['taxes'] = $noticeData['taxationStatutorySources'];
417 unset($noticeData['taxationStatutorySources']);
418 $noticeData['taxTypes'] = [];
419 foreach ($notice->getTaxTypes() as $type) {
420 $type = (string)$type;
421 $noticeData['taxTypes'][$taxKeys[$type]] = $taxNames[$type];
422 }
423 $taxes = [];
424 foreach ($noticeData['taxes'] as $type => &$tax) {
425 $tax['taxType'] = $taxKeys[$type];
426 $tax['name'] = $taxNames[$type];
427 $taxes[$taxKeys[$type]] = $tax;
428 }
429 unset($tax);
430 $noticeData['taxes'] = $taxes;
431 $substitutions['org']['taxAuthorities']['exemptionNotices'][$taxKeys[(string)$taxType]] = $noticeData;
432 }
433
434 return $substitutions;
