130 public function set(string $hash, array $data): void
131 {
132 $urlPath = $data[self::FRONTEND_URL_PATH_KEY];
133 foreach (self::EXCLUDED_KEYS as $key) {
134 unset($data[$key]);
135 }
136 $historyData = $this->sessionRetrieveValue(self::SESSION_HISTORY_KEY);
137 if (($historyData[$hash] ?? null) === null) {
138 $historyData[$hash] = $data;
139 $this->sessionStoreValue(self::SESSION_HISTORY_KEY, $historyData);
140 }
141 if (empty($urlPath) || $urlPath === '/') {
142 $urlPath = null;
143 }
144 $this->sessionStoreValue(self::SESSION_LAST_URL_PATH_KEY, $urlPath);
145 }
