126 public function set(string $hash, array $data):void
127 {
128 $urlPath = $data[self::FRONTEND_URL_PATH_KEY];
129 foreach (self::EXCLUDE_KEYS as $key) {
130 unset($data[$key]);
131 }
132 if (!empty($data)) {
133 $historyData = $this->sessionRetrieveValue(self::SESSION_HISTORY_KEY);
134 if (empty($historyData[$hash])) {
135 $historyData[$hash] = $data;
136 $this->sessionStoreValue(self::SESSION_HISTORY_KEY, $historyData);
137 }
138 }
139 if (empty($urlPath) || $urlPath === '/') {
140 $urlPath = null;
141 }
142 $this->sessionStoreValue(self::SESSION_LAST_URL_PATH_KEY, $urlPath);
143 }
