666 protected function getDocumentTemplatesPath(?string $templateName = null, bool $dirName = false)
667 {
668 $pathComponents = [''];
669 $sharedFolder = $this->getSharedFolderPath();
670 if (empty($sharedFolder)) {
671 return null;
672 }
673 $pathComponents[] = $sharedFolder;
674 $templatesFolder = $this->getConfigValue(ConfigConstants::DOCUMENT_TEMPLATES_FOLDER);
675 if (empty($templatesFolder)) {
676 return null;
677 }
678 $pathComponents[] = $templatesFolder;
679 if (!empty($templateName)) {
680 $subFolder = ConfigConstants::DOCUMENT_TEMPLATES[$templateName]['folder']??null;
681 if (!empty($subFolder)) {
682 $subFolder = $this->getConfigValue($subFolder);
683 if (!empty($subFolder)) {
684 $pathComponents[] = $subFolder;
685 }
686 }
687 $templateFileName = $this->getConfigValue($templateName);
688 if (empty($templateFileName)) {
689 return null;
690 }
691 $pathComponents[] = $dirName ? '' : $templateFileName ;
692 }
693 return implode(Constants::PATH_SEP, $pathComponents);
