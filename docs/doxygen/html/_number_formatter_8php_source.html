107 protected function provideLanguage(string $locale):void
108 {
109 $this->lang = locale_get_primary_language($locale);
110 $this->numberToWords = new NumberToWords();
111 $this->numberTransformer = $this->numberToWords->getNumberTransformer($this->lang);
112 $this->currencyTransformer = $this->numberToWords->getCurrencyTransformer($this->lang);
113
114 $this->numberFormatter = new PhpNumberFormatter($locale, PhpNumberFormatter::DECIMAL);
115 $this->numberSpeller = new PhpNumberFormatter($locale, PhpNumberFormatter::SPELLOUT);
116
117 $this->currencyFormatter = new PhpNumberFormatter($locale, PhpNumberFormatter::CURRENCY);
118 $this->currencyIsoCode = $this->currencyFormatter->getTextAttribute(PhpNumberFormatter::CURRENCY_CODE);
119 $this->currencySymbol = $this->currencyFormatter->getSymbol(PhpNumberFormatter::CURRENCY_SYMBOL);
120
121 // The following may not work for all locales, the idea is to "ask" the
122 // PHP NumberFormatter class about the word of the fractional separator
123 // and the ordering of the spelled out fractional part.
124
125 $nullWord = $this->numberSpeller->format(0);
126 $oneWord = $this->numberSpeller->format(1);
127 $zeroDotOne = explode(' ', $this->numberSpeller->format(0.1));
128 if (count($zeroDotOne) != 3) {
129 throw new UnexpectedValueException('0.1 to be spelled out as a three digit word, but got "' . implode(' ', $zeroDotOne) . '".');
130 }
131 if ($zeroDotOne[0] == $nullWord && $zeroDotOne[2] == $oneWord) {
132 $this->fractionalSeparatorWord = $zeroDotOne[1];
133 $this->fractionalPartLast = true;
134 } elseif ($zeroDotOne[0] == $nullWord && $zeroDotOne[2] == $oneWord) {
135 $this->fractionalSeparatorWord = $zeroDotOne[1];
136 $this->fractionalPartLast = false;
137 } else {
138 throw new UnexpectedValueException('Expected the fractional part to be located either in front or after the decimal separator word.');
139 }
140 $zeroDotZeroOne = explode(' ', $this->numberSpeller->format(0.01));
141 if (count($zeroDotOne) < 3) {
142 throw new UnexpectedValueException('Expected 0.01 to be spelled out with at least three words, but got "' . implode(' ', $zeroDotZeroOne) . '".');
143 }
144 $this->fractionalPartAscending =
145 $zeroDotZeroOne[0] == $this->fractionalSeparatorWord
146 || ($zeroDotZeroOne[0] == $nullWord
147 && $zeroDotZeroOne[1] == $this->fractionalSeparatorWord);
148 }

220 protected function numberToWordsKWN(float $value, int $minDigits = 1, int $maxDigits = 2):string
221 {
222 $minDigits = min($minDigits, $maxDigits);
223 $maxDigits = max($minDigits, $maxDigits);
224 $value = round($value, $maxDigits);
225 $factor = pow(10.0, $maxDigits);
226 $digitsPart = abs((int)($value * $factor) % $factor);
227 $majorPart = (int)$value;
228 $majorPartWords = $this->numberTransformer->toWords($majorPart);
229 $digits = [];
230 while ($maxDigits-- > 0) {
231 $digits[] = $digitsPart % 10;//$this->numberTransformer->toWords($digitsPart % 10);
232 $digitsPart /= 10;
233 }
234 // strip trailing zeros
235 while (true) {
236 $value = reset($digits);
237 if ($value > 0 || count($digits) <= $minDigits) {
238 break;
239 }
240 array_shift($digits);
241 }
242 if ($this->fractionalPartAscending) {
243 $digits = array_pad(array_reverse($digits), $minDigits, 0);
244 } else {
245 $digits = array_reverse(array_pad(array_reverse($digits), $minDigits, 0));
246 }
247 $digits = array_map(fn(int $digit) => $this->numberTransformer->toWords($digit), $digits);
248 if ($this->fractionalPartLast) {
249 $parts = [ $majorPartWords ];
250 if (count($digits) > 0) {
251 $parts[] = $this->fractionalSeparatorWord;
252 $parts = array_merge($parts, $digits);
253 }
254 } else {
255 if (count($digits) > 0) {
256 $parts = $digits;
257 $parts[] = $this->fractionalSeparatorWord;
258 }
259 $parts[] = $majorPartWords;
260 }
261 return $this->kwnSanitizer(implode(' ', $parts));
262 }

275 protected function kwnSanitizer(string $words, ?string $currencyISOCode = null):string
276 {
277 switch ($this->lang) {
278 case 'de':
279 $words = lcfirst($words);
280 if ($currencyISOCode !== null) {
281 $nullCurrency = $this->currencyTransformer->toWords(0, $currencyISOCode);
282 $nullWord = $this->numberTransformer->toWords(0);
283 $currencyWords = trim(str_replace($nullWord, '', $nullCurrency));
284 $parts = [];
285 if (str_starts_with($words, 'minus')) {
286 $parts[] = 'minus';
287 $words = substr($words, strlen('minus'));
288 }
289 $currencyPos = strpos($words, $currencyWords);
290 $number = trim(substr($words, 0, $currencyPos));
291 $rest = trim(substr($words, $currencyPos));
292 if (str_starts_with($number, 'eins')) {
293 $number = 'ein' . substr($number, strlen('eins'));
294 }
295 $number = str_replace(' ', '', $number);
296 if (str_ends_with($rest, 'cent')) {
297 $rest = substr($rest, 0, -strlen('cent')) . 'Cent';
298 }
299 $parts[] = $number;
300 $parts[] = $rest;
301 $words = implode(' ', $parts);
302 }
303 break;
304 default:
305 break;
306 }
307 return $words;
308 }

323 protected function numberToWordsPhp(float $value, int $minDigits = 1, int $maxDigits = 2):string
324 {
325 $minDigits = min($minDigits, $maxDigits);
326 $maxDigits = max($minDigits, $maxDigits);
327 $value = round($value, $maxDigits);
328 $factor = pow(10.0, $maxDigits);
329 $value = ((float)(int)$value) + (((int)($value * $factor) % $factor) / $factor);
330
331 $optionalFactor = pow(10.0, $maxDigits - $minDigits + 1);
332 $fractionalPart = abs((int)($value * $factor) % $factor);
333 $forceDigits = $fractionalPart % $optionalFactor == 0;
334
335 if ($forceDigits) {
336 $excessDigit = pow(10.0, -($minDigits + 1));
337 $value = $value >= 0 ? $value + $excessDigit : $value - $excessDigit;
338 }
339 $words = explode(' ', str_replace("\xc2\xad", '', $this->numberSpeller->format($value)));
340 if ($forceDigits) {
341 $separatorPos = array_search($this->fractionalSeparatorWord, $words);
342 if ($this->fractionalPartLast) {
343 if ($this->fractionalPartAscending) {
344 $words = array_slice($words, 0, $separatorPos + $minDigits + 1);
345 } else {
346 $words = array_merge(
347 array_slice($words, 0, $separatorPos + 1),
348 array_slice($words, - $minDigits),
349 );
350 }
351 } else {
352 if ($this->fractionalPartAscending) {
353 $words = array_merge(
354 array_slice($words, 0, $minDigits),
355 array_slice($words, -$separatorPos - 1),
356 );
357 } else {
358 $words = array_merge(
359 array_slice($words, $separatorPos - $minDigits, $minDigits),
360 array_slice($words, -$separatorPos - 1),
361 );
362 }
363 }
364 }
365 return implode(' ', $words);
366 }
