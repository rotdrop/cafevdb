68 public function onChangeDecision(SluggableAdapter $eventAdapter, array &$config, $object, &$slug, &$needToChangeSlug, $otherSlugs)
69 {
70 if (empty($object->getInvoiceNumber())) {
71 $needToChangeSlug = true;
72 return;
73 }
74 $entityManager = $eventAdapter->getObjectManager();
75 $uow = $entityManager->getUnitOfWork();
76 $changeSet = $eventAdapter->getObjectChangeSet($uow, $object);
77 if (isset($changeSet['invoiceNumber'])) {
78 // manually set to something, just keep it, the user has to reset the
79 // invoice number to empty if the person wants to establish
80 // auto-generation again.
81 $needToChangeSlug = false;
82 return;
83 }
84 foreach (Entities\Invoice::INVOICE_NUMBER_FIELDS as $field) {
85 if (isset($changeSet[$field])) {
86 $needToChangeSlug = true;
87 return;
88 }
89 }

93 public function postSlugBuild(SluggableAdapter $eventAdapter, array &$config, $object, &$slug)
94 {
95 if (!$object instanceof Entities\Invoice) {
96 throw new InvalidArgumentException('This handler can only act on "' . Entities\Invoice::class . '" but this object is an instance of "' . get_class($object) . '".');
97 }
98
100 $slug = $object->generateInvoiceNumber();
101 $slugs = explode(Entities\Invoice::INVOICE_NUMBER_SEPARATOR, $slug);
102
103 $appLocale = $this->sluggable->getAppContainer()->get(Registration::APP_LOCALE);
104
105 // transliterate them separately, again using $config['separator']
106 $slugs = array_map(function($slugPart) use ($appLocale) {
107 // use iconv for transliteration first place
108 return $this->transliterate($slugPart, $appLocale);
109 }, $slugs);
110
111 // implode again using the outer separator
112 $slug = implode(Entities\Invoice::INVOICE_NUMBER_SEPARATOR, $slugs);
113 }
