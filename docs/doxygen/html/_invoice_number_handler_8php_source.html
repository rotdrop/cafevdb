78 $changeSet = $eventAdapter->getObjectChangeSet($uow, $object);
79 if (isset($changeSet['invoiceNumber'])) {
80 // manually set to something, just keep it, the user has to reset the

82 // auto-generation again.
83 $needToChangeSlug = false;
84 return;
85 }
86 foreach (Entities\Invoice::INVOICE_NUMBER_FIELDS as $field) {
87 if (isset($changeSet[$field])) {
88 $needToChangeSlug = true;
