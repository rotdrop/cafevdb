84 public static function create(
85 null|int|float|string|Rational $integralPartOrAny = null,
86 ?int $numerator = null,
87 ?int $denominator = null,
88 ): static {
89 if (!is_int($integralPartOrAny)) {
90 if ($numerator !== null && $denominator !== null) {
91 throw new InvalidArgumentException(
92 'Too many arguments: only 1 is expected: "'
93 . implode('", "', [$integralPartOrAny, $numerator, $denominator])
94 . '".'
95 );
96 }

106 return static::zero();
107 }
108 }

131 public function round(int $precision = 0): static
132 {
133 $rollIn = pow(10, $precision + 1);
134 $roundInc = ($this->getWholePart() + $this->getNumerator() < 0) ? -5 : 5;
135 $integralPart = $this->getWholePart();
136 // this should in principle reduce the chance of overflow ...
137 $fractional = $this->sub($integralPart);
138 return new static($integralPart, intdiv($fractional->mul($rollIn)->whole + $roundInc, 10), $rollIn / 10);

172 public function toDecimal(int $scale = -1, int $precision = 0):string
173 {
174 $sign = $this->sign();
175 $abs = $this->abs();
176 if ($scale === -1) {
177 $result = $abs->whole;
178 $fractionalLimit = static::DECIMAL_DIGITS_MAX - max($abs->numerator > 1 ? strlen($abs->numerator) : 0, ($result == 0 ? 0 : strlen($result)));
179 $abs = $abs->round($fractionalLimit);
180 $abs->subEq($result);
181 $fractionalPart = '';
182 while ($abs->numerator != 0 && strlen($fractionalPart) < $fractionalLimit) {
183 $abs->mulEq(10);
184 $digit = $abs->whole;
185 $abs->subEq($digit);
186 $fractionalPart .= $digit;
187 }
188 $result .= '.' . $fractionalPart;
189 return $sign < 0 ? '-' . $result : $result;
190 }
191 $rollIn = pow(10, $scale + 1);
192 $fixedPoint = str_pad(intdiv($abs->mul($rollIn)->whole + 5, 10), $scale + 1, '0', STR_PAD_LEFT);
193 $integralPart = substr($fixedPoint, 0, -$scale);
194 $fractionalPart = substr($fixedPoint, -$scale);
195 $result = $integralPart . '.' . $fractionalPart;
196 if ($precision > 0 && (strlen(ltrim($integralPart, '0')) + strlen($fractionalPart)) > $precision) {
197 $bound = str_pad('', $precision - $scale, '9') . '.' . str_pad('', $scale, '9');
198 throw new OutOfBoundsException(
199 'The rational number ' . (string)$this . ' (' . $this->toFloat() . ') does not fit into the range [-' . $bound . ', ' . $bound . '].'
200 );
201 }
202 return $sign < 0 ? '-' . $result : $result;
203 }

219 public static function fromDecimal(string|int $decimal): static
220 {
221 if (is_int($decimal)) {
222 return new static($decimal, 0, 1);
223 }
224 $matches = [];
225 if (!preg_match_all('/^(-)?([1-9]\d*|0?)(?:\.(\d*))?$/', $decimal, $matches)) {
226 throw new InvalidArgumentException('Unable to parse input string "' . $decimal . '".');
227 }
228 $sign = $matches[1][0] == '-' ? -1 : 1;
229 $integralPart = empty($matches[2]) ? 0 : (int)$matches[2][0];
230 $numerator = empty($matches[3]) ? 0 : (int)$matches[3][0];
231 $denominator = pow(10, strlen($matches[3][0] ?? ''));
232 return new static($sign * $integralPart, $sign * $numerator, $denominator);
233 }

246 public static function fromFractionString(string $input): static
247 {
248 $originalInput = $input;
249 $input = trim($input);
250 $negate = false;
251 if (str_starts_with($input, '-')) {
252 $negate = true;
253 $input = substr($input, 1);
254 }
255 $isInteger = filter_var($input, FILTER_VALIDATE_INT, ['options' => ['min_range' => 0]]);
256 if (!$isInteger && !str_contains($input, '/')) {
257 throw new InvalidArgumentException("The given string \"{$originalInput}\" does not appear to be a valid fraction.");
258 }
259 // replace super- and subscripts back to ordinary numbers.
260 $from = array_merge(self::SUPER_SCRIPTS, self::SUB_SCRIPTS);
261 $from[] = ' ';
262 $to = array_merge(array_keys(self::SUPER_SCRIPTS), array_keys(self::SUB_SCRIPTS));
263 $to[] = '+';
264 $input = str_replace($from, $to, $input);
265 // now we have at any rate the jsonSerialize() representation.
266 $parts = array_filter(explode('+', $input));
267 $numParts = count($parts);
268 switch ($numParts) {
269 case 2:
270 [$whole, $fraction] = $parts;
271 break;
272 case 1:
273 if (str_contains($parts[0], '/')) {
274 $whole = 0;
275 $fraction = $parts[0];
276 } else {
277 $whole = $parts[0];
278 $fraction = '0/1';
279 }
280 break;
281 default:
282 case 0:
283 throw new InvalidArgumentException("The given string \"{$originalInput}\" does not appear to be a valid fraction.");
284 }
285 [$numerator, $denominator] = explode('/', $fraction);
286 $number = new static((int)$whole, (int)$numerator, (int)$denominator);
287 return $negate ? $number->negate() : $number;
288 }
