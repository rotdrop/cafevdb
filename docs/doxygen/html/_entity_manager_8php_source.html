405 public function connected():bool
406 {
407 try {
408 $connection = $this->getConnection();
409 } catch (Exceptions\DatabaseNotConnectedException $e) {
410 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
411 $this->logException($e);
412 }
413 return false;
414 }
415 $params = $connection->getParams();
416 $impossible = false;
417 foreach ([ 'host', 'user', 'password', 'dbname' ] as $key) {
418 if (empty($params[$key])) {
419 $impossible = true;
420 }
421 }
422 if ($impossible) {
423 $this->logError('Unable to access database, connection parameters are unset');
424 return false;
425 }
426 try {
427 if (!$connection->isConnected()) {
428 if (!$connection->connect()) {
429 $this->logError('db cannot connect');
430 return false;
431 }
432 }
433 } catch (Throwable $t) {
434 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
435 $this->logException($t, 'Caught execption checking connection to database server ' . $params['user'] . '@' . $params['host'] . ':' . $params['dbname']);
436 }
437 return false;
438 }
439 return true;

874 public function persist(mixed $entity):mixed
875 {
876 $meta = $this->getClassMetadata(get_class($entity));
877 if ($meta->containsForeignIdentifier) {
878 $columnValues = $meta->getIdentifierColumnValues($entity);
879
880 foreach ($meta->associationMappings as $property => $association) {
881
882 if (!$meta->isSingleValuedAssociation($property)) {
883 // only to-one makes sense
884 continue;
885 }
886
887 if (array_search($property, $meta->identifier) !== false) {
888 // all primary keys must already be there
889 continue;
890 }
891
892 if (!empty($meta->getFieldValue($entity, $property))) {
