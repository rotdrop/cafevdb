427 public function connected():bool
428 {
429 try {
430 $connection = $this->getConnection();
431 } catch (Exceptions\DatabaseNotConnectedException $e) {
432 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
433 $this->logException($e);
434 }
435 return false;
436 }
437 $params = $connection->getParams();
438 $impossible = false;
439 foreach ([ 'host', 'user', 'password', 'dbname' ] as $key) {
440 if (empty($params[$key])) {
441 $impossible = true;
442 }
443 }
444 if ($impossible) {
445 $this->logError('Unable to access database, connection parameters are unset');
446 return false;
447 }
448 try {
449 if (!$connection->isConnected()) {
450 if (!$connection->connect()) {
451 $this->logError('db cannot connect');
452 return false;
453 }
454 }
455 } catch (Throwable $t) {
456 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
457 $this->logException($t, 'Caught execption checking connection to database server ' . $params['user'] . '@' . $params['host'] . ':' . $params['dbname']);
458 }
459 return false;
460 }
461 return true;
462 }
