415 public function connected():bool
416 {
417 try {
418 $connection = $this->getConnection();
419 } catch (Exceptions\DatabaseNotConnectedException $e) {
420 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
421 $this->logException($e);
422 }
423 return false;
424 }
425 $params = $connection->getParams();
426 $impossible = false;
427 foreach ([ 'host', 'user', 'password', ConfigConstants::APP_DB_NAME ] as $key) {
428 if (empty($params[$key])) {
429 $impossible = true;
430 }
431 }
432 if ($impossible) {
433 $this->logError('Unable to access database, connection parameters are unset');
434 return false;
435 }
436 try {
437 if (!$connection->isConnected()) {
438 if (!$connection->connect()) {
439 $this->logError('db cannot connect');
440 return false;
441 }
442 }
443 } catch (Throwable $t) {
444 if (str_contains($this->request->getPathInfo(), 'apps/' . $this->appName)) {
445 $this->logException($t, 'Caught execption checking connection to database server ' . $params['user'] . '@' . $params['host'] . ':' . $params[ConfigConstants::APP_DB_NAME]);
446 }
447 return false;
448 }
449 return true;

899 public function persist(object $entity):void
900 {
901 $meta = $this->getClassMetadata(get_class($entity));
902 if ($meta->containsForeignIdentifier) {
903 $columnValues = $meta->getIdentifierColumnValues($entity);

906

908 // only to-one makes sense
909 continue;
910 }

913 // all primary keys must already be there
914 continue;
915 }
916
917 if (!empty($meta->getFieldValue($entity, $property))) {
918 // don't override values already present
919 continue;

921

923 $targetEntity = $association['targetEntity'];

934 $targetColumnValues[$targetColumn] = $value;
935 }
936 }

939 if (empty($targetEntityId)) {
940 $reference = null;
941 } else {
942 $reference = $this->getReference($targetEntity, $targetEntityId);
943 }
944
945 $meta->setFieldValue($entity, $property, $reference);
946
947 } catch (Throwable $t) {

949 // $this->logException($t);
