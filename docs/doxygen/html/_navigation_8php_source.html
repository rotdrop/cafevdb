77 public static function selectOptions(array $options, mixed $selectedValues = []):string
78 {
79 if (empty($selectedValues)) {
80 $selectedValues = [];
81 }
82 $result = '';
83 $indent = '';
84 if (!is_array($options) || count($options) == 0) {
85 return $result;
86 }
87 if (!is_array($selectedValues)) {
88 $selectedValues = [ $selectedValues ];
89 }
90 $option = $options[0]; // initialize option groups
91 $oldGroup = isset($option['group']) ? Util::htmlEscape($option['group']) : false;
92 if ($oldGroup) {
93 $groupClass = isset($option['groupClass']) ? ' class="'.$option['groupClass'].'"' : '';
94 $groupId = 0;
95 $groupInfoData = [
96 'id' => $groupId,
97 'default' => false,
98 ];
99 $groupData = " data-group-info='".json_encode($groupInfoData, JSON_FORCE_OBJECT)."'"
100 . ' data-group-id="'.$groupId.'"'
101 . (isset($option['groupData']) ? " data-group='".json_encode($option['groupData'], JSON_FORCE_OBJECT)."'" : '');
102 $result .= '<optgroup label="'.$oldGroup.'"'.$groupClass.$groupData.'>
103 ';
104 $indent = ' ';
105 }
106 foreach ($options as $option) {
107 $value = $option['value'];
108 $flags = isset($option['flags']) ? $option['flags'] : 0;
109 $disabled = $flags & self::DISABLED ? ' disabled' : '';
110 if (($flags & self::SELECTED) || in_array($value, $selectedValues)) {
111 $selected = ' selected="selected"';
112 } else {
113 $selected = '';
114 }
115 $label = isset($option['label']) ? ' label="'.Util::htmlEscape($option['label']).'"' : '';
116 $title = isset($option['title']) ? ' title="'.Util::htmlEscape($option['title']).'"' : '';
117 $group = isset($option['group']) ? Util::htmlEscape($option['group']) : false;
118 $cssClass = isset($option['class']) ? ' class="'.Util::htmlEscape($option['class']).'"' : '';
119 $data = '';
120 if (isset($option['data'])) {
121 $optionData = $option['data'];
122 if (!is_array($optionData)) {
123 $optionData = [ $optionData ];
124 }
125 foreach ($optionData as $key => $dataValue) {
126 $data .= ' data-'.Util::htmlEscape($key)."='";
127 if (is_array($dataValue) || ($dataValue instanceof \JsonSerializable)) {
128 $data .= json_encode($dataValue, JSON_FORCE_OBJECT);
129 } else {
130 // convert to string
131 $data .= Util::htmlEscape((string)$dataValue);
132 }
133 $data .= "'";
134 }
135 }
136 if ($group != $oldGroup) {
137 $result .= '</optgroup>
138 ';
139 $oldGroup = $group;
140 ++$groupId;
141 $indent = '';
142 if ($group) {
143 $groupClass = isset($option['groupClass']) ? ' class="'.$option['groupClass'].'"' : '';
144 $groupInfoData = [
145 'id' => $groupId,
146 'default' => false,
147 ];
148 $groupData = " data-group-info='".json_encode($groupInfoData, JSON_FORCE_OBJECT)."'"
149 . ' data-group-id="'.$groupId.'"'
150 . (isset($option['groupData']) ? " data-group='".json_encode($option['groupData'], JSON_FORCE_OBJECT)."'" : '');
151 $result .= '<optgroup label="'.$group.'"'.$groupClass.$groupData.'>
152 ';
153 $indent = ' ';
154 }
155 }
156 $result .= $indent.'<option value="'.Util::htmlEscape((string)$value).'"'
157 . $cssClass.$disabled.$selected.$label.$title.(isset($groupId) ? ' data-group-id="'.$groupId.'"' : '').$data
158 . '>'.
159 Util::htmlEscape($option['name']).
160 '</option>
161 ';
162 }
163 return $result;
164 }

287 public function prependTableButtons(array $buttons, mixed $misc = false)
288 {
289 $defaultButtonsNoB = [
290 'L' => [
291 'add',
292 '<<', '<',
293 'goto',
294 '>', '>>',
295 'rows_per_page',
296 'reload',
297 ],
298 'F' => [
299 'add',
300 '<<', '<',
301 'goto',
302 '>', '>>',
303 'rows_per_page',
304 'reload',
305 ],
306 'A' => ['save', 'apply', 'more', 'cancel'],
307 'C' => ['save', 'more', 'cancel', 'reload'],
308 'P' => ['save', 'apply', 'cancel'],
309 'D' => ['save', 'cancel', 'reload'],
310 'V' => ['change', 'copy', 'delete', 'cancel', 'reload']
311 ];
312
313 if ($misc === true) {
314 $misc = [ 'misc', 'placeholder' ];
315 foreach ($buttons as &$modButton) {
316 $modButton['name'] = 'placeholder';
317 }
318 } elseif ($misc === false && !empty($buttons)) {
319 $misc = [ 'placeholder' ];
320 foreach ($buttons as &$modButton) {
321 $modButton['name'] = 'placeholder';
322 }
323 } elseif (!is_array($misc)) {
324 $misc = [];
325 }
326
327 if (!isset($misc['up']) && !isset($misc['down'])) {
328 $misc = [ 'up' => $misc, 'down' => $misc ];
329 }
330 $misc = array_merge([ 'up' => [], 'down' => [] ], $misc);
331 foreach ($misc as $vPos => &$miscDef) {
332 $vPos = $vPos;
333 if (!isset($miscDef['left']) && !isset($miscDef['right'])) {
334 $miscDef = [ 'left' => $miscDef, 'right' => [] ];
335 }
336 $miscDef = array_merge([ 'left' => [], 'right' => [] ], $miscDef);
337 }
338
339 $result = [];
340 $cssPositions = [ 'up' => 'top', 'down' => 'bottom' ];
341 foreach (['up', 'down'] as $verticalPosition) {
342 $miscVPos = $misc[$verticalPosition];
343 $defaultButtons = $defaultButtonsNoB;
344 $defaultButtons['L'] = array_merge($miscVPos['left'], $defaultButtons['L'], $miscVPos['right']);
345 $defaultButtons['F'] = array_merge($miscVPos['left'], $defaultButtons['F'], $miscVPos['right']);
346
347 foreach ($defaultButtons as $key => $value) {
348 $positionValue = [];
349 foreach ($value as $oneButton) {
350 $replacement = false;
351 foreach ($buttons as $button) {
352 if ($button['name'] == $oneButton) {
353 $replacement = true;
354 if (isset($button['code'])) { // 'code' is a magic PME thing
355 $button = preg_replace('/\sid="([^"]*)"/', ' id="$1-'.$verticalPosition.'"', $button);
356 $button = preg_replace('/\sfor="([^"]*)"/', ' for="$1-'.$verticalPosition.'"', $button);
357 $button = preg_replace('/class="([^"]*)"/', 'class="$1 '.$cssPositions[$verticalPosition].' '.$verticalPosition.'"', $button);
358 $button = str_replace('{POSITION}', $cssPositions[$verticalPosition], $button);
359 }
360 $positionValue[] = $button;
361 }
362 }
363 if (!$replacement) {
364 $positionValue[] = $oneButton;
365 }
366 }
367 $result[$key][$verticalPosition] = $positionValue;
368 }
369 }
370 return $result;
371 }

393 public function htmlTagsFromArray(array $tags):string
394 {
395 // Global setup, if any
396 $pre = $post = $between = '';
397 if (isset($tags['pre'])) {
398 $pre = $tags['pre'];
399 unset($tags['pre']);
400 }
401 if (isset($tags['post'])) {
402 $post = $tags['post'];
403 unset($tags['post']);
404 }
405 if (isset($tags['between'])) {
406 $between = $tags['between'];
407 unset($tags['between']);
408 }
409
410 // Per element setup
411 $html = $pre;
412 foreach ($tags as $key => $tag) {
413 $type = isset($tag['type']) ? $tag['type'] : 'button';
414 $name = $tag['name'];
415 $value = ' value="'.Util::htmlEscape((isset($tag['value']) ? $tag['value'] : $name)).'"';
416 $title = ' title="'.(isset($tag['title']) ? $tag['title'] : $name).'"';
417 $id = isset($tag['id']) ? ' id="'.$tag['id'].'"' : '';
418 $class = empty($tag['class']) ? '' : ' class="'.$tag['class'].'"';
419 $disabled = (isset($tag['disabled']) && $tag['disabled']) ? ' disabled' : '';
420 $formAction = empty($tag['formaction']) ? '' : ' formaction="'.$tag['formaction'].'"';
421 $data = '';
422 if (isset($tag['data'])) {
423 $dataArray = $tag['data'];
424 if (!is_array($dataArray)) {
425 $dataArray = ['value' => $dataArray];
426 }
427 foreach ($dataArray as $key => $dataValue) {
428 $key = Util::camelCaseToDashes($key);
429 $data .= ' data-'.$key.'="'.Util::htmlEscape($dataValue).'"';
430 }
431 }
432 switch ($type) {
433 case 'resetbutton':
434 case 'submitbutton':
435 case 'button':
436 if ($type == 'resetbutton') {
437 $buttonType = 'reset';
438 } elseif ($type == 'submitbutton') {
439 $buttonType = 'submit';
440 } else {
441 $buttonType = 'button';
442 }
443 $method = isset($tag['method']) ? ' formmethod="'.$tag['method'].'"' : '';
444 $style = isset($tag['style']) ? ' style="'.$tag['style'].'"' : '';
445 $html .= ''
446 .'<button type="'.$buttonType.'" '.$method.$formAction.$disabled.$class.$value.$title.$data.$id.$style.'>';
447 if (isset($tag['image'])) {
448 $images = false;
449 if (!is_array($tag['image'])) {
450 $images = [$tag['image']];
451 } else {
452 $images = $tag['image'];
453 }
454 $count = 0;
455 foreach ($images as $image) {
456 if (isset($tag['id'])) {
457 $id = $tag['id'].'-img';
458 if ($count > 0) {
459 $id .= '-'.$count;
460 }
461 $id = ' id="'.$id.'" ';
462 } else {
463 $id = '';
464 }
465 $class = ' number-'.$count;
466 $html .= ''.
467 '<img class="svg'.$class.'" '.
468 $id.
469 'src="'.$image.'" alt="'.$name.'" '.
470 ' />';
471 ++$count;
472 }
473 } else {
474 $html .= $name;
475 }
476 $html .= '</button>
477 ';
478 break;
479 case 'input':
480 $style = !empty($tag['style']) ? $tag['style'] : '';
481 if (isset($tag['image'])) {
482 $style = 'background:url(\''.$tag['image'].'\') no-repeat center;'.$style;
483 $value = '';
484 }
485 $style = $style ? ' style="'.$style.'"' : '';
486 $name = $name != '' ? ' name="'.Util::htmlEscape($name).'"' : '';
487 $html .= ''
488 .'<input type="button" '.$class.$value.$title.$data.$id.$style.$name.'/>
489 ';
490 break;
491 case 'option':
492 $style = isset($tag['style']) ? ' style="'.$tag['style'].'"' : '';
493 $name = empty($name) ? '' : Util::htmlEscape($name);
494 $selected = '';
495 if (isset($tag['selected']) && $tag['selected'] !== false) {
496 $selected = ' selected="selected"';
497 }
498 $html .= ''
499 .'<option '.$class.$value.$title.$data.$id.$style.$selected.$disabled.'>'.$name.'</option>
500 ';
501 break;
502 default:
503 $html .= '<span>'.$this->l->t('Error: Unknonwn Button Type').'</span>'."\n";
504 break;
505 }
506 $html .= $between;
507 }
508 $html .= $post;
509 return $html;
510 }
