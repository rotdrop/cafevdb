183 public function testAddDocumentConflictRename(): void
184 {
185 $dirEntryFiles = [
186 $this->dirEntryFile,
187 ];
188 // addDocuments() calls flush().
189 $entityManager = $this->getMockBuilder(EntityManager::class)
190 ->disableOriginalConstructor()
191 ->getMock();
192 $instance = new ReflectionProperty($entityManager, 'instance');
193 $instance->setValue($entityManager, $entityManager);
194 $entityManager->expects($this->atLeastOnce())->method('flush')->willReturnCallback(function() {
195 });
196
197 $conflictAction = EnumAddDocumentConflictAction::RENAME;
198 $rounds = 2;
199 while ($rounds-- > 0) {
200 $newFile = new Entities\EncryptedFile(self::FILE_NAME, mimeType: self::FILE_MIMETYPE);
201 $dirEntryFiles[] = $dirEntryFile = $this->folder->addDocument($newFile, conflictAction: $conflictAction);
202 $this->assertEquals($newFile, $dirEntryFile->getFile());
203 $this->assertEquals(count($dirEntryFiles), $this->folder->getDirectoryEntries()->count());
204 foreach ($this->folder->getDirectoryEntries() as $entry) {
205 $dirEntryIndex = array_search($entry, $dirEntryFiles, strict: true);
206 $this->assertNotFalse($dirEntryIndex);
207 if ($dirEntryIndex == count($dirEntryFiles) - 1) {
208 // the most recent has the correct name
209 $this->assertEquals(self::FILE_NAME, $entry->getName());
210 } else {
211 // the other get new names, but unordered, only the conflicting node
212 // is renamed.
213 $base = basename($entry->getName(), self::FILE_EXTENSION);
214 $backupNumber = $dirEntryIndex + 1;
215 $this->assertStringEndsWith('~' . $backupNumber, $base);
216 }
217 }
218 }
