67 {
68 if (self::$appFolderPath ?? null) {

72 // Extract the directory nesting level from the class-name, so this counts
73 // the part after OCA\APP_NAME_SPACE. Thus OCA\APP_NAME_SPACE\AppInfo\Application.php
74 // yields a nesting-level of 2 and yes, the info file is
75 // lib/AppInfo/../../appinfo/info.xml
76 $nestingLevel = count(explode('\\', __CLASS__)) - 2;
77
78 $pathPrefix = str_repeat(Constants::PATH_SEP . '..', $nestingLevel);
79
80 self::$appFolderPath = __DIR__ . $pathPrefix;
81
82 return self::$appFolderPath;
83 }
