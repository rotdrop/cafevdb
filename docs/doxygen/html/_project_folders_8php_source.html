120 protected function execute(InputInterface $input, OutputInterface $output): int
121 {
122 $folder = $input->getOption('folder');
123 $projectName = $input->getOption('project');
124 $years = $input->getOption('years');
125 $all = $input->getOption('all');
126 $skel = $input->getOption('skeleton');
127 $dry = $input->getOption('dry');
128 $check = $input->getOption('check');
129 if ($check) {
130 $dry = true;
131 }
132
133 if (empty($years) && empty($folder) && empty($projectName) && empty($all)) {
134 $output->writeln('<error>' . $this->l->t('One of the options "years=RANGE", "--folder=FOLDER", "--project=PROJECT" or "--all" has to be specified.') . '</error>');
135 $output->writeln('');
136 (new DescriptorHelper)->describe($output, $this);
137 return 1;
138 }
139 if (!empty($all) && (!empty($folder) || !empty($projectName))) {
140 $output->writeln('<error>' . $this->l->t('"--all" cannot be compbined with "years=RANGE", "--folder=FOLDER" or "--project=PROJECT".') . '</error>');
141 $output->writeln('');
142 return 1;
143 }
144 if (!empty($skel) && !in_array($skel, ['omit', 'overwrite', 'clean'])) {
145 $output->writeln('<error>' . $this->l->t('SKELETON must be one of "omit", "overwrite" or "clean".'));
146 $output->writeln('');
147 return 1;
148 }
149 if (!empty($folder) && !in_array($folder, ProjectService::PROJECT_FOLDER_CONFIG_KEYS)) {
150 $output->writeln('<error>' . $this->l->t('FOLDER must be one of "%s"', implode('", "', ProjectService::PROJECT_FOLDER_CONFIG_KEYS)));
151 $output->writeln('');
152 return 1;
153 }
154
155 if (empty($years)) {
156 $firstYear = 1000;
157 $lastYear = 9999;
158 } elseif (preg_match('/([0-9]{4})?\\s*-?\\s*([0-9]{4})?/', $years, $matches)) {
159 $output->writeln('YEARS ' . print_r($matches, true));
160 $firstYear = $matches[1] ?? 1000;
161 $lastYear = $matches[2] ?? 9999;
162 }
163
164 $result = $this->authenticate($input, $output);
165 if ($result != 0) {
166 return $result;
167 }
168
169 $this->disableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
170
172 $projectService = $this->appContainer->get(ProjectService::class);
173
175 $userStorage = $this->appContainer->get(UserStorage::class);
176 $skeletonFolders = $projectService->getProjectSkeletonPaths();
177 foreach ($skeletonFolders as $skeletonFolder) {
178 $userStorage->ensureFolder($skeletonFolder);
179 }
180
182 $projectsRepository = $this->getDatabaseRepository(Entities\Project::class);
183
184 $projectCriteria = [
185 '>=year' => $firstYear,
186 '<=year' => $lastYear,
187 ];
188 if (!empty($projectName)) {
189 $projectCriteria['name'] = $projectName;
190 }
191 $output->writeln('PROJECT CRIT ' . print_r($projectCriteria, true));
192 $projects = $projectsRepository->findBy($projectCriteria, [ 'year' => 'DESC', 'name' => 'ASC' ]);
193
194 $group = $this->appConfig->getValueString($this->appName, ConfigConstants::USER_GROUP_KEY);
195 if ($group) {
196 $groupMembers = array_map(fn(IUSER $user) => $user->getUID(), $this->groupManager->get($group)->getUsers());
197 $sharingService = $this->appContainer->get(SimpleSharingService::class);
198 $shareOwner = $this->appConfig->getValueString($this->appName, ConfigConstants::SHARE_OWNER_KEY);
199 } else {
200 $groupMembers = [];
201 }
202 $output->writeln(print_r($groupMembers, true) . ' ' . $group . ' ' . $shareOwner);
203
204 foreach ($projects as $project) {
205 $folders = $projectService->ensureProjectFolders($project, $folder, $dry);
206 $output->writeln('PROJECT ' . $project->getName() . ': ' . implode(', ', $folders));
207
208 // ensure the music sheet downloads are owned by the correct owner
209 if (!$dry && !empty($groupMembers) && !empty($shareOwner)) {
210 $downloadsPath = $folders[ProjectService::FOLDER_TYPE_DOWNLOADS];
211 $userStorage->folderWalk(
212 $downloadsPath,
213 function(Node $node, int $depth) use ($groupMembers, $sharingService, $shareOwner, $project, $output) {
214 if ($node->getType() == Node::TYPE_FILE) {
215 return;
216 }
217 $shareIds = [];
218 foreach ($groupMembers as $shareOwnerCandidate) {
219 ['files_sharing' => $url, 'share' => $share] = $sharingService->linkShare(
220 $node,
221 shareOwner: $shareOwnerCandidate,
222 newShareOwner: $shareOwner,
223 sharePerms: \OCP\Constants::PERMISSION_READ|\OCP\Constants::PERMISSION_SHARE,
224 expirationDate: false,
225 noCreate: true,
226 );
227 if ($share && !in_array($share->getId(), $shareIds)) {
228 $shareIds[] = $share->getId();
229 $output->writeln('PROJECT ' . $project->getName() . ': download share is ' . $url);
230 }
231 }
232 },
233 );
234 }
235 }
