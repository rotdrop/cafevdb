143 ):IMAPService {
144 foreach (self::CONFIG_KEYS as $topic => $configKey) {
145 $key = 'imap' . ucfirst($topic);
146 $value = ${$topic} ?? null;
147 if ($value !== null) {

173 public function connect(
174 ?Closure $progressCallback = null,
175 ?int $chunkSize = null,
176 ):void {
177 if (!empty($this->client)) {
178 return;
179 }
180 $imapSecurity = $this->imapSecurity;
181 switch ($imapSecurity) {
182 case 'starttls':
183 $imapSecurity = 'tls';
184 break;
185 case 'tls':
186 break;
187 case 'ssl':
188 break;
189 default:
190 $imapSecurity = false;
191 break;
192 }
193
194 // $this->logInfo('IMAP ACCOUNT ' . $imapSecurity . '://' . urlencode($this->imapUser) . ':' . $this->imapPassword[0] . 'XXX' . ':' . $this->imapHost);
195
196 $context = [
197 'ssl' => [
198 'verify_peer' => true,
199 'verify_peer_name' => true,
200 ],
201 ];
202 if ($chunkSize !== null) {
203 $context['chunkSize'] = $chunkSize;
204 }
205 if ($progressCallback !== null) {
206 $context['progressCallback'] = $progressCallback;
207 }
208
209 $params = [
210 'username' => $this->imapUser,
211 'password' => $this->imapPassword,
212 'hostspec' => $this->imapHost,
213 'port' => (int)$this->imapPort,
214 'secure' => $imapSecurity,
215 'timeout' => 20,
216 'context' => $context,
217 ];
218 $this->client = new Horde_Imap_Client_Socket($params);
219 try {
220 $this->client->login();
221 } catch (Horde_Imap_Client_Exception $e) {
222 $this->logError('LOGIN FAILED ' . $imapSecurity . '://' . urlencode($this->imapUser) . ':' . $this->imapPassword[0] . 'XXX' . ':' . $this->imapHost);
223 throw $e;
224 }

247 protected static function decodeSubject(Horde_Imap_Client_Data_Envelope $envelope):string
248 {
249 // Try a soft conversion first (some installations, eg: Alpine linux,
250 // have issues with the '//IGNORE' option)
251 $subject = $envelope->subject;
252 $utf8 = iconv('UTF-8', 'UTF-8', $subject);
253 if ($utf8 !== false) {
254 return $utf8;
255 }
256 return iconv("UTF-8", "UTF-8//IGNORE", $subject);

264 private function fetchMailboxes():array
265 {
266 if (empty($this->mailboxes)) {
267 $this->connect();
268 $mailboxes = $this->client->listMailboxes('*', Horde_Imap_Client::MBOX_ALL, [
269 'delimiter' => true,
270 'attributes' => true,
271 'special_use' => true,
272 ]);
273 $mailboxes = array_filter(
274 $mailboxes,
275 fn(array $folder) => !in_array('\\nexistent', $folder['attributes']),
276 );
277
278 $mboxNames = array_map(fn(array $mailbox) => $mailbox['mailbox']->utf8, $mailboxes);
279 $this->mailboxes = array_combine($mboxNames, $mailboxes);
280
281 foreach ($mailboxes as $folderName => $folder) {
282 $attributes = array_map('strtolower', $folder['attributes']);
283 foreach (self::SPECIAL_USE_ATTRIBUTES as $attribute) {
284 if (in_array(strtolower($attribute), $attributes)) {
285 $this->specialUseMailboxes[$attribute] = $folderName;
286 }
287 }
288 }
289 }
290 return $this->mailboxes;
291 }

378 public function searchHeaders(
379 array $and = [],
380 array $or = [],
381 ?string $mbox = null,
382 bool $fuzzy = false,
383 bool $useFirst = false,
384 ):array|IMAPMessage|null {
385 $mailboxes = $this->fetchMailboxes();
386 if ($mbox !== null) {
387 $mailboxes = array_filter([ $mbox => $mailboxes[$mbox] ?? null ]);
388 }
389 $andQueries = [];
390 foreach ($and as $header => $text) {
391 if (is_numeric($header) && is_array($text)) {
392 // [ [ HEADER1 => VALUE1 ], [ HEADER2 => VALUE2 ], ... ]
393 $header = array_key_first($text);
394 $text = $text[$header];
395 } elseif (is_array($text)) {
396 $values = $text;
397 foreach ($values as $text) {
398 $andQueries[] = $query = new Horde_Imap_Client_Search_Query();
399 $query->headerText($header, $text);
400 }
401 continue;
402 }
403 $andQueries[] = $query = new Horde_Imap_Client_Search_Query();
404 $query->headerText($header, $text);
405 }
406 $orQueries = [];
407 foreach ($or as $header => $text) {
408 if (is_numeric($header) && is_array($text)) {
409 // [ [ HEADER1 => VALUE1 ], [ HEADER2 => VALUE2 ], ... ]
410 $header = array_key_first($text);
411 $text = $text[$header];
412 } elseif (is_array($text)) {
413 $values = $text;
414 foreach ($values as $text) {
415 $orQueries[] = $query = new Horde_Imap_Client_Search_Query();
416 $query->headerText($header, $text);
417 }
418 continue;
419 }
420 $orQueries[] = $query = new Horde_Imap_Client_Search_Query();
421 $query->headerText($header, $text);
422 }
423 if (!empty($andQueries) && count($andQueries) > 1) {
424 $andQuery = $query = new Horde_Imap_Client_Search_Query();
425 $query->andSearch($andQueries);
426 // $this->logInfo('DOING AND QUERIES ' . print_r($query, true));
427 }
428 if (!empty($orQueries) && count($orQueries) > 1) {
429 $orQuery = $query = new Horde_Imap_Client_Search_Query();
430 $query->orSearch($orQueries);
431 // $this->logInfo('DOING OR QUERIES ' . print_r($query, true));
432 }
433 if (!empty($orQueries) && !empty($andQueries)) {
434 $query = new Horde_Imap_Client_Search_Query();
435 $query->andSearch([ $andQuery, $orQuery ]);
436 }
437 $results = [];
438 foreach ($mailboxes as $folderName => $folder) {
439 $mbox = $folder['mailbox'];
440 try {
441 $searchResults = $this->client->search($mbox, $query);
442 } catch (Throwable $t) {
443 // don't care
444 continue;
445 }
446 if ($searchResults['count'] == 0) {
447 continue;
448 }
449 // $this->logInfo('FOUND MATCHES ' . $searchResults['count']);
450 $ids = $searchResults['match'];
451 $query = new Horde_Imap_Client_Fetch_Query();
452 $query->envelope();
453 $query->flags();
454 $query->uid();
455 $query->imapDate();
456 $query->headerText([
457 'cache' => true,
458 'peek' => true,
459 ]);
460 $fetchResults = $this->client->fetch($mbox, $query, [ 'ids' => $ids ]);
461
462 $results = array_merge(
463 $results,
464 array_map(
465 fn(Horde_Imap_Client_Data_Fetch $fetchResult)
466 => (new ImapMessageFetcher(
467 $fetchResult->getUid(),
468 $folderName,
469 $this->client,
470 ))
471 ->withBody(true)
472 ->fetchMessage($fetchResult),
473 iterator_to_array($fetchResults),
474 ),
475 );
476 if (!empty($results) && $useFirst) {
477 return reset($results);
478 }
479 }
480 return $useFirst ? null : $results;
481 }
