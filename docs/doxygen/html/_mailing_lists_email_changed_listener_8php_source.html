92 'added' => [],

99 switch (true) {
100 case $event instanceof Events\PostChangeMusicianEmail:
102 $subscriptionRequest['changed'] = [
103 'old' => $event->getOldEmail()->getAddress(),
104 'new' => $event->getNewEmail()->getAddress(),
105 ];
106 break;
107 case $event instanceof Events\PostRemoveMusicianEmail:
109 $subscriptionRequest['removed'][] = $event->getEntity()->getAddress();
110 break;
111 case $event instanceof Events\PostPersistMusicianEmail:
113 $subscriptionRequest['added'][] = $event->getEntity()->getAddress();
114 break;

166 MailingListsService::MEMBER_DISPLAY_NAME => $musician->getPublicName(firstNameFirst: true),

194 // the old primary address was subscribed.
195 if ($listId == $announcementsListId) {
196 $removedAddresses = [];
197 $addedAddresses = [];
198 if ($oldPrimary != $newPrimary) {
199 $removedAddresses[] = $oldPrimary;

203 // step 2: remove all address which are to be removed
204 foreach ($removedAddresses as $removedAddress) {

206 }
207
208 // step 3: add the old and new primary address to the set of added address to simplify things.
209 if (!empty($newPrimary) && array_search($newPrimary, $addedAddresses) === false) {
210 $addedAddresses[] = $newPrimary;
211 }
212 if (array_search($oldPrimary, $removedAddresses) === false
213 && array_search($oldPrimary, $addedAddresses) === false) {
214 $addedAddresses[] = $oldPrimary;
215 }
216
217 // step 4: add or patch all new addresses with the proper delivery status
218 foreach ($addedAddresses as $addedAddress) {
219 $subscription = $listsService->getSubscription($listId, $addedAddress);
220 if (empty($subscription)) {
221 // generate a new subscription
222 $subscriptionData = $primarySubscription;
223 $subscriptionData[MailingListsService::SUBSCRIBER_EMAIL] = $addedAddress;
224 if ($addedAddress != $newPrimary) {
225 $subscriptionData[MailingListsService::MEMBER_DELIVERY_STATUS] = MailingListsService::DELIVERY_STATUS_DISABLED_BY_USER;
226 }
227 try {
228 $listsService->subscribe($listId, subscriptionData: $subscriptionData);
229 } catch (\Throwable $t) {
230 $this->logException($t, sprintf('Subscribing email "%1$s" to list "%2$s" failed.', $addedAddress, $listId));
231 }
232 } else {
233 // patch the existing subscription
234 $deliveryStatus = ($addedAddress == $newPrimary)

236 : MailingListsService::DELIVERY_STATUS_DISABLED_BY_USER;
237 try {
238 $listsService->setSubscriptionPreferences($listId, $addedAddress, [
239 MailingListsService::MEMBER_DELIVERY_STATUS => $deliveryStatus,
240 ]);
241 } catch (\Throwable $t) {
242 $this->logException($t, sprintf('Setting delivery status of email "%1$s" in list "%2$s" failed.', $addedAddress, $listId));
