136 private function isContainerDirectory(string $path):?int
137 {
138 switch ($this->project->getType()) {
139 case ProjectType::TEMPLATE:
140 break;
141 case ProjectType::PERMANENT:
142 // Folder have a year prefix and the supporting document has the year in its name
143 if (preg_match('|^/?' . '\d{4}' . '/' . $this->project->getName() . '-' . '\d{4}' . '-' . '(\d{3})/?$|', $path, $matches)) {
144 return $matches[1];
145 }
146 break;
147 case ProjectType::TEMPORARY:
148 // project-name contains already the year
149 if (preg_match('|^/?' . $this->project->getName() . '-' . '(\d{3})/?$|', $path, $matches)) {
150 return $matches[1];
151 }
152 break;
153 }
154 return null;

434 public function unlink(string $path): bool
435 {
436 if ($this->is_dir($path)) {
437 return false;
438 }
440 $dirEntry = $this->fileFromFileName($path);
441 if (empty($dirEntry)) {
442 throw new Exceptions\DatabaseStorageException(
443 $this->l->t('Unable to find database entity for path "%s".', $path)
444 );
445 }
446 if ($dirEntry instanceof Entities\DatabaseStorageFolder) {
447 throw new Exceptions\DatabaseStorageException(
448 $this->l->t('Path "%s" is a directory.', $path)
449 );
450 }
451
452 $parent = $dirEntry->getParent();
453 if (empty($parent)) {
454 throw new Exceptions\DatabaseStorageException(
455 $this->l->t('Unable to find document container for path "%s".', $path)
456 );
457 }
458 $file = $dirEntry->getFile();
459 if (empty($file)) {
460 throw new Exceptions\DatabaseStorageException(
461 $this->l->t('The directory entry "%s" is not linked to a file.', $path)
462 );
463 }
464
465 $this->entityManager->beginTransaction();
466 try {
467 // break the link to the supporting documents of payments
468 // @todo Maybe better use a find-by instead of bloating the entities with too many associations.
469
471 $repository = $this->entityManager->getRepository(Entities\CompositePayment::class);
472 $compositePayments = $repository->findBy([
473 'balanceDocumentsFolder' => $parent,
474 'supportingDocument' => $dirEntry,
475 ]);
477 foreach ($compositePayments as $compositePayment) {
478 $compositePayment->setBalanceDocumentsFolder(null);
479 }
480
482 $repository = $this->entityManager->getRepository(Entities\ProjectPayment::class);
483 $projectPayments = $repository->findBy([
484 'balanceDocumentsFolder' => $parent,
485 ]);
487 foreach ($projectPayments as $projectPayment) {
488 $supportingDocument = $projectPayment->getReceivable() ? $projectPayment->getReceivable()->getSupportingDocument() : null;
489 if ($supportingDocument == $dirEntry) {
490 $projectPayment->setBalanceDocumentsFolder(null);
491 }
492 }
493
494 $dirEntry->setParent(null);
495 $dirEntry->setFile(null);
496 $this->entityManager->remove($dirEntry);
497
498 $this->flush();
499
500 $this->entityManager->commit();
501
502 $this->unsetFileNameCache($path);
503 } catch (\Throwable $t) {
504 $this->logException($t);
505 if ($this->entityManager->isTransactionActive()) {
506 $this->entityManager->rollback();
507 }
508 return false;
509 }

511 return true;
