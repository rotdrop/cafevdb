123 private function request(string $command, array $parameters, int $type = self::JSON):mixed
124 {
125 if (isset($parameters['postalCode']) && !isset($parameters['postalcode'])) {
126 $parameters['postalcode'] = $parameters['postalCode'];
127 unset($parameters['postalCode']);
128 }
129 if (isset($parameters['country']) && is_array($parameters['country'])) {
130 $countries = $parameters['country'];
131 unset($parameters['country']);
132 $query = http_build_query($parameters, '', '&');
133 foreach ($countries as $country) {
134 $query .= '&'.http_build_query(array('country' => $country));
135 }
136 } else {
137 $query = http_build_query($parameters, '', '&');
138 }
139 if ($type === self::JSON || $type === self::DRY) {
140 $url = self::PROVIDER_URL . '/' .$command . 'JSON' . '?username=' . $this->userName . '&' . $query;
141 if ($type === self::DRY) {
142 return $url;
143 } else {
144 $this->debug($url);
145 }
146 // $start = hrtime(true);
147 $ctx = stream_context_create([
148 'http'=> [
149 'timeout' => 240,
150 ]
151 ]);
152 $response = file_get_contents($url, false, $ctx);
153 // $this->logInfo('DURATION ' . (float)(hrtime(true) - $start) / 1e9);
154 if ($response !== false) {
155 $json = json_decode($response, true, 512, JSON_BIGINT_AS_STRING);
156 $this->debug(print_r($json, true));
157 return $json;
158 }
159 }
160 return null;

194 public function autoCompleteStreet(
195 ?string $country = null,
196 ?string $city = null,
197 ?string $postalCode = null,
198 ):array {
199 if (empty($country) && empty($city) && empty($postalCode)) {
200 return [];
201 }
202
203 $countryArea =
204 $postalCodeArea =
205 $cityArea = '';
206
207 $oql = '[out:csv("name";false)];
208';
209 if (!empty($country)) {
210 $oql .= sprintf('area["ISO3166-1"="%s"]->.country;
211', $country);

215 $oql .= sprintf('area[postal_code="%s"]->.postalCode;

218 } elseif (!empty($city)) {

220 $oql .= sprintf('area[name~"%s"]->.city;
221', $city);
222 $cityArea = '(area.city)';
223 }
224 $oql .= sprintf('way%1$s%2$s%3$s[highway][name%4$s];

226 $oql .= 'for (t["name"])
227(
228 make x name=_.val;
229 out;
230);
231';
232
233 $queryUrl = self::OVERPASS_URL . '?' . \http_build_query([ 'data' => $oql ]);
234 $this->debug('OQL ' . print_r(preg_split('/\r\n|\r|\n/', $oql), true));
235
236 $curlHandle = curl_init();
237 curl_setopt($curlHandle, CURLOPT_URL, $queryUrl);
238 curl_setopt($curlHandle, CURLOPT_RETURNTRANSFER, 1);
239 $response = curl_exec($curlHandle);
240 curl_close($curlHandle);
241

244 return $this->autoCompleteStreet($country, $city);

247 } else {
248 // overpass may return an HTML page with errors
250 $mimeTypeDetector = $this->di(\OCP\Files\IMimeTypeDetector::class);
251 $mimeType = $mimeTypeDetector->detectString($response);
252 if ($mimeType != 'text/plain') {
253 // assume it is an error
254 return [];
255 }
256 }
257 $results = array_filter(preg_split('/\r\n|\r|\n/', $response));
258 $this->debug(print_r($results, true));

629 private function stampPostalCode(string $postalCode, string $country):void
630 {
631 // Remember we tried ... note that due to typos
632 // (e.g. comma w/o following space) we may actually fail
633 // to update some of those beasts. We simply pretend to
634 // update all, e.g. for the timestamp we give a damn on
635 // the name.
636
637 $expr = $this->expr();
638 $qb = $this->queryBuilder()
639 ->update(GeoPostalCode::class, 'gpc')
640 ->set('gpc.updated', "'".(new DateTimeImmutable)->format('Y-m-d H:i:s.u')."'")
641 ->where(
642 $expr->andX(
643 $expr->eq('gpc.country', ':country'),
644 $expr->eq('gpc.postalCode', ':postalCode')
645 ))
646 ->setParameter('country', $country)
647 ->setParameter('postalCode', $postalCode);
648 $this->debug($qb->getDql());
649 $qb->getQuery()
650 ->execute();
651 $this->flush();
652 $this->debug('Stamped postal code ' . $postalCode . '@' . $country);
653 }

996 {
