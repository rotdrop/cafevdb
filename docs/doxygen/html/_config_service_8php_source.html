987 public function encryptConfigValues(array $override = []):void
988 {
989 $this->encryptionCache = array_merge($this->encryptionCache, $override);
990 $appKeys = $this->getAppKeys();
991 $cacheKeys = array_keys($this->encryptionCache);
992 foreach (array_diff($appKeys, $cacheKeys) as $uncached) {
993 if (preg_match('/::[0-9]+$/', $uncached)) {
994 // skip backup keys
995 continue;
996 }
997 $this->logWarn("Found un-cached configuration key $uncached");
998 $this->getConfigValue($uncached, ignoreLock: true);
999 }
1000 foreach (array_diff($cacheKeys, $appKeys) as $unstored) {
1001 $this->logWarn("Found un-persisted configuration key $unstored");
1002 }
1003 $cacheKeys = array_keys($this->encryptionCache);
1004 //$this->logInfo('keys: '.print_r($cacheKeys, true));
1005 foreach ($cacheKeys as $key) {
1006 $this->setConfigValue($key, $this->encryptionCache[$key], ignoreLock: true);
1007 }
1008 }
