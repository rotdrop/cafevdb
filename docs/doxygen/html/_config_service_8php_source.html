708 public function encryptConfigValues(array $override = []):void
709 {
710 $this->encryptionCache = array_merge($this->encryptionCache, $override);
711 $appKeys = $this->getAppKeys();
712 $cacheKeys = array_keys($this->encryptionCache);
713 foreach (array_diff($appKeys, $cacheKeys) as $uncached) {
714 if (preg_match('/::[0-9]+$/', $uncached)) {
715 // skip backup keys
716 continue;
717 }
718 $this->logWarn("Found un-cached configuration key $uncached");
719 $this->getConfigValue($uncached, ignoreLock: true);
720 }
721 foreach (array_diff($cacheKeys, $appKeys) as $unstored) {
722 $this->logWarn("Found un-persisted configuration key $unstored");
723 }
724 $cacheKeys = array_keys($this->encryptionCache);
725 //$this->logInfo('keys: '.print_r($cacheKeys, true));
726 foreach ($cacheKeys as $key) {
727 $this->setConfigValue($key, $this->encryptionCache[$key], ignoreLock: true);
728 }
