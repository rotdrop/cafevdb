635 public function encryptConfigValues(array $override = []):void
636 {
637 $this->encryptionCache = array_merge($this->encryptionCache, $override);
638 $appKeys = $this->getAppKeys();
639 $cacheKeys = array_keys($this->encryptionCache);
640 foreach (array_diff($appKeys, $cacheKeys) as $uncached) {
641 if (preg_match('/::[0-9]+$/', $uncached)) {
642 // skip backup keys
643 continue;
644 }
645 $this->logWarn("Found un-cached configuration key $uncached");
646 $this->getConfigValue($uncached, ignoreLock: true);
647 }
648 foreach (array_diff($cacheKeys, $appKeys) as $unstored) {
649 $this->logWarn("Found un-persisted configuration key $unstored");
650 }
651 $cacheKeys = array_keys($this->encryptionCache);
652 //$this->logInfo('keys: '.print_r($cacheKeys, true));
653 foreach ($cacheKeys as $key) {
654 $this->setConfigValue($key, $this->encryptionCache[$key], ignoreLock: true);
655 }

754 public function getLocale(?string $lang = null, bool $addEncoding = false):string
755 {
756 if (!empty($this->localeLanguageCache[$lang . $addEncoding])) {
757 return $this->localeLanguageCache[$lang . $addEncoding];
758 }
759
760 if (empty($lang)) {
761 $locale = $this->appContainer->get(Registration::USER_LOCALE);
762 $this->logDebug('Locale seems to be ' . $locale);
763 $this->logDebug('Language seems to be ' . $lang);
764 } else {
765 $primary = locale_get_primary_language($lang);
766 $region = locale_get_region($lang);
767 if (empty($region)) {
768 $locale = $primary . '_' . strtoupper($primary);
769 } else {
770 $locale = $lang;
771 }
772 }
773
774 if (strpos($locale, '.') === false) {
775 if ($addEncoding) {
776 $locale .= '.UTF-8';
777 }
778 } else {
779 if (!$addEncoding) {
780 $locale = locale_get_primary_language($locale) . '_' . locale_get_region($locale);
781 }
782 }
783
784 $this->localeLanguageCache[$lang . $addEncoding] = $locale;
785
786 $this->logDebug('Generated locale string: ' . $locale);
787
788 return $locale;
789 }
