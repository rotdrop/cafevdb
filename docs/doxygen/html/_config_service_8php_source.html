646 public function encryptConfigValues(array $override = []):void
647 {
648 $this->encryptionCache = array_merge($this->encryptionCache, $override);
649 $appKeys = $this->getAppKeys();
650 $cacheKeys = array_keys($this->encryptionCache);
651 foreach (array_diff($appKeys, $cacheKeys) as $uncached) {
652 if (preg_match('/::[0-9]+$/', $uncached)) {
653 // skip backup keys
654 continue;
655 }
656 $this->logWarn("Found un-cached configuration key $uncached");
657 $this->getConfigValue($uncached, ignoreLock: true);
658 }
659 foreach (array_diff($cacheKeys, $appKeys) as $unstored) {
660 $this->logWarn("Found un-persisted configuration key $unstored");
661 }
662 $cacheKeys = array_keys($this->encryptionCache);
663 //$this->logInfo('keys: '.print_r($cacheKeys, true));
664 foreach ($cacheKeys as $key) {
665 $this->setConfigValue($key, $this->encryptionCache[$key], ignoreLock: true);
666 }

765 public function getLocale(?string $lang = null, bool $addEncoding = false):string
766 {
767 if (!empty($this->localeLanguageCache[$lang . $addEncoding])) {
768 return $this->localeLanguageCache[$lang . $addEncoding];
769 }
770
771 if (empty($lang)) {
772 $locale = $this->appContainer->get(Registration::USER_LOCALE);
773 $this->logDebug('Locale seems to be ' . $locale);
774 $this->logDebug('Language seems to be ' . $lang);
775 } else {
776 $primary = locale_get_primary_language($lang);
777 $region = locale_get_region($lang);
778 if (empty($region)) {
779 $locale = $primary . '_' . strtoupper($primary);
780 } else {
781 $locale = $lang;
782 }
783 }
784
785 if (strpos($locale, '.') === false && $addEncoding) {
786 $locale .= '.UTF-8';
787 } elseif (!$addEncoding) {
788 $locale = locale_get_primary_language($locale) . '_' . locale_get_region($locale);
789 }
790
791 $this->localeLanguageCache[$lang . $addEncoding] = $locale;
792
793 $this->logDebug('Generated locale string: ' . $locale);
794
795 return $locale;
796 }
