255 private function generatePurpose(Entities\CompositePayment $payment): array
256 {
257 $subject = $payment->getSubject();
258 if (strlen($subject) > FinanceService::SEPA_PURPOSE_LENGTH) {
259 $autoSubject = $payment->generateSubject(transliterate: fn(string $x) => $this->sepaTransliterator->translit($x));
260 if ($autoSubject == $this->sepaTransliterator->translit($subject)) {
261 $subject = $payment->generateSubject(
262 maxLength: FinanceService::SEPA_PURPOSE_LENGTH,
263 transliterate: fn(string $x) => $this->sepaTransliterator->translit($x),
264 );
265 } else {
266 $subject = $this->camelCaseString->shorten(
267 $subject,
268 limit: FinanceService::SEPA_PURPOSE_LENGTH,
269 minLen: 4,
270 );
271 }
272 }
273
274 $purpose = [$subject, '', '', ''];
275 return $purpose;
276 }
