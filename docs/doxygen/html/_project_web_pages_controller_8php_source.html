70 ?int $projectId = null,
71 ?int $articleId = null,
72 array $articleData = [],
73 ):Http\Response {
74 if ($topic != 'ping' && $projectId <= 0) {
75 return self::grumble($this->l->t('Invalid or unset project-id: "%s".', [ $projectId ]));
76 }
77
78 if (count($articleData) > 0 &&
79 $articleId >= 0 &&
80 $articleData[EnumProjectWebPageParam::ARTICLE_ID->value] != $articleId) {
81 return self::grumble(
82 $this->l->t(
83 'Submitted article id "%1$d" does not match the id "%2$d" stored in the article-data.',
84 [ $articleId, $articleData[EnumProjectWebPageParam::ARTICLE_ID->value] ]));
85 }
86
87 $topic = EnumProjectWebPagesAction::get($topic);
88
89 if ($topic != EnumProjectWebPagesAction::ADD && $topic != EnumProjectWebPagesAction::PING) {
90 // require both, articleId and articleData
91 if ($articleId < 0) {
92 return self::grumble($this->l->t('Invalid or unset article-id: "%s".', [ $articleId ]));
93 }
94 if (count($articleData) == 0) {
95 return self::grumble(
96 $this->l->t(
97 'Cannot perform action "%1$s" with article with id "%2$d", project with id "%3$d": '.
98 'missing article-data.',
99 [ $topic, $articleId, $projectId ]));
100 }
101 }
102
103 switch ($topic) {
104 case EnumProjectWebPagesAction::PING:
105 if ($this->projectService->pingWebPages() === false) {
106 return self::grumble($this->l->t('Unable to ping project web-pages CMS'));
107 } else {
108 return self::response('OK');
109 }
110 break;
111 case EnumProjectWebPagesAction::ADD:
112 try {
113 // This simply means: create a new page for the project.
114 $article = $this->projectService->createProjectWebPage($projectId);
115 } catch (Throwable $t) {
116 throw new Exceptions\EnduserNotificationException(
117 $this->l->t('Unable to create new project web pages for the project with id "%s".', $projectId),
118 previous: $t,
119 httpStatusCode: Http::STATUS_INTERNAL_SERVER_ERROR,
120 );
121 }
122 $message = $this->l->t(
123 'Created a new public web page with name "%1$s" and id "%2$d" for the project with id "%3$d".',
124 [ $article['articleName'], $article[EnumProjectWebPageParam::ARTICLE_ID->value], $projectId ],
125 );
126 // If there is no rehearsal page attached to the project, then attach one
127 $rehearsalsCat = $this->getConfigValue('redaxoRehearsals');
128 $rehearsal = null;
129 try {
130 $articles = $this->projectService->fetchProjectWebPages($projectId);
131 foreach ($articles as $article) {
132 if ($article['categoryId'] == $rehearsalsCat) {
133 $rehearsal = $article;
134 break;
135 }
136 }
137 } catch (Throwable $t) {
138 // ignore
139 }
140 if ($rehearsal === null) {
141 // create one, but ignore anypotential error
142 try {
143 $this->projectService->createProjectWebPage($projectId, 'rehearsals');
144 $message .= ' '.$this->l->t('Created additionally a new rehearsal web page.');
145 } catch (Throwable $t) {
146 $message .= ' '.$this->l->t('Failed to create additionally a new rehearsal web-page.');
147 }
148 }
149 return self::response($message);
150 case EnumProjectWebPagesAction::LINK:
151 try {
152 $this->projectService->attachProjectWebPage($projectId, $articleData);
153 } catch (Throwable $t) {
154 throw new Exceptions\EnduserNotificationException(
155 $this->l->t(
156 'Unable to link the existing public web-article "%1$s" (id "%2$d") to the project with id "%3$d".',
157 [ $articleData['articleName'], $articleId, $projectId ],
158 ),
159 previous: $t,
160 httpStatusCode: Http::STATUS_INTERNAL_SERVER_ERROR,
161 );
162 }
163 return self::response(
164 $this->l->t(
165 'Linked the existing public web-article "%1$s" (id "%2$d") to the project with id "%3$d".',
166 [ $articleData['articleName'], $articleId, $projectId ]));
167 case EnumProjectWebPagesAction::UNLINK:
168 try {
169 $this->projectService->detachProjectWebPage($projectId, $articleId);
170 } catch (Throwable $t) {
171 throw new Exceptions\EnduserNotificationException(
172 $this->l->t(
173 'Unable to detach the public web-article "%s" (id "%d") from the project with id "%d".',
174 [ $articleData['articleName'], $articleId, $projectId ],
175 ),
176 previous: $t,
177 httpStatusCode: Http::STATUS_INTERNAL_SERVER_ERROR,
178 );
179 }
180 return self::response(
181 $this->l->t(
182 'Detached the public web-article "%1$s" (id "%2$d") from the project with id "%3$d".',
183 [ $articleData['articleName'], $articleId, $projectId ],
184 ));
185 case EnumProjectWebPagesAction::DELETE:
186 try {
187 $this->projectService->deleteProjectWebPage($projectId, $articleData);
188 } catch (Throwable $t) {
189 throw new Exceptions\EnduserNotificationException(
190 $this->l->t(
191 'Unable to remove the public web page "%1$s" (id "%2$d") from the project with id "%3$d".',
192 [ $articleData['articleName'], $articleId, $projectId ],
193 ),
194 previous: $t,
195 httpStatusCode: Http::STATUS_INTERNAL_SERVER_ERROR,
196 );
197 }
198 return self::response(
199 $this->l->t(
200 'Removed the public web page "%1$s" (id "%2$d") from the project with id "%3$d".',
201 [ $articleData['articleName'], $articleId, $projectId ],
202 ));
203 break;
204 }
205
206 return self::grumble($this->l->t('Unknown Request'));
