79 public function recordUntranslated(string $phrase, string $locale, string $file, int $line):void
80 {
81 if (array_search($locale, $this->availableLanguages) === false) {
82 return;
83 }
84 $this->entityManager->suspendLogging();
85 $this->setDataBaseRepository(TranslationKey::class);
86 $translationKey = $this->findOneBy([ 'phraseHash' => md5($phrase) ]);
87 if (empty($translationKey)) {
88 $translationKey = TranslationKey::create()->setPhrase($phrase);
89 try {
90 $this->persist($translationKey);
91 $this->flush();
92 } catch (Throwable $t) {
93 $this->logException($t);
94 }
95 $this->logDebug('Translation key for "'.$phrase.'" was empty, new id '.$translationKey->getId());
96 } else {
97 $this->logDebug('Existing translation key for "'.$phrase.'" has id '.$translationKey->getId());
98 }
99
100 $this->setDataBaseRepository(TranslationLocation::class);
101 $location = $this->findOneBy([
102 'translationKey' => $translationKey,
103 'file' => $file,
104 'line' => $line,
105 ]);
106 if (empty($location)) {
107 $this->logDebug('Empty location for key '.$translationKey->getId());
108 $location = TranslationLocation::create()
109 ->setTranslationKey($translationKey)
110 ->setFile($file)
111 ->setLine($line);
112 try {
113 $this->persist($location);
114 $this->flush();
115 } catch (Throwable $t) {
116 $this->logException($t);
117 }
118 }
119 $this->setDatabaseRepository(MissingTranslation::class);
120 $missingLocale = $this->findOneBy([
121 'translationKey' => $translationKey,
122 'locale' => $locale,
123 ]);
124 if (empty($missingLocale)) {
125 $missingLocale = MissingTranslation::create()
126 ->setTranslationKey($translationKey)
127 ->setLocale($locale);
128 try {
129 $this->persist($missingLocale);
130 $this->flush();
131 } catch (Throwable $t) {
132 $this->logException($t);
133 }
134 }
135 try {
136 $this->flush();
137 } catch (Throwable $t) {
138 $this->logException($t);
139 }
140 $this->entityManager->resumeLogging();

154 public function recordTranslation(string $phrase, string $translatedPhrase, string $locale):bool
155 {
156 if (empty(Util::normalizeSpaces($translatedPhrase))) {
157 throw new Exceptions\TanslationException(
158 $this->l->t('Translation for %1$s is empty.', $phrase),
159 );
160 }
161 $this->setDataBaseRepository(TranslationKey::class);
162 $translationKey = $this->findOneBy([ 'phrase' => $phrase ]);
163 if (empty($translationKey)) {
164 $translationKey = TranslationKey::create()->setPhrase($phrase);
165 try {
166 $this->persist($translationKey);
167 $this->flush();
168 } catch (Throwable $t) {
169 $this->logException($t);
170 return false;
171 }
172 }
173
174 $this->setDataBaseRepository(Translation::class);
175 $changed = false;
176 $translation = $this->findOneBy([
177 'translationKey' => $translationKey,
178 'locale' => $locale ]);
179 if (empty($translation)) {
180 $translation = Translation::create()
181 ->setTranslationKey($translationKey)
182 ->setTranslation($translatedPhrase)
183 ->setLocale($locale);
184 $changed = true;
185 } elseif ($translation->getTranslation() != $translatedPhrase) {
186 $translation->setTranslation($translatedPhrase);
187 $changed = true;
188 }
189 if ($changed) {
190 try {
191 $this->persist($translation);
192 $this->flush();
193 } catch (Throwable $t) {
194 $this->logException($t);
195 return false;
196 }
197 }
198 $this->setDatabaseRepository(MissingTranslation::class);
199 $missingLocale = $this->findOneBy([
200 'translationKey' => $translationKey,
201 'locale' => $locale,
202 ]);
203 if (!empty($missingLocale)) {
204 $this->remove($missingLocale, flush: true);
205 }
206 return true;
207 }

244 public function generateCatalogueTemplates():string
245 {
246 $this->setDataBaseRepository(TranslationKey::class);
247 $translationKeys = $this->findAll();
248 $catalogue = [];
249 foreach ($translationKeys as $translationKey) {
250 $entry = [];
251 foreach ($translationKey['locations'] as $location) {
252 $entry[] = "#: {$location['file']}:{$location['line']}";
253 }
254 $entry[] = "#, php-format";
255 $entry[] = 'msgid "'.str_replace('"', '\\"', $translationKey['phrase']).'"';
256 $entry[] = 'msgstr ""';
257 $entry[] = '';
258 $catalogue[] = implode("\n", $entry);
259 }
260 $contents = implode("\n", $catalogue);
261
262 return $contents;

270 public function eraseTranslationKeys(string $phrase):bool
