122 public function round(int $precision = 0):RationalNumber
123 {
124 $rollIn = pow(10, $precision + 1);
125 $roundInc = ($this->getWholePart() + $this->getNumerator() < 0) ? -5 : 5;
126 $integralPart = $this->getWholePart();
127 // this should in principle reduce the chance of overflow ...
128 $fractional = $this->sub($integralPart);
129 return new RationalNumber($integralPart, intdiv($fractional->mul($rollIn)->whole + $roundInc, 10), $rollIn / 10);
130 }

163 public function toDecimal(int $scale = -1, int $precision = 0):string
164 {
165 $sign = $this->sign();
166 $abs = $this->abs();
167 if ($scale === -1) {
168 $result = $abs->whole;
169 $fractionalLimit = self::DECIMAL_DIGITS_MAX - max($abs->numerator > 1 ? strlen($abs->numerator) : 0, ($result == 0 ? 0 : strlen($result)));
170 $abs = $abs->round($fractionalLimit);
171 $abs->subEq($result);
172 $fractionalPart = '';
173 while ($abs->numerator != 0 && strlen($fractionalPart) < $fractionalLimit) {
174 $abs->mulEq(10);
175 $digit = $abs->whole;
176 $abs->subEq($digit);
177 $fractionalPart .= $digit;
178 }
179 $result .= '.' . $fractionalPart;
180 return $sign < 0 ? '-' . $result : $result;
181 }
182 $rollIn = pow(10, $scale + 1);
183 $fixedPoint = str_pad(intdiv($abs->mul($rollIn)->whole + 5, 10), $scale + 1, '0', STR_PAD_LEFT);
184 $integralPart = substr($fixedPoint, 0, -$scale);
185 $fractionalPart = substr($fixedPoint, -$scale);
186 $result = $integralPart . '.' . $fractionalPart;
187 if ($precision > 0 && (strlen(ltrim($integralPart, '0')) + strlen($fractionalPart)) > $precision) {
188 $bound = str_pad('', $precision - $scale, '9') . '.' . str_pad('', $scale, '9');
189 throw new OutOfBoundsException(
190 'The rational number ' . (string)$this . ' (' . $this->toFloat() . ') does not fit into the range [-' . $bound . ', ' . $bound . '].'
191 );
192 }
193 return $sign < 0 ? '-' . $result : $result;
194 }

210 public static function fromDecimal(string|int $decimal):RationalNumber
211 {
212 if (is_int($decimal)) {
213 return new RationalNumber($decimal, 0, 1);
214 }
215 $matches = [];
216 if (!preg_match_all('/^(-)?([1-9]\d*|0?)(?:\.(\d*))?$/', $decimal, $matches)) {
217 throw new InvalidArgumentException('Unable to parse input string "' . $decimal . '".');
218 }
219 $sign = $matches[1][0] == '-' ? -1 : 1;
220 $integralPart = empty($matches[2]) ? 0 : (int)$matches[2][0];
221 $numerator = empty($matches[3]) ? 0 : (int)$matches[3][0];
222 $denominator = pow(10, strlen($matches[3][0] ?? ''));
223 return new RationalNumber($sign * $integralPart, $sign * $numerator, $denominator);
224 }
