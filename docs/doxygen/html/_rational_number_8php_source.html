81 public static function create(
82 int|float|string|Rational $integralPartOrAny,
83 ?int $numerator = null,
84 ?int $denominator = null,

86 if (!is_int($integralPartOrAny)) {

126 public function round(int $precision = 0):RationalNumber
127 {
128 $rollIn = pow(10, $precision + 1);
129 $roundInc = ($this->getWholePart() + $this->getNumerator() < 0) ? -5 : 5;
130 $integralPart = $this->getWholePart();
131 // this should in principle reduce the chance of overflow ...
132 $fractional = $this->sub($integralPart);
133 return new RationalNumber($integralPart, intdiv($fractional->mul($rollIn)->whole + $roundInc, 10), $rollIn / 10);
134 }

167 public function toDecimal(int $scale = -1, int $precision = 0):string
168 {
169 $sign = $this->sign();
170 $abs = $this->abs();
171 if ($scale === -1) {
172 $result = $abs->whole;
173 $fractionalLimit = self::DECIMAL_DIGITS_MAX - max($abs->numerator > 1 ? strlen($abs->numerator) : 0, ($result == 0 ? 0 : strlen($result)));
174 $abs = $abs->round($fractionalLimit);
175 $abs->subEq($result);
176 $fractionalPart = '';
177 while ($abs->numerator != 0 && strlen($fractionalPart) < $fractionalLimit) {
178 $abs->mulEq(10);
179 $digit = $abs->whole;
180 $abs->subEq($digit);
181 $fractionalPart .= $digit;
182 }
183 $result .= '.' . $fractionalPart;
184 return $sign < 0 ? '-' . $result : $result;
185 }
186 $rollIn = pow(10, $scale + 1);
187 $fixedPoint = str_pad(intdiv($abs->mul($rollIn)->whole + 5, 10), $scale + 1, '0', STR_PAD_LEFT);
188 $integralPart = substr($fixedPoint, 0, -$scale);
189 $fractionalPart = substr($fixedPoint, -$scale);
190 $result = $integralPart . '.' . $fractionalPart;
191 if ($precision > 0 && (strlen(ltrim($integralPart, '0')) + strlen($fractionalPart)) > $precision) {
192 $bound = str_pad('', $precision - $scale, '9') . '.' . str_pad('', $scale, '9');
193 throw new OutOfBoundsException(
194 'The rational number ' . (string)$this . ' (' . $this->toFloat() . ') does not fit into the range [-' . $bound . ', ' . $bound . '].'
195 );
196 }
197 return $sign < 0 ? '-' . $result : $result;
198 }

214 public static function fromDecimal(string|int $decimal):RationalNumber
215 {
216 if (is_int($decimal)) {
217 return new RationalNumber($decimal, 0, 1);
218 }
219 $matches = [];
220 if (!preg_match_all('/^(-)?([1-9]\d*|0?)(?:\.(\d*))?$/', $decimal, $matches)) {
221 throw new InvalidArgumentException('Unable to parse input string "' . $decimal . '".');
222 }
223 $sign = $matches[1][0] == '-' ? -1 : 1;
224 $integralPart = empty($matches[2]) ? 0 : (int)$matches[2][0];
225 $numerator = empty($matches[3]) ? 0 : (int)$matches[3][0];
226 $denominator = pow(10, strlen($matches[3][0] ?? ''));
227 return new RationalNumber($sign * $integralPart, $sign * $numerator, $denominator);
228 }
