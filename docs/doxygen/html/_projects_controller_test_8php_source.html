341 public function testValidateProjectName(): void
342 {
343 $this->entityManager->expects($this->atLeastOnce())->method('getRepository')->with(Entities\Project::class);
344 foreach (self::NAME_VALIDATION as $testCase) {
345 $this->postData = [];
346 if (!empty($testCase['control'])) {
347 $this->postData['control'] = $testCase['control'];
348 }
349 foreach ($testCase['input'] as $key => $value) {
350 $this->postData[$this->pme->cgiDataName($key)] = $value;
351 }
352 if ($testCase['result'] === true) {
353 $response = $this->projectsController->validate(EnumProjectValidationTopic::NAME);
354 $this->assertInstanceOf(JSONResponse::class, $response);
355 $data = $response->getData();
356 $this->assertInstanceOf(ProjectValidationResponse::class, $data);
357 $this->assertEquals([], $data->messages);
358 $this->assertNull($data->hints);
359 } elseif (is_array($testCase['result']['output'])) {
360 $response = $this->projectsController->validate(EnumProjectValidationTopic::NAME);
361 $this->assertInstanceOf(JSONResponse::class, $response);
362 $data = $response->getData();
363 $this->assertInstanceOf(ProjectValidationResponse::class, $data);
364 foreach ($testCase['result']['output'] as $key => $value) {
365 $this->assertEquals($value, $data->{$key});
366 }
367 } elseif (is_string($testCase['result']['exception'])) {
368 try {
369 $this->projectsController->validate(EnumProjectValidationTopic::NAME);
370 $this->assertTrue(false, 'The case should throw "' . $testCase['result']['exception'] . '".');
371 } catch (Throwable $t) {
372 $this->assertInstanceOf($testCase['result']['exception'], $t);
373 }
374 }
375 }
