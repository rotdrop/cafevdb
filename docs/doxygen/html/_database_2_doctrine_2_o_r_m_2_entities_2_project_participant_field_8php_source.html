291 public function getSelectableOptions(bool $includeDeleted = false):Collection
292 {
293 // this unfortunately just does not work.
294 // return $this->dataOptions->matching(DBUtil::criteriaWhere([ '!key' => Uuid::NIL, 'deleted' => null, ]));
295 $filtered = $this->dataOptions->filter(function($option) use ($includeDeleted) {
297 return ($includeDeleted || empty($option->getDeleted()))
298 && (string)$option->getKey() != ProjectParticipantFieldDataOption::GENERATOR_KEY;
299 });
300 $iterator = $filtered->getIterator();
301 $iterator->uasort(function(ProjectParticipantFieldDataOption $a, ProjectParticipantFieldDataOption $b) {
302 $cmp = strcmp($a->getLabel(), $b->getLabel());
303 return $cmp !== 0 ? $cmp : strcmp($a->getKey(), $b->getKey());
304 });
305 return new ArrayCollection(iterator_to_array($iterator));
