670 public function setBalanceDocumentsFolder(?DatabaseStorageFolder $balanceDocumentsFolder):Invoice
671 {
672 if (!empty($this->balanceDocumentsFolder)) {
674 foreach ($this->invoiceItems as $part) {
675 if ($part->getBalanceDocumentsFolder() == $this->balanceDocumentsFolder) {
676 $part->setBalanceDocumentsFolder(null);
677 }
678 }
679 if (!empty($this->writteninvoice)) {
680 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
681
682 $this->balanceDocumentsFolder->removeDocument($this->writteninvoice->getFile(), $fileName);
683 }
684 }
685
686 $this->balanceDocumentsFolder = $balanceDocumentsFolder;
687
688 if (!empty($this->balanceDocumentsFolder)) {
689 if (!empty($this->writteninvoice)) {
690 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
691 $this->balanceDocumentsFolder->addDocument($this->writteninvoice->getFile(), $fileName);
692 }
694 foreach ($this->invoiceItems as $part) {
695 if (empty($part->getBalanceDocumentsFolder())) {
696 $part->setBalanceDocumentsFolder($this->balanceDocumentsFolder);
697 }
698 }
699 }
700
701 return $this;
