699 public function setBalanceDocumentsFolder(?DatabaseStorageFolder $balanceDocumentsFolder):Invoice
700 {
701 if (!empty($this->balanceDocumentsFolder)) {
703 foreach ($this->invoiceItems as $part) {
704 if ($part->getBalanceDocumentsFolder() == $this->balanceDocumentsFolder) {
705 $part->setBalanceDocumentsFolder(null);
706 }
707 }
708 if (!empty($this->writteninvoice)) {
709 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
710
711 $this->balanceDocumentsFolder->removeDocument($this->writteninvoice->getFile(), $fileName);
712 }
713 }
714
715 $this->balanceDocumentsFolder = $balanceDocumentsFolder;
716
717 if (!empty($this->balanceDocumentsFolder)) {
718 if (!empty($this->writteninvoice)) {
719 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
720 $this->balanceDocumentsFolder->addDocument($this->writteninvoice->getFile(), $fileName);
721 }
723 foreach ($this->invoiceItems as $part) {
724 if (empty($part->getBalanceDocumentsFolder())) {
725 $part->setBalanceDocumentsFolder($this->balanceDocumentsFolder);
726 }
727 }
728 }
729
730 return $this;
