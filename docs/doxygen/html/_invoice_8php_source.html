648 public function setBalanceDocumentsFolder(?DatabaseStorageFolder $balanceDocumentsFolder):Invoice
649 {
650 if (!empty($this->balanceDocumentsFolder)) {
652 foreach ($this->invoiceItems as $part) {
653 if ($part->getBalanceDocumentsFolder() == $this->balanceDocumentsFolder) {
654 $part->setBalanceDocumentsFolder(null);
655 }
656 }
657 if (!empty($this->writteninvoice)) {
658 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
659
660 $this->balanceDocumentsFolder->removeDocument($this->writteninvoice->getFile(), $fileName);
661 }
662 }
663
664 $this->balanceDocumentsFolder = $balanceDocumentsFolder;
665
666 if (!empty($this->balanceDocumentsFolder)) {
667 if (!empty($this->writteninvoice)) {
668 $fileName = $this->getPaymentRecordFileName($this, $this->writteninvoice->getExtension());
669 $this->balanceDocumentsFolder->addDocument($this->writteninvoice->getFile(), $fileName);
670 }
672 foreach ($this->invoiceItems as $part) {
673 if (empty($part->getBalanceDocumentsFolder())) {
674 $part->setBalanceDocumentsFolder($this->balanceDocumentsFolder);
675 }
676 }
677 }
678
679 return $this;
680 }
