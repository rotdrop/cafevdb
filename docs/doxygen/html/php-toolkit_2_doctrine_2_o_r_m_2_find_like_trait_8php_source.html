336 {
337 $options = [
338 self::QUERY_OPTION_WILDCARDS => false,
339 ];
340 if (!empty($criteria)
341 && array_keys($criteria)[0] === self::QUERY_OPTIONS_KEY
342 && is_array($criteria[self::QUERY_OPTIONS_KEY])) {
343 $options = array_merge($options, $criteria[self::QUERY_OPTIONS_KEY]);
344 unset($criteria[self::QUERY_OPTIONS_KEY]);
345 }
346
347 $orderBy = $orderBy ?: [];
348
349 // filter out instances of criteria
350 $collectionCriteria = [];
351 foreach ($criteria as $key => $value) {
352 if ($value instanceof Collections\Criteria) {
353 $collectionCriteria[] = $value;
354 unset($criteria[$key]);
355 }
356 }
357
358 $whereCriteria = [];
359 $havingCriteria = [];
360 foreach ($criteria as $key => $value) {
361 // if key is numeric we assume $value is a pair [ EXPR => VALUE ]
362 if (is_numeric($key)) {
363 $key = array_key_first($value);
364 $value = $value[$key];
365 }
366 if ($value instanceof BackedEnum) {
367 $value = $value->value;
368 } elseif (is_array($value)) {
369 $value = array_map(fn(mixed $data) => $data instanceof BackedEnumd ? $data->value : $data, $value);
370 }
371 $groupFunction = null;
372 $fctPos = strpos($key, '@');
373 if ($fctPos !== false) {
374 $groupFunction = substr($key, $fctPos + 1);
375 $key = substr($key, 0, $fctPos);
376 }
377 $sqlFunction = null;
378 $fctPos = strpos($key, '#');
379 if ($fctPos !== false) {
380 $sqlFunction = substr($key, $fctPos +1);
381 $key = substr($key, 0, $fctPos);
382 }
383 $fieldType = null;
384 $typePos = strpos($key, ':');
385 if ($typePos !== false) {
386 $fieldType = substr($key, $typePos + 1);
387 $key = substr($key, 0, $typePos);
388 }
389
390 $criterion = [
391 'field' => $key,
392 'fieldType' => $fieldType,
393 'value' => $value,
394 'modifiers' => [],
395 'junctors' => [],
396 'comparator' => null,
397 'literal' => false,
398 'index' => count($whereCriteria) + count($havingCriteria),
399 'groupFunction' => $groupFunction,
400 'sqlFunction' => $sqlFunction,
401 ];
402
403 if (preg_match('/^[!=<>&|()]+/', $key, $matches)) {
404
405 $operators = $matches[0];
406 $key = substr($key, strlen($operators));
407
408 $criterion['field'] = $key;
409
410 $count = 0;
411 while (!empty($operators)) {
412 ++$count;
413 if ($count > 100) {
414 throw new DatabaseException('INFINITE LOOP "' . $operators . '" ' . print_r($criterion, true) . ' ' . print_r($criteria, true));
415 }
416 // reduce multiple ! signs
417 while (strpos($operators, '!!') !== false) {
418 $operators = str_replace('!!', '', $operators);
419 }
420 // (&, (|, !(|, !(&
421 foreach (self::JUNCTORS as $abbr => $junctor) {
422 $pos = strpos($operators, $abbr);
423 if ($pos === 0) {
424 $operators = substr($operators, strlen($abbr));
425 $criterion['junctors'] = array_merge($criterion['junctors'], $junctor);
426 }
427 if (empty($operators)) {
428 break 2;
429 }
430 }
431 // this is only !
432 foreach (self::MODIFIERS as $abbr => $modifier) {
433 $pos = strpos($operators, $abbr);
434 if ($pos === 0) {
435 $operators = substr($operators, strlen($abbr));
436 $criterion['modifiers'][] = $modifier;
437 }
438 if (empty($operators)) {
439 break 2;
440 }
441 }
442 foreach (self::COMPARISONS as $abbr => $comparator) {
443 $pos = strpos($operators, $abbr);
444 if ($pos === 0) {
445 if (!empty($criterion['comparator'])) {
446 throw new DatabaseException('Comparison for key "' . $key . '" already set to "' . $criterion['comparator'] . '".');
447 }
448 if ($abbr == '=' && $value === null) {
449 throw new DatabaseException('Comparison with null is undefined, only equality is allowed.');
450 }
451 if (is_array($value)) {
452 throw new DatabaseException('Array-valued comparisons are not allowed.');
453 }
454 $operators = substr($operators, strlen($abbr));
455 $criterion['comparator'] = $comparator;
456 }
457 if (empty($operators)) {
458 break 2;
459 }
460 }
461 }
462 } // !empty($operators)
463
464 if ($groupFunction === null) {
465 $whereCriteria[] = $criterion;
466 } else {
467 $havingCriteria[] = $criterion;
468 }
469 }
470
471 // walk through criteria and find associations ASSOCIATION.FIELD
472 $joinEntities = [];
473 foreach (array_merge($whereCriteria, $havingCriteria) as $criterion) {
474 $field = $criterion['field'];
475 $dotPos = strpos($field, '.');
476 $joinParent = 'mainTable';
477 $joinAlias = [];
478 while ($dotPos !== false) {
479 $joinTable = substr($field, 0, $dotPos);
480 $joinAlias[] = $joinTable;
481 $aliasKey = implode('_', $joinAlias);
482 $joinEntities[$aliasKey] = [
483 'parent' => $joinParent,
484 'association' => $joinTable,
485 'alias' => $aliasKey,
486 ];
487 $field = substr($field, $dotPos + 1);
488 $dotPos = strpos($field, '.');
489 $joinParent = $joinTable;
490 }
491 }
492 $indexBy = [];
493 foreach ($orderBy as $key => $ordering) {
494 if (is_string($ordering)) {
495 $ordering = EnumOrderByOptions::get(strtoupper($ordering));
496 }
497 $dotPos = strpos($key, '.');
498 if ($dotPos !== false) {
499 $joinParent = 'mainTable';
500 $joinAlias = [];
501 while ($dotPos !== false) {
502 $joinTable = substr($field, 0, $dotPos);
503 $joinAlias[] = $joinTable;
504 $aliasKey = implode('_', $joinAlias);
505 $joinEntities[] = [
506 'parent' => $joinParent,
507 'association' => $joinTable,
508 'alias' => $aliasKey,
509 ];
510 $joinAlias[] = $joinTable;
511 $field = substr($field, $dotPos + 1);
512 $dotPos = strpos($field, '.');
513 $joinParent = $joinTable;
514 }
515 } else {
516 $tableAlias = 'mainTable';
517 $field = $tableAlias.'.'.$key;
518 }
519 if ($ordering == EnumOrderByOptions::INDEX) {
520 $indexBy[$tableAlias] = $field;
521 unset($orderBy[$key]);
522 }
523 }
524
525 // modify and also return
526 $queryParts = [
527 'options' => $options,
528 'criteria' => [
529 'where' => $whereCriteria,
530 'having' => $havingCriteria,
531 ],
532 'orderBy' => $orderBy,
533 'joinEntities' => $joinEntities,
534 'indexBy' => $indexBy,
535 'collectionCriteria' => $collectionCriteria,
536 ];
537
538 return $queryParts;

552 {
553 $indexBy = $queryParts['indexBy']?:['mainTable' => null];

623 protected function generateFindByWhere(
624 ORM\QueryBuilder $qb,
625 array $queryParts,
626 ?int $limit = null,
627 ?int $offset = null,
628 ):ORM\QueryBuilder {
629
631 $class = $this->getClassMetadata();
632
634 $entityManager = $this->getEntityManager();
635
636 // unpack parameter array
637 // foreach ($queryParts as $key => $part) {
638 // ${$key} = $part;
639 // }
640 extract($queryParts);
641
642 foreach (['where', 'having'] as $conditionType) {
643
644 if (!empty($criteria[$conditionType])) {
645 $groupExpression = [
646 0 => [
647 'junctor' => 'andX',
648 'components' => [],
649 ]
650 ];
651 $groupLevel = 0;
652
653 foreach ($criteria[$conditionType] as &$criterion) {
654
655 $field = $criterion['field'];
656 $joinParts = explode('.', $field);
657 if (count($joinParts) > 1) {
658 $col = array_pop($joinParts);
659 $field = implode('_', $joinParts) . '.' . $col;
660 }
661
662 $value = $criterion['value'];
663
664 // $this->log('FIELD ' . $conditionType . ': ' . $field);
665
666 foreach ($criterion['junctors'] as $junctor) {
667 if ($junctor !== ')') {
668 $expression = [
669 'junctor' => $junctor,
670 'components' => [],
671 ];
672 $groupExpression[++$groupLevel] = $expression;
673 } else {
674 $expression = array_pop($groupExpression);
675 $groupLevel--;
676 $junctor = $expression['junctor'];
677 $components = $expression['components'];
678 $compositeExpr = call_user_func_array([ $qb->expr(), $junctor ], $components);
679 $groupExpression[$groupLevel]['components'][] = $compositeExpr;
680 }
681 }
682 if (empty($field)) {
683 continue;
684 }
685 // $this->log('FIELD ' . $field);
686 $fieldIsCollection = false;
687 $dotPos = strpos($field, '.');
688 if ($dotPos !== false) {
689 $tableAlias = substr($field, 0, $dotPos);
690 $column = substr($field, $dotPos + 1);
691 } else {
692 $tableAlias = 'mainTable';
693 $column = $field;
694 $field = $tableAlias . '.' . $field;
695 }
696 if ($tableAlias == 'mainTable') {
697 $fieldIsCollection = $class->isCollectionValuedAssociation($column);
698 } elseif ($class->hasAssociation($tableAlias)) {
699 $targetClass = $entityManager->getClassMetadata($class->getAssociationTargetClass($tableAlias));
700 $fieldIsCollection = $targetClass->isCollectionValuedAssociation($column);
701 }
702 $param = str_replace('.', '_', $field) . '_' . $criterion['index'];
703 if (!empty($criterion['groupFunction'])) {
704 $field = sprintf($criterion['groupFunction'], $field);
705 } elseif (!empty($criterion['sqlFunction'])) {
706 $field = sprintf($criterion['sqlFunction'], $field);
707 }
708 $comparator = $criterion['comparator'] ?? 'eq';
709 if ($value === null) {
710 $criterion['literal'] = true;
711 $expr = $qb->expr()->isNull($field);
712 } elseif (is_array($value)) {
713 // special case empty array:
714 // - in principle always FALSE (nothing is in an empty set)
715 // - FIELD == NULL ? NULL in any given set would be FALSE, we
716 // keep it that way, even if the set is empty
717 if (empty($value)) {
718 $criterion['literal'] = true;
719 // unfortunately, a literal 0 just cannot be modelled with the query builder
720 $expr = $qb->expr()->eq(1, 0);
721 } else {
722 // array values could contain wildcards
723 if (($options[self::QUERY_OPTION_WILDCARDS] ?? false) === true
724 && !empty(array_filter($value, fn($x) => self::containsWildcards($x)))) {
725 $value = implode('|', array_map(fn($x) => self::wildcardsToRegexp($x), $value));
726 $value = '^' . $value . '$';
727 $expr = $qb->expr()->eq(new Expr\Func('REGEXP', [ $field, ':' . $param ]), 1);
728 $criterion['value'] = $value;
729 } else {
730 $expr = $qb->expr()->in($field, ':' . $param);
731 }
732 }
733 } elseif (is_string($value)) {
734 if (($options[self::QUERY_OPTION_WILDCARDS] ?? false) === true && self::containsWildcards($value)) {
735 $expr = $qb->expr()->like($field, ':' . $param);
736 $criterion['value'] = $value;
737 } else {
738 if ($fieldIsCollection && $comparator == 'eq') {
739 $expr = $qb->expr()->isMemberOf(':' . $param, $field);
740 } else {
741 $expr = $qb->expr()->$comparator($field, ':' . $param);
742 }
743 }
744 } elseif ($fieldIsCollection && $comparator == 'eq') {
745 $expr = $qb->expr()->isMemberOf(':' . $param, $field);
746 } else {
747 $expr = $qb->expr()->$comparator($field, ':' . $param);
748 }
749 foreach ($criterion['modifiers'] as $modifier) {
750 $expr = $qb->expr()->$modifier($expr);
751 }
752 $groupExpression[$groupLevel]['components'][] = $expr;
753 }
754
755 // closing parenthesis maybe omitted at end, can be arbitrary many
756 while ($groupLevel >= 0) {
757 $expression = array_pop($groupExpression);
758 $groupLevel--;
759 $junctor = $expression['junctor'];
760 $components = $expression['components'];
761 $compositeExpr = call_user_func_array([ $qb->expr(), $junctor ], $components);
762 if ($groupLevel >= 0) {
763 $groupExpression[$groupLevel]['components'][] = $compositeExpr;
764 } else {
765 $qb->{$conditionType}($compositeExpr);
766 }
767 }
768 unset($criterion); // break reference
769 }
770
771 foreach ($criteria[$conditionType] as $criterion) {
772 if ($criterion['literal']) {
773 continue;
774 }
775 $field = $criterion['field'];
776 if (empty($field)) {
777 continue;
778 }
779 // $this->log('PARAMETER ' . $field);
780 $dotPos = strpos($field, '.');
781 if ($dotPos !== false) {
782 $tableAlias = substr($field, 0, $dotPos);
783 $column = substr($field, $dotPos+1);
784 } else {
785 $tableAlias = 'mainTable';
786 $column = $field;
787 $field = $tableAlias . '.' . $column;
788 }
789 $param = str_replace('.', '_', $field) . '_' . $criterion['index'];
790
791 $value = $criterion['value'];
792 $fieldType = null;
793 if ($criterion['fieldType'] !== null) {
794 $fieldType = $criterion['fieldType'];
795 } elseif ($tableAlias == 'mainTable') {
796 if (!is_array($value)
797 && empty($criterion['groupFunction'])
798 && empty($criterion['sqlFunction'])) {
799 // try to deduce the type as this is NOT done by ORM here
800 $fieldType = $this->getClassMetadata()->getTypeOfField($column);
801 }
802 }
803 $qb->setParameter($param, $value, $fieldType);
804 }
805 }
806
807 self::addOrderBy($qb, $orderBy, $limit, $offset, 'mainTable');
808
809 foreach ($collectionCriteria as $selectableCriteria) {
810 $qb->addCriteria($selectableCriteria);
811 }
812
813 return $qb;
814 }
