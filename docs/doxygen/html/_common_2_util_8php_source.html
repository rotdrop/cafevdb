148 public static function normalizeSpaces(?string $name, string $singleSpace = ' ', bool $stripLinebreaks = false):?string
149 {
150 if ($name === null) {
151 return null;
152 }
153 $name = str_replace("\xc2\xad", '', $name); // soft hyphen
154 $name = str_replace("\xc2\xa0", "\x20", $name);
155 $name = trim($name);
156 $name = str_replace("\r\n", "\n", $name);
157 if ($stripLinebreaks) {
158 $name = str_replace("\n", $singleSpace, $name);
159 }
160 $name = preg_replace('/\h+/u', $singleSpace, $name);
161 $name = preg_replace('/\h+([\n.,;:?!])/u', '$1', $name);
162
163 return $name;
164 }

265 public static function explode(string $delim, ?string $string, int $flags = self::OMIT_EMPTY_FIELDS|self::ESCAPED, string $escape = '\\'):array
266 {
267 if (empty($flags)) {
268 return explode($delim, $string);
269 }
270 $pregFlags = ($flags & self::OMIT_EMPTY_FIELDS) ? PREG_SPLIT_NO_EMPTY : 0;
271 $trimExpr = ($flags & self::TRIM) ? '\s*' : '';
272 if (($flags & self::ESCAPED) && !empty($escape)) {
273 return
274 self::unescapeDelimiter(
275 preg_split('/'.$trimExpr.preg_quote($escape, '/').'.'.'(*SKIP)(*FAIL)|'.preg_quote($delim, '/').$trimExpr.'/s', $string, -1, $pregFlags),
276 $delim,
277 $escape);
278 } else {
279 return preg_split('/'.$trimExpr.preg_quote($delim, '/').$trimExpr.'/', $string, -1, $pregFlags);
280 }

492 public static function humanFileSize(int $bytes, ?string $locale = null, bool $binary = true, int $digits = 2):string
493 {
494 $prefix = [ '', 'K', 'M', 'G', 'T', 'P', 'E', 'Z' ];
495 $locale = $locale ?? 'en_US_POSIX';
496 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
497 $multiplier = $binary ? 1024.0 : 1000.0;
498 $bytes = (float)$bytes;
499 $exp = 0;
500 $prefixCount = count($prefix);
501 while ($exp < $prefixCount && $bytes > $multiplier) {
502 ++$exp;
503 $bytes /= $multiplier;
504 }
505 $postfix = $prefix[$exp];
506 if ($binary && $postfix !== '') {
507 $postfix .= 'i';
508 }
509 $postfix .= 'B';
510 return $fmt->format(round($bytes, $digits)) . ' ' . $postfix;
511 }

601 {
602 $excess = strlen($string) - $limit;

604 $parts = explode(' ', Util::camelCaseToDashes($string, ' '));

639 public static function hexDump(string $string, ?array $options = null)
640 {
641 if (!is_scalar($string)) {
642 throw new InvalidArgumentException('$string argument must be a string');
643 }
644 if (!is_array($options)) {
645 $options = array();
646 }
647 $lineSep = isset($options['line_sep']) ? $options['line_sep'] : "\n";
648 $bytesPerLine = $options['bytes_per_line'] ? $options['bytes_per_line'] : 16;
649 $padChar = isset($options['pad_char']) ? $options['pad_char'] : '.'; // padding for non-readable characters
650
651 $textLines = str_split($string, $bytesPerLine);
652 $hexLines = str_split(bin2hex($string), $bytesPerLine * 2);
653
654 $offset = 0;
655 $output = array();
656 $bytesPerLineDiv2 = (int)($bytesPerLine / 2);
657 foreach ($hexLines as $i => $hexLine) {
658 $textLine = $textLines[$i];
659 $output [] =
660 sprintf('%08X', $offset) . ' ' .
661 str_pad(
662 strlen($textLine) > $bytesPerLineDiv2
663 ?
664 implode(' ', str_split(substr($hexLine, 0, $bytesPerLine), 2)) . ' ' .
665 implode(' ', str_split(substr($hexLine, $bytesPerLine), 2))
666 :
667 implode(' ', str_split($hexLine, 2)), $bytesPerLine * 3) .
668 ' |' . preg_replace('/[^\x20-\x7E]/', $padChar, $textLine) . '|';
669 $offset += $bytesPerLine;
670 }
671 $output[] = sprintf('%08X', strlen($string));
672 return $options['want_array'] ? $output : join($lineSep, $output) . $lineSep;
673 }
