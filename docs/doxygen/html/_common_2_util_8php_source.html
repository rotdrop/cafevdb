113 private static function arrayMergeTwoRecursive(array $dest, array $override):array
114 {
115 $isList = array_is_list($override);
116 foreach ($override as $key => $value) {
117 if ($isList && is_integer($key)) {
118 $dest[] = $value;
119 } elseif (isset($dest[$key]) && is_array($dest[$key]) && is_array($value)) {
120 $dest[$key] = self::arrayMergeTwoRecursive($dest[$key], $value);
121 } else {
122 $dest[$key] = $value;
123 }
124 }

153 public static function normalizeSpaces(?string $name, string $singleSpace = ' ', bool $stripLinebreaks = false):?string
154 {
155 if ($name === null) {
156 return null;
157 }
158 $name = str_replace("\xc2\xad", '', $name); // soft hyphen
159 $name = str_replace("\xc2\xa0", "\x20", $name);
160 $name = trim($name);
161 $name = str_replace("\r\n", "\n", $name);
162 if ($stripLinebreaks) {
163 $name = str_replace("\n", $singleSpace, $name);
164 }
165 $name = preg_replace('/\h+/u', $singleSpace, $name);
166 $name = preg_replace('/\h+([\n.,;:?!])/u', '$1', $name);
167
168 return $name;
169 }

502 public static function humanFileSize(int $bytes, ?string $locale = null, bool $binary = true, int $digits = 2):string
503 {
504 $prefix = [ '', 'K', 'M', 'G', 'T', 'P', 'E', 'Z' ];
505 $locale = $locale ?? 'en_US_POSIX';
506 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
507 $multiplier = $binary ? 1024.0 : 1000.0;
508 $bytes = (float)$bytes;
509 $exp = 0;
510 $prefixCount = count($prefix);
511 while ($exp < $prefixCount && $bytes > $multiplier) {
512 ++$exp;
513 $bytes /= $multiplier;
514 }
515 $postfix = $prefix[$exp];
516 if ($binary && $postfix !== '') {
517 $postfix .= 'i';
518 }
519 $postfix .= 'B';
520 return $fmt->format(round($bytes, $digits)) . ' ' . $postfix;
521 }

565 public static function shortenCamelCaseString(string $string, int $limit, int $minLen = 2):string
566 {
567 $excess = strlen($string) - $limit;
568 if ($excess > 0) {
569 $parts = explode(' ', Util::camelCaseToDashes($string, ' '));
570 // \OCP\Util::writeLog('cafevdb', print_r($parts, true), \OCP\Util::INFO);

582 }

604 public static function hexDump(string $string, ?array $options = null)
605 {
606 if (!is_scalar($string)) {
607 throw new InvalidArgumentException('$string argument must be a string');
608 }
609 if (!is_array($options)) {
610 $options = array();
611 }
612 $lineSep = isset($options['line_sep']) ? $options['line_sep'] : "\n";
613 $bytesPerLine = $options['bytes_per_line'] ? $options['bytes_per_line'] : 16;
614 $padChar = isset($options['pad_char']) ? $options['pad_char'] : '.'; // padding for non-readable characters
615
616 $textLines = str_split($string, $bytesPerLine);
617 $hexLines = str_split(bin2hex($string), $bytesPerLine * 2);
618
619 $offset = 0;
620 $output = array();
621 $bytesPerLineDiv2 = (int)($bytesPerLine / 2);
622 foreach ($hexLines as $i => $hexLine) {
623 $textLine = $textLines[$i];
624 $output [] =
625 sprintf('%08X', $offset) . ' ' .
626 str_pad(
627 strlen($textLine) > $bytesPerLineDiv2
628 ?
629 implode(' ', str_split(substr($hexLine, 0, $bytesPerLine), 2)) . ' ' .
630 implode(' ', str_split(substr($hexLine, $bytesPerLine), 2))
631 :
632 implode(' ', str_split($hexLine, 2)), $bytesPerLine * 3) .
633 ' |' . preg_replace('/[^\x20-\x7E]/', $padChar, $textLine) . '|';
634 $offset += $bytesPerLine;
635 }
636 $output[] = sprintf('%08X', strlen($string));
637 return $options['want_array'] ? $output : join($lineSep, $output) . $lineSep;
638 }
