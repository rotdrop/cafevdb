113 private static function arrayMergeTwoRecursive(array $dest, array $override):array
114 {
115 $isList = array_is_list($override);
116 foreach ($override as $key => $value) {
117 if ($isList && is_integer($key)) {
118 $dest[] = $value;
119 } elseif (isset($dest[$key]) && is_array($dest[$key]) && is_array($value)) {
120 $dest[$key] = self::arrayMergeTwoRecursive($dest[$key], $value);
121 } else {
122 $dest[$key] = $value;
123 }
124 }
125 return $dest;
126 }

153 public static function normalizeSpaces(?string $name, string $singleSpace = ' ', bool $stripLinebreaks = false):?string
154 {
155 if ($name === null) {
156 return null;
157 }
158 $name = str_replace("\xc2\xad", '', $name); // soft hyphen
159 $name = str_replace("\xc2\xa0", "\x20", $name);
160 $name = trim($name);
161 $name = str_replace("\r\n", "\n", $name);
162 if ($stripLinebreaks) {
163 $name = str_replace("\n", $singleSpace, $name);
164 }
165 $name = preg_replace('/\h+/u', $singleSpace, $name);
166 $name = preg_replace('/\h+([\n.,;:?!])/u', '$1', $name);
167
168 return $name;
169 }

513 public static function humanFileSize(int $bytes, ?string $locale = null, bool $binary = true, int $digits = 2):string
514 {
515 $prefix = [ '', 'K', 'M', 'G', 'T', 'P', 'E', 'Z' ];
516 $locale = $locale ?? 'en_US_POSIX';
517 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
518 $multiplier = $binary ? 1024.0 : 1000.0;
519 $bytes = (float)$bytes;
520 $exp = 0;
521 $prefixCount = count($prefix);
522 while ($exp < $prefixCount && $bytes > $multiplier) {
523 ++$exp;
524 $bytes /= $multiplier;
525 }
526 $postfix = $prefix[$exp];
527 if ($binary && $postfix !== '') {
528 $postfix .= 'i';
529 }
530 $postfix .= 'B';
531 return $fmt->format(round($bytes, $digits)) . ' ' . $postfix;
532 }

580 public static function hexDump(string $string, ?array $options = null)
581 {
582 if (!is_scalar($string)) {
583 throw new InvalidArgumentException('$string argument must be a string');
584 }
585 if (!is_array($options)) {
586 $options = array();
587 }
588 $lineSep = isset($options['line_sep']) ? $options['line_sep'] : "\n";
589 $bytesPerLine = $options['bytes_per_line'] ? $options['bytes_per_line'] : 16;
590 $padChar = isset($options['pad_char']) ? $options['pad_char'] : '.'; // padding for non-readable characters
591
592 $textLines = str_split($string, $bytesPerLine);
593 $hexLines = str_split(bin2hex($string), $bytesPerLine * 2);
594
595 $offset = 0;
596 $output = array();
597 $bytesPerLineDiv2 = (int)($bytesPerLine / 2);
598 foreach ($hexLines as $i => $hexLine) {
599 $textLine = $textLines[$i];
600 $output [] =
601 sprintf('%08X', $offset) . ' ' .
602 str_pad(
603 strlen($textLine) > $bytesPerLineDiv2
604 ?
605 implode(' ', str_split(substr($hexLine, 0, $bytesPerLine), 2)) . ' ' .
606 implode(' ', str_split(substr($hexLine, $bytesPerLine), 2))
607 :
608 implode(' ', str_split($hexLine, 2)), $bytesPerLine * 3) .
609 ' |' . preg_replace('/[^\x20-\x7E]/', $padChar, $textLine) . '|';
610 $offset += $bytesPerLine;
611 }
612 $output[] = sprintf('%08X', strlen($string));
613 return $options['want_array'] ? $output : join($lineSep, $output) . $lineSep;
614 }
