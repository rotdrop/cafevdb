145 public static function normalizeSpaces(?string $name, string $singleSpace = ' ', bool $stripLinebreaks = false):?string
146 {
147 if ($name === null) {
148 return null;
149 }
150 $name = str_replace("\xc2\xad", '', $name); // soft hyphen
151 $name = str_replace("\xc2\xa0", "\x20", $name);
152 $name = trim($name);
153 $name = str_replace("\r\n", "\n", $name);
154 if ($stripLinebreaks) {
155 $name = str_replace("\n", $singleSpace, $name);
156 }
157 $name = preg_replace('/\h+/u', $singleSpace, $name);
158 $name = preg_replace('/\h+([\n.,;:?!])/u', '$1', $name);
159
160 return $name;
161 }

262 public static function explode(string $delim, ?string $string, int $flags = self::OMIT_EMPTY_FIELDS|self::ESCAPED, string $escape = '\\'):array
263 {
264 if (empty($flags)) {
265 return explode($delim, $string);
266 }
267 $pregFlags = ($flags & self::OMIT_EMPTY_FIELDS) ? PREG_SPLIT_NO_EMPTY : 0;
268 $trimExpr = ($flags & self::TRIM) ? '\s*' : '';
269 if (($flags & self::ESCAPED) && !empty($escape)) {
270 return
271 self::unescapeDelimiter(
272 preg_split('/'.$trimExpr.preg_quote($escape, '/').'.'.'(*SKIP)(*FAIL)|'.preg_quote($delim, '/').$trimExpr.'/s', $string, -1, $pregFlags),
273 $delim,
274 $escape);
275 } else {
276 return preg_split('/'.$trimExpr.preg_quote($delim, '/').$trimExpr.'/', $string, -1, $pregFlags);
277 }

489 public static function humanFileSize(int $bytes, ?string $locale = null, bool $binary = true, int $digits = 2):string
490 {
491 $prefix = [ '', 'K', 'M', 'G', 'T', 'P', 'E', 'Z' ];
492 $locale = $locale ?? 'en_US_POSIX';
493 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
494 $multiplier = $binary ? 1024.0 : 1000.0;
495 $bytes = (float)$bytes;
496 $exp = 0;
497 $prefixCount = count($prefix);
498 while ($exp < $prefixCount && $bytes > $multiplier) {
499 ++$exp;
500 $bytes /= $multiplier;
501 }
502 $postfix = $prefix[$exp];
503 if ($binary && $postfix !== '') {
504 $postfix .= 'i';
505 }
506 $postfix .= 'B';
507 return $fmt->format(round($bytes, $digits)) . ' ' . $postfix;
508 }

597 public static function shortenCamelCaseString(string $string, int $limit, int $minLen = 2):string
598 {
599 $excess = strlen($string) - $limit;
600 if ($excess > 0) {
601 $parts = explode(' ', Util::camelCaseToDashes($string, ' '));
602 // \OCP\Util::writeLog('cafevdb', print_r($parts, true), \OCP\Util::INFO);
603 do {
604 $shortened = false;
605 foreach ($parts as &$part) {
606 if (strlen($part) > $minLen) {
607 $part = substr($part, 0, -1);
608 $excess --;
609 $shortened = true;
610 }
611 }
612 } while ($excess > 0 && $shortened);
613 $string = Util::dashesToCamelCase(implode(' ', $parts), capitalizeFirstCharacter: true, dashes: ' ');
614 }
615
616 return $string;
617 }

636 public static function hexDump(string $string, ?array $options = null)
637 {
638 if (!is_scalar($string)) {
639 throw new InvalidArgumentException('$string argument must be a string');
640 }
641 if (!is_array($options)) {
642 $options = array();
643 }
644 $lineSep = isset($options['line_sep']) ? $options['line_sep'] : "\n";
645 $bytesPerLine = $options['bytes_per_line'] ? $options['bytes_per_line'] : 16;
646 $padChar = isset($options['pad_char']) ? $options['pad_char'] : '.'; // padding for non-readable characters
647
648 $textLines = str_split($string, $bytesPerLine);
649 $hexLines = str_split(bin2hex($string), $bytesPerLine * 2);
650
651 $offset = 0;
652 $output = array();
653 $bytesPerLineDiv2 = (int)($bytesPerLine / 2);
654 foreach ($hexLines as $i => $hexLine) {
655 $textLine = $textLines[$i];
656 $output [] =
657 sprintf('%08X', $offset) . ' ' .
658 str_pad(
659 strlen($textLine) > $bytesPerLineDiv2
660 ?
661 implode(' ', str_split(substr($hexLine, 0, $bytesPerLine), 2)) . ' ' .
662 implode(' ', str_split(substr($hexLine, $bytesPerLine), 2))
663 :
664 implode(' ', str_split($hexLine, 2)), $bytesPerLine * 3) .
665 ' |' . preg_replace('/[^\x20-\x7E]/', $padChar, $textLine) . '|';
666 $offset += $bytesPerLine;
667 }
668 $output[] = sprintf('%08X', strlen($string));
669 return $options['want_array'] ? $output : join($lineSep, $output) . $lineSep;
670 }

777 return $l->t('Failed to write to disk');
