277 private function getTemplate(?string $template, string $renderAs):string
278 {
279 // Replace colons back to path separators. All our template parameters are
280 // alpha-numeric with the exception that they man contain path separators
281 // (i.e. '/') which optionally are replaced by colons (i.e. ':') in order
282 // to avoid url en-/decoding. Here we need to convert back to path
283 // separators.
284 if (!$this->configCheck['summary']) {
285 return PageRenderer\ConfigCheck::TEMPLATE;
286 }
287 if (empty($template)) {
288 $template = self::DEFAULT_TEMPLATE;
289 }
290
292 $blogRenderer = $this->appContainer->get(PageRenderer\Blog::class);
293 if ($blogRenderer->notificationsPending()) {
294 $template = PageRenderer\BLOG::TEMPLATE;
295 }
296 $template = str_replace(':', Constants::PATH_SEP, $template);
297
298 return $template;
299 }
