269 private function getTemplate(?string $template, string $renderAs):string
270 {
271 // Replace colons back to path separators. All our template parameters are
272 // alpha-numeric with the exception that they man contain path separators
273 // (i.e. '/') which optionally are replaced by colons (i.e. ':') in order
274 // to avoid url en-/decoding. Here we need to convert back to path
275 // separators.
276 if (!$this->configCheck['summary']) {
277 return PageRenderer\ConfigCheck::TEMPLATE;
278 }
279 if (empty($template)) {
280 $template = self::DEFAULT_TEMPLATE;
281 }
282
284 $blogRenderer = $this->appContainer->get(PageRenderer\Blog::class);
285 if ($blogRenderer->notificationsPending()) {
286 $template = PageRenderer\BLOG::TEMPLATE;
287 }
288 $template = str_replace(':', Constants::PATH_SEP, $template);
289
290 return $template;
291 }
