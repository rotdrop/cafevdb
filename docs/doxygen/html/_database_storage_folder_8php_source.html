201 public function addDocument(EncryptedFile $file, ?string $fileName = null, string $conflict = self::ADD_DOCUMENT_CONFLICT_FAIL):?DatabaseStorageFile
202 {
203 $fileName = $fileName ?? $file->getFileName();
204 $fileName = trim($fileName, Constants::PATH_SEP);
205 $existing = $this->directoryEntries->filter(
206 fn(DatabaseStorageDirEntry $dirEntry) => $dirEntry->name === $fileName
207 );
208 if ($existing->count() > 1) {
209 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists multiple times in directory ' . $this->id);
210 }
211 if ($existing->count() == 1) {
213 $dirEntry = $existing->first();
214 switch ($conflict) {
215 case self::ADD_DOCUMENT_CONFLICT_RENAME:
216 $pathInfo = pathinfo($fileName);
217 $namePart = $pathInfo['filename'] . '~';
218 $extension = '.' . $pathInfo['extension'];
219 $backups = $this->directoryEntries
220 ->map(fn(DatabaseStorageDirEntry $value) => $value->getName())
221 ->filter(fn(string $name) => str_starts_with($name, $namePart) && str_ends_with($name, $extension))
222 ->toArray();
223 $count = count($backups);
224 do {
225 ++$count;
226 $backupName = $namePart . $count . $extension;
227 } while (in_array($backupName, $backups));
228 $dirEntry
229 ->setName($backupName)
230 ->setUpdated(new DateTimeImmutable);
231 $dirEntry = null;
232 // We now need to do a flush() to the database as there is a unique
233 // constraint on the dir-entry names. Might be an anti-pattern to
234 // put this into the entity ...
235 EntityManager::getInstance()->flush();
236 break;
237 case self::ADD_DOCUMENT_CONFLICT_REPLACE:
238 if (!($dirEntry instanceof DatabaseStorageFile)) {
239 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' but the existing entry is not a file.');
240 }
241 $dirEntry
242 ->setFile($file)
243 ->setUpdated($dirEntry->getUpdated());
244 break;
245 default:
246 case self::ADD_DOCUMENT_CONFLICT_FAIL:
247 if (!($dirEntry instanceof DatabaseStorageFile)) {
248 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' but the existing entry is not a file.');
249 }
250 if ($dirEntry->getFile() !== $file) {
251 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' and the existing entry does not point to the same file.');
252 }
253 break;
254 }
255 }
256
257 if (empty($dirEntry)) {
258 // need a new one
259 $dirEntry = (new DatabaseStorageFile)
260 ->setName($fileName)
261 ->setParent($this)
262 ->setFile($file);
263 }
264
265 return $dirEntry;
266 }
