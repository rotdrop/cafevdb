199 public function addDocument(EncryptedFile $file, ?string $fileName = null, string $conflict = self::ADD_DOCUMENT_CONFLICT_FAIL):?DatabaseStorageFile
200 {
201 $fileName = $fileName ?? $file->getFileName();
202 $fileName = trim($fileName, Constants::PATH_SEP);
203 $existing = $this->directoryEntries->filter(
204 fn(DatabaseStorageDirEntry $dirEntry) => $dirEntry->name === $fileName
205 );
206 if ($existing->count() > 1) {
207 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists multiple times in directory ' . $this->id);
208 }
209 if ($existing->count() == 1) {
211 $dirEntry = $existing->first();
212 switch ($conflict) {
213 case self::ADD_DOCUMENT_CONFLICT_RENAME:
214 $pathInfo = pathinfo($fileName);
215 $namePart = $pathInfo['filename'] . '~';
216 $extension = '.' . $pathInfo['extension'];
217 $backups = $this->directoryEntries
218 ->map(fn(DatabaseStorageDirEntry $value) => $value->getName())
219 ->filter(fn(string $name) => str_starts_with($name, $namePart) && str_ends_with($name, $extension))
220 ->toArray();
221 $count = count($backups);
222 do {
223 ++$count;
224 $backupName = $namePart . $count . $extension;
225 } while (in_array($backupName, $backups));
226 $dirEntry
227 ->setName($backupName)
228 ->setUpdated(new DateTimeImmutable);
229 $dirEntry = null;
230 // We now need to do a flush() to the database as there is a unique
231 // constraint on the dir-entry names. Might be an anti-pattern to
232 // put this into the entity ...
233 EntityManager::getInstance()->flush();
234 break;
235 case self::ADD_DOCUMENT_CONFLICT_REPLACE:
236 if (!($dirEntry instanceof DatabaseStorageFile)) {
237 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' but the existing entry is not a file.');
238 }
239 $dirEntry
240 ->setFile($file)
241 ->setUpdated($dirEntry->getUpdated());
242 break;
243 default:
244 case self::ADD_DOCUMENT_CONFLICT_FAIL:
245 if (!($dirEntry instanceof DatabaseStorageFile)) {
246 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' but the existing entry is not a file.');
247 }
248 if ($dirEntry->getFile() !== $file) {
249 throw new Exceptions\DatabaseException('Directory entry "' . $fileName . '" already exists in directory ' . $this->id . ' and the existing entry does not point to the same file.');
250 }
251 break;
252 }
253 }
254
255 if (empty($dirEntry)) {
256 // need a new one
257 $dirEntry = (new DatabaseStorageFile)
258 ->setName($fileName)
259 ->setParent($this)
260 ->setFile($file);
261 }
262
263 return $dirEntry;
