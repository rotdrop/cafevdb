154 public function searchFolders(string $regexp, string $topic = self::SEARCH_TOPIC_MOUNT):array
155 {
156 if ($this->sharedFolders === null) {
157 $this->fetchFolders();
158 }
159 if ($regexp[0] != $regexp[-1]) {
160 $regexp = '|^' . $regexp . '$|';
161 }
162 $this->logDebug('REGEXP ' . $regexp . ' / ' . $topic);
163 switch ($topic) {
164 case self::SEARCH_TOPIC_MOUNT:
165 return array_filter($this->sharedFolders, function(FolderWithMappingsAndCache $folderInfo) use ($regexp) {
166 return preg_match($regexp, $folderInfo->mountPoint);
167 });
168 case self::SEARCH_TOPIC_GROUP:
169 return array_filter($this->sharedFolders, function(FolderWithMappingsAndCache $folderInfo) use ($regexp) {
170 foreach (array_keys($folderInfo->groups) as $group) {
171 if (preg_match($regexp, $group)) {
172 return true;
173 }
174 return false;
175 }
176 });
177 default:
178 throw new RuntimeException($this->l->t('Unknown search topic "%1$s"', $topic));
179 }
180 }

336 private function changeFolderManager(string $mountPoint, string $managerId, string $type, bool $canManage):void
337 {
338 // POST BASEURL/groupfolders/folders/8/acl
339 // [ acl => 0 / 1 ]
340 //
341 // POST BASEURL/groupfolders/folders/4/manageACL
342 // [ mappingType => MANAGER_TYPE, mappingId => MANAGER_ID, manageAcl=> 1 ]
343 //
345 $folderInfo = $this->getFolder($mountPoint);
346 if ($folderInfo === null) {
347 throw new RuntimeException($this->l->t('Shared folder for mount-point "%1$s" does not exist, cannot modify manager "%2$s".', [ $mountPoint, $managerId ]));
348 }
349
350 // $this->logDebug('FOLDER INFO ' . print_r($folderInfo->toArray(), true));
351
352 // first check if anything needs to be done
353 $aclEnabled = !!($folderInfo->acl ?? false);
354 $isManager = 0 < count(array_filter($folderInfo->manage ?? [], function(array $manager) use ($managerId, $type) {
355 return $manager['type'] == $type && $manager['id'] == $managerId;
356 }));
357
358 if ($isManager == $canManage && ($canManage || $isManager == $canManage)) {
359 return;
360 }
361
362 $folderId = $folderInfo->id;
363 if (!$aclEnabled) {
364 $this->folderManager->setFolderACL($folderId, true);
365 }
366
367 if ($isManager != $canManage) {
368 $this->folderManager->setManageACL($folderId, $type, $managerId, $canManage);
369 }
370 if (!$aclEnabled || $isManager != $canManage) {
371 $this->getFolderById($folderId);
372 }
373 }

449 public function changeMountPoint(string $mountPoint, string $targetMountPoint, bool $moveChildren = true):void
450 {
451 if ($mountPoint == $targetMountPoint) {
452 return;
453 }
455 $folderInfo = $this->getFolder($mountPoint);
456 if (null === $folderInfo) {
457 throw new RuntimeException($this->l->t('Shared folder for mount-point "%1$s" does not exist, cannot change its mount-point to "%2$s".', [ $mountPoint, ]));
458 }
459 // POST BASE_URL/groupfolders/folders/FOLDER_ID/mountpoint
460 // DATA [ mountpoint => NEW_MOUNT_POINT ]
461
462 $this->folderManager->renameFolder($folderInfo->id, $targetMountPoint);
463 $this->sharedFolders[$targetMountPoint] = $this->sharedFolders[$mountPoint];
464 unset($this->sharedFolders[$mountPoint]);
465 $this->getFolderById($folderInfo->id);
466
467 if ($moveChildren) {
468 $children = $this->searchFolders('|^' . $mountPoint . '/.*$|');
470 foreach ($children as $childInfo) {
471 $oldChildMount = $childInfo->mountPoint;
472 $newChildMount = $targetMountPoint . substr($oldChildMount, strlen($mountPoint));
473 try {
474 $this->changeMountPoint($oldChildMount, $newChildMount, moveChildren: false);
475 } catch (Throwable $t) {
476 $this->logException($t, 'Failed to move child mount from ' . $oldChildMount . ' to ' . $newChildMount);
477 }
478 }
479 }
