196 public function fetchLastModifiedDate(array $criteria = [], ?string $indexBy = null):?array
197 {
198 $selects = [];
199 $selects[] = "GREATEST(
200 MAX(COALESCE(m.updated, '')),
201 MAX(COALESCE(pp.updated, '')),
202 MAX(COALESCE(mi.updated, '')))
203 AS lastModified";
204
205 if ($indexBy !== null) {
206 $selects[] = 'm.' . $indexBy . ' AS ' . $indexBy;
207 $qb = $this->createQueryBuilder('m', 'm.' . $indexBy);
208 $qb->groupBy('m.id');
209 } else {
210 $qb = $this->createQueryBuilder('m', 'm.uuid');
211 }
212 $qb->select($selects)
213 ->leftJoin('m.projectParticipation', 'pp')
214 ->leftJoin('m.instruments', 'mi');
215
216 if (!empty($criteria)) {
217 $andX = $qb->expr()->andX();
218 foreach ($criteria as $key => &$value) {
219 if (is_string($value)) {
220 $value = str_replace('*', '%', $value);
221 if (strpos($value, '%') !== false) {
222 $andX->add($qb->expr()->like('m.' . $key, ':' . $key));
223 } else {
224 $andX->add($qb->expr()->eq('m.' . $key, ':' . $key));
225 }
226 } elseif ($value === null) {
227 $andX->add($qb->expr()->isNull('m.' . $key));
228 } else {
229 $andX->add($qb->expr()->eq('m.' . $key, ':' . $key));
230 }
231 }
232 $qb->where($andX);
233 foreach ($criteria as $key => $value) {
234 if ($value !== null) {
235 $type = ($key == 'uuid') ? 'uuid_binary' : null;
236 $qb->setParameter($key, $value, $type);
237 }
238 }
239 }

242 // $this->log('SQL ' . $qb->getQuery()->getSql());
243 // $this->log('PARAM ' . print_r($qb->getQuery()->getParameters(), true));
244
245 if ($indexBy === null ) {
246 return $qb->getQuery()->getOneOrNullResult();
247 } else {
