449 public function checkShareOwner(string $shareOwnerId, ?string $shareOwnerPassword = null):bool
450 {
451 $shareGroupId = $this->getAppValue(ConfigService::USER_GROUP_KEY, false);
452 if (empty($shareGroupId)) {
453 return false; // need at least this group!
454 }
455
456 $created = false;
457 $shareOwner = null;
458
459 // Create the user if necessary
460 if (!$this->userManager()->userExists($shareOwnerId)) {
461 $this->logError("User does not exist");
462 if (empty($shareOwnerPassword)) {
463 $shareOwnerPassword = $this->generateRandomBytes(30);
464 }
465
466 if (!$this->isSubAdminOfGroup()) {
467 $this->logError("Permission denied: ".$this->userId()." is not a group admin of ".$shareGroupId.".");
468 return false;
469 }
470 $userBackend = $this->user()->getBackend();
471
472 $shareOwner = $this->userManager()->createUserFromBackend($shareOwnerId, $shareOwnerPassword, $userBackend);
473 if (!empty($shareOwner)) {
474 $this->logInfo("User created");
475 // trigger address book creation and things
476 $this->eventDispatcher->dispatch(\OCP\IUser::class . '::firstLogin', new GenericEvent($shareOwner));
477 $created = true;
478 } else {
479 $this->logError("User could not be created");
480 return false;
481 }
482 } else {
483 $shareOwner = $this->userManager()->get($shareOwnerId);
484 }
485
486 if (empty($shareOwner)) {
487 $this->logError("Share-owner " . $shareOwnerId . " could not be found or created.");
488 return false;
489 }
490
491 // Sutff the user in its appropriate group
492 if (!$this->inGroup($shareOwnerId, $shareGroupId)) {
493 $this->logError("not in group");
494 $shareGroup = $this->group($shareGroupId);
495 if (empty($shareGroup)) {
496 $this->logError("Could not get group " . $shareGroupId . ".");
497 } else {
498 $shareGroup->addUser($shareOwner);
499 }
500 // check again, addUser() has no return value
501 if (!$this->inGroup($shareOwnerId, $shareGroupId)) {
502 $this->logError("Could not add " . $shareOwnerId . " to group " . $shareGroupId . ".");
503 if ($created) {
504 $this->logError("Deleting just created user " . $shareOwnerId . ".");
505 $shareOwner->delete();
506 }
507 return false;
508 }
509 $this->logDebug("added to group");
510 }
511
512 return $this->shareOwnerExists($shareOwnerId);
