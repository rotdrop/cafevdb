50const pmeFormInit = function(containerSel, parameters, resizeCB) {
51 containerSel = PHPMyEdit.selector(containerSel);
52 const $container = PHPMyEdit.container(containerSel);
53 const $form = $container.find(pmeFormSelector);
54
55 if (!PHPMyEdit.hasEditableData($form)) {
56 // no need to do further work
57 return;
58 }
59
60 $container
61 .off('click', 'a.download-link.ajax-download')
62 .on('click', 'a.download-link.ajax-download', function(event) {
63 const $this = $(this);
64 const post = $this.data('post');
65 fileDownload($this.attr('href'), post);
66 return false;
67 });
68
69 // upload supporting document(s)
70 $container
71 .find('tr.supporting-document ' + pmeValueSelector + ' .file-upload-row')
72 .each(function() {
73 initFileUploadRow.call(
74 this,
75 -1, // projectId
76 -1, // musicianId,
77 resizeCB, {
78 upload: 'documents/finance/' + templateName + '/upload',
79 delete: 'documents/finance/' + templateName + '/delete',
80 });
81 const ambientContainerSelector = parameters?.tableOptions?.ambientContainerSelector;
82 if (ambientContainerSelector) {
83 $(this).on('pme:upload-done pme:upload-deleted', (event) => {
84 event.stopImmediatePropagation();
85 $(ambientContainerSelector).trigger('pmedialog:changed');
86 PHPMyEdit.submitOuterFormNoThrow(ambientContainerSelector);
87 });
88 }
89 });
90
91 let inputLock = false;
92
93 $container.on('change', pmeSelectInputSelector + '.musician-id', function(event) {
94 // avoid ping-pong
95 if (inputLock) {
96 return false;
97 }
98 inputLock = true;
99 const $this = $(this);
100 const musicianId = +$this.val() || 0;
101 const $musicianOption = getSelectOptionByValue($this, musicianId);
102 const projects = $musicianOption.data('projects') || [];
103 console.info('MUSICIAN', $musicianOption, projects);
104
105 const $projectSelect = $container.find(pmeSelectInputSelector + '.project-id');
106 const $projectOptions = getSelectOptions($projectSelect);
107 const selectableProjects = [];
108 $projectOptions.each(function() {
109 const $projectOption = $(this);
110 const projectId = +$projectOption.val();
111 if (musicianId > 0 && projects.find((id) => id === projectId) === undefined) {
112 $projectOption.prop('selected', false).prop('disabled', true);
113 } else {
114 $projectOption.prop('disabled', false);
115 selectableProjects.push($projectOption);
116 }
117 });
118 if (selectableProjects.length === 1) {
119 $projectOptions.each(function() { $(this).prop('disabled', false); });
120 selectableProjects[0].prop('selected', true);
121 }
122 refreshSelectWidget($projectSelect);
123
124 const $paymentsSelect = $container.find(pmeSelectInputSelector + '.composite-payment-id');
125 const $paymentOptions = getSelectOptions($paymentsSelect);
126 const selectablePayments = [];
127 $paymentOptions.each(function() {
128 const $paymentOption = $(this);
129 const thisMusicianId = +$paymentOption.data('musicianId');
130 if (musicianId > 0 && musicianId !== thisMusicianId) {
131 $paymentOption.prop('disabled', true);
132 } else {
133 $paymentOption.prop('disabled', false);
134 selectablePayments.push($paymentOption);
135 }
136 });
137 if (selectablePayments.length === 1) {
138 $paymentOptions.each(function() { $(this).prop('disabled', false); });
139 selectablePayments[0].prop('selected', true);
140 $container.find(pmeInputClassSelector() + '.composite-payment-subject').val(
141 selectablePayments[0].data('subject'),
142 );
143 $form.toggleClass($form.data('selfTestFailure'), !selectablePayments[0].data('status'));
144 }
145 refreshSelectWidget($paymentsSelect);
146
147 setTimeout(() => { inputLock = false; }, 0);
148
149 return false;
150 });
151

211 $container.on('change', pmeSelectInputSelector + '.composite-payment-id', function(event) {
212 // avoid ping-pong
213 if (inputLock) {
214 return false;
215 }
216 inputLock = true;
217
218 const $this = $(this);
219 const paymentId = +$this.val() || 0;
220 const $paymentOption = getSelectOptionByValue($this, paymentId);
221 const projectId = $paymentOption.data('projectId');
222 const musicianId = $paymentOption.data('musicianId');
223 const amount = $paymentOption.data('amount');
224 const amountWaived = $paymentOption.data('amountWaived');
225 const status = $paymentOption.data('status');
226 const subject = $paymentOption.data('subject');
227
228 $container.find(pmeInputSelector + '.project-payment.amount').val(amount);
229 $container.find(pmeInputSelector + '.project-payment.amount-waived').val(amountWaived);
230 $container.find(pmeInputSelector + '.project-payment.status').val([status]);
231
232 $form.toggleClass($form.data('selfTestFailure'), !status);
233
234 // we re-enable all options but select the respective project and
235 // person
236
237 const $musicianInput = $container.find(pmeInputClassSelector() + '.musician-id');
238 if ($musicianInput.is('select')) {
239 const $musicianOptions = getSelectOptions($musicianInput);
240 $musicianOptions.each(function() {
241 const $musicianOption = $(this);
242 const thisMusicianId = +$musicianOption.val();
243 $musicianOption.prop('disabled', false).prop('selected', thisMusicianId === musicianId);
244 });
245 refreshSelectWidget($musicianInput);
246 } else if ($musicianInput.is('input')) {
247 $musicianInput.val($musicianInput.data('pmeValues').values[musicianId]);
248 }
249
250 const $projectInput = $container.find(pmeInputClassSelector() + '.project-id');
251 if ($projectInput.is('select')) {
252 const $projectOptions = getSelectOptions($projectInput);
253 $projectOptions.each(function() {
254 const $projectOption = $(this);
255 const thisProjectId = +$projectOption.val();
256 $projectOption.prop('disabled', false).prop('selected', thisProjectId === projectId);
257 });
258 refreshSelectWidget($projectInput);
259 } else if ($projectInput.is('input')) {
260 $projectInput.val($projectInput.data('pmeValues').values[projectId]);
261 }
262
263 const $subjectInput = $container.find(pmeInputClassSelector() + '.composite-payment-subject');
264 console.info('SUBJECT INPUT', pmeInputClassSelector() + '.composite-payment-subject', $subjectInput);
265 $subjectInput.val(subject);
266
267 const $options = getSelectOptions($this);
268 $options.each(function() { $(this).prop('disabled', false); });
269
270 setTimeout(() => { inputLock = false; }, 0);
271
272 return false;
273 });

276const documentReady = function() {
277
278 PHPMyEdit.addTableLoadCallback(
279 templateName, {
280 callback(selector, parameters, resizeCB) {
281 if (parameters.reason === 'dialogOpen') {
282 pmeFormInit(selector, parameters, resizeCB);
283 }
284 console.info('RESIZE DB');
285 resizeCB();
286 },
287 context: globalState,
288 parameters: [],
289 });
290
291 CAFEVDB.addReadyCallback(function() {
292
293 const container = PHPMyEdit.container();
294
295 if (!container.hasClass(templateName)) {
296 return;
297 }
298
299 const renderer = $(PHPMyEdit.defaultSelector).find('form.pme-form input[name="templateRenderer"]').val();
300 if (renderer === templateRenderer(templateName)) {
301 pmeFormInit(PHPMyEdit.defaultSelector, undefined, () => null);
302 }
303 });
304};
