91 private static function normalizeSpaces(string $text): string
92 {
93 $text = str_replace("\xc2\xad", '', $text); // soft hyphen
94 $text = str_replace("\xc2\xa0", "\x20", $text);
95 $text = trim($text);
96 $text = preg_replace('/\h+([\n.,;:?!])/u', '$1', $text);

98 return $text;

106 public static function convert(string $html):string
107 {
108 foreach (self::HTML_CONVERTERS as $converter => $arguments) {
109 $text = self::callConverter($converter, $arguments, $html);
110 if ($text !== null) {

113 }
114 // Fallback. The PHP converter cannot handle tables, so render them
115 // ourselves as preformatted text.
116 $html2Text = new PhpHtml2Text(self::renderTables($html));
117 return self::normalizeSpaces($html2Text->getText());

129 private static function renderTables(string $html):string
130 {
131 if (stripos($html, '<table') === false) {
132 return $html;
133 }
134 $document = new DOMDocument;
135 $useErrors = libxml_use_internal_errors(true);
136 $loaded = $document->loadHTML('<?xml encoding="utf-8"?>' . $html);
137 libxml_clear_errors();
138 libxml_use_internal_errors($useErrors);
139 if (!$loaded) {
140 return $html;
141 }
142
143 do {
144 $table = null;
146 foreach ($document->getElementsByTagName('table') as $candidate) {
147 if ($candidate->getElementsByTagName('table')->length === 0) {
148 $table = $candidate;
149 break;
150 }
151 }
152 if ($table === null) {
153 break;
154 }
155 $pre = $document->createElement('pre');
156 $pre->appendChild($document->createTextNode(self::renderTable($table)));
157 $table->parentNode->replaceChild($pre, $table);
158 } while (true);
159
160 $result = '';
161 $body = $document->getElementsByTagName('body')->item(0);
162 foreach ($body?->childNodes ?? [] as $child) {
163 $result .= $document->saveHTML($child);
164 }
165 return $result;
166 }

173 private static function renderTable(DOMElement $table):string
174 {
175 $rows = [];
176 foreach ($table->getElementsByTagName('tr') as $tableRow) {
177 $row = [];
178 foreach ($tableRow->childNodes as $cell) {
179 if (!($cell instanceof DOMElement) || !in_array(strtolower($cell->tagName), [ 'td', 'th' ])) {
180 continue;
181 }
182 $row[] = new TableCell(
183 OutputFormatter::escape(self::cellText($cell)),
184 [
185 'colspan' => max(1, (int)$cell->getAttribute('colspan')),
186 'style' => new TableCellStyle([ 'align' => self::cellAlignment($cell) ]),
187 ],
188 );
189 }
190 if (!empty($row)) {
191 $rows[] = $row;
192 $rows[] = new TableSeparator;
193 }
194 }
195 if (empty($rows)) {
196 return '';
197 }
198 array_pop($rows);
199
200 // Mimic the layout of w3m: no cell padding, lines between all rows.
201 $style = (new TableStyle)
202 ->setHorizontalBorderChars('─')
203 ->setVerticalBorderChars('│')
204 ->setCrossingChars('┼', '┌', '┬', '┐', '┤', '┘', '┴', '└', '├')
205 ->setCellRowContentFormat('%s');
206
207 $output = new BufferedOutput(decorated: false);
208 (new Table($output))->setStyle($style)->setRows($rows)->render();
209
210 return "\n" . rtrim($output->fetch()) . "\n";
211 }

218 private static function cellText(DOMNode $cell):string
219 {
220 foreach (iterator_to_array($cell->getElementsByTagName('br')) as $lineBreak) {
221 $lineBreak->parentNode->replaceChild($cell->ownerDocument->createTextNode("\n"), $lineBreak);
222 }
223 $lines = array_map(
224 fn(string $line) => trim(preg_replace('/\s+/u', ' ', str_replace("\xc2\xa0", ' ', $line))),
225 explode("\n", $cell->textContent),
226 );
227 return trim(implode("\n", $lines));
228 }

235 private static function cellAlignment(DOMElement $cell):string
236 {
237 $style = $cell->getAttribute('style');
238 if (preg_match('/text-align\s*:\s*(left|center|right)/i', $style, $matches)) {
239 return strtolower($matches[1]);
240 }
241 $align = strtolower($cell->getAttribute('align'));
242 return in_array($align, [ 'center', 'right' ]) ? $align : 'left';

254 private static function callConverter(string $converter, array $arguments, string $html):?string
255 {
256 $converter = (self::$executableFinder ?? new ExecutableFinder)->find($converter);
257 if (empty($converter)) {
258 return null;
259 }
260 array_unshift($arguments, $converter);
261 $htmlConvert = new Process($arguments, null, [ 'LC_ALL' => 'C.UTF-8' ]);
262 $htmlConvert->setInput($html);
263 $htmlConvert->run();
264 $text = $htmlConvert->getOutput();
265 return $text;
266 }
