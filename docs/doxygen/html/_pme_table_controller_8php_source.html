157 {

159 if (empty($exportFormat)) {

168
170 $exporter = $this->appContainer->query('export'.':'.$template);
171 if (empty($exporter)) {
172 return self::response(

174 Http::STATUS_BAD_REQUEST);

177 $tmpFile = $this->tempManager->getTemporaryFile($this->appName());
