259 public function canOpen(File $fileNode):bool
260 {
261 $this->setProcessEnvironment();
262
263 $localPath = self::getLocalPath($fileNode);
264
265 $format = ArchiveFormats::detectArchiveFormat($localPath);
266 $canOpen = $format !== null && ArchiveFormats::canOpen($format);
267
268 $this->restoreProcessEnvironment();
269
270 if ($format === null) {
271 throw new Exceptions\ArchiveCannotOpenException(
272 $this->l->t('Unable to detect the archive format of "%1$s".', $fileNode->getName())
273 );
274 }
275 if (!$canOpen) {
276 $messages = [];
277 if (str_starts_with($format, 't') && !str_starts_with($format, 'tar')) {
278 $innerFormat = 'tar';
279 $compositeFormat = $format;
280 $format = substr($format, 1);
281 $this->setProcessEnvironment();
282 $canDecompress = $format !== null && ArchiveFormats::canOpen($format);
283 $canUntar = ArchiveFormats::canOpen($innerFormat);
284 $this->restoreProcessEnvironment();
285 if (!$canDecompress) {
286 $messages[] = $this->l->t('Archive format of "%1$s" detected as "%2$s", but there is no backend driver installed which can decompress ".%3$s" files.', [
287 $fileNode->getName(),
288 $compositeFormat,
289 $format,
290 ]);
291 }
292 if (!$canUntar) {
293 // this should never be the case ...
294 $messages[] = $this->l->t(
295 'Unable to deal with tar-files. Please check the installation of the app.'
296 );
297 }
298 } else {
299 $messages[] = $this->l->t('The archive format of "%1$s" has been detected as "%2$s", but there is no backend driver installed which can deal with this format.', [
300 $fileNode->getName(),
301 $format,
302 ]);
303 }
304 $formats = ArchiveFormats::getDeclaredDriverFormats();
305 foreach ($formats[$format] as $driverClass) {
306 $shortDriver = substr($driverClass, strrpos($driverClass, '\\') + 1);
307 if (!$driverClass::isInstalled()) {
308 $messages[] = $this->l->t('The "%1$s" driver could handle this format, but it is not installed.', $shortDriver);
309 $typeLabel = BasicDriver::$typeLabels[$driverClass::TYPE];
310 $instructions = $driverClass::getInstallationInstruction();
311 $messages[] = $this->l->t('Installation instructions: ')
312 . ucfirst($typeLabel)
313 . '. '
314 . ucfirst($instructions)
315 . '.';
316 continue;
317 }
318 $abilities = $driverClass::getFormatAbilities($format);
319 $requiredAbilities = DriverAbilities::OPEN|DriverAbilities::EXTRACT_CONTENT;
320 if (($abilities & $requiredAbilities) != $requiredAbilities) {
321 $messages[] = $this->l->t('The "%1$s" driver claims to handle this format, but cannot extract the archive content.', $shortDriver);
322 }
323 }
324 throw new Exceptions\ArchiveCannotOpenException(
325 implode(PHP_EOL, $messages)
326 );
327 }
328
329 return true;
330 }

365 public function open(File $fileNode, ?int $sizeLimit = null, ?string $password = null):?ArchiveService
366 {
367 if (!$this->canOpen($fileNode)) {
368 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
369 $fileNode->getPath(), self::getLocalPath($fileNode),
370 ]));
371 }
372
373 $this->setProcessEnvironment();
374
375 $this->archiver = ArchiveBackend::open(self::getLocalPath($fileNode), password: $password);
376
377 $this->restoreProcessEnvironment();
378
379 if (empty($this->archiver)) {
380 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
381 $fileNode->getPath(), self::getLocalPath($fileNode),
382 ]));
383 }
384 if ($sizeLimit === null) {
385 $sizeLimit = $this->sizeLimit;
386 }
387 $this->fileNode = $fileNode;
388 $archiveInfo = $this->getArchiveInfo();
389 $archiveSize = $archiveInfo['originalSize'];
390 if ($sizeLimit !== null && $archiveSize > $sizeLimit) {
391 $this->archiver = null;
392 $this->fileNode = null;
393 throw new Exceptions\ArchiveTooLargeException(
394 $this->t('Uncompressed size of archive "%1$s" is too large: %2$s > %3$s', [
395 $fileNode->getInternalPath(), CloudUtil::humanFileSize($archiveSize), CloudUtil::humanFileSize($sizeLimit),
396 ]),
397 $sizeLimit,
398 $archiveSize,
399 );
400 }
401
402 $this->unicodeNormalization = Normalizer::NFC;
403 foreach ($this->archiver->getFileNames() as $fileName) {
404 if (!Normalizer::isNormalized($fileName)) {
405 $this->unicodeNormalization = Normalizer::NFD;
406 break;
407 }
408 }
409
410 return $this;

463 public static function getArchiveFolderName(string $archiveFileName):?string
464 {
465 // double to account for "nested" archive types
466 $archiveFolderName = pathinfo($archiveFileName, PATHINFO_FILENAME);
467 $secondExtension = pathinfo($archiveFolderName, PATHINFO_EXTENSION);
468
469 // as a rule of thumb we only strip the second extension if it contains no
470 // spaces and is no longer that 4 characters.
471 if (strlen($secondExtension) <= 4 && str_replace(' ', '', $secondExtension) === $secondExtension) {
472 $archiveFolderName = pathinfo($archiveFolderName, PATHINFO_FILENAME);
473 }
474
475 return $archiveFolderName;
476 }

585 protected function restoreProcessEnvironment():void
586 {
587 foreach (array_keys(self::OVERRIDE_ENVIRONMENT) as $key) {
588 if (!isset($this->savedProcessEnvironment[$key])) {
589 continue;
590 }
591 if ($this->savedProcessEnvironment[$key] === null) {
592 unset($_ENV[$key]);
593 } else {
594 $_ENV[$key] = $this->savedProcessEnvironment[$key];
595 }
596 }
