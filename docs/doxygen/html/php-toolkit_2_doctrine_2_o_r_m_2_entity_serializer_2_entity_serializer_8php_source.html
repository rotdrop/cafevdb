182 public function addEntity(mixed $entity, int $depth = 1, bool $principal = true): void
183 {
184 // disable the filter here in order to have all entities available
185 $softDeleteableState = $this->entityManager->setFilterEnabled(EntityManager::SOFT_DELETEABLE_FILTER, false);
186 try {
187 $metaData = $this->entityManager->getClassMetadata(get_class($entity));
188 $entityClassName = $this->stripCommonPrefix($metaData->getName());
189 $id = $metaData->getIdentifierValues($entity);
190 if (empty($id)) {
191 throw new Exceptions\DatabaseMissingIdentifierException(
192 $this->l->t('Unable to determine the identifier values for an instance of "%s".', get_class($entity)),
193 entityClassName: get_class($entity),
194 );
195 }
196 $flatIdentifier = $this->flattenIdentifier($metaData, $id);
197 $existing = false;
198 if (!empty($this->repositories[$entityClassName][$flatIdentifier])) {
199 if ($principal) {
200 if (empty($this->entities[$entityClassName])
201 || !in_array($flatIdentifier, $this->entities[$entityClassName])) {
202 $this->entities[$entityClassName][] = $flatIdentifier;
203 }
204 }
205 if (($this->entityDepths[$entityClassName][$flatIdentifier] ?? -1) >= $depth) {
206 return;
207 }
208 $existing = true;
209 }
210
211 // must be set before recursing
212 $this->entityDepths[$entityClassName][$flatIdentifier] = $depth;
213
214 $flatEntity = [];
215
216 // Make sure lazy objects are initialized
217 $this->entityManager->initializeObject($entity);
218
219 // ordinary non-associative fields
221 foreach (array_keys($metaData->fieldMappings) as $field) {
222 $flatEntity[$field] = $metaData->getFieldValue($entity, $field);
223 }
225 foreach ($metaData->associationMappings as $field => $associationMapping) {
226 switch (true) {
227 case $associationMapping instanceof Mapping\ToOneAssociationMapping:
228 $targetEntity = $metaData->getFieldValue($entity, $field);
229 if ($targetEntity === null) {
230 $flatEntity[$field] = null;
231 break;
232 }
233 $targetMetaData = $this->entityManager->getClassMetadata(get_class($targetEntity));
234 $targetClassName = $this->stripCommonPrefix($targetMetaData->getName());
235 $targetId = $targetMetaData->getIdentifierValues($targetEntity);
236 if (empty($targetId)) {
237 throw new Exceptions\DatabaseMissingIdentifierException(
238 $this->l->t('Unable to determine the identifier values for an instance of "%s".', $targetClassName),
239 entityClassName: $targetClassName,
240 );
241 }
242 $flatTargetIdentifier = $this->flattenIdentifier($targetMetaData, $targetId);
243 $flatEntity[$field] = new EntityReference(
244 flatIdentifier: $flatTargetIdentifier,
245 entityClassName: $targetClassName,
246 );
247 if ($depth > 0
248 && (empty($this->repositories[$targetClassName][$flatTargetIdentifier])
249 || ($this->entityDepths[$targetClassName][$flatTargetIdentifier] ?? -1) < ($depth - 1))) {
250 $this->addEntity($targetEntity, $depth - 1, false);
251 }
252 break;
253 case $associationMapping instanceof Mapping\ToManyAssociationMapping:
254 $targetCollection = $metaData->getFieldValue($entity, $field);
255 if (null === $targetCollection) {
256 throw new UnexpectedValueException($this->l->t('Collection "%1$s" in entity of type "%2$s" is null.', [
257 $field, $entityClassName,
258 ]));
259 }
260 $targetClassName = $this->stripCommonPrefix($associationMapping->targetEntity);
261 $flatCollection = [];
262 $keyConvert = fn(mixed $value, mixed $metaData) => $value;
263 if ($associationMapping->isIndexed()) {
264 $targetMetaData = $this->entityManager->getClassMetadata($associationMapping->targetEntity);
265 $indexField = $associationMapping->indexBy();
266 $keyConvert = function(mixed $value, mixed $metaData) use ($indexField, $targetMetaData, $field, $entityClassName) {
267 $fieldType = PersisterHelper::getTypeOfColumn(
268 $indexField,
269 $this->getWrappedObject($targetMetaData),
270 $this->entityManager->getWrappedObject(),
271 );
272 $phpValue = $this->entityManager->getConnection()->convertToPHPValue($value, $fieldType);
273 if (!is_string($phpValue) && !is_integer($phpValue)) {
274 $phpValue = (string)$phpValue;
275 }
276 return $phpValue;
277 };
278 }
279 foreach ($targetCollection as $key => $targetEntity) {
281 $targetMetaData = $this->entityManager->getClassMetadata(get_class($targetEntity));
282 $key = $keyConvert($key, $targetMetaData);
283 $targetId = $targetMetaData->getIdentifierValues($targetEntity);
284 if (empty($targetId)) {
285 throw new Exceptions\DatabaseMissingIdentifierException(
286 $this->l->t('Unable to determine the identifier values for an instance of "%s".', $targetClassName),
287 entityClassName: $targetClassName,
288 );
289 }
290 $entityName = $this->stripCommonPrefix($targetMetaData->getName());
291 if ($entityName == $targetClassName) {
292 $entityName = null;
293 }
294 $flatTargetIdentifier = $this->flattenIdentifier($targetMetaData, $targetId);
295 $flatCollection[$key] = new CollectionEntityReference(
296 flatIdentifier: $flatTargetIdentifier,
297 entityClassName: $entityName,
298 );
299 if ($depth > 0
300 && (empty($this->repositories[$targetClassName][$flatTargetIdentifier])
301 || ($this->entityDepths[$targetClassName][$flatTargetIdentifier] ?? -1) < $depth - 1)) {
302 $this->addEntity($targetEntity, $depth - 1, false);
303 }
304 }
305 $flatEntity[$field] = new EntityReferenceCollection(
306 entityClassName: $targetClassName,
307 entities: $flatCollection,
308 );
309 break;
310 }
311 }
312 if ($entity instanceof JsonSerializable) {
313 // add further properties if advertised by the entity. This is now
314 // somewhat inefficient, but perhaps we will find a better way some
315 // day ....
316 $jsonData = $entity->jsonSerialize();
317 $missingProperties = array_diff(array_keys($jsonData), array_keys($flatEntity));
318 foreach ($missingProperties as $property) {
319 $flatEntity[$property] = $jsonData[$property];
320 }
321 foreach ($jsonData as $key => $value) {
322 if (is_scalar($value) && $value !== $flatEntity[$key]) {
323 $flatEntity[$key] = $value;
324 }
325 }
326 }
327 ksort($flatEntity);
328 $this->repositories[$entityClassName][$flatIdentifier] = $flatEntity;
329 if ($principal) {
330 if (!$existing) {
331 $this->entities[$entityClassName][] = $flatIdentifier;
332 }
333 }
334 } catch (Throwable $t) {
335 $this->entityManager->setFilterEnabled(EntityManager::SOFT_DELETEABLE_FILTER, $softDeleteableState);
336 throw new Exceptions\EntitySerializationException(
337 $this->l->t('Unable to compute a serialization for an instance of "%s".', get_class($entity)),
338 $entity,
339 previous: $t,
340 );
341 }
342 $this->entityManager->setFilterEnabled(EntityManager::SOFT_DELETEABLE_FILTER, $softDeleteableState);
343 }
