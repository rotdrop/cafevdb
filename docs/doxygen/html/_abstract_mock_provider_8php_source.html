138 private static function snapshotContainer(SimpleContainer $container): InnerContainer
139 {
140 $innerContainer = new ReflectionProperty(SimpleContainer::class, 'container')->getValue($container);
141 $reflectionContainer = new ReflectionClass($innerContainer);
142 $snapshot = new InnerContainer;
144 foreach ($reflectionContainer->getProperties() as $propertyAccessor) {
145 $property = $propertyAccessor->getValue($innerContainer);
146 if (is_object($property) && get_class($property) === SplObjectStorage::class) {
147 $snapshotProperty = new SplObjectStorage();
148 $snapshotProperty->addAll($property);
149 $property = $snapshotProperty;
150 }
151 $propertyAccessor->setValue($snapshot, $property);
152 }
153 $snapshot->offsetUnset(\OC\DateTimeZone::class);
154 $snapshot->offsetUnset(\OC\URLGenerator::class);
155 // print_r($snapshot->keys());
156 return $snapshot;
157 }
