92 public function preUpdate(Entity $entity, ORMEvent\PreUpdateEventArgs $event)
93 {
94 $this->l = $this->appContainer->get(IL10N::class);
95 $field = 'participationStatus';
96 if ($event->hasChangedField($field)) {
97 $participationStatus = (string)$event->getNewValue($field);
98 switch ($participationStatus) {
99 case ParticipationStatus::PASSIVE:
100 // confirmed partiticipants must not be "passive"
101 if ($entity->getRegistration()) {
102 throw new Exceptions\EnduserNotificationException(
103 $this->l->t(
104 'Confirmed project participants cannot have a participation status of "%s".',
105 ParticipationStatus::getL10NValues($this->l)[$participationStatus],
106 ),
107 );
108 }
109 break;
110 case ParticipationStatus::ASSOCIATED:
111 // participants playing real instruments must not be associated
112 $realInstruments = $entity->getRealInstruments();
113 if (!$realInstruments->isEmpty()) {
114 throw new Exceptions\EnduserNotificationException(
115 $this->l->t(
116 'The participation-status cannot be set to "%1$s" as long as the participant is registered to perform with real instruments (%2$s)',
117 [
118 ParticipationStatus::getL10NValues($this->l)[$participationStatus],
119 implode(', ', array_map(fn(Entities\ProjectInstrument $instrument) => $instrument->getInstrument()->getName(), $realInstruments->toArray())),
120 ],
121 ),
122 );
123 }
124 break;
125 }
126 }
127 $field = 'registration';
128 if ($event->hasChangedField($field)) {
129 $oldValue = $event->getOldValue($field);
130 $this->entityManager->dispatchEvent(
131 new Events\PreChangeRegistrationConfirmation(
132 $entity,
133 !empty($oldValue),
134 !empty($event->getNewValue($field)),
135 )
136 );
137
138 $key = self::entityId($entity, $field);
139 $this->preUpdateValues[$key] = $oldValue;
140 }
141 $field = 'deleted';
142 if ($event->hasChangedField($field)) {
143 $oldValue = $event->getOldValue($field);
144 $key = self::entityId($entity, $field);
145 $this->preUpdateValues[$key] = $oldValue;
146 }

153 {
154 $field = 'registration';

157 if (array_key_exists($key, $this->preUpdateValues)) {
158 $this->entityManager->dispatchEvent(
159 new Events\PostChangeRegistrationConfirmation(

161 !empty($this->preUpdateValue[$field]),
162 )
163 );
164 unset($this->preUpdateValues[$key]);
165 }
166 $field = 'deleted';
167 $key = self::entityId($entityId, $field);
168 if (array_key_exists($key, $this->preUpdateValues)) {
169 // This is just like persist and remove with the respective actions to perform
170 if (empty($this->preUpdateValues[$key]) && !$entity->isDeleted()) {
171 $this->registerPreCommitAction($entity, remove: false);
172 } elseif ($entity->isDeleted() && empty($this->preUpdateValues[$key])) {
173 $this->registerPreCommitAction($entity, remove: true);

241 private function registerPreCommitAction(Entity $entity, bool $remove):void
242 {
243 $entityId = self::entityId($entity);
244 if (!empty($this->preCommitActions[$entityId])) {
245 // avoid recursion
246 return;
247 }
248 $this->preCommitActions[$entityId] = true;
249
251 $contactsService = $this->appContainer->get(ContactsService::class);
252 $musician = $entity->getMusician();
253 switch ($contactsService->registerContactSynchronization($musician)) {
254 case null:
255 $this->logInfo('Contacts synchronization is already scheduled for ' . $musician->getPublicName());
256 break;
257 case false:
258 $this->logError('Contacts-synchronization could not be registered for ' . $musician->getPublicName());
259 break;
260 default:
261 // be silent
262 break;
263 }
264
266 $cloudConfig = $this->appContainer->get(IConfig::class);
267
268 $appName = $this->appContainer->get('appName');
269
270 $orchestraGroupId = $cloudConfig->getAppValue($appName, ConfigConstants::USER_GROUP_KEY);
271 if (empty($orchestraGroupId)) {
272 return;
273 }
274
276 $encryptionService = $this->appContainer->get(EncryptionService::class);
277 $executiveBoardProjectId = $encryptionService->getConfigValue(ConfigConstants::EXECUTIVE_BOARD_PROJECT_ID_KEY, -1);
278 if ($entity->getParticipationContext() == ParticipationContext::ASSOCIATES
279 || $entity->getProject()->getId() != $executiveBoardProjectId) {
280 // nothing to do
281 return;
282 }
283
285 $groupManager = $this->appContainer->get(IGroupManager::class);
286
287 $managementGroupId = $orchestraGroupId . AuthorizationService::MANAGEMENT_GROUP_SUFFIX;
289 $managementGroup = $groupManager->get($managementGroupId);
291 $orchestraGroup = $groupManager->get($orchestraGroupId);
292
294 $userManager = $this->appContainer->get(IUserManager::class);
295 $userId = $musician->getUserIdSlug();
296 $user = $userManager->get($userId);
297
298 $closureArguments = compact([
299 'userId',
300 'user',
301 'managementGroup',
302 'orchestraGroup',
303 'groupManager',
304 'userManager',
305 ]);
306
307 $this->entityManager->registerPostCommitAction(
308 new GenericUndoable(
309 function() use ($closureArguments) {
310 extract($closureArguments);
311 $previousStatus = [
312 'orchestraGroup' => $orchestraGroup->inGroup($user),
313 'managementGroup' => $managementGroup->inGroup($user),
314 ];
315 if ($remove) {
316 $managementGroup->removeUser($user);
317 } else {
318 $managementGroup->addUser($user);
319 $orchestraGroup->addUser($user);
320 }
321 return $previousStatus;
322 },
323 function(array $previousStatus) use ($closureArguments) {
324 extract($closureArguments);
325 foreach ($previousStatus as $groupKey => $status) {
327 $group = $closureArguments[$groupKey];
328 if ($status && !$group->inGroup($user)) {
329 $group->addUser($user);
330 }
331 if (!$status && $group->inGroup($user)) {
332 $group->removeUser($user);
333 }
334 }
335 },
336 ),
337 );
338 }
