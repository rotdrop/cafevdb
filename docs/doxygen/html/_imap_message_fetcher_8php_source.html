108 public function fetchMessage(?Horde_Imap_Client_Data_Fetch $fetch = null): IMAPMessage {
109 $ids = new Horde_Imap_Client_Ids($this->uid);
110
111 $isSigned = false;
112 $isEncrypted = false;
113
114 if ($this->loadBody) {
115 // Ignore given query because lots of data needs to be fetched anyway
116 // TODO: reuse given query if beneficial for performance and worth the refactoring effort
117 $query = new Horde_Imap_Client_Fetch_Query();
118 $query->envelope();
119 $query->structure();
120 $query->flags();
121 $query->imapDate();
122 $query->headerText([
123 'peek' => true,
124 ]);
125
126 $headers = $this->client->fetch($this->mailbox, $query, ['ids' => $ids]);
128 $fetch = $headers[$this->uid];
129 if (is_null($fetch)) {
130 throw new DoesNotExistException("This email ($this->uid) can't be found. Probably it was deleted from the server recently. Please reload.");
131 }
132
133 // analyse the body part
134 $structure = $fetch->getStructure();
135
136 $this->hasAnyAttachment = $this->hasAttachments($structure);
137
138 $isEncrypted = $this->isEncrypted($fetch);
139 if ($isEncrypted) {
140 throw new UnexpectedValueException('Encrypted messages are not supported.');
141 }
142
143 // debugging below
144 $structureType = $structure->getPrimaryType();
145 if ($structureType === 'multipart') {
146 $i = 1;
147 foreach ($structure->getParts() as $p) {
148 $this->getPart($p, (string)$i++, $isEncrypted || $isSigned);
149 }
150 } else {
151 $bodyPartId = $structure->findBody();
152 if (!is_null($bodyPartId)) {
153 $this->getPart($structure[$bodyPartId], $bodyPartId, $isEncrypted || $isSigned);
154 }
155 }
156 } elseif (is_null($fetch)) {
157 // Reuse given query or construct a new minimal one
158 $query = new Horde_Imap_Client_Fetch_Query();
159 $query->envelope();
160 $query->flags();
161 $query->imapDate();
162 $query->headerText([
163 'peek' => true,
164 ]);
165
166 $result = $this->client->fetch($this->mailbox, $query, ['ids' => $ids]);
167 $fetch = $result[$this->uid];
168 if (is_null($fetch)) {
169 throw new DoesNotExistException("This email ($this->uid) can't be found. Probably it was deleted from the server recently. Please reload.");
170 }
171 }
172
173 $this->parseHeaders($fetch);
174
175 $envelope = $fetch->getEnvelope();
176 return new IMAPMessage(
177 $this->uid,
178 $envelope->message_id,
179 $fetch->getFlags(),
180 AddressList::fromHorde($envelope->from),
181 AddressList::fromHorde($envelope->to),
182 AddressList::fromHorde($envelope->cc),

215 private function getPart(Horde_Mime_Part $p, string $partNo, bool $isFetched): void {
216 // iMIP messages
217 // Handle text/calendar parts first because they might be attachments at the same time.
218 // Otherwise, some of the following if-conditions might break the handling and treat iMIP
219 // data like regular attachments.
220 $allContentTypeParameters = $p->getAllContentTypeParameters();
221 if ($p->getType() === 'text/calendar') {
222 // Handle event data like a regular attachment
223 // Outlook doesn't set a content disposition
224 // We work around this by checking for the name only
225 if ($p->getName() !== null) {
226 $this->attachments[] = [
227 'id' => $p->getMimeId(),
228 'messageId' => $this->uid,
229 'fileName' => $p->getName(),
230 'mime' => $p->getType(),
231 'size' => $p->getBytes(),
232 'cid' => $p->getContentId(),
233 'disposition' => $p->getDisposition()
234 ];
235 }
236
237 // return if this is an event attachment only
238 // the method parameter determines if this is a iMIP message
239 if (!isset($allContentTypeParameters['method'])) {
240 return;
241 }
242
243 if (in_array(strtoupper($allContentTypeParameters['method']), ['REQUEST', 'REPLY', 'CANCEL'])) {
244 $this->scheduling[] = [
245 'id' => $p->getMimeId(),
246 'messageId' => $this->uid,
247 'method' => strtoupper($allContentTypeParameters['method']),
248 'contents' => $this->loadBodyData($p, $partNo, $isFetched),
249 ];
250 return;
251 }
252 }
253
254 // Regular attachments
255 if ($p->isAttachment() || $p->getType() === 'message/rfc822') {
256 $this->attachments[] = [
257 'id' => $p->getMimeId(),
258 'messageId' => $this->uid,
259 'fileName' => $p->getName(),
260 'mime' => $p->getType(),
261 'size' => $p->getBytes(),
262 'cid' => $p->getContentId(),
263 'disposition' => $p->getDisposition()
264 ];
265 return;
266 }
267
268 // Inline attachments
269 // Horde doesn't consider parts with content-disposition set to inline as
270 // attachment so we need to use another way to get them.
271 // We use these inline attachments to render a message's html body in $this->getHtmlBody()
272 $filename = $p->getName();
273 if ($p->getType() === 'message/rfc822' || isset($filename)) {
274 if (in_array($filename, $this->attachmentsToIgnore)) {
275 return;
276 }
277 $this->inlineAttachments[] = [
278 'id' => $p->getMimeId(),
279 'messageId' => $this->uid,
280 'fileName' => $filename,
281 'mime' => $p->getType(),
282 'size' => $p->getBytes(),
283 'cid' => $p->getContentId()
284 ];
285 return;
286 }
287
288 if ($p->getPrimaryType() === 'multipart') {
289 $this->handleMultiPartMessage($p, $partNo, $isFetched);
290 return;
291 }
292
293 if ($p->getType() === 'text/plain') {
294 $this->handleTextMessage($p, $partNo, $isFetched);
295 return;
296 }
297
298 if ($p->getType() === 'text/html') {
299 $this->handleHtmlMessage($p, $partNo, $isFetched);
300 return;
301 }
302
303 // EMBEDDED MESSAGE
304 // Many bounce notifications embed the original message as type 2,
305 // but AOL uses type 1 (multipart), which is not handled here.
306 // There are no PHP functions to parse embedded messages,
307 // so this just appends the raw source to the main message.
308 if ($p[0] === 'message') {
309 $data = $this->loadBodyData($p, $partNo, $isFetched);
310 $this->plainMessage .= trim($data) . "\n\n";
311 }
312 }

373 private function loadBodyData(Horde_Mime_Part $p, string $partNo, bool $isFetched): string {
374 if (!$isFetched) {
375 $fetch_query = new Horde_Imap_Client_Fetch_Query();
376 $ids = new Horde_Imap_Client_Ids($this->uid);
377
378 $fetch_query->bodyPart($partNo, [
379 'peek' => true
380 ]);
381 $fetch_query->bodyPartSize($partNo);
382 $fetch_query->mimeHeader($partNo, [
383 'peek' => true
384 ]);
385
386 $headers = $this->client->fetch($this->mailbox, $fetch_query, ['ids' => $ids]);
387 /* @var $fetch Horde_Imap_Client_Data_Fetch */
388 $fetch = $headers[$this->uid];
389 if (is_null($fetch)) {
390 throw new DoesNotExistException("Mail body for this mail($this->uid) could not be loaded");
391 }
392
393 $mimeHeaders = $fetch->getMimeHeader($partNo, Horde_Imap_Client_Data_Fetch::HEADER_PARSE);
394 if ($enc = $mimeHeaders->getValue('content-transfer-encoding')) {
395 $p->setTransferEncoding($enc);
396 }
397
398 $data = $fetch->getBodyPart($partNo);
399
400 $p->setContents($data);
401 }
402
403 $data = $p->getContents();
404 if ($data === null) {
405 return '';
406 }
407
408 // Only convert encoding if it is explicitly specified in the header because text/calendar
409 // data is utf-8 by default.
410 $charset = $p->getContentTypeParameter('charset');
411 if ($charset !== null && strtoupper($charset) !== 'UTF-8') {
412 $data = mb_convert_encoding($data, 'UTF-8', $charset);
413 }
414 return (string)$data;
415 }

430 private function decodeSubject(Horde_Imap_Client_Data_Envelope $envelope): string {
431 // Try a soft conversion first (some installations, eg: Alpine linux,
432 // have issues with the '//IGNORE' option)
433 $subject = $envelope->subject;
434 $utf8 = iconv('UTF-8', 'UTF-8', $subject);
435 if ($utf8 !== false) {
436 return $utf8;
437 }
438 return iconv("UTF-8", "UTF-8//IGNORE", $subject);
439 }

441 private function parseHeaders(Horde_Imap_Client_Data_Fetch $fetch): void {
443 $headersStream = $fetch->getHeaderText('0', Horde_Imap_Client_Data_Fetch::HEADER_STREAM);
444 $this->rawHeaders = stream_get_contents($headersStream);
445 rewind($headersStream);
446 $parsedHeaders = Horde_Mime_Headers::parseHeaders($headersStream);
447 fclose($headersStream);
448
449 $references = $parsedHeaders->getHeader('references');
450 if ($references !== null) {
451 $this->rawReferences = $references->value_single;
452 }
453
454 $dispositionNotificationTo = $parsedHeaders->getHeader('disposition-notification-to');
455 if ($dispositionNotificationTo !== null) {
456 $this->dispositionNotificationTo = $dispositionNotificationTo->value_single;
457 }
458
459 $dkimSignatureHeader = $parsedHeaders->getHeader('dkim-signature');
460 $this->hasDkimSignature = $dkimSignatureHeader !== null;
461
462 $listUnsubscribeHeader = $parsedHeaders->getHeader('list-unsubscribe');
463 if ($listUnsubscribeHeader !== null) {
464 $listHeaders = new Horde_ListHeaders();
465 $headers = $listHeaders->parse($listUnsubscribeHeader->name, $listUnsubscribeHeader->value_single);
466 if (!$headers) {
467 // Unable to parse headers
468 return;
469 }
470 foreach ($headers as $header) {
471 if (str_starts_with($header->url, 'http')) {
472 $this->unsubscribeUrl = $header->url;
473
474 $unsubscribePostHeader = $parsedHeaders->getHeader('List-Unsubscribe-Post');
475 if ($unsubscribePostHeader !== null) {
476 $this->isOneClickUnsubscribe = strtolower($unsubscribePostHeader->value_single) === 'list-unsubscribe=one-click';
477 }
478 break;
479 }
480 if (str_starts_with($header->url, 'mailto')) {
481 $this->unsubscribeMailto = $header->url;
482 break;
483 }
484 }
485 }

488 private function isEncrypted(Horde_Imap_Client_Data_Fetch $message): bool {
489 $headers = $message->getHeaderText('0', Horde_Imap_Client_Data_Fetch::HEADER_PARSE);
490 if (!isset($headers['content-type'])) {
491 return false;
492 }
493
495 $contentType = $headers['content-type'];
496 return $contentType->ptype === 'application'
497 && str_ends_with($contentType->stype, 'pkcs7-mime')
498 && isset($contentType['smime-type'])
499 && $contentType['smime-type'] === 'enveloped-data';
500 }
