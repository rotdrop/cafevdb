330 public function getSelectableOptions(bool $includeDeleted = false):Collection
331 {
332 // this unfortunately just does not work.
333 // return $this->dataOptions->matching(DBUtil::criteriaWhere([ '!key' => Uuid::NIL, 'deleted' => null, ]));
334 $filtered = $this->dataOptions->filter(function($option) use ($includeDeleted) {
336 return ($includeDeleted || empty($option->getDeleted()))
337 && (string)$option->getKey() != ProjectParticipantFieldDataOption::GENERATOR_KEY;
338 });
339 $iterator = $filtered->getIterator();
340 $iterator->uasort(function(ProjectParticipantFieldDataOption $a, ProjectParticipantFieldDataOption $b) {
341 $cmp = strcmp($a->getLabel(), $b->getLabel());
342 return $cmp !== 0 ? $cmp : strcmp($a->getKey(), $b->getKey());
343 });
344 return new ArrayCollection(iterator_to_array($iterator));
