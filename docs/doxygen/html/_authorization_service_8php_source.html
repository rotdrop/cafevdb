197 public function getUserPermissions(?string $userId = null):int
198 {
199 if (empty($userId)) {
200 $userId = $this->userId;
201 }
202 if (empty($userId) || empty($this->userGroupId)) {
203 return self::PERMISSION_NONE;
204 }
205 $userPermissions = self::PERMISSION_NONE;
206 foreach (self::GROUP_SUFFIX_LIST as $permissions => $suffix) {
207 $groupId = $this->userGroupId . $suffix;
208 if ($this->groupManager->isInGroup($userId, $groupId)) {
209 $userPermissions |= $permissions|self::IMPLIED_PERMISSIONS[$permissions];
210 }
211 if ($userPermissions == self::PERMISSION_ALL) {
212 break;
213 }
214 }
215 $this->logPermissions($userId, $userPermissions, \OCP\ILogger::DEBUG);
216 return $userPermissions;
217 }
