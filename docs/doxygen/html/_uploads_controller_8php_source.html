145 #[CoreAttributes\NoAdminRequired]
146 #[CoreAttributes\FrontpageRoute(verb: 'POST', url: '/' . self::END_POINT_STASH)]
147 public function stash(
148 array $cloudPaths = [],
149 string|EnumFileUploadMode $uploadMode = EnumFileUploadMode::COPY,
150 string $uploadFolder = AppStorage::UPLOAD_FOLDER,
151 ): DataResponse {
152 $uploadMaxFileSize = \OCP\Util::computerFileSize(ini_get('upload_max_filesize'));
153 $postMaxSize = \OCP\Util::computerFileSize(ini_get('post_max_size'));
154 $maxUploadFileSize = min($uploadMaxFileSize, $postMaxSize);
155 $maxHumanFileSize = \OCP\Util::humanFileSize($maxUploadFileSize);
156
157 $uploadMode = EnumFileUploadMode::get($uploadMode);
158
159 $uploads = [];
160 if (!empty($cloudPaths)) {
161 foreach ($cloudPaths as $path) {
162 $files = [];
163
165 $cloudFile = $this->userStorage->getFile($path);
166
168 $dbFile = $this->userStorage->getDatabaseFile($path);
169 if (!empty($dbFile) && !($dbFile instanceof Entities\DatabaseStorageFile)) {
170 return self::grumble($this->l->t('File "%s" does not exist or refers to a database storage folder, we expected a plain file.', $path));
171 }
172 if (empty($cloudFile)) {
173 if ($this->userStorage->getFolder($path)) {
174 return self::grumble($this->l->t('The path "%s/" refers to a folder. This is not yet implemented.', $path));
175 }
176 return self::grumble($this->l->t('File "%s" could not be found in cloud storage.', $path));
177 }
178
179 $fileName = $cloudFile->getName();
180 $fileEntityId = !empty($dbFile) ? $dbFile->getFile()->getId() : null;
181
182 switch ($uploadMode) {
183 case EnumFileUploadMode::TEST:
184 $uploadModes = [ EnumFileUploadMode::COPY, ];
185 if ($cloudFile->getPermissions() & CloudConstants::PERMISSION_DELETE) {
186 $uploadModes[] = EnumFileUploadMode::MOVE;
187 }
188 if (!empty($dbFile)) {
189 $uploadModes[] = EnumFileUploadMode::LINK;
190 }
191 $uploads[] = new DTO\UploadModeTest(
192 originalName: $path,
193 availableUploadModes: $uploadModes,
194 );
195 continue 2;
196 case EnumFileUploadMode::LINK:
197 if (empty($dbFile)) {
198 return self::grumble($this->l->t(
199 'File "%s" is not backed by database-storage and thus cannot be linked.',
200 $cloudFile->getName()
201 ));
202 }
203 $originalName = $fileEntityId;
204 $tmpName = null;
205 break;
206 case EnumFileUploadMode::MOVE:
207 if (!($cloudFile->getPermissions() & CloudConstants::PERMISSION_DELETE)) {
208 return self::grumble($this->l->t(
209 'File "%s" cannot be deleted, moving it is therefor not possible.',
210 $fileName
211 ));
212 }
213 // the actual deletion should be post-poned until the stashed file
214 // has been moved into place.
215 // no break
216 case EnumFileUploadMode::COPY:
217 try {
218 $uploadFile = $this->appStorage->newTemporaryFile($uploadFolder);
219 $uploadFile->putContent($cloudFile->getContent());
220 } catch (Throwable $t) {
221 return self::grumble($this->l->t('Could not copy cloud file "%s" to upload storage.', $fileName));
222 }
223 $originalName = $uploadMode == EnumFileUploadMode::MOVE ? $path : $fileName;
224 $tmpName = $uploadFile->getName();
225 break;
226 }
227
228 // We emulate an uploaded file here:
229 $fileRecord = [
230 'name' => $fileName,
231 'error' => UPLOAD_ERR_OK,
232 'str_error' => Util::fileUploadError(UPLOAD_ERR_OK, $this->l),
233 'tmp_name' => $tmpName,
234 'type' => $cloudFile->getMimetype(),
235 'size' => $cloudFile->getSize(),
236 'upload_max_file_size' => $maxUploadFileSize,
237 'max_human_file_size' => $maxHumanFileSize,
238 'upload_mode' => $uploadMode,
239 'original_name' => $originalName,
240 ];
241 $uploads[] = DTO\UploadFileData::fromArray($fileRecord);
242 }
243 } else {
244 if ($uploadMode != EnumFileUploadMode::COPY) {
245 return self::grumble($this->l->t(
246 'For client-uploads the only supported upload-mode is "copy", "%s" is not possible.',
247 $uploadMode
248 ));
249 }
250
251 $files = $this->request->files[self::UPLOAD_KEY];
252 if (empty($files)) {
253 // may be caused by PHP restrictions which are not caught by
254 // error handlers.
255 $contentLength = $this->request->server['CONTENT_LENGTH'];
256 $limit = \OCP\Util::uploadLimit();
257 if ($contentLength > $limit) {
258 return self::grumble(
259 $this->l->t('Upload size %s exceeds limit %s, contact your server administrator.', [
260 \OCP\Util::humanFileSize($contentLength),
261 \OCP\Util::humanFileSize($limit),
262 ]));
263 }
264 $error = error_get_last();
265 if (!empty($error)) {
266 return self::grumble(
267 $this->l->t('No file was uploaded, error message was "%s".', $error['message']));
268 }
269 return self::grumble($this->l->t('No file was uploaded. Unknown error'));
270 }
271
272 $files = Util::transposeArray($files);
273
274 $totalSize = 0;
275 foreach ($files as $file) {
276
277 $totalSize += $file['size'];
278
279 if ($maxUploadFileSize >= 0 and $totalSize > $maxUploadFileSize) {
280 return self::grumble([
281 'message' => $this->l->t('Not enough storage available'),
282 'upload_max_file_size' => $maxUploadFileSize,
283 'max_human_file_size' => $maxHumanFileSize,
284 ]);
285 }
286
287
288 $file['str_error'] = Util::fileUploadError($file['error'], $this->l);
289 if ($file['error'] != UPLOAD_ERR_OK) {
290 continue;
291 }
292
293 $file['upload_max_file_size'] = $maxUploadFileSize;
294 $file['max_human_file_size'] = $maxHumanFileSize;
295 $file['original_name'] = $file['name']; // clone
296 $file['upload_mode'] = EnumFileUploadMode::COPY;
297
298 try {
299 $uploadFile = $this->appStorage->newTemporaryFile($uploadFolder);
300 $this->appStorage->moveFileSystemFile($file['tmp_name'], $uploadFile);
301 $file['name'] = $uploadFile->getName();
302 $file['tmp_name'] = $file['name'];
303 } catch (Throwable $t) {
304 $file['error'] = 99;
305 $file['str_error'] = $this->l->t('Couldn\'t save temporary file for: %s', $file['name']);
306 continue;
307 }
308 $uploads[] = DTO\UploadFileData::fromArray($file);
309 }
310 }
311 return new DataResponse($uploads);
312 }
