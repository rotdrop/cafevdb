186 private function matchCIDR(string $ip, string $range): bool
187 {
188 list($subnet, $bits) = array_pad(explode('/', $range), 2, null);
189 $subnetIsIPv6 = $this->domainNameService->isIPv6($subnet);
190
191 if ($bits === null) {
192 $bits = $subnetIsIPv6 ? 128 : 32;
193 }
194 $bits = (int)$bits;
195
196 if ($this->domainNameService->isIpv4($ip) && !$subnetIsIPv6) {
197 $mask = -1 << (32 - $bits);
198
199 $ip = ip2long($ip);
200 $subnet = ip2long($subnet);
201
202 return ($ip & $mask) === ($subnet & $mask);
203 }
204
205 if ($this->domainNameService->isIpv6($ip) && $subnetIsIPv6) {
206 $subnet = inet_pton($subnet);
207 $ip = inet_pton($ip);
208
209 $binMask = str_repeat("f", (int)($bits / 4));
210 switch ($bits % 4) {
211 case 0:
212 break;
213 case 1:
214 $binMask .= "8";
215 break;
216 case 2:
217 $binMask .= "c";
218 break;
219 case 3:
220 $binMask .= "e";
221 break;
222 }
223
224 $binMask = str_pad($binMask, 32, '0');
225 $binMask = pack("H*", $binMask);
226
227 return ($ip & $binMask) === ($subnet & $binMask);
228 }
229 return false;
