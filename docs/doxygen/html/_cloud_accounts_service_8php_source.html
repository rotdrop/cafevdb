131 public function addUserToBackend(IUser $user, string|UserInterface $newBackend): bool
132 {
133 if (is_string($newBackend)) {
134 $newBackend = array_filter(
135 $this->userManager->getBackends(),
136 fn(IUserBackend $backend) => $backend->getBackendName() == $newBackend,
137 );
138 if (count($newBackend) != 1) {
139 return false;
140 }
141 $newBackend = reset($newBackend);
142 }
143
144 $oldBackend = $user->getBackend();
145
146 if (!($oldBackend instanceof IUserBackend) || !($newBackend instanceof IUserBackend)) {
147 $this->logError('Only support moving between sane user-backends.');
148 return false;
149 }
150
151 $oldBackendName = $oldBackend->getBackendName();
152 $newBackendName = $newBackend->getBackendName();
153
154 if ($oldBackendName == $newBackendName) {
155 return true;
156 }
157
158 $email = $user->getSystemEMailAddress() ?? null;
159
160 if (empty($email)) {
161 // This also means that the setup is somehow totally broken ...
162 $this->logError('User "' . $userId . '" does not have an email.');
163 return false;
164 }
165
166 $userId = $user->getUID();
167 $displayName = $user->getDisplayName();
168 $quota = $user->getQuota();
169
170 // try to transfer the avatar
172 $avatar = $this->avatarManager->getAvatar($userId);
173 if ($avatar->isCustomAvatar()) {
174 $avatarImage = $avatar->get(-1);
175 }
176
177 $newUser = $this->doAddUserToBackend(
178 userId: $userId,
179 backend: $newBackend,
180 displayName: $displayName,
181 email: $email,
182 quota: $quota,
183 avatarImage: $avatarImage ?? null,
184 );
185
186 if (empty($newUser)) {
187 $this->logger->error('Failed to transfer "' . $userId . '" to the user-backend "' . $newBackend->getBackendName());
188 return false;
189 }
190
191 // If this has succeeded, perhaps the user should be deleted in the old backend, if possible
192 try {
194 $oldBackend->deleteUser($user->getUID());
195 $this->logInfo('Deleted user "' . $userId . '" in old backend "' . $oldBackendName . '"');
196 } catch (Throwable $t) {
197 $this->logException($t, 'Cannot delete user "' . $userId . '" in old backend "' . $oldBackendName . '"');
198 }
199
200 return true;
201 }

218 protected function doAddUserToBackend(
219 string $userId,
220 UserInterface $backend,
221 string $displayName,
222 string $email,
223 string $quota,
224 ?IAvatar $avatarImage,
225 ):?IUser {
226 $backendName = $backend->getBackendName();
227 if ($backend->userExists($userId)) {
228 $this->logWarn('User "' . $userId . '" already exists in backend "' . $backendName . '"');
229 if (!($this->userManager instanceof \OC\User\Manager) || !method_exists($this->userManager, 'getUserObject')) {
230 $this->logError('The IUserManager implementation seems to have changed, cannot retrieve the user object from the backend.');
231 return null;
232 }
234 $userManager = $this->userManager;
235 return $userManager->getUserObject($userId, $backend);
236 }
237
238 $passwordEvent = new GenerateSecurePasswordEvent();
239 $this->eventDispatcher->dispatchTyped($passwordEvent);
240 $password = $passwordEvent->getPassword();
241
242 if ($password === null) {
243 // Fallback: ensure to pass password_policy in any case
244 $password = $this->secureRandom->generate(10)
245 . $this->secureRandom->generate(1, ISecureRandom::CHAR_UPPER)
246 . $this->secureRandom->generate(1, ISecureRandom::CHAR_LOWER)
247 . $this->secureRandom->generate(1, ISecureRandom::CHAR_DIGITS)
248 . $this->secureRandom->generate(1, ISecureRandom::CHAR_SYMBOLS);
249 }
250 $generatePasswordResetToken = true;
251
252 try {
253 $newUser = $this->userManager->createUserFromBackend($userId, $password, $backend);
254
255 $this->logger->info('Successful addUser call with userid: ' . $userId, ['app' => 'ocs_api']);
256
257 $newUser->setDisplayName($displayName);
258 $newUser->setQuota($quota);
259
260 if (!empty($avatarImage)) {
261 try {
262 $avatar = $this->avatarManager->getAvatar($userId);
263 $avatar->set($avatarImage);
264 } catch (Throwable $t) {
265 $this->logException($t, 'Unable to restore the avatar image after moving "' . $userId . '" to the new backend "' . $backendName . '".');
266 }
267 }
268
269 $newUser->setSystemEMailAddress($email);
270 if ($this->cloudConfig->getAppValue('core', 'newUser.sendEmail', 'yes') === 'yes') {
271 try {
272 $emailTemplate = $this->newUserMailHelper->generateTemplate($newUser, $generatePasswordResetToken);
273 $this->newUserMailHelper->sendMail($newUser, $emailTemplate);
274 } catch (Throwable $e) {
275 // Mail could be failing hard or just be plain not configured
276 // Logging error as it is the hardest of the two
277 $this->logger->error(
278 "Unable to send the invitation mail to $email",
279 [
280 'app' => 'ocs_api',
281 'exception' => $e,
282 ]
283 );
284 }
285 }
286 return $newUser;
287 } catch (HintException $e) {
288 $this->logger->warning(
289 'Failed addUser attempt with hint exception.',
290 [
291 'app' => 'ocs_api',
292 'exception' => $e,
293 ]
294 );
295 return null;
296 } catch (InvalidArgumentException $e) {
297 $this->logger->error(
298 'Failed addUser attempt with invalid argument exception.',
299 [
300 'app' => 'ocs_api',
301 'exception' => $e,
302 ]
303 );
304 return null;
305 } catch (Throwable $e) {
306 $this->logger->error(
307 'Failed addUser attempt with exception.',
308 [
309 'app' => $this->appName,
310 'exception' => $e
311 ]
312 );
313 return null;
314 }

357 public function promoteGroupMembership(IUser $user, IGroup $group, array $requiredBackends = [])
358 {
359 if (empty($requiredBackends)) {
360 $requiredBackends = [ 'Database' ];
361 $orchestraUserAndGroupBackend = $this->cloudConfig->getAppValue(
362 $this->appName,
363 ConfigConstants::USER_AND_GROUP_BACKEND_KEY,

366 if ($orchestraUserAndGroupBackend !== null) {
367 $requiredBackends[] = $orchestraUserAndGroupBackend;

390 $this->logException($t, 'Backend ' . get_class($backend) . ' cannot add user ' . $user->getUID());
391 }

416 public function ensureGroupBackends(IGroup $group, array $requiredBackends = []):array
417 {
418 $this->logInfo('TRY ENSURE GROUP BACKENDS');
419 if (empty($requiredBackends)) {
420 $requiredBackends = [ 'Database' ];
421 $orchestraUserAndGroupBackend = $this->cloudConfig->getAppValue(
422 $this->appName,
423 ConfigConstants::USER_AND_GROUP_BACKEND_KEY,
424 null,
425 );
426 if ($orchestraUserAndGroupBackend !== null) {
427 $requiredBackends[] = $orchestraUserAndGroupBackend;
428 }
429 }
430
431 $status = array_fill_keys($requiredBackends, true); // set to \false on failure below
432 $gid = $group->getGID();
433 $displayName = $group->getDisplayName();
434 $groupBackendNames = $group->getBackendNames();
435 $missingBackends = array_diff($requiredBackends, $groupBackendNames);
437 foreach ($this->groupManager->getBackends() as $backend) {
438 if (!($backend instanceof INamedBackend) || !($backend instanceof GroupInterface)) {
439 $this->logInfo('SKIPPING BACKEND ' . get_class($backend));
440 continue;
441 }
442 $backendName = $backend->getBackendName();
443 if (!in_array($backendName, $missingBackends)) {
444 continue;
445 }
446 if (!$backend->implementsActions(GroupInterface::CREATE_GROUP)) {
447 $this->logError('Group ' . $gid . ' is missing in backend "' . $backendName . '", but the backend is not able to create groups.');
448 $status[$backendName] = false;
449 continue;
450 }
451 if ($backend->groupExists($gid)) {
452 $this->logError('Backend "' . $backendName . '" already provides the group "' . $gid . '", but the cloud group does not reflect this.');
453 $status[$backendName] = false;
454 continue;
455 } elseif (!$backend->createGroup($gid)) {
456 $this->logError('Group ' . $gid . ' is missing in backend "' . $backendName . '", but the backend failed to create it.');
457 $status[$backendName] = false;
458 continue;
459 }
460 if (!$backend->groupExists($gid)) {
461 $this->logError('Backend "' . $backendName . '" did not error out creating the group "' . $gid . '" but the group does not exist in the backend.');
462 $status[$backendName] = false;
463 continue;
464 }
465 if (!($backend instanceof ISetDisplayNameBackend)) {
466 continue;
467 }
468 if (($backend instanceof IGetDisplayNameBackend) && $backend->getDisplayName($gid) == $displayName) {
469 continue;
470 }
471 $backend->setDisplayName($gid, $displayName);
472 // ignore the result, not so important
473 }
474 if (($this->groupManager instanceof \OC\Group\Manager) && method_exists($this->groupManager, 'clearCaches')) {
475 new ReflectionMethod($this->groupManager, 'clearCaches')->invoke($this->groupManager);
476 }
477 return $status;
478 }
