134 public function addUserToBackend(IUser $user, string|UserInterface $newBackend)
135 {
136 if (is_string($newBackend)) {
137 $newBackend = array_filter(
138 $this->userManager->getBackends(),
139 fn(IUserBackend $backend) => $backend->getBackendName() == $newBackend,
140 );
141 if (count($newBackend) != 1) {
142 return false;
143 }
144 $newBackend = reset($newBackend);
145 }
146
147 $oldBackend = $user->getBackend();
148
149 if (!($oldBackend instanceof IUserBackend) || !($newBackend instanceof IUserBackend)) {
150 $this->logError('Only support moving between sane user-backends.');
151 return false;
152 }
153
154 $oldBackendName = $oldBackend->getBackendName();
155 $newBackendName = $newBackend->getBackendName();
156
157 if ($oldBackendName == $newBackendName) {
158 return true;
159 }
160
161 $email = $user->getEMailAddress() ?? '';
162
163 if ($email === '') {
164 // This also means that the setup is somehow totally broken ...
165 $this->logError('User "' . $userId . '" does not have an email.');
166 return false;
167 }
168
169 $userId = $user->getUID();
170 $displayName = $user->getDisplayName();
171 $quota = $user->getQuota();
172
173 // try to transfer the avatar
175 $avatar = $this->avatarManager->getAvatar($userId);
176 if ($avatar->isCustomAvatar()) {
177 $avatarImage = $avatar->get(-1);
178 }
179
180 $newUser = $this->doAddUserToBackend(
181 userId: $userId,
182 UserInterface: $newBackend,
183 displayName: $displayName,
184 email: $email,
185 quota: $quota,
186 avatarImage: $avatarImage ?? null,
187 );
188
189 if (empty($newUser)) {
190 $this->logger->error('Failed to transfer "' . $userId . '" to the user-backend "' . $newBackend->getBackendName());
191 return false;
192 }
193
194 // If this has succeeded, perhaps the user should be deleted in the old backend, if possible
195 try {
197 $oldBackend->deleteUser($user->getUID());
198 $this->logInfo('Deleted user "' . $userId . '" in old backend "' . $oldBackendName . '"');
199 } catch (Throwable $t) {
200 $this->logException($t, 'Cannot delete user "' . $userId . '" in old backend "' . $oldBackendName . '"');
201 }
202
203 return true;
204 }

221 protected function doAddUserToBackend(
222 string $userId,
223 UserInterface $backend,
224 string $displayName,
225 string $email,
226 string $quota,
227 ?IAvatar $avatarImage,
228 ):?IUser {
229 $backendName = $backend->getBackendName();
230 if ($backend->userExists($userId)) {
231 $this->logWarn('User "' . $userId . '" already exists in backend "' . $backendName . '"');
232 if (!($this->userManager instanceof \OC\User\Manager) || !method_exists($this->userManager, 'getUserObject')) {
233 $this->logError('The IUserManager implementation seems to have changed, cannot retrieve the user object from the backend.');
234 return null;
235 }
237 $userManager = $this->userManager;
238 return $userManager->getUserObject($userId, $backend);
239 }
240
241 $passwordEvent = new GenerateSecurePasswordEvent();
242 $this->eventDispatcher->dispatchTyped($passwordEvent);
243 $password = $passwordEvent->getPassword();
244
245 if ($password === null) {
246 // Fallback: ensure to pass password_policy in any case
247 $password = $this->secureRandom->generate(10)
248 . $this->secureRandom->generate(1, ISecureRandom::CHAR_UPPER)
249 . $this->secureRandom->generate(1, ISecureRandom::CHAR_LOWER)
250 . $this->secureRandom->generate(1, ISecureRandom::CHAR_DIGITS)
251 . $this->secureRandom->generate(1, ISecureRandom::CHAR_SYMBOLS);
252 }
253 $generatePasswordResetToken = true;
254
255 try {
256 $newUser = $this->userManager->createUserFromBackend($userId, $password, $backend);
257
258 $this->logger->info('Successful addUser call with userid: ' . $userId, ['app' => 'ocs_api']);
259
260 $newUser->setDisplayName($displayName);
261 $newUser->setQuota($quota);
262
263 if (!empty($avatarImage)) {
264 try {
265 $avatar = $this->avatarManager->getAvatar($userId);
266 $avatar->set($avatarImage);
267 } catch (Throwable $t) {
268 $this->logException($t, 'Unable to restore the avatar image after moving "' . $userId . '" to the new backend "' . $backendName . '".');
269 }
270 }
271
272 $newUser->setEMailAddress($email);
273 if ($this->cloudConfig->getAppValue('core', 'newUser.sendEmail', 'yes') === 'yes') {
274 try {
275 $emailTemplate = $this->newUserMailHelper->generateTemplate($newUser, $generatePasswordResetToken);
276 $this->newUserMailHelper->sendMail($newUser, $emailTemplate);
277 } catch (Throwable $e) {
278 // Mail could be failing hard or just be plain not configured
279 // Logging error as it is the hardest of the two
280 $this->logger->error(
281 "Unable to send the invitation mail to $email",
282 [
283 'app' => 'ocs_api',
284 'exception' => $e,
285 ]
286 );
287 }
288 }
289 return $newUser;
290 } catch (HintException $e) {
291 $this->logger->warning(
292 'Failed addUser attempt with hint exception.',
293 [
294 'app' => 'ocs_api',
295 'exception' => $e,
296 ]
297 );
298 return null;
299 } catch (InvalidArgumentException $e) {
300 $this->logger->error(
301 'Failed addUser attempt with invalid argument exception.',
302 [
303 'app' => 'ocs_api',
304 'exception' => $e,
305 ]
306 );
307 return null;
308 } catch (Throwable $e) {
309 $this->logger->error(
310 'Failed addUser attempt with exception.',
311 [
312 'app' => $this->appName,
313 'exception' => $e
314 ]
315 );
316 return null;
317 }

328 public function promoteGroupDisplayName(IGroup $group):void
329 {
330 $gid = $group->getGID();

333 foreach ($this->groupManager->getBackends() as $backend) {
334 if (!($backend instanceof ISetDisplayNameBackend)

360 public function promoteGroupMembership(IUser $user, IGroup $group, array $requiredBackends = [])
361 {
362 if (empty($requiredBackends)) {
363 $requiredBackends = [ 'Database' ];
364 $orchestraUserAndGroupBackend = $this->cloudConfig->getAppValue(
365 $this->appName,
366 ConfigService::USER_AND_GROUP_BACKEND_KEY,
367 null,
368 );
369 if ($orchestraUserAndGroupBackend !== null) {
370 $requiredBackends[] = $orchestraUserAndGroupBackend;
371 }
372 }
373
374 $groupId = $group->getGID();
375 foreach ($this->groupManager->getBackends() as $backend) {
376 if (!$backend->implementsActions(\OC\Group\Backend::ADD_TO_GROUP)) {
377 continue;
378 }
379 if (!($backend instanceof INamedBackend)) {
380 continue;
381 }
382 $backendName = $backend->getBackendName();
383 if (!in_array($backendName, $requiredBackends)) {
384 continue;
385 }
386 $userId = $user->getUID();
387 try {
388 if (!$backend->inGroup($userId, $groupId)) {
389 $backend->addToGroup($userId, $groupId);
390 }
391 } catch (Throwable $t) {
392 // Can happen if the user does not exist in the backend
393 $this->logException($t, 'Backend ' . get_class($backend) . ' cannot add user ' . $user->getUID());
394 }
395 }

419 public function ensureGroupBackends(IGroup $group, array $requiredBackends = []):array
420 {
421 if (empty($requiredBackends)) {
422 $requiredBackends = [ 'Database' ];
423 $orchestraUserAndGroupBackend = $this->cloudConfig->getAppValue(
424 $this->appName,
425 ConfigService::USER_AND_GROUP_BACKEND_KEY,
426 null,
427 );
428 if ($orchestraUserAndGroupBackend !== null) {
429 $requiredBackends[] = $orchestraUserAndGroupBackend;
430 }
431 }
432
433 $status = array_fill_keys($requiredBackends, true); // set to \false on failure below
434 $gid = $group->getGID();
435 $displayName = $group->getDisplayName();
436 $groupBackendNames = $group->getBackendNames();
437 $missingBackends = array_diff($requiredBackends, $groupBackendNames);
438 foreach ($this->groupManager->getBackends() as $backend) {
439 if (!($backend instanceof INamedBackend)) {
440 continue;
441 }
442 $backendName = $backend->getBackendName();
443 if (!in_array($backendName, $missingBackends)) {
444 continue;
445 }
446 if (!($backend instanceof ICreateGroupBackend)) {
447 $this->logError('Group ' . $gid . ' is missing in backend "' . $backendName . '", but the backend is not able to create groups.');
448 $status[$backendName] = false;
449 continue;
450 }
451 if (!$backend->createGroup($gid)) {
452 $this->logError('Group ' . $gid . ' is missing in backend "' . $backendName . '", but the backend failed to create it.');
453 $status[$backendName] = false;
454 continue;
455 }
456 if (!$backend->groupExists($gid)) {
457 $this->logError('Backend "' . $backendName . '" did not error out creating the group "' . $gid . '" but the group does not exist in the backend.');
458 $status[$backendName] = false;
459 continue;
460 }
461 if (!($backend instanceof ISetDisplayNameBackend)) {
462 continue;
463 }
464 if (($backend instanceof IGetDisplayNameBackend) && $backend->getDisplayName($gid) == $displayName) {
465 continue;
466 }
467 $backend->setDisplayName($gid, $displayName);
468 // ignore the result, not so important
469 }
470 if (($this->groupManager instanceof \OC\Group\Manager) && method_exists($this->groupManager, 'clearCaches')) {
471 $this->groupManager->clearCaches();
472 }
473 return $status;
474 }
