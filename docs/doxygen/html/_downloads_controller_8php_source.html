89 public function fetch(string $section, string $object, array $items = [], ?string $fileName = null)
90 {
91 switch ($section) {
92 case self::SECTION_DATABASE:
93 if ($object == self::OBJECT_COLLECTION) {
94 $this->logInfo('ITEMS ' . print_r($items, true));
95
96 if (empty($fileName)) {
97 $fileName = $this->timeStamp() . '-' . $this->appName() . '-' . 'download' . '.zip';
98 }
99 $fileName = basename($fileName, '.zip');
100
101 $fileData = $this->di(DatabaseStorageUtil::class)->getCollectionArchive($items, $fileName);
102
103 return $this->dataDownloadResponse($fileData, $fileName . '.zip', 'application/zip');
104 } else {
105 $fileId = $object;
107 $file = $this->getDatabaseRepository(Entities\DatabaseStorageFile::class)->find($fileId);
108 if (empty($file)) {
109 return self::grumble($this->l->t('File width id %d not found in database-storage.', $fileId));
110 }
111 $mimeType = $file->getMimeType();
112 if (empty($fileName)) {
113 $fileName = $file->getFileName();
114 if (empty($fileName)) {
115 $fileName = $this->appName() . '-' . 'download' . $fileId;
116 }
117 }
118 return $this->dataDownloadResponse($file->getFileData()->getData(), $fileName, $mimeType);
119 }
120 case self::SECTION_FILECACHE:
121 $cacheKey = $object;
123 $fileCache = $this->di(\OCP\ICache::class);
124 $fileData = $fileCache->get($cacheKey);
125 if (empty($fileData)) {
126 return self::grumble($this->l->t('File with cache-key "%s" not found in user\'s file-cache.', $cacheKey));
127 }
128 $meta = $fileCache->get($cacheKey . '-meta');
129 if (!empty($meta)) {
130 $fileName = $meta['name']??null;
131 $mimeType = $meta['mimeType']??null;
132 }
133 if (empty($mimeType)) {
135 $mimeTypeDetector = $this->di(\OCP\Files\IMimeTypeDetector::class);
136 $mimeType = $mimeTypeDetector->detectString($fileData);
137 }
138 if (empty($fileName)) {
139 $fileName = implode('-', [
140 $this->appName(),
141 $this->userId(),
142 $cacheKey,
143 ]) . '.' . Util::fileExtensionFromMimeType($mimeType);
144 }
145 return $this->dataDownloadResponse($fileData, $fileName, $mimeType);
146 default:
147 break;
148 }
149 return self::grumble($this->l->t('Unknown Request'));
150 }
