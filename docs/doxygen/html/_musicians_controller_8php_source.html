125 public function search(
126 string $pattern,
127 ?int $limit = null,
128 ?int $offset = null,
129 ?string $projectName = null,
130 ?int $projectId = null,
131 array $ids = [],
132 string $scope = self::SCOPE_MUSICIANS
133 ):DataResponse {
134
135 switch ($scope) {
136 case self::SCOPE_ADDRESSBOOK:
137 case self::SCOPE_CLOUD_USERS:
138 return self::grumble($this->l->t('Picking users from the cloud or the address-books is not yet supported, sorry.'));
139 case self::SCOPE_EXECUTIVE_BOARD:
140 $projectId = $this->getExecutiveBoardProjectId();
141 break;
142 case self::SCOPE_CLUB_MEMBERS:
143 $projectId = $this->getClubMembersProjectId();
144 break;
145 case self::SCOPE_MUSICIANS:
146 // just go
147 break;
148 }
149
150 if ($projectName !== null && $projectId === null) {
151 // our findLikeTrait cannot iterate to projectParticipation.project.name
152 $project = $this->getDatabaseRepository(Entities\Project::class)->findOneBy([ 'name' => $projectName ]);
153 $project = $project->getId();
154 } else {
155 $project = $projectId;
156 }
157
158 if (empty($pattern)) {
159 $criteria = [];
160 } else {
161 $pattern = str_replace('*', '%', $pattern);
162
163 if (strpos($pattern, '%') === false) {
164 if ($pattern[0] != '^') {
165 $pattern = '%' . $pattern;
166 }
167 if (substr($pattern, -1) != '$') {
168 $pattern = $pattern . '%';
169 }
170 }
171
172 // Attention: userIdSlug can only be compared against ASCII, so a
173 // transliteration is necessary here.
174
175 $criteria = [
176 '(|surName' => $pattern,
177 'firstName' => $pattern,
178 'displayName' => $pattern,
179 'nickName' => $pattern,
180 'userIdSlug' => $this->transliterate($pattern),
181 ')' => true,
182 ];
183 }
184
185 if ($project !== null) {
186 $criteria[] = [ 'projectParticipation.project' => $project ];
187 }
188
189 $musicians = $this->musiciansRepository->findBy($criteria, [
190 'surName' => 'ASC',
191 'firstName' => 'ASC'
192 ], $limit, $offset);
193
194 if ($limit !== null && count($ids) > 0) {
195 $criteria = [ [ 'id' => $ids ] ];
196 if ($project !== null) {
197 $criteria[] = [ 'projectParticipation.project' => $project ];
198 }
199 $byIdMusicians = $this->musiciansRepository->findBy($criteria);
200 $musicians = array_merge($musicians, $byIdMusicians);
201 }
202
203 $musiciansData = [];
205 foreach ($musicians as $musician) {
206 if (Uuid::asUuid($musician->getSurName()) !== null) {
207 continue; // skip dummy musicians
208 }
209 $musiciansData[] = $this->getFlatMusician($musician, only: []);
210 }
211
212 return self::dataResponse($musiciansData);
213 }
