134 public function testDescribeAll(): void
135 {
136 $this->getRepository();
137 $instrumentInfo = $this->repository->describeALL();
138 $instruments = $this->repository->findAll();
139 $this->assertEqualsCanonicalizing(self::EXPECTED_INFO_KEYS, array_keys($instrumentInfo));
140 $allGroups = [];
142 foreach ($instruments as $instrument) {
143 $name = $instrument->getName();
144 $id = $instrument->getId();
145 $this->assertNotEmpty($instrumentInfo['nameGroups'][$name]);
146 $this->assertEquals($name, $instrumentInfo['byName'][$name]);
147 $this->assertEquals($name, $instrumentInfo['byId'][$id]);
148 $families = $instrument->getFamilies()->map(fn(Entities\InstrumentFamily $family) => $family->getFamily())->toArray();
149 sort($families);
150 $families = implode(',', $families);
151 $this->assertEquals($families, $instrumentInfo['nameGroups'][$name]);
152 $this->assertEquals($families, $instrumentInfo['idGroups'][$id]);
153 $allGroups[] = $families;
154 }
155 $allGroups = array_values(array_unique($allGroups));
156
157 $this->assertEqualsCanonicalizing($allGroups, $instrumentInfo['families']);
158 }

172 }
173 $this->assertTrue(count($result) > 1);
174 $only = reset($result)->getUntranslatedName();
175 $onlyResult = $this->repository->findNonInstruments(only: [$only]);

177 $this->assertEquals($result[0], $onlyResult[0]);
