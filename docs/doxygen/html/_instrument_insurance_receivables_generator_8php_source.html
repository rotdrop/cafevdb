297 protected function updateOne(
298 Entities\ProjectParticipantFieldDataOption $receivable,
299 Entities\ProjectParticipant $participant,
300 string $updateStrategy = self::UPDATE_STRATEGY_EXCEPTION,
301 ):array {
302 // cook-book:
303 // * find list of insurance years
304 // * walk years from start until now
305 // - add missing items if insurance fee != 0
306 // - remove items without payment when insurance fee == 0
307 // - update all existing items with newly computed insurance sum
308
309 $data = $receivable->getData();
310 if (!str_starts_with($data, '{')) {
311 // legacy item
312 $year = $data;
313 $brokerShortName = null;
314 } else {
315 list('year' => $year, 'broker' => $brokerShortName) = json_decode($data, true);
316 }
317
318 $openingBalance = $year === '0000';
319
320 $removed = false;
321 $added = false;
322 $changed = false;
323 $skipped = false;
324 $notices = [];
325
327 $musician = $participant->getMusician();
329 $project = $participant->getProject();
331 $fileSystemStorage = $this->storageFactory->getProjectParticipantsStorage($participant);
332
333 if (!$openingBalance) {
334 // "now" should in principle just do ...
335 $referenceDate = new DateTime($year.'-06-01');
336
337 // Compute the actual fee
338 $dueInterval = null;
339 $fee = $this->insuranceService->insuranceFee($musician, $brokerShortName, $referenceDate, $dueInterval);
340
341 // Generate the overview letter as supporting document
342 $overview = $this->insuranceService->musicianOverview($musician, $brokerShortName, $referenceDate);
343 } else {
344 if (0 == count($this->insuranceService->billableInsurances($musician, $brokerShortName))) {
345 // bail out early, DO NOT ADD an opening balance
346 return [
347 'added' => 0,
348 'removed' => 0,
349 'changed' => 0,
350 'skipped' => 1,
351 'notices' => [], // no insurance, no message
352 ];
353 }
354 $fee = null;
355 $updateStrategy = self::UPDATE_STRATEGY_SKIP; // set only manually
356 }
357
358 $participantFieldsData = $participant->getParticipantFieldsData();
359 $optionKey = $receivable->getKey();
360 $datum = $participant->getParticipantFieldsDatum($optionKey);
362 if (empty($datum)) {
363 if ($openingBalance || !$fee->equals(0)) {
364 // add a new option
366 $datum = (new Entities\ProjectParticipantFieldDatum)
367 ->setDataOption($receivable)
368 ->setProjectParticipant($participant)
369 ->setOptionValue($fee);
370
371 if (!$openingBalance) {
372 // store overview letter
373 $overviewFilename = $this->insuranceService->musicianOverviewFileName($overview);
374 $overviewLetter = $this->insuranceService->musicianOverviewLetter($overview);
375 $supportingDocumentFile = new Entities\EncryptedFile(
376 $overviewFilename, $overviewLetter, 'application/pdf', $musician);
377 $supportingDocument = $fileSystemStorage->addFieldDatumDocument($datum, $supportingDocumentFile, flush: false);
378 $datum->setSupportingDocument($supportingDocument);
379 }
380
381 // @todo Too much connectivity
382 $participantFieldsData->set($optionKey->getBytes(), $datum);
383 $musician->getProjectParticipantFieldsData()->set($optionKey->getBytes(), $datum);
384 $receivable->getFieldData()->set($musician->getId(), $datum);
385 $project->getParticipantFieldsData()->add($datum);
386 $added = true;
387 }
388 } else { // !empty($datum)
389 $optionValue = $datum->getOptionValue();
390 if (!$datum->isDeleted() && (!$fee || $fee->toDecimal(2) != $optionValue)) {
391 if ($openingBalance) {
392 $notices[] = $this->l->t('Keeping opening balance of %s.', $this->moneyValue($optionValue));
393 } else {
394 $notices[] = $this->l->t('Data inconsistency for musician %s in year %d: old fee %s, new fee %s.', [
395 $musician->getPublicName(true),
396 $year,
397 $this->moneyValue($optionValue),
398 $this->moneyValue($fee),
399 ]);
400 }
401 switch ($updateStrategy) {
402 case self::UPDATE_STRATEGY_REPLACE:
403 break;
404 case self::UPDATE_STRATEGY_EXCEPTION:
405 throw new Exceptions\EnduserNotificationException(end($notices));
406 break;
407 case self::UPDATE_STRATEGY_SKIP:
408 $skipped = true;
409 break;
410 default:
411 throw new RuntimeException($this->l->t('Unknonw update strategy: "%s".', $updateStrategy));
412 }
413 }
414 if (!$skipped) {
415 if (!$openingBalance && $fee->equals(0)) {
416 // remove current option
417 $this->remove($datum);
418 $this->remove($datum);
419 $participantFieldsData->removeElement($datum);
420 $musician->getProjectParticipantFieldsData()->removeElement($datum);
421 $receivable->getFieldData()->removeElement($datum);
422 $project->getParticipantFieldsData()->removeElement($datum);
423 $removed = true;
424 } else {
425 if (!$openingBalance) {
426 $overviewFilename = $this->insuranceService->musicianOverviewFileName($overview);
427 $overviewLetter = $this->insuranceService->musicianOverviewLetter($overview);
428 $this->logInfo('OVERVIEW ' . print_r($overview, true));
430 $supportingDocument = $datum->getSupportingDocument();
431 if (empty($supportingDocument)) {
432 // create overview letter
433 $supportingDocumentFile = new Entities\EncryptedFile(
434 fileName: $overviewFilename,
435 data: $overviewLetter,
436 mimeType: 'application/pdf',
437 owner: $musician
438 );
439 $supportingDocument = $fileSystemStorage->addFieldDatumDocument($datum, $supportingDocumentFile, flush: false);
440 $datum->setSupportingDocument($supportingDocument);
441 } elseif (true || $updateStrategy == self::UPDATE_STRATEGY_REPLACE || $fee->toDecimal(2) != $datum->getOptionValue()) {
442 $supportingDocument
443 ->setName($overviewFilename)
444 ->getFile()
445 ->setFileName($overviewFilename)
446 ->setMimeType('application/pdf')
447 ->setSize(strlen($overviewLetter))
448 ->getFileData()->setData($overviewLetter);
449 }
450 }
451 // just update current data to the computed value
452 if ($datum->isDeleted()) {
453 $datum->setDeleted(null);
454 $datum->setOptionValue($fee);
455 $added = true;
456 } elseif ($fee->toDecimal(2) != $datum->getOptionValue()) {
457 $datum->setOptionValue($fee);
458 $changed = true;
459 }
460 }
461 }
462 }
463 return [
464 'added' => (int)$added,
465 'removed' => (int)$removed,
466 'changed' => (int)$changed,
467 'skipped' => (int)$skipped,
468 'notices' => $notices,
469 ];
470 }
