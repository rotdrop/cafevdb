186 public function shortTitle()
187 {
188 if ($this->deleteOperation()) {
189 return $this->l->t('Remove the person from %s?', [ $this->projectName ]);
190 } elseif ($this->viewOperation()) {
191 return $this->l->t('Display of all stored data for the shown person.');
192 } elseif ($this->changeOperation()) {
193 return $this->l->t('Edit the data of the displayed person.');
194 }
195 return $this->l->t('Business Contacts and Associates for Project "%s"', [ $this->projectName ]);
196 }
