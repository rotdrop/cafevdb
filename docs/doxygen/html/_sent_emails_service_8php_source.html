86 ->setBulkRecipients(implode(';', $bulkRecipients))
87 ->setCc(implode(';', $carbonCopy))
88 ->setMessageId($imapMessage->getMessageId())
89 ->setHtmlBody($imapMessage->htmlMessage)

91 ->setSubjectHash(hash('md5', $sentEmail->getSubject()))

93 ;
