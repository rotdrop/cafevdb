93 public function svgFromFile(File $file, int $options = self::SVG_OPTIMIZE, ?string $hash = null):string
94 {
95 $options &= self::SVG_OPTIONS;
96 $data = $file->getContent();
97 if ($options === self::SVG_AS_IS) {
98 return $data; // just pass through
99 }
100 $this->filesCacheService->validate($file); // will erase all outdated cache data
101 $cacheKey = self::SVG_FROM_FILE_TAG . $options;
102 $cachedData = $this->filesCacheService->get($file, $cacheKey);
103 if ($cachedData) {
104 return $cachedData;
105 }
106 $data = $this->svgFromImageData($data, $file->getMimeType()); // no-op if already SVG
107 $data = $this->svgToSvg($data, $options);
108 $this->filesCacheService->set($file, $cacheKey, $data);
109 return $data;
110 }

121 public function dataUriFromFile(File $file, int $options = self::SVG_OPTIMIZE):string
122 {
123 $mimeType = $file->getMimeType();
124 // sanitize options in order to have a proper cache key suffix
125 if ($mimeType == self::SVG_MIME_TYPE) {
126 $options &= ~self::CONVERT_TO_SVG;
127 } elseif (!($options & CONVERT_TO_SVG)) {
128 $options &= ~SVG_OPTIONS;
129 }
130 $hash = $this->filesCacheService->validate($file);
131 $cacheKey = self::DATA_URI_TAG . $options;
132 $cachedData = $this->filesCacheService->get($file, $cacheKey);
133 if ($cachedData) {
134 return $cachedData;
135 }
136 if (($options & self::CONVERT_TO_SVG) || $mimeType == self::SVG_MIME_TYPE) {
137 $data = $this->svgFromFile($file, $options, $hash);
138 $mimeType = self::SVG_MIME_TYPE;
139 } else {
140 $data = $file->getContent();
141 }
142 $dataUri = $this->generateDataUri($data, $mimeType);
143 $this->filesCacheService->set($file, $cacheKey, $dataUri);
144 return $dataUri;
145 }

159 public function svgToSvg(string $data, int $options = self::SVG_OPTIMIZE):string
160 {
161 $mimeType = $this->mimeTypeDetector->detectString($data);
162 if ($mimeType !== self::SVG_MIME_TYPE) {
163 throw new InvalidArgumentException($this->l->t('Given image data is not an SVG image. "%s".', $data));
164 }
165 $throwArgv = null;
166 if ($options & self::SVG_TEXT_TO_PATH) {
167 // libreoffice has a bug with SVGs which contain text.
168 $inkscape = $this->executableFinder->find('inkscape');
169 if (!empty($inkscape)) {
170 $processArgv = [
171 $inkscape,
172 '--export-plain-svg',
173 '--export-text-to-path',
174 '-o', '-',
175 '-p',
176 ];
177 $process = new Process($processArgv);
178 try {
179 $process->setInput($data)->run();
180 $data = $process->getOutput();
181 } catch (Throwable $t) {
182 $throwArgv = $processArgv;
183 }
184 }
185 }
186 if ($options & self::SVG_OPTIMIZE) {
187 $svgo = $this->executableFinder->find('svgo');
188 if (!empty($svgo)) {
189 $svgoConfigFile = $this->tempManager->getTemporaryFile();
190 $svgoConfig = <<<'EOD'
191export default {
192 multipass: true,
193 js2svg: {
194 indent: 2,
195 pretty: true,
196 },
197 plugins: [
198 {
199 name: 'preset-default',
200 params: {
201 overrides: {
202 // viewBox is required to resize SVGs with CSS.
203 // @see https://github.com/svg/svgo/issues/1128
204 removeViewBox: false,
205 },
206 },
207 },
208 ],
209};
210EOD;
211 file_put_contents($svgoConfigFile, $svgoConfig);
212 $processArgv = [
213 $svgo,
214 '--input', '-',
215 '--output', '-',
216 '--config', $svgoConfigFile,
217 ];
218 $process = new Process($processArgv);
219 try {
220 $process->setInput($data)->run();
221 $data = $process->getOutput();
222 } catch (Throwable $t) {
223 $throwArgv = $processArgv;
224 // provide the config string as the temp file name has no value
225 array_pop($throwArgv);
226 array_push($throwArgv, $svgoConfig);
227 }
228 } elseif (false) {
229 // The PHP optimizer does not work ATM.
230 try {
231 $svgOptimizer = SvgOptimizerService::fromString($data)->optimize();
232 $data = $svgOptimizer->getContent();
233 } catch (Throwable $t) {
234 $throwArgv = [
235 SvgOptimizerService::class,
236 ];
237 }
238 }
239 }
240 if ($throwArgv) {
241 throw new Exceptions\FilterFailedException(
242 $processArgv,
243 $this->l->t('Conversion of SVG image data failed.'),
244 0,
245 $t,
246 );
247 }
248 return $data;
249 }

261 public function svgFromImageData(string $data, ?string $mimeType = null):string
262 {
263 if (str_starts_with($data, 'data:')) {
264 $data = file_get_contents($data);
265 if ($mimeType === null) {
266 $mimeType = mime_content_type($data);
267 }
268 }
269 if ($mimeType === null) {
270 $mimeType = $this->mimeTypeDetector->detectString($data);
271 }
272 if ($mimeType == self::SVG_MIME_TYPE) {
273 return $data;
274 }
275 $cloudImage = new Image;
276 $cloudImage->loadFromData($data);
277 $width = $cloudImage->width();
278 $height = $cloudImage->height();
279 $dataUri = $this->generateDataUri($data, $mimeType);
280 $svg =<<<EOD
281<svg xmlns="http://www.w3.org/2000/svg"
282 xmlns:xlink="http://www.w3.org/1999/xlink"
283 viewBox="0 0 {$width} {$height}"
284>
285 <image width="{$width}" height="{$height}"
286 xlink:href="{$dataUri}"
287 />
288</svg>
289EOD;
290 return $svg;
291 }
