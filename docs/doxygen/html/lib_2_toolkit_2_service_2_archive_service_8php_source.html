255 public function canOpen(File $fileNode):bool
256 {
257 $this->setProcessEnvironment();
258
259 $localPath = self::getLocalPath($fileNode);
260
261 $format = ArchiveFormats::detectArchiveFormat($localPath);
262 $canOpen = $format !== null && ArchiveFormats::canOpen($format);
263
264 $this->restoreProcessEnvironment();
265
266 if ($format === null || $format === false) {
267 throw new Exceptions\ArchiveCannotOpenException(
268 $this->l->t('Unable to detect the archive format of "%1$s".', $fileNode->getName())
269 );
270 }
271 if (!$canOpen) {
272 $messages = [];
273 if (str_starts_with($format, 't') && !str_starts_with($format, 'tar')) {
274 $innerFormat = 'tar';
275 $compositeFormat = $format;
276 $format = substr($format, 1);
277 $this->setProcessEnvironment();
278 $canDecompress = $format !== null && ArchiveFormats::canOpen($format);
279 $canUntar = ArchiveFormats::canOpen($innerFormat);
280 $this->restoreProcessEnvironment();
281 if (!$canDecompress) {
282 $messages[] = $this->l->t('Archive format of "%1$s" detected as "%2$s", but there is no backend driver installed which can decompress ".%3$s" files.', [
283 $fileNode->getName(),
284 $compositeFormat,
285 $format,
286 ]);
287 }
288 if (!$canUntar) {
289 // this should never be the case ...
290 $messages[] = $this->l->t(
291 'Unable to deal with tar-files. Please check the installation of the app.'
292 );
293 }
294 } else {
295 $messages[] = $this->l->t('The archive format of "%1$s" has been detected as "%2$s", but there is no backend driver installed which can deal with this format.', [
296 $fileNode->getName(),
297 $format,
298 ]);
299 }
300 $formats = ArchiveFormats::getDeclaredDriverFormats();
301 foreach ($formats[$format] as $driverClass) {
302 $shortDriver = substr($driverClass, strrpos($driverClass, '\\') + 1);
303 if (!$driverClass::isInstalled()) {
304 $messages[] = $this->l->t('The "%1$s" driver could handle this format, but it is not installed.', $shortDriver);
305 $typeLabel = BasicDriver::$typeLabels[$driverClass::TYPE];
306 $instructions = $driverClass::getInstallationInstruction();
307 $messages[] = $this->l->t('Installation instructions:')
308 . ' '
309 . ucfirst($typeLabel)
310 . '. '
311 . ucfirst($instructions)
312 . '.';
313 continue;
314 }
315 $abilities = $driverClass::getFormatAbilities($format);
316 $requiredAbilities = DriverAbilities::OPEN|DriverAbilities::EXTRACT_CONTENT;
317 if (($abilities & $requiredAbilities) != $requiredAbilities) {
318 $messages[] = $this->l->t('The "%1$s" driver claims to handle this format, but cannot extract the archive content.', $shortDriver);
319 }
320 }
321 throw new Exceptions\ArchiveCannotOpenException(
322 implode(PHP_EOL, $messages)
323 );
324 }
325
326 return true;
327 }

362 public function open(File $fileNode, ?int $sizeLimit = null, #[SensitiveParameter] ?string $password = null):?ArchiveService
363 {
364 if (!$this->canOpen($fileNode)) {
365 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
366 $fileNode->getPath(), self::getLocalPath($fileNode),
367 ]));
368 }
369
370 $this->setProcessEnvironment();
371
372 $this->archiver = ArchiveBackend::open(self::getLocalPath($fileNode), password: $password);
373
374 $this->restoreProcessEnvironment();
375
376 if (empty($this->archiver)) {
377 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
378 $fileNode->getPath(), self::getLocalPath($fileNode),
379 ]));
380 }
381 if ($sizeLimit === null) {
382 $sizeLimit = $this->sizeLimit;
383 }
384 $this->fileNode = $fileNode;
385 $archiveInfo = $this->getArchiveInfo();
386 $archiveSize = $archiveInfo->originalSize;
387 if ($sizeLimit !== null && $archiveSize > $sizeLimit) {
388 $this->archiver = null;
389 $this->fileNode = null;
390 throw new Exceptions\ArchiveTooLargeException(
391 $this->t('Uncompressed size of archive "%1$s" is too large: %2$s > %3$s', [
392 $fileNode->getInternalPath(), CloudUtil::humanFileSize($archiveSize), CloudUtil::humanFileSize($sizeLimit),
393 ]),
394 $sizeLimit,
395 $archiveSize,
396 );
397 }

399 // Build a per-member map from the NFC-normalized name to the raw name as
400 // stored in the archive. Nextcloud normalizes every path to NFC, so this
401 // is the key by which members will be requested again on read. Resolving
402 // each member on its own avoids choosing a single normalization for the
403 // whole archive, which breaks archives that mix NFC and NFD names.
404 $this->memberNameMap = [];
405 $this->collidingMembers = [];
406 foreach ($this->archiver->getFileNames() as $rawName) {
407 $key = Normalizer::normalize($rawName, Normalizer::NFC);
408 if ($key === false) {
409 // The name is not valid UTF-8 (e.g. a legacy DOS/codepage name stored
410 // without the UTF-8 flag). Keep the raw bytes as key so the member
411 // stays addressable instead of being dropped.
412 $key = $rawName;
413 }
414 if (array_key_exists($key, $this->memberNameMap) && $this->memberNameMap[$key] !== $rawName) {
415 $this->collidingMembers[$key] ??= [ $this->memberNameMap[$key] ];
416 $this->collidingMembers[$key][] = $rawName;
417 }
418 $this->memberNameMap[$key] = $rawName;
419 }
420
421 return $this;

474 public static function getArchiveFolderName(string $archiveFileName):?string
475 {
476 // double to account for "nested" archive types
477 $archiveFolderName = pathinfo($archiveFileName, PATHINFO_FILENAME);
478 $secondExtension = pathinfo($archiveFolderName, PATHINFO_EXTENSION);
479
480 // as a rule of thumb we only strip the second extension if it contains no
481 // spaces and is no longer that 4 characters.
482 if (strlen($secondExtension) <= 4 && str_replace(' ', '', $secondExtension) === $secondExtension) {
483 $archiveFolderName = pathinfo($archiveFolderName, PATHINFO_FILENAME);
484 }
485
486 return $archiveFolderName;
487 }
