390 public function sql_fetch(&$stmt, $type = 'a')
391 {
392 if (!$this->resultValid($stmt)) {
393 $this->logError('RESULT INVALID ' . get_class($stmt));
394 return false;
395 }
396 $pdo = $this->dbh->getWrappedConnection()->getNativeConnection();
397 $pdo->setAttribute(PDO::ATTR_STRINGIFY_FETCHES, true);
398 $type = $type === 'n' ? FetchMode::NUMERIC : FetchMode::ASSOCIATIVE;
399 $result = $stmt->fetch($type);
400 // Work around bug https://jira.mariadb.org/browse/MDEV-27323
401 if ($result !== false
402 && $type == FetchMode::ASSOCIATIVE
403 && $this->queryHash == $this->generatedQueryHash
404 && !empty($this->columnAliases)) {
405 $missingColumns = array_diff($this->columnAliases, array_keys($result));
406 if (!empty($missingColumns)) {
407 // We assume that this is caused by an optimization bug and that
408 // either the corresponding qfNN_idx column or the corresponding
409 // non-..._idx column is there and contains the correct data
410 $this->logError('Potential MariaDB missing-columns bug (https://jira.mariadb.org/browse/MDEV-27323): ' . print_r($missingColumns, true));
411 $postfix = '_idx';
412 foreach ($missingColumns as $column) {
413 $dataColumn = strrpos($column, $postfix) === false ? $column . $postfix : substr($column, -strlen($postfix));
414 if (empty($result[$dataColumn])) {
415 $this->logError('MariaDB bug https://jira.mariadb.org/browse/MDEV-27323): unable to reconstruct data for column ' . $column);
416 }
417 $result[$column] = $result[$dataColumn];
418 }
419 }
420 }
421 $pdo->setAttribute(PDO::ATTR_STRINGIFY_FETCHES, false);
422
423 return $result;
424 }

460 public function myquery($query, $line = 0, $debug = false)
461 {
462 $this->queryHash = md5($query);
463 if ($debug || $this->debug) {
464 $this->logInfo("DEBUG QUERY: ".$query, [], 2);
465 }
466 $logEntry = [
467 'query' => $query,
468 ];
469 $startTime = hrtime(true); // [ns]
470 try {
471 $stmt = $this->dbh->executeQuery($query);
472 $endTime = hrtime(true);
473 $this->affectedRows = $stmt->rowCount();
474 $logEntry['affectedRows'] = $this->affectedRows;
475 // $this->errorCode = $stmt->errorCode();
476 // $this->errorInfo = $stmt->errorInfo();
477 $logEntry['errorCode'] = 0;
478 $logEntry['errorInfo'] = null;
479 $logEntry['duration'] = ($endTime - $startTime) / 1e6; // [ms]
480 $this->queryLog[] = $logEntry;
481 } catch (DriverException $t) {
482 $endTime = hrtime(true);
483 $this->logException($t);
484 throw $t;
485 $this->exception = $t;
486 $this->errorCode = $t->getSQLState();
487 $this->errorInfo = $t->getMessage();
488 $logEntry['affectedRows'] = $this->affectedRows;
489 $logEntry['errorCode'] = $this->errorCode;
490 $logEntry['errorInfo'] = $this->errorInfo;
491 $logEntry['duration'] = ($endTime - $startTime) / 1e6;
492 $this->queryLog[] = $logEntry;
493 return false;
494 }
495 return $stmt;
496 }

645 if ($timeStamp % (24 * 60 * 60) != 0 || preg_match("/\d{1,2}\:\d{1,2}/", $databaseValue)) {

660 $this->logError('Timezone adjustment failed for data-base input ' . $databaseValue);
661 }

671 public function makeUserTimeString($k, $row)
672 {
673 $value = '';
674 $data = $row["qf$k"."_timestamp"];
675 switch ($data) {
676 case '':
677 case 0:
678 case '0000-00-00':
679 // Invalid date and time
680 break;
681 default:
682 if (!is_numeric($data)) {
683 // Convert whatever is contained in the timestamp field to
684 // seconds since the epoch.
685 $data = $this->makeTimeStampFromDatabase($data);
686 }
687 $timeStamp = intval($data);
688 $dateFormat = $timeFormat = $this->fdd[$k]['datetimeformat'] ?? false;
689 $dateFormat = $this->fdd[$k]['dateformat'] ?? false;
690 $timeFormat = $this->fdd[$k]['timeformat'] ?? false;
691
692 if (!empty($dateFormat) && !empty($timeFormat)) {
693 if ($dateFormat === true && $timeFormat === true) {
694 return $this->dateTimeFormatter->formatDateTime($timeStamp);
695 } elseif ($timeFormat === true) {
696 return $this->dateTimeFormatter->formatDateTime($timeStamp, $dateFormat);
697 } else {
698 return $this->dateTimeFormatter->formatDateTime($timeStamp, $dateFormat, $timeFormat);
699 }
700 } elseif (!empty($dateFormat)) {
701 return $dateFormat === true
702 ? $this->dateTimeFormatter->formatDate($timeStamp)
703 : $this->dateTimeFormatter->formatDate($timeStamp, $dateFormat);
704 } elseif (!empty($timeFormat)) {
705 return $timeFormat === true
706 ? $this->timeTimeFormatter->formatTime($timeStamp)
707 : $this->timeTimeFormatter->formatTime($timeStamp, $timeFormat);
708 } elseif (!empty($this->fdd[$k]['datemask'])) {
709 return date($this->fdd[$k]['datemask'], $timeStamp);
710 } elseif (!empty($this->fdd[$k]['strftimemask'])) {
711 return strftime($this->fdd[$k]['strftimemask'], $timeStamp);
712 }
713 }
714 return $value;
715 }
