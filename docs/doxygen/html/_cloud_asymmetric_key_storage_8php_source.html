64 public function getKeyPair(string $ownerId, string $keyPassphrase)
65 {
66 $privKeyMaterial = $this->getUserValue($ownerId, self::PRIVATE_ENCRYPTION_KEY);
67 if (!empty($privKeyMaterial)) {
68 try {
69 $this->cryptor->setEncryptionKey($keyPassphrase);
70 $privKeyMaterial = $this->cryptor->decrypt($privKeyMaterial);
71 } catch (Throwable $t) {
72 // exceptions are ok, but we want to stick to Exceptions\EncryptionException
73 throw new Exceptions\EncryptionKeyException(
74 $this->l->t('Unable to decrypt private key for owner "%s".', $ownerId),
75 $t->getCode(), $t
76 );
77 }
78 $privKey = $this->unserializeKey($privKeyMaterial, self::PRIVATE_ENCRYPTION_KEY);
79 $pubKey = $this->getPublicKey($ownerId);
80 }
81 return [
82 self::PRIVATE_ENCRYPTION_KEY => $privKey ?? null,
83 self::PUBLIC_ENCRYPTION_KEY => $pubKey ?? null,
84 ];

88 public function generateKeyPair(string $ownerId, string $keyPassphrase)

195 private function getUserValue(string $ownerId, string $key, mixed $default = null):mixed
196 {
197 $value = $this->getUserValueCloud($key, userId: $ownerId);
198 if (!empty($value)) {
199 $separatorPos = strpos($value, self::NAME_SEPARATOR);
200 if ($separatorPos !== false) {
201 $name = substr($value, 0, $separatorPos);
202 if ($name != static::$name) {
203 throw new InvalidArgumentException($this->l->t('Key-storage mismatch: %1$s / %2$s', [ $name, static::$name ]));
204 }
205 $value = substr($value, $separatorPos + 1);
206 }
207 } else {
208 $value = $default;
209 }
210 return $value;
211 }
