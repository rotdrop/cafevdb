107 private function registerPreCommitAction(Entities\ProjectEvent $projectEvent):void
108 {
109 if (!empty($this->preCommitActions[$projectEvent->getId()])) {
110 return;
111 }
112 if (empty($this->eventsService)) {
113 $this->eventsService = $this->appContainer->get(EventsService::class);
114 }
115 $recordAbsenceCategory = $this->eventsService->getRecordAbsenceCategory();
116 $calendarId = $projectEvent->getCalendarId();
117 $eventUri = $projectEvent->getEventUri();
118 $recurrenceId = $projectEvent->getRecurrenceId();
119 $this->preCommitActions[$projectEvent->getId()] = new GenericUndoable(
120 function() use ($projectEvent, $recordAbsenceCategory, $calendarId, $eventUri, $recurrenceId) {
121 if (empty($projectEvent->getAbsenceField())) {
122 $additions = [];
123 $removals[] = $recordAbsenceCategory;
124 } else {
125 $additions[] = $recordAbsenceCategory;
126 $removals = [];
127 }
128 try {
129 $changed = $this->eventsService->changeCategories(
130 $calendarId,
131 $eventUri,
132 $recurrenceId,
133 $additions,
134 $removals,
135 );
136 } catch (Exceptions\CalendarEntryNotFoundException $e) {
137 // this can happen if the project events table got out of sync
138 $changed = false;
139 }
140 return $changed;
141 },
142 function(bool $changed) use ($projectEvent, $recordAbsenceCategory, $calendarId, $eventUri, $recurrenceId) {
143 if (!$changed) {
144 return;
145 }
146 // just exchange additions and removals
147 if (!empty($projectEvent->getAbsenceField())) {
148 $additions = [];
149 $removals[] = $recordAbsenceCategory;
150 } else {
151 $additions[] = $recordAbsenceCategory;
152 $removals = [];

154 $this->eventsService->changeCategories(

156 $eventUri,

159 $removals,
160 );

162 );
