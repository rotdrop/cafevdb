462 private function getVEventSiblings(
463 int $calendarId,
464 VCalendar $vCalendar,
465 ?DateTimeInterface $futureLimit = null,
466 ): array {
467 if ($futureLimit === null) {
468 $futureLimit = $this->timeFactory->getDateTimeImmutable()->modify('+' . self::PROJECT_EVENTS_RECURRING_FUTURE_YEARS . ' years');
469 }
470 $vObject = VCalendarService::getVObject($vCalendar);
471 $uid = (string)$vObject->UID;
472 $sequence = (int)(string)($vObject->SEQUENCE ?? 0);
473 $siblings = $this->eventSiblings[$calendarId][$uid][$sequence] ?? null;
474 if ($siblings !== null) {
475 return $siblings;
476 }
477 if (!VCalendarService::isEventRecurring($vObject)) {
478 $siblings = [ 0 => $vObject ];
479 } else {
480 $vEvents = VCalendarService::getAllVObjects($vCalendar);
481 try {
482 // there is also the expand() method on the VCalendar ...
483 $siblings = [];
484 $eventIterator = new EventIterator($vEvents);
485 while ($eventIterator->valid()) {
486 $sibling = $eventIterator->getEventObject();
487 // the siblings are ordered by start data, bail out if this is
488 // beyond limit.
489 if ($sibling->DTSTART->getDateTime() > $futureLimit) {
490 $this->logInfo('Stopping sibling generation at date ' . $futureLimit->format('Y-m-d'));
491 break;
492 }
493 $recurrenceId = $sibling->{'RECURRENCE-ID'}->getDateTime()->getTimestamp();
494 if (empty($siblings[$recurrenceId]) || $siblings[$recurrenceId]->{'LAST-MODIFIED'}->getDateTime() < $sibling->{'LAST-MODIFIED'}->getDateTime()) {
495 // This might have been intended for an undo-feature. At any rate:
496 // there may be mutiple siblings for the same recurrence id, and
497 // the one with the most recent mtime is the one to consider.
498 $siblings[$recurrenceId] = $sibling;
499 // $this->logInfo('REPLACED SIBLING ' . $uid . ' -> ' . $recurrenceId . ' mtime ' . $lastModified);
500 }
501 $eventIterator->next();
502 }
503 } catch (NoInstancesException $e) {
504 $this->logError('No event instances for "' . $uid . '@' . $calendarId . '".');
505 // This event is recurring, but it doesn't have a single
506 // instance. We are skipping this event from the output
507 // entirely.
508 $siblings = [];
509 } catch (MaxInstancesExceededException $e) {
510 $this->logError('Maximum number of recurring event siblings exceeded for "' . $uid . '@' . $calendarId . '" after generating ' . count($siblings) . ' siblings.');
511 // hopefully happens never, but ... just live with the sequence we
512 // have.
513 }
514 }
515 $this->eventSiblings[$calendarId][$uid][$sequence] = $siblings;
516 return $siblings;
517 }

782 private function eventTimes(array $eventObject, ?string $timezone = null, ?string $locale = null):array
783 {
784 if ($timezone === null) {
785 $timezone = $this->getTimezone();
786 }
787 if ($locale === null) {
788 $locale = $this->getLocale();
789 }
790 if (!empty($eventObject['times']) && $eventObject['times']['locale'] == $locale && $eventObject['times']['timezone'] == $timezone) {
791 return $eventObject['times'];
792 }
793
794 $language = locale_get_primary_language($locale);
795 $region = locale_get_region($locale);
796 $locale = $language . ($region ? '_' . $region : ''); // strip coding from locale
797 $dateTimeZone = new DateTimeZone($timezone);
798
799 // Variant using Carbon. We could do the same using Nextcloud's
800 // IDateTimeFormatter by constructing an UTC locale.
802 $start = CarbonImmutable::createFromInterface($eventObject['start'])->locale($locale);
804 $end = CarbonImmutable::createFromInterface($eventObject['end'])->locale($locale);
805 $allDay = $eventObject['allDay'];
806
807 $startStamp = $start->getTimestamp();
808 $endStamp = $end->getTimestamp();
809
810 if (!$allDay) {
811 // all-day event dates have UTC timezone and THAT JUST SHOULD BE LEFT AS IS
812 // $startDate = $start->format;
813 $start = $start->setTimezone($dateTimeZone);
814 $end = $end->setTimezone($dateTimeZone);
815 }
816 $startDate = $start->isoFormat('L');
817 $startTime = $start->isoFormat('LT');
818 $endTime = $end->isoFormat('LT');
819
820 // make events ending at midnight more readable for known locales.
821 switch ($endTime) {
822 case '00:00':
823 $endDate = $end->setTimestamp($endStamp - 1)->isoFormat('L');
824 $endTime = '24:00';
825 break;
826 default:
827 // leave as is
828 $endDate = $end->isoFormat('L');
829 break;
830 }
831
832 return [
833 'timezone' => $timezone,
834 'locale' => $locale,
835 'allDay' => $allDay,
836 'start' => [
837 'stamp' => $startStamp,
838 'date' => $startDate,
839 'time' => $startTime,
840 ],
841 'end' => [
842 'stamp' => $endStamp,
843 'date' => $endDate,
844 'time' => $endTime,
845 ],
846 ];
847 }
