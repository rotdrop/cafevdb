427 private function getVEventSiblings(int $calendarId, VCalendar $vCalendar):array
428 {
429 $vObject = VCalendarService::getVObject($vCalendar);
430 $uid = (string)$vObject->UID;
431 $sequence = (int)(string)($vObject->SEQUENCE ?? 0);
432 $siblings = $this->eventSiblings[$calendarId][$uid][$sequence] ?? null;
433 if ($siblings !== null) {
434 return $siblings;
435 }
436 if (!VCalendarService::isEventRecurring($vObject)) {
437 $siblings = [ 0 => $vObject ];
438 } else {
439 $vEvents = VCalendarService::getAllVObjects($vCalendar);
440 try {
441 // there is also the expand() method on the VCalendar ...
442 $siblings = [];
443 $eventIterator = new EventIterator($vEvents);
444 while ($eventIterator->valid()) {
445 $sibling = $eventIterator->getEventObject();
446 $recurrenceId = $sibling->{'RECURRENCE-ID'}->getDateTime()->getTimestamp();
447 if (empty($siblings[$recurrenceId]) || $siblings[$recurrenceId]->{'LAST-MODIFIED'}->getDateTime() < $sibling->{'LAST-MODIFIED'}->getDateTime()) {
448 // This might have been intended for an undo-feature. At any rate:
449 // there may be mutiple siblings for the same recurrence id, and
450 // the one with the most recent mtime is the one to consider.
451 $siblings[$recurrenceId] = $sibling;
452 // $this->logInfo('REPLACED SIBLING ' . $uid . ' -> ' . $recurrenceId . ' mtime ' . $lastModified);
453 }
454 $eventIterator->next();
455 }
456 } catch (NoInstancesException $e) {
457 // This event is recurring, but it doesn't have a single
458 // instance. We are skipping this event from the output
459 // entirely.
460 $siblings = [];
461 } catch (MaxInstancesExceededException $e) {
462 // hopefully happens never, but ... just live with the sequence we
463 // have.
464 }
465 }
466 $this->eventSiblings[$calendarId][$uid][$sequence] = $siblings;
467 return $siblings;

728 private function eventTimes(array $eventObject, ?string $timezone = null, ?string $locale = null):array
729 {
730 if ($timezone === null) {
731 $timezone = $this->getTimezone();
732 $this->getDateTimeZone();
733 }
734 if ($locale === null) {
735 $locale = $this->getLocale();
736 }
737 if (!empty($eventObject['times']) && $eventObject['times']['locale'] == $locale && $eventObject['times']['timezone'] == $timezone) {
738 return $eventObject['times'];
739 }
740
741 $language = locale_get_primary_language($locale);
742 $region = locale_get_region($locale);
743 $locale = $language . ($region ? '_' . $region : ''); // strip coding from locale
744 $dateTimeZone = new DateTimeZone($timezone);
745
746 // Variant using Carbon. We could do the same using Nextcloud's
747 // IDateTimeFormatter by constructing an UTC locale.
749 $start = CarbonImmutable::createFromInterface($eventObject['start'])->locale($locale);
751 $end = CarbonImmutable::createFromInterface($eventObject['end'])->locale($locale);
752 $allDay = $eventObject['allday'];
753
754 $startStamp = $start->getTimestamp();
755 $endStamp = $end->getTimestamp();
756
757 if (!$allDay) {
758 // all-day event dates have UTC timezone and THAT JUST SHOULD BE LEFT AS IS
759 // $startDate = $start->format;
760 $start = $start->setTimezone($dateTimeZone);
761 $end = $end->setTimezone($dateTimeZone);
762 }
763 $startDate = $start->isoFormat('L');
764 $startTime = $start->isoFormat('LT');
765 $endTime = $end->isoFormat('LT');
766
767 // make events ending at midnight more readable for known locales.
768 switch ($endTime) {
769 case '00:00':
770 $endDate = $end->setTimestamp($endStamp - 1)->isoFormat('L');
771 $endTime = '24:00';
772 break;
773 default:
774 // leave as is
775 $endDate = $end->isoFormat('L');
776 break;
777 }
778
779 return [
780 'timezone' => $timezone,
781 'locale' => $locale,
782 'allday' => $allDay,
783 'start' => [
784 'stamp' => $startStamp,
785 'date' => $startDate,
786 'time' => $startTime,
787 ],
788 'end' => [
789 'stamp' => $endStamp,
790 'date' => $endDate,
791 'time' => $endTime,
792 ],
793 ];
794 }
