357 public function getShareFromUrl(string $url):?IShare
358 {
359 $baseUrl = $this->urlGenerator->getBaseUrl();
360 if (str_starts_with($url, $baseUrl)) {
361 $url = substr($url, strlen($baseUrl));
362 }
363 $webRoot = $this->urlGenerator->getWebroot();
364 if (str_starts_with($url, $webRoot)) {
365 $url = substr($url, strlen($webRoot));
366 }
367 if (str_starts_with($url, self::INDEX_PHP)) {
368 $url = substr($url, strlen(self::INDEX_PHP));
369 }
370 $urlParts = parse_url($url);
371 $urlPath = $urlParts['path'];
372 if (str_starts_with($urlPath, self::LINK_SHARE_PREFIX)) {
373 $token = explode(Constants::PATH_SEPARATOR, substr($urlPath, strlen(self::LINK_SHARE_PREFIX)))[0];
374 } else {
375 $token = $urlPath;
376 }
377
378 $share = $this->shareManager->getShareByToken($token);
379
380 return $share;
381 }
