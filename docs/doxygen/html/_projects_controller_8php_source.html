85 public function validate(string $topic):DataResponse
86 {
87 $projectValues = $this->getPrefixParams($this->pme->cgiDataName());
88 switch ($topic) {
89 case 'name':
90 $required = [
91 'year' => $this->l->t("project-year"),
92 'name' => $this->l->t("project-name"),
93 'type' => $this->l->t("project-kind"),
94 ];
95 foreach ($required as $key => $subject) {
96 if (empty($projectValues[$key])) {
97 return self::grumble($this->l->t("The %s must not be empty.", [$subject]));
98 }
99 }
100 $control = $this->request->getParam('control', 'name');
101 $record = $this->pme->getCGIRecordId();
102 $projectId = $record['id']??null;
103 $projectName = $projectValues['name'];
104 $projectYear = $projectValues['year'];
105 $attachYear = !empty($projectValues['type']) && $projectValues['type'] == ProjectType::TEMPORARY;
106
107 $infoMessage = "";
108 switch ($control) {
109 case "submit":
110 case "name":
111 // No whitespace, s.v.p., and CamelCase
112 $origName = $projectName;
113
115 $projectService = $this->di(ProjectService::class);
116 $projectName = $projectService->sanitizeName($projectName);
117 if ($origName != $projectName) {
118 $infoMessage .= $this->l->t(
119 'The project name has been simplified from "%s" to "%s".',
120 [ $origName, $projectName ]
121 );
122 }
123 $matches = [];
124 // Get the year from the name, if set
125 if (preg_match('/^(.*\D)?(\d{4})$/', $projectName, $matches) == 1) {
126 $projectName = $matches[1];
127 if ($control != "submit" && $attachYear) {
128 // the year-control wins when submitting the form
129 $projectYear = $matches[2];
130 }
131 if ($projectName == "") {
132 return self::grumble($this->l->t("The project-name must not only consist of the year-number."));
133 }
134 } elseif ($projectName == "") {
135 return self::grumble($this->l->t("No project-name given."));
136 }
137 if (mb_strlen($projectName) > Renderer::NAME_LENGTH_MAX) {
138 return self::grumble($this->l->t(
139 "The project-name is too long, ".
140 "please use something less than %d characters ".
141 "(excluding the attached year). Thanks",
142 [ Renderer::NAME_LENGTH_MAX ]
143 ));
144 }
145 // fallthrough
146 case "year":
147 if ($projectYear == "") {
148 return self::grumble($this->l->t("No project-year given."));
149 }
150 if (preg_match('/^\d{4}$/', $projectYear) !== 1) {
151 return self::grumble($this->l->t("The project-year has to consist of four digits, e.g. ``1984''."));
152 }
153
154 // Strip the year from the name and replace with the given year
155 $origName = $origName??$projectName;
156 $strippedName = preg_replace('/\d+$/', '', $projectName);
157 if ($attachYear) {
158 $projectName = $strippedName . $projectYear;
159 if ($projectName != $origName) {
160 $infoMessage .= $this->l->t("The year %s has been appended to the project-slug %s.", [ $projectYear, $strippedName ]);
161 }
162 }
163 // Project name may be empty at this point. Why not
164 break;
165 default:
166 return self::grumble($this->l->t('Unknown Request'));
167 }
168
169 $repository = $this->getDatabaseRepository(Entities\Project::class);
170 $projects = $repository->shortDescription();
171 foreach ($projects['projects'] as $id => $nameYear) {
172 if ($id != $projectId && $nameYear['name'] == $projectName && $nameYear['year'] == $projectYear) {
173 return self::grumble($this->l->t(
174 'A project with the name "%1$s" already exists in the year %2$s with the id %3$d (new: %4$d). Please choose a different name or year.',
175 [ $projectName, $projectYear, $id, $projectId ]
176 ));
177 }
178 }
179 return self::dataResponse(['projectYear' => $projectYear,
180 'projectName' => $projectName,
181 'message' => $infoMessage ]);
182 default:
183 break;
184 }
185 return self::grumble($this->l->t('Unknown Request'));
186 }

197 public function changeInstrumentation(string $instruments, string $voices):DataResponse
198 {
199 $instrumentsKey = str_replace('[]', '', $instruments);
200 $instruments = array_filter($this->request[$instrumentsKey]??[]);
201
202 $voicesKey = str_replace('[]', '', $voices);
203 $voices = array_filter($this->request[$voicesKey]??[]);
204
205 $instrumentInfo =
206 $this->getDatabaseRepository(Entities\Instrument::class)->describeALL();
207
208 // convert voices array to instrumentId => [ voices ]
209 $voicesByInstrument = Util::explodeIndexedMulti(implode(',', $voices));
210
211 $voicesSelectArray = [];
212 foreach ($instruments as $instrument) {
213 $highestConfigured = empty($voicesByInstrument[$instrument]) ? 0 : max($voicesByInstrument[$instrument]);
214 $highest = max(Renderer::NUM_VOICES_MIN, $highestConfigured + Renderer::NUM_VOICES_EXTRA);
215 for ($i = 1; $i <= $highest+1; ++$i) {
216 $voiceIndicator = ($i <= $highest) ? $i : '?';
217 $value = $instrument . Renderer::JOIN_KEY_SEP . $voiceIndicator;
218 $voiceOption = [
219 'value' => $value,
220 'name' => $instrumentInfo['byId'][$instrument] . ' ' . $voiceIndicator,
221 'group' => $instrumentInfo['idGroups'][$instrument],
222 ];
223 $voicesSelectArray[] = $voiceOption;
224 // make sure there are no holes, all voices up to the maximum are
225 // selected
226 if ($i <= $highestConfigured) {
227 $voices[] = $value;
228 }
229 }
230 }
231
232 $voicesSelectOptions = PageNavigation::selectOptions($voicesSelectArray, $voices);
233
234 return self::dataResponse([ 'voices' => $voicesSelectOptions ]);

372 'message' => $messages,

374 'status' => 'unchanged',

741 public function searchProjects(string $pattern, ?int $limit = null, ?int $offset = null, ?int $year = null):DataResponse
742 {
743 $repository = $this->getDatabaseRepository(Entities\Project::class);
744
745 if (empty($pattern)) {
746 $criteria = [];
747 } else {
748 $pattern = str_replace('*', '%', $pattern);
749
750 if (strpos($pattern, '%') === false) {
751 if ($pattern[0] != '^') {
752 $pattern = '%' . $pattern;
753 } else {
754 $pattern = substr($pattern, 1);
755 }
756 if (substr($pattern, -1) != '$') {
757 $pattern = $pattern . '%';
758 } else {
759 $pattern = substr($pattern, 0, -1);
760 }
761 }
762 $criteria = [
763 '(|name' => $pattern,
764 'id' => $pattern,
765 ')' => true,
766 ];
767 if ($year !== null) {
768 $criteria[] = [ 'year' => $year ];
769 }
770 }
771
772 $projects = $repository->findBy($criteria, [
773 'year' => 'DESC',
774 'name' => 'ASC',
775 ], $limit, $offset);
776
777 return self::dataResponse(array_map(fn($project) => $this->flattenProject($project), $projects));
778 }
