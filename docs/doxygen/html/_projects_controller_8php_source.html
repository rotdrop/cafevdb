96 public function validate(string|EnumProjectValidationTopic $topic): DataResponse|JSONResponse
97 {
98 $topic = EnumProjectValidationTopic::get($topic);
99 $projectValues = $this->getPrefixParams($this->pme->cgiDataName());
100 switch ($topic) {
101 case EnumProjectValidationTopic::NAME:
102 $required = [
103 'year' => $this->l->t('project-year'),
104 'name' => $this->l->t('project-name'),
105 'type' => $this->l->t('project-kind'),
106 ];
107 foreach ($required as $key => $subject) {
108 if (empty($projectValues[$key])) {
109 throw new Exceptions\EnduserNotificationException(
110 $this->l->t('The "%s" value must not be empty.', [$subject]),
111 );
112 }
113 }
114 $control = $this->request->getParam('control', 'name');
115 $record = $this->pme->getCGIRecordId();
116 $projectId = $record['id']??null;
117 $projectName = $projectValues['name'];
118 $projectYear = $projectValues['year'];
119 $attachYear = !empty($projectValues['type']) && $projectValues['type'] == ProjectType::TEMPORARY->value;
120
121 $messages = [];
122 switch ($control) {
123 case 'submit':
124 case 'name':
125 // No whitespace, s.v.p., and CamelCase
126 $origName = $projectName;
127
129 $projectService = $this->di(ProjectService::class);
130 $projectName = $projectService->sanitizeName($projectName);
131 if ($origName != $projectName) {
132 $messages[] = $this->l->t(
133 'The project name has been simplified from "%s" to "%s".',
134 [ $origName, $projectName ]
135 );
136 }
137 $matches = [];
138 // Get the year from the name, if set
139 if (preg_match('/^(.*\D)?(\d{4})$/', $projectName, $matches) == 1) {
140 $projectName = $matches[1];
141 if ($control != 'submit' && $attachYear) {
142 // the year-control wins when submitting the form
143 $projectYear = $matches[2];
144 }
145 if ($projectName == '') {
146 throw new Exceptions\EnduserNotificationException(
147 $this->l->t('The project-name must not only consist of the year-number.'),
148 );
149 }
150 } elseif ($projectName == '') {
151 throw new Exceptions\EnduserNotificationException(
152 $this->l->t('No project-name given.'),
153 );
154 }
155 if (mb_strlen($projectName) > Renderer::NAME_LENGTH_MAX) {
156 throw new Exceptions\EnduserNotificationException(
157 $this->l->t(
158 'The project-name is too long, '.
159 'please use something less than %d characters '.
160 '(excluding the attached year). Thanks',
161 [ Renderer::NAME_LENGTH_MAX ]
162 ),
163 );
164 }
165 // fallthrough
166 case 'year':
167 if ($projectYear == '') {
168 throw new Exceptions\EnduserNotificationException(
169 $this->l->t('No project-year given.'),
170 );
171 }
172 if (preg_match('/^\d{4}$/', $projectYear) !== 1) {
173 throw new Exceptions\EnduserNotificationException(
174 $this->l->t('The project-year has to consist of four digits, e.g. "1984".'),
175 );
176 }
177
178 // Strip the year from the name and replace with the given year
179 $origName = $origName??$projectName;
180 $strippedName = preg_replace('/\d+$/', '', $projectName);
181 if ($attachYear) {
182 $projectName = $strippedName . $projectYear;
183 if ($projectName != $origName) {
184 $messages[] = $this->l->t('The year %s has been appended to the project-slug %s.', [ $projectYear, $strippedName ]);
185 }
186 }
187 // Project name may be empty at this point. Why not
188 break;
189 default:
190 throw new Exceptions\EnduserNotificationException(
191 $this->l->t('Unknown Request'),
192 );
193 }
194
195 $repository = $this->getDatabaseRepository(Entities\Project::class);
196 $projects = $repository->shortDescription();
197 foreach ($projects['projects'] as $id => $nameYear) {
198 if ($id != $projectId && $nameYear['name'] == $projectName && $nameYear['year'] == $projectYear) {
199 throw new Exceptions\EnduserNotificationException(
200 $this->l->t(
201 'A project with the name "%1$s" already exists in the year %2$s with the id %3$d (new: %4$d). Please choose a different name or year.',
202 [ $projectName, $projectYear, $id, $projectId ]
203 ),
204 );
205 }
206 }
207 return new DTO\ProjectValidationResponse(
208 messages: $messages,
209 year: $projectYear,
210 name: $projectName,
211 )->response();
212 default:
213 break;
214 }
215 throw new Exceptions\EnduserNotificationException(
216 $this->l->t('Unknown Request'),
217 );
218 }

232 public function changeInstrumentation(string $instruments, string $voices): DataResponse|JSONResponse
233 {
234 $instrumentsKey = str_replace('[]', '', $instruments);
235 $instruments = array_filter($this->request[$instrumentsKey]??[]);
236
237 $voicesKey = str_replace('[]', '', $voices);
238 $voices = array_filter($this->request[$voicesKey]??[]);
239
240 $instrumentInfo =
241 $this->getDatabaseRepository(Entities\Instrument::class)->describeALL();
242
243 // convert voices array to instrumentId => [ voices ]
244 $voicesByInstrument = Util::explodeIndexedMulti(implode(',', $voices));
245
246 $voicesSelectArray = [];
247 foreach ($instruments as $instrument) {
248 $highestConfigured = empty($voicesByInstrument[$instrument]) ? 0 : max($voicesByInstrument[$instrument]);
249 $highest = max(Renderer::NUM_VOICES_MIN, $highestConfigured + Renderer::NUM_VOICES_EXTRA);
250 for ($i = 1; $i <= $highest+1; ++$i) {
251 $voiceIndicator = ($i <= $highest) ? $i : '?';
252 $value = $instrument . DataConstants::JOIN_KEY_SEP . $voiceIndicator;
253 $voiceOption = [
254 'value' => $value,
255 'name' => $instrumentInfo['byId'][$instrument] . ' ' . $voiceIndicator,
256 'group' => $instrumentInfo['idGroups'][$instrument],
257 ];
258 $voicesSelectArray[] = $voiceOption;
259 // make sure there are no holes, all voices up to the maximum are
260 // selected
261 if ($i <= $highestConfigured) {
262 $voices[] = $value;
263 }
264 }
265 }
266
267 $voicesSelectOptions = PageNavigation::selectOptions($voicesSelectArray, $voices);
268
269 return new DTO\ChangeProjectInstrumentationResponse(
270 voices: $voicesSelectOptions,
271 )->response();
272 }

285 public function mailingLists(string $operation, int $projectId, bool $force = false): DataResponse|JSONResponse
286 {
287 switch ($operation) {
288 case self::LIST_OPERATION_CREATE:
290 $projectService = $this->di(ProjectService::class);
291 $listInfo = $projectService->createProjectMailingList($projectId);
292 $listId = $listInfo['list_id'];
294 $listsService = $this->di(MailingListsService::class);
295 if (empty($listsService->getListConfig($listId, 'allow_list_posts'))) {
296 $l10nStatus = $this->l->t($status = 'closed');
297 } else {
298 $l10nStatus = $this->l->t($status = 'active');
299 }
300 $listInfo['message'] = $this->l->t('Mailing-list "%s" successfully created.', $listInfo['fqdn_listname']);
301 $listInfo['status'] = $status;
302 $listInfo['l10nStatus'] = $l10nStatus;
303 return self::dataResponse($listInfo);
304
305 case self::LIST_OPERATION_REOPEN:
306 case self::LIST_OPERATION_CLOSE:
308 $projectService = $this->di(ProjectService::class);
310 $project = $projectService->findById($projectId);
311 $listId = $project->getMailingListId();
313 $listsService = $this->di(MailingListsService::class);
314 $listsService->setListConfig($listId, 'emergency', $operation === 'close');
315 $listInfo = $listsService->getListInfo($listId);
316 if ($operation == 'close') {
317 $l10nStatus = $this->l->t($status = 'closed');
318 $listInfo['message'] = $this->l->t('Successfully closed "%s".', $listId);
319 } else {
320 $l10nStatus = $this->l->t($status = 'active');
321 $listInfo['message'] = $this->l->t('Successfully re-opened "%s".', $listId);
322 }
323 $listInfo['status'] = $status;
324 $listInfo['l10nStatus'] = $l10nStatus;
325 return self::dataResponse($listInfo);
326
327 case self::LIST_OPERATION_DELETE:
329 $projectService = $this->di(ProjectService::class);
331 $project = $projectService->findById($projectId);
332 $listId = $project->getMailingListId();
333 if (!$force) {
334 return new DataResponse([
335 'status' => 'unconfirmed',
336 'feedback' => $this->l->t('Really delete the mailing list "%1$s" for the project "%2$s"?', [ $listId, $project->getName() ]),
337 ]);
338 }
339 $projectService->deleteProjectMailingList($project);
340 return self::dataResponse([
341 'message' => $this->l->t('Successfully deleted "%s".', $listId),
342 'list_id' => $listId,
343 'fqdn_listname' => preg_replace('/\./', '@', $listId, 1),
344 'status' => 'unset',
345 'l10nStatus' => $this->l->t('unset'),
346 ]);
347
348 case self::LIST_OPERATION_SUBSCRIBE:
350 $listsService = $this->di(MailingListsService::class);
352 $projectService = $this->di(ProjectService::class);
354 $project = $projectService->findById($projectId);
355 $listId = $project->getMailingListId();
356
357 $failures = [];
358 $newCount = 0;
359 $keptCount = 0;
360 $noEmailCount = 0;
361 $notConfirmedCount = 0;
363 foreach ($project->getParticipants() as $participant) {
364 if (!$participant->getRegistration()) {
365 ++$notConfirmedCount;
366 continue; // only subscribe confirmed participants
367 }
368 $email = $participant->getMusician()->getEmail();
369 if (empty($email)) {
370 ++$noEmailCount;
371 continue; // after all, it is a mailing list ...
372 }
373 $displayName = $participant->getPublicName(firstNameFirst: true);
374 try {
375 $result = $projectService->ensureMailingListSubscription($participant);
376 $newCount += (int)$result;
377 $keptCount += (int)!$result;
378 } catch (\Throwable $t) {
379 $failures[] = [
380 'email' => $email,
381 'displayName' => $displayName,
382 'message' => $t->getMessage(),
383 ];
384 }
385 }
386
387 $messages = [
388 $this->l->t('%1$d new subscriptions, %2$d existing subscriptions have been kept.', [ $newCount, $keptCount ]),
389 ];
390 if ($notConfirmedCount > 0) {
391 $messages[] = $this->l->n(
392 '%n person has not been subscribed because of unconfirmed participation.',
393 '%n persons have not been subscribed because their participation is not confirmed.',
394 $notConfirmedCount
395 );
396 }
397 if ($noEmailCount > 0) {
398 $messages[] = $this->l->t('%d persons have not been subscribed because we do not have their email-address.', $noEmailCount);
399 }
400 if (count($failures) > 0) {
401 $messages[] = $this->l->t('%d subscription requests have failed.', [ count($failures) ]);
402 }
403 foreach ($failures as $failure) {
404 $messages[] = $this->l->t('Error for "%1$s <%2$s>": %3$s', [
405 $failure['displayName'], $failure['email'], $failure['message'],
406 ]);
407 }
408 return self::dataResponse([
409 'message' => $messages,
410 'listId' => $listId,
411 'status' => 'unchanged',
412 ]);
413 }
414 return self::grumble($this->l->t('Unknown Request: "%s".', $operation));
415 }
