115 $count = $this->recurseDirectory($directory, function($node) use ($uid) {

117 // this will never return expired shares and event delete expired shares
118 $shares = $this->shareManager->getSharesBy($uid, IShare::TYPE_LINK, $node);
119 $this->debug('Found ' . count($shares) . ' shares');
120 // initialize with file mtime in order to implement a grace period
121 $expirationStampMax = $node->stat()['mtime'];
123 foreach ($shares as $share) {
124 $expirationDate = $share->getExpirationDate();
125 if (empty($expirationDate)) {
126 $this->debug('Expiration date for share is null');
127 $expirationDate = new DateTime; // never expire?
128 } else {
129 $this->debug('Expiration date ' . print_r($expirationDate, true));
130 }
131 $expirationStampMax = max($expirationStampMax, $expirationDate->getTimestamp());
132 }
133 $ttlSeconds = $expirationStampMax + $this->oldAge - $this->time->getTime();
134 if ($ttlSeconds <= 0) {
135 $this->debug('Deleting expired download share ' . $node->getName() . ', TTL ' . $ttlSeconds);
136 $node->delete();
137 return 0;
138 } else {
139 $this->debug('Keeping active share ' . $node->getName() . ', TTL ' . $ttlSeconds);
140 return 1;
141 }
142 });
143 $this->debug('Found ' . $count . ' entries.');
144 // If no entries remain we can remove ourselves. We will be re-added by the email form as necessary

165 private function recurseDirectory(Folder $folder, callable $callback, int $level = 0):int
166 {
167 $count = 0;
169 foreach ($folder->getDirectoryListing() as $node) {
170 if ($node->getType() == Node::TYPE_FOLDER) {
171 $count += $this->recurseDirectory($node, $callback, $level + 1);
172 } else {
173 $count += call_user_func($callback, $node);
174 }
175 }
176 if ($count == 0 && $level > 0) {
177 $this->debug('Deleting empty download folder ' . $folder->getName());
178 $folder->delete();
179 }
180 return $count;
181 }
