372 public function participantFieldSurcharge(?string $key, ?string $value, Entities\ProjectParticipantField $participantField):float
373 {
374 switch ($participantField->getMultiplicity()) {
375 case Multiplicity::SIMPLE():
376 return (float)$value;
377 case Multiplicity::GROUPOFPEOPLE():
378 if (empty($key)) {
379 break;
380 }
381 return (float)$participantField->getManagementOption()->getData();
382 case Multiplicity::SINGLE():
383 if (empty($key)) {
384 break;
385 }
387 $dataOption = $participantField->getDataOptions()->first();
388 // Non empty value means "yes".
389 if ((string)$dataOption['key'] != $key) {
390 $this->logWarn('Stored value "'.$key.'" unequal to stored key "'.$dataOption['key'].'"');
391 }
392 return (float)$dataOption['data'];
393 case Multiplicity::GROUPSOFPEOPLE():
394 case Multiplicity::MULTIPLE():
395 if (empty($key)) {
396 break;
397 }
398 foreach ($participantField->getDataOptions() as $dataOption) {
399 if ((string)$dataOption['key'] == $key) {
400 return (float)$dataOption['data'];
401 }
402 }
403 $this->logError('No data item for multiple choice key "'.$key.'"');
404 return 0.0;
405 case Multiplicity::PARALLEL():
406 if (empty($key)) {
407 break;
408 }
409 $keys = Util::explode(',', $key);
410 $found = false;
411 $amount = 0.0;
412 foreach ($participantField->getDataOptions() as $dataOption) {
413 if (array_search((string)$dataOption['key'], $keys) !== false) {
414 $amount += (float)$dataOption['data'];
415 $found = true;
416 }
417 }
418 if (!$found) {
419 $this->logError('No data item for parallel choice key "'.$key.'"');
420 }
421 return $amount;
422 case Multiplicity::RECURRING():
423 if (empty($key) || empty($value)) {
424 break;
425 }
426 // $keys = Util::explode(',', $key);
427 $values = Util::explodeIndexed($value);
428
429 $amount = 0.0;
430 foreach ($values as $key => $value) {
431 $amount += $value;
432 }
433 return $amount;
434 }
435 return 0.0;
436 }

735 public function explodeDataOptions(mixed $values, bool $addProto = true, bool $trimInactive = false):array
736 {
737 $options = empty($values) ? [] : (is_array($values) ? $values : json_decode($values, true));
738 if (is_string($options)) {
739 // perhaps double encoded, try ...
740 $options = json_decode($options, true);
741 }
742 if (is_string($values) && $values != '[]' && !empty($values) && empty($options)) {
743 $options = [
744 array_merge(
745 $this->dataOptionPrototype(),
746 [ 'key' => Uuid::create(), 'label' => $values, ]),
747 ];
748 }
749 if (isset($options[-1])) {
750 throw new Exception(
751 $this->l->t(
752 'Option index -1 should not be present here, options: %s',
753 print_r($options, true)));
754 }
755 $options = array_values($options);
756 $protoType = $this->dataOptionPrototype();
757 $protoKeys = array_keys($protoType);
758 $result = [];
759 foreach ($options as $option) {
760 $keys = array_keys($option);
761 if ($keys !== $protoKeys) {
762 throw new InvalidArgumentException(
763 $this->l->t(
764 'Prototype keys "%s" and options keys "%s" differ',
765 [ implode(',', $protoKeys), implode(',', $keys) ])
766 );
767 }
768 if ($trimInactive && !empty($option['deleted'])) {
769 continue;
770 }
771 $result[(string)$option['key']] = $option;
772 }
773 if ($addProto) {
774 $result[$protoType['key']] = $protoType;
775 }
776 return $result;
777 }
