376 ?string $key,
377 ?string $value,
378 Entities\ProjectParticipantField $participantField,
379 ): MonetaryNumberType {
380 switch ($participantField->getMultiplicity()) {
381 case FieldMultiplicity::SIMPLE:
382 return MonetaryNumberType::create($value);
383 case FieldMultiplicity::GROUPOFPEOPLE:
384 if (empty($key)) {
385 break;
386 }
387 return MonetaryNumberType::create($participantField->getManagementOption()->getData());
388 case FieldMultiplicity::SINGLE:
389 if (empty($key)) {
390 break;
391 }
393 $dataOption = $participantField->getDataOptions()->first();

395 if ((string)$dataOption['key'] != $key) {
396 $this->logWarn('Stored value "' . $key . '" unequal to stored key "' . $dataOption['key'] . '"');
397 }
398 return MonetaryNumberType::create($dataOption['data']);
399 case FieldMultiplicity::GROUPSOFPEOPLE:
400 case FieldMultiplicity::MULTIPLE:
401 if (empty($key)) {
402 break;
403 }
404 foreach ($participantField->getDataOptions() as $dataOption) {
405 if ((string)$dataOption['key'] == $key) {
406 return MonetaryNumberType::create($dataOption['data']);
407 }
408 }
409 $this->logError('No data item for multiple choice key "'.$key.'"');
410 return MonetaryNumberType::zero();
411 case FieldMultiplicity::PARALLEL:
412 if (empty($key)) {
413 break;
414 }
415 $keys = Util::explode(',', $key);
416 $found = false;
418 $amount = MonetaryNumberType::zero();
419 foreach ($participantField->getDataOptions() as $dataOption) {
420 if (array_search((string)$dataOption['key'], $keys) !== false) {
421 $amount->addEq($dataOption['data']);
422 $found = true;
423 }
424 }
425 if (!$found) {
426 $this->logError('No data item for parallel choice key "'.$key.'"');
427 }
428 return $amount;
429 case FieldMultiplicity::RECURRING:
430 if (empty($key) || empty($value)) {
431 break;
432 }
433 // $keys = Util::explode(',', $key);
434 $values = Util::explodeIndexed($value);
435
436 $amount = MonetaryNumberType::zero();
437 foreach ($values as $key => $value) {
438 if (!empty($value)) {
439 $amount->addEq($value);
440 }
441 }
442 return $amount;
443 }
444 return MonetaryNumberType::zero();

755 public function explodeDataOptions(mixed $values, bool $addProto = true, bool $trimInactive = false):array
756 {
757 $options = empty($values) ? [] : (is_array($values) ? $values : json_decode($values, true));
758 if (is_string($options)) {
759 // perhaps double encoded, try ...
760 $options = json_decode($options, true);
761 }
762 if (is_string($values) && $values != '[]' && !empty($values) && empty($options)) {
763 $options = [
764 array_merge(
765 $this->dataOptionPrototype(),
766 [ 'key' => Uuid::create(), 'label' => $values, ]),
767 ];
768 }
769 if (isset($options[-1])) {
770 throw new Exception(
771 $this->l->t(
772 'Option index -1 should not be present here, options: %s',
773 print_r($options, true)));
774 }
775 $options = array_values($options);
776 $protoType = $this->dataOptionPrototype();
777 $protoKeys = array_keys($protoType);
778 $result = [];
779 foreach ($options as $option) {
780 $keys = array_keys($option);
781 if ($keys !== $protoKeys) {
782 throw new InvalidArgumentException(
783 $this->l->t(
784 'Prototype keys "%s" and options keys "%s" differ',
785 [ implode(',', $protoKeys), implode(',', $keys) ])
786 );
787 }
788 if ($trimInactive && !empty($option['deleted'])) {
789 continue;
790 }
791 $result[(string)$option['key']] = $option;
792 }
793 if ($addProto) {
794 $result[$protoType['key']] = $protoType;
795 }
796 return $result;
797 }
