365 public function participantFieldSurcharge(?string $key, ?string $value, Entities\ProjectParticipantField $participantField):RationalNumber
366 {
367 switch ($participantField->getMultiplicity()) {
368 case Multiplicity::SIMPLE():
369 return RationalNumber::create($value);
370 case Multiplicity::GROUPOFPEOPLE():
371 if (empty($key)) {
372 break;
373 }
374 return RationalNumber::create($participantField->getManagementOption()->getData());
375 case Multiplicity::SINGLE():
376 if (empty($key)) {
377 break;
378 }
380 $dataOption = $participantField->getDataOptions()->first();
381 // Non empty value means "yes".
382 if ((string)$dataOption['key'] != $key) {
383 $this->logWarn('Stored value "' . $key . '" unequal to stored key "' . $dataOption['key'] . '"');
384 }
385 return RationalNumber::create($dataOption['data']);
386 case Multiplicity::GROUPSOFPEOPLE():
387 case Multiplicity::MULTIPLE():
388 if (empty($key)) {
389 break;
390 }
391 foreach ($participantField->getDataOptions() as $dataOption) {
392 if ((string)$dataOption['key'] == $key) {
393 return RationalNumber::create($dataOption['data']);
394 }
395 }
396 $this->logError('No data item for multiple choice key "'.$key.'"');
397 return RationalNumber::zero();
398 case Multiplicity::PARALLEL():
399 if (empty($key)) {
400 break;
401 }
402 $keys = Util::explode(',', $key);
403 $found = false;
405 $amount = RationalNumber::zero();
406 foreach ($participantField->getDataOptions() as $dataOption) {
407 if (array_search((string)$dataOption['key'], $keys) !== false) {
408 $amount->addEq($dataOption['data']);
409 $found = true;
410 }
411 }
412 if (!$found) {
413 $this->logError('No data item for parallel choice key "'.$key.'"');
414 }
415 return $amount;
416 case Multiplicity::RECURRING():
417 if (empty($key) || empty($value)) {
418 break;
419 }
420 // $keys = Util::explode(',', $key);
421 $values = Util::explodeIndexed($value);
422
423 $amount = RationalNumber::zero();
424 foreach ($values as $key => $value) {
425 if (!empty($value)) {
426 $amount->addEq($value);
427 }
428 }
429 return $amount;
430 }
431 return RationalNumber::zero();
432 }

732 public function explodeDataOptions(mixed $values, bool $addProto = true, bool $trimInactive = false):array
733 {
734 $options = empty($values) ? [] : (is_array($values) ? $values : json_decode($values, true));
735 if (is_string($options)) {
736 // perhaps double encoded, try ...
737 $options = json_decode($options, true);
738 }
739 if (is_string($values) && $values != '[]' && !empty($values) && empty($options)) {
740 $options = [
741 array_merge(
742 $this->dataOptionPrototype(),
743 [ 'key' => Uuid::create(), 'label' => $values, ]),
744 ];
745 }
746 if (isset($options[-1])) {
747 throw new Exception(
748 $this->l->t(
749 'Option index -1 should not be present here, options: %s',
750 print_r($options, true)));
751 }
752 $options = array_values($options);
753 $protoType = $this->dataOptionPrototype();
754 $protoKeys = array_keys($protoType);
755 $result = [];
756 foreach ($options as $option) {
757 $keys = array_keys($option);
758 if ($keys !== $protoKeys) {
759 throw new InvalidArgumentException(
760 $this->l->t(
761 'Prototype keys "%s" and options keys "%s" differ',
762 [ implode(',', $protoKeys), implode(',', $keys) ])
763 );
764 }
765 if ($trimInactive && !empty($option['deleted'])) {
766 continue;
767 }
768 $result[(string)$option['key']] = $option;
769 }
770 if ($addProto) {
771 $result[$protoType['key']] = $protoType;
772 }
773 return $result;
774 }
