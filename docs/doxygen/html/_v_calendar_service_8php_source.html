137 private function createVEventFromRequest(mixed $objectData):?VCalendar
138 {
139 if (!isset($objectData['repeat'])) {
140 $objectData['repeat'] = 'doesnotrepeat';
141 }
143 $start = $objectData['start'] ?? null;
144 if ($start instanceof DateTimeInterface) {
145 $timezone = $this->getDateTimeZone($start->getTimestamp());
146 $start = $start->setTimezone($timezone);
147 $objectData['from'] = $objectData['start']->format('d-m-Y');
148 if (!isset($objectData['allday'])) {
149 $objectData['fromtime'] = $start->format('H:i:s');
150 }
151 }
153 $end = $objectData['end'] ?? null;
154 if ($end instanceof DateTimeInterface) {
155 $timezone = $this->getDateTimeZone($end->getTimestamp());
156 $end = $end->setTimezone($timezone);
157 $objectData['to'] = $objectData['end']->format('d-m-Y');
158 if (!isset($objectData['allday'])) {
159 $objectData['totime'] = $end->format('H:i:s');
160 }
161 }
162 if (is_array($objectData['categories'])) {
163 $objectData['categories'] = implode(',', $objectData['categories']);
164 }
165 $vObject = $this->legacyCalendarObject->createVCalendarFromRequest($objectData);
166 if (empty($vObject) || empty($vObject->VEVENT)) {
167 return $vObject;
168 }
169 $this->addVAlarmsFromRequest($vObject, $vObject->VEVENT, $objectData);
170 $this->addParticipants($vObject, $vObject->VEVENT, $objectData);

251 public function addRelations(VCalendar $vRoot, VComponent $vComponent, mixed $objectData):void
252 {
253 if (empty($objectData['related'])) {
254 return;
255 }
256 $related = $objectData['related'];
257 if (empty(array_intersect(array_keys($related), ['PARENT', 'CHILD', 'SIBLING']))) {
258 $related = [ 'PARENT' => $related ];
259 }
260 $ownUid = $vComponent->UID;
261 unset($vComponent->{'RELATED-TO'});

263 foreach ($uids as $uid) {

268 'RELATED-TO',

645 {

755 private function updateVTodoFromRequest(VCalendar $vCalendar, array $request):VCalendar
756 {
757 if (!empty($this->validateVTodoRequest($request))) {
758 return null;
759 }
760
761 $vTodo = $vCalendar->VTODO;
762 $timezone = $this->getDateTimeZone();
763
764 $vTodo->{'LAST-MODIFIED'} = new DateTimeImmutable('now', new DateTimeZone('UTC'));
765 $vTodo->DTSTAMP = new DateTimeImmutable('now', new DateTimeZone('UTC'));
766
767 $vTodo->SUMMARY = $request['summary'];
768
769 if (!empty($request['due'])) {
770 $due = new DateTimeImmutable($request['due'], $timezone);
771 if ($request['allday']) {
772 $due = $due->setTime(0, 0, 0);
773 $vTodo->DUE = $due;
774 $vTodo->DUE['VALUE'] = 'DATE';
775 } else {
776 $vTodo->DUE = $due;
777 }
778 }
779 if (!empty($request['start'])) {
780 $dtStart = new DateTimeImmutable($request['start'], $timezone);
781 if ($request['allday']) {
782 $dtStart = $dtStart->setTime(0, 0, 0);
783 $vTodo->DTSTART = $dtStart;
784 $vTodo->DTSTART['VALUE'] = 'DATE';
785 } else {
786 $vTodo->DTSTART = $dtStart;
787 }
788 }
789
790 $optionalKeys = [
791 'priority' => 'PRIORITY',
792 'description' => 'DESCRIPTION',
793 'location' => 'LOCATION',
794 'status' => 'STATUS',
795 ];
796 foreach ($optionalKeys as $rqKey => $vKey) {
797 if (!empty($request[$rqKey])) {
798 $vTodo->{$vKey} = $request[$rqKey];
799 } else {
800 unset($vTodo->{$vKey});
801 }
802 }
803
804 $categories = [];
805 if (!empty($request['categories'])) {
806 $categories = explode(',', $request['categories']);
807 }
808 if (count($categories)) {
809 $vTodo->CATEGORIES = $categories;
810 } else {
811 unset($vTodo->{'CATEGORIES'});
812 }
813
814 $this->addVAlarmsFromRequest($vCalendar, $vCalendar->VTODO, $request);
815 $this->addParticipants($vCalendar, $vCalendar->VTODO, $request);
816 $this->addRelations($vCalendar, $vCalendar->VTODO, $request);
817
818 return $vCalendar;
819 }

835 private function createVAlarmFromSeconds(
836 VCalendar $vCalendar,
837 int $seconds,
838 ?string $related = null,
839 ?string $description = null,
840 string $action = self::ALARM_ACTION_DISPLAY,
841 ):?VAlarm {
842 if (empty($seconds)) {
843 return null;
844 }
845 /*
846BEGIN:VALARM
847DESCRIPTION:
848ACTION:DISPLAY
849TRIGGER;VALUE=DURATION:-P1D
850X-KDE-KCALCORE-ENABLED:TRUE
851END:VALARM
852 */
853 $inverted = $seconds < 0;
854 if ($inverted) {
855 $seconds = -$seconds;
856 $related = $related ?? self::VALARM_FROM_END;
857 } else {
858 $related = $related ?? self::VALARM_FROM_START;
859 }
860 $vAlarm = $vCalendar->createComponent('VALARM');
861 $vAlarm->DESCRIPTION = !empty($description) ? $description : $this->l->t('Default Event Notification');
862 $vAlarm->ACTION = $action;
863 if ($inverted) {
864 $dinterval = (new DateTimeImmutable)->add(new DateInterval('PT' . ($seconds + 1) . 'S'));
865 } else {
866 $dinterval = (new DateTimeImmutable)->sub(new DateInterval('PT' . $seconds . 'S'));
867 }
868 $interval = $dinterval->diff(new DateTimeImmutable);
869 $alarmValue = sprintf(
870 '%sP%s%s%s%s',
871 $interval->format('%r'),
872 $interval->format('%d') > 0 ? $interval->format('%dD') : null,
873 ($interval->format('%h') > 0 || $interval->format('%i') > 0) ? 'T' : null,
874 $interval->format('%h') > 0 ? $interval->format('%hH') : null,
875 $interval->format('%i') > 0 ? $interval->format('%iM') : null);
876 $vAlarm->add('TRIGGER', $alarmValue, [ 'VALUE' => 'DURATION', 'RELATED' => $related ]);
877
878 // $this->logInfo('TRY CREATE ALARM ' . $vAlarm->TRIGGER . ' SECONDS ' . ($inverted ? '-' : '') . $seconds);
879
880 return $vAlarm;
881 }
