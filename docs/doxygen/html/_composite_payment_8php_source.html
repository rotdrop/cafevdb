702 public function generateSubject(?Closure $transliterate = null):string
703 {
704 if (empty($transliterate)) {
705 $transliterate = fn(string $x) => $x;
706 }
707
708 $projectName = $this->project->getName();
709 $projectYear = $this->project->getYear();
710 if (substr($projectName, -4) == (string)$projectYear) {
711 $projectName = substr($projectName, 0, -4);
712 $yearSeparator = '';
713 } else {
714 $yearSeparator = '-';
715 }
716
717 $projectName = Util::dashesToCamelCase(
718 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
719 capitalizeFirstCharacter: true,
720 dashes: ' ',
721 );
722
723 // collect the DatabaseStorageFolder's
724 $balanceDocuments = [];
725 if (!empty($this->balanceDocumentsFolder)) {
726 $folderName = $this->balanceDocumentsFolder->getName();
727 if (!empty($folderName)) {
728 $balanceDocuments[$folderName] = true;
729 }
730 }
732 foreach ($this->projectPayments as $projectPayment) {
733 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
734 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
735 if (!empty($folderName)) {
736 $balanceDocuments[$folderName] = true;
737 }
738 }
739
740 $yearSequences = [];
741 foreach (array_keys($balanceDocuments) as $folderName) {
742 $sequence = substr($folderName, -3);
743 $year = substr($folderName, -8, 4);
744 $yearSequences[$year] = $yearSequences[$year] ?? [];
745 $yearSequences[$year][] = $sequence;
746 }
747 ksort($yearSequences);
748
749 $balanceSequences = [];
750 if (count($yearSequences) == 1) {
751 $projectYear = array_keys($yearSequences)[0];
752 $projectName .= $yearSeparator . ($projectYear % 100);
753 $sequences = $yearSequences[$projectYear];
754 sort($sequences, SORT_NUMERIC);
755 $len = (int)log10(end($sequences)) + 1;
756 foreach ($sequences as $sequence) {
757 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
758 }
759 } else {
760 foreach ($yearSequences as $year => $sequences) {
761 sort($sequences, SORT_NUMERIC);
762 $len = (int)log10(end($sequences)) + 1;
763 foreach ($sequences as $sequence) {
764 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
765 }
766 }
767 }
768
769 $subjectPrefix = $projectName;
770 if (count($balanceSequences) >= 1) {
771 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
772 }
773
774 $subjects = $this->projectPayments
775 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
776 ->toArray();
777 natsort($subjects);
778
779 // try to compactify the composite subject by attempting to group similar subjects
780 $oldPrefix = false;
781 $postfix = [];
782 $purpose = '';
783 foreach ($subjects as $subject) {
784 $parts = array_map(
785 function(string $part) {
786 // This should remain human-readable, therefore "camel-casing"
787 // should only be performed on "real words".
788 $part = Util::normalizeSpaces($part);
789 $words = explode(' ', $part);
790 $result = '';
791 $prevIsWord = true;
792 foreach ($words as $word) {
793 // Difficult, sequences of 4 digits may as well form part of important booking ids.
794 // $word = preg_replace('/(?<![0-9])[0-9]{2}([0-9]{2})(?![0-9])/', '\1', $word);
795 if (preg_match('/^[a-zA-Z-_]+$/', $word)) {
796 $word = Util::dashesToCamelCase($word, capitalizeFirstCharacter: true, dashes: '-_');
797 if (!$prevIsWord) {
798 $result .= ' ';
799 }
800 $prevIsWord = true;
801 } else {
802 $result .= ' ';
803 $prevIsWord = false;
804 }
805 $result .= $word;
806 }
807 return $result;
808 },
809 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
810 );
811 $prefix = $parts[0];
812 if (count($parts) < 2 || $oldPrefix != $prefix) {
813 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, array_unique($postfix));
814 if (strlen($purpose) > 0) {
815 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
816 }
817 $purpose .= $prefix;
818 if (count($parts) >= 2) {
819 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
820 $oldPrefix = $prefix;
821 } else {
822 $oldPrefix = false;
823 }
824 $postfix = [];
825 }
826 if (count($parts) >= 2) {
827 $postfix = array_merge($postfix, array_splice($parts, 1));
828 }
829 }
830 if (!empty($postfix)) {
831 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, array_unique($postfix));
832 }
833 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
834
835 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
836
837 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
838 $subject = substr(
839 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
840 0,
841 self::SUBJECT_FIELD_LENGTH,
842 );
843 }
844
845 return $subject;
846 }
