733 public function generateSubject(?Closure $transliterate = null):string
734 {
735 if (empty($transliterate)) {
736 $transliterate = fn(string $x) => $x;
737 }
738
739 $projectName = $this->project->getName();
740 $projectYear = $this->project->getYear();
741 if (substr($projectName, -4) == (string)$projectYear) {
742 $projectName = substr($projectName, 0, -4);
743 $yearSeparator = '';
744 } else {
745 $yearSeparator = '-';
746 }
747
748 $projectName = Util::dashesToCamelCase(
749 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
750 capitalizeFirstCharacter: true,
751 dashes: ' ',
752 );
753
754 // collect the DatabaseStorageFolder's
755 $balanceDocuments = [];
756 if (!empty($this->balanceDocumentsFolder)) {
757 $folderName = $this->balanceDocumentsFolder->getName();
758 if (!empty($folderName)) {
759 $balanceDocuments[$folderName] = true;
760 }
761 }
763 foreach ($this->projectPayments as $projectPayment) {
764 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
765 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
766 if (!empty($folderName)) {
767 $balanceDocuments[$folderName] = true;
768 }
769 }
770
771 $yearSequences = [];
772 foreach (array_keys($balanceDocuments) as $folderName) {
773 $sequence = substr($folderName, -3);
774 $year = substr($folderName, -8, 4);
775 $yearSequences[$year] = $yearSequences[$year] ?? [];
776 $yearSequences[$year][] = $sequence;
777 }
778 ksort($yearSequences);
779
780 $balanceSequences = [];
781 if (count($yearSequences) == 1) {
782 $projectYear = array_keys($yearSequences)[0];
783 $projectName .= $yearSeparator . ($projectYear % 100);
784 $sequences = $yearSequences[$projectYear];
785 sort($sequences, SORT_NUMERIC);
786 $len = (int)log10(end($sequences)) + 1;
787 foreach ($sequences as $sequence) {
788 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
789 }
790 } else {
791 foreach ($yearSequences as $year => $sequences) {
792 sort($sequences, SORT_NUMERIC);
793 $len = (int)log10(end($sequences)) + 1;
794 foreach ($sequences as $sequence) {
795 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
796 }
797 }
798 }
799
800 $subjectPrefix = $projectName;
801 if (count($balanceSequences) >= 1) {
802 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
803 }
804
805 $subjects = $this->projectPayments
806 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
807 ->toArray();
808 natsort($subjects);
809
810 // try to compactify the composite subject by attempting to group similar subjects
811 $oldPrefix = false;
812 $postfix = [];
813 $purpose = '';
814 foreach ($subjects as $subject) {
815 $parts = array_map(
816 fn($part) => Util::dashesToCamelCase(
817 preg_replace('/([^0-9]|^)[0-9]{2}([0-9]{2})([^0-9]|$)/', '\1\2\3', $part),
818 capitalizeFirstCharacter: true,
819 dashes: ' -_',
820 ),
821 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
822 );
823 $prefix = $parts[0];
824 if (count($parts) < 2 || $oldPrefix != $prefix) {
825 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
826 if (strlen($purpose) > 0) {
827 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
828 }
829 $purpose .= $prefix;
830 if (count($parts) >= 2) {
831 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
832 $oldPrefix = $prefix;
833 } else {
834 $oldPrefix = false;
835 }
836 $postfix = [];
837 }
838 if (count($parts) >= 2) {
839 $postfix = array_merge($postfix, array_splice($parts, 1));
840 }
841 }
842 if (!empty($postfix)) {
843 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
844 }
845 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
846
847 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
848
849 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
850 $subject = substr(
851 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
852 0,
853 self::SUBJECT_FIELD_LENGTH,
854 );
855 }
856
857 return $subject;
858 }
