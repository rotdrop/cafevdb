706 public function generateSubject(?Closure $transliterate = null): string
707 {
708 if (empty($transliterate)) {
709 $transliterate = fn(string $x) => $x;
710 }
711
712 $projectName = $this->project->getName();
713 $projectYear = $this->project->getYear();
714 if (substr($projectName, -4) == (string)$projectYear) {
715 $projectName = substr($projectName, 0, -4);
716 $yearSeparator = '';
717 } else {
718 $yearSeparator = '-';
719 }
720
721 $projectName = Util::dashesToCamelCase(
722 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
723 capitalizeFirstCharacter: true,
724 dashes: ' ',
725 );
726
727 // collect the DatabaseStorageFolder's
728 $balanceDocuments = [];
729 if (!empty($this->balanceDocumentsFolder)) {
730 $folderName = $this->balanceDocumentsFolder->getName();
731 if (!empty($folderName)) {
732 $balanceDocuments[$folderName] = true;
733 }
734 }
736 foreach ($this->projectPayments as $projectPayment) {
737 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
738 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
739 if (!empty($folderName)) {
740 $balanceDocuments[$folderName] = true;
741 }
742 }
743
744 $yearSequences = [];
745 foreach (array_keys($balanceDocuments) as $folderName) {
746 $sequence = substr($folderName, -3);
747 $year = substr($folderName, -8, 4);
748 $yearSequences[$year] = $yearSequences[$year] ?? [];
749 $yearSequences[$year][] = $sequence;
750 }
751 ksort($yearSequences);
752
753 $balanceSequences = [];
754 if (count($yearSequences) == 1) {
755 $projectYear = array_keys($yearSequences)[0];
756 $projectName .= $yearSeparator . ($projectYear % 100);
757 $sequences = $yearSequences[$projectYear];
758 sort($sequences, SORT_NUMERIC);
759 $len = (int)log10(end($sequences)) + 1;
760 foreach ($sequences as $sequence) {
761 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
762 }
763 } else {
764 foreach ($yearSequences as $year => $sequences) {
765 sort($sequences, SORT_NUMERIC);
766 $len = (int)log10(end($sequences)) + 1;
767 foreach ($sequences as $sequence) {
768 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
769 }
770 }
771 }
772
773 $subjectPrefix = $projectName;
774 if (count($balanceSequences) >= 1) {
775 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
776 }
777
778 $subjects = $this->projectPayments
779 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
780 ->toArray();
781 natsort($subjects);
782
783 // try to compactify the composite subject by attempting to group similar subjects
784 $oldPrefix = false;
785 $postfix = [];
786 $purpose = '';
787 foreach ($subjects as $subject) {
788 $parts = array_map(
789 function(string $part) {
790 // This should remain human-readable, therefore "camel-casing"
791 // should only be performed on "real words".
792 $part = Util::normalizeSpaces($part);
793 $words = explode(' ', $part);
794 $result = '';
795 $prevIsWord = true;
796 foreach ($words as $word) {
797 // Difficult, sequences of 4 digits may as well form part of important booking ids.
798 // $word = preg_replace('/(?<![0-9])[0-9]{2}([0-9]{2})(?![0-9])/', '\1', $word);
799 if (preg_match('/^[a-zA-Z-_]+$/', $word)) {
800 $word = Util::dashesToCamelCase($word, capitalizeFirstCharacter: true, dashes: '-_');
801 if (!$prevIsWord) {
802 $result .= ' ';
803 }
804 $prevIsWord = true;
805 } else {
806 $result .= ' ';
807 $prevIsWord = false;
808 }
809 $result .= $word;
810 }
811 return $result;
812 },
813 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
814 );
815 $prefix = $parts[0];
816 if (count($parts) < 2 || $oldPrefix != $prefix) {
817 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, array_unique($postfix));
818 if (strlen($purpose) > 0) {
819 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
820 }
821 $purpose .= $prefix;
822 if (count($parts) >= 2) {
823 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
824 $oldPrefix = $prefix;
825 } else {
826 $oldPrefix = false;
827 }
828 $postfix = [];
829 }
830 if (count($parts) >= 2) {
831 $postfix = array_merge($postfix, array_splice($parts, 1));
832 }
833 }
834 if (!empty($postfix)) {
835 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, array_unique($postfix));
836 }
837 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
838
839 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
840
841 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
842 $subject = substr(
843 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
844 0,
845 self::SUBJECT_FIELD_LENGTH,
846 );
847 }
848
849 return $subject;
850 }
