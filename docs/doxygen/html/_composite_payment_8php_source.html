743 public function generateSubject(?Closure $transliterate = null):string
744 {
745 if (empty($transliterate)) {
746 $transliterate = fn(string $x) => $x;
747 }
748
749 $projectName = $this->project->getName();
750 $projectYear = $this->project->getYear();
751 if (substr($projectName, -4) == (string)$projectYear) {
752 $projectName = substr($projectName, 0, -4);
753 $yearSeparator = '';
754 } else {
755 $yearSeparator = '-';
756 }
757
758 $projectName = Util::dashesToCamelCase(
759 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
760 capitalizeFirstCharacter: true,
761 dashes: ' ',
762 );
763
764 // collect the DatabaseStorageFolder's
765 $balanceDocuments = [];
766 if (!empty($this->balanceDocumentsFolder)) {
767 $folderName = $this->balanceDocumentsFolder->getName();
768 if (!empty($folderName)) {
769 $balanceDocuments[$folderName] = true;
770 }
771 }
773 foreach ($this->projectPayments as $projectPayment) {
774 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
775 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
776 if (!empty($folderName)) {
777 $balanceDocuments[$folderName] = true;
778 }
779 }
780
781 $yearSequences = [];
782 foreach (array_keys($balanceDocuments) as $folderName) {
783 $sequence = substr($folderName, -3);
784 $year = substr($folderName, -8, 4);
785 $yearSequences[$year] = $yearSequences[$year] ?? [];
786 $yearSequences[$year][] = $sequence;
787 }
788 ksort($yearSequences);
789
790 $balanceSequences = [];
791 if (count($yearSequences) == 1) {
792 $projectYear = array_keys($yearSequences)[0];
793 $projectName .= $yearSeparator . ($projectYear % 100);
794 $sequences = $yearSequences[$projectYear];
795 sort($sequences, SORT_NUMERIC);
796 $len = (int)log10(end($sequences)) + 1;
797 foreach ($sequences as $sequence) {
798 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
799 }
800 } else {
801 foreach ($yearSequences as $year => $sequences) {
802 sort($sequences, SORT_NUMERIC);
803 $len = (int)log10(end($sequences)) + 1;
804 foreach ($sequences as $sequence) {
805 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
806 }
807 }
808 }
809
810 $subjectPrefix = $projectName;
811 if (count($balanceSequences) >= 1) {
812 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
813 }
814
815 $subjects = $this->projectPayments
816 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
817 ->toArray();
818 natsort($subjects);
819
820 // try to compactify the composite subject by attempting to group similar subjects
821 $oldPrefix = false;
822 $postfix = [];
823 $purpose = '';
824 foreach ($subjects as $subject) {
825 $parts = array_map(
826 fn($part) => Util::dashesToCamelCase(
827 preg_replace('/(?<![0-9])[0-9]{2}([0-9]{2})(?![0-9])/', '\1', $part),
828 capitalizeFirstCharacter: true,
829 dashes: ' -_',
830 ),
831 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
832 );
833 $prefix = $parts[0];
834 if (count($parts) < 2 || $oldPrefix != $prefix) {
835 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
836 if (strlen($purpose) > 0) {
837 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
838 }
839 $purpose .= $prefix;
840 if (count($parts) >= 2) {
841 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
842 $oldPrefix = $prefix;
843 } else {
844 $oldPrefix = false;
845 }
846 $postfix = [];
847 }
848 if (count($parts) >= 2) {
849 $postfix = array_merge($postfix, array_splice($parts, 1));
850 }
851 }
852 if (!empty($postfix)) {
853 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
854 }
855 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
856
857 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
858
859 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
860 $subject = substr(
861 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
862 0,
863 self::SUBJECT_FIELD_LENGTH,
864 );
865 }
866
867 return $subject;
868 }
