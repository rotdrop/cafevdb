735 public function generateSubject(?Closure $transliterate = null):string
736 {
737 if (empty($transliterate)) {
738 $transliterate = fn(string $x) => $x;
739 }
740
741 $projectName = $this->project->getName();
742 $projectYear = $this->project->getYear();
743 if (substr($projectName, -4) == (string)$projectYear) {
744 $projectName = substr($projectName, 0, -4);
745 $yearSeparator = '';
746 } else {
747 $yearSeparator = '-';
748 }
749
750 $projectName = Util::dashesToCamelCase(
751 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
752 capitalizeFirstCharacter: true,
753 dashes: ' ',
754 );
755
756 // collect the DatabaseStorageFolder's
757 $balanceDocuments = [];
758 if (!empty($this->balanceDocumentsFolder)) {
759 $folderName = $this->balanceDocumentsFolder->getName();
760 if (!empty($folderName)) {
761 $balanceDocuments[$folderName] = true;
762 }
763 }
765 foreach ($this->projectPayments as $projectPayment) {
766 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
767 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
768 if (!empty($folderName)) {
769 $balanceDocuments[$folderName] = true;
770 }
771 }
772
773 $yearSequences = [];
774 foreach (array_keys($balanceDocuments) as $folderName) {
775 $sequence = substr($folderName, -3);
776 $year = substr($folderName, -8, 4);
777 $yearSequences[$year] = $yearSequences[$year] ?? [];
778 $yearSequences[$year][] = $sequence;
779 }
780 ksort($yearSequences);
781
782 $balanceSequences = [];
783 if (count($yearSequences) == 1) {
784 $projectYear = array_keys($yearSequences)[0];
785 $projectName .= $yearSeparator . ($projectYear % 100);
786 $sequences = $yearSequences[$projectYear];
787 sort($sequences, SORT_NUMERIC);
788 $len = (int)log10(end($sequences)) + 1;
789 foreach ($sequences as $sequence) {
790 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
791 }
792 } else {
793 foreach ($yearSequences as $year => $sequences) {
794 sort($sequences, SORT_NUMERIC);
795 $len = (int)log10(end($sequences)) + 1;
796 foreach ($sequences as $sequence) {
797 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
798 }
799 }
800 }
801
802 $subjectPrefix = $projectName;
803 if (count($balanceSequences) >= 1) {
804 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
805 }
806
807 $subjects = $this->projectPayments
808 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
809 ->toArray();
810 natsort($subjects);
811
812 // try to compactify the composite subject by attempting to group similar subjects
813 $oldPrefix = false;
814 $postfix = [];
815 $purpose = '';
816 foreach ($subjects as $subject) {
817 $parts = array_map(
818 fn($part) => Util::dashesToCamelCase(
819 preg_replace('/([^0-9]|^)[0-9]{2}([0-9]{2})([^0-9]|$)/', '\1\2\3', $part),
820 capitalizeFirstCharacter: true,
821 dashes: ' -_',
822 ),
823 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
824 );
825 $prefix = $parts[0];
826 if (count($parts) < 2 || $oldPrefix != $prefix) {
827 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
828 if (strlen($purpose) > 0) {
829 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
830 }
831 $purpose .= $prefix;
832 if (count($parts) >= 2) {
833 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
834 $oldPrefix = $prefix;
835 } else {
836 $oldPrefix = false;
837 }
838 $postfix = [];
839 }
840 if (count($parts) >= 2) {
841 $postfix = array_merge($postfix, array_splice($parts, 1));
842 }
843 }
844 if (!empty($postfix)) {
845 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
846 }
847 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
848
849 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
850
851 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
852 $subject = substr(
853 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
854 0,
855 self::SUBJECT_FIELD_LENGTH,
856 );
857 }
858
859 return $subject;
860 }
