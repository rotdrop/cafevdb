713 public function generateSubject(?Closure $transliterate = null):string
714 {
715 if (empty($transliterate)) {
716 $transliterate = fn(string $x) => $x;
717 }
718
719 $projectName = $this->project->getName();
720 $projectYear = $this->project->getYear();
721 if (substr($projectName, -4) == (string)$projectYear) {
722 $projectName = substr($projectName, 0, -4);
723 $yearSeparator = '';
724 } else {
725 $yearSeparator = '-';
726 }
727
728 $projectName = Util::dashesToCamelCase(
729 $transliterate(strtolower(Util::camelCaseToDashes($projectName, separator: ' '))),
730 capitalizeFirstCharacter: true,
731 dashes: ' ',
732 );
733
734 // collect the DatabaseStorageFolder's
735 $balanceDocuments = [];
736 if (!empty($this->balanceDocumentsFolder)) {
737 $folderName = $this->balanceDocumentsFolder->getName();
738 if (!empty($folderName)) {
739 $balanceDocuments[$folderName] = true;
740 }
741 }
743 foreach ($this->projectPayments as $projectPayment) {
744 $documentFolder = $projectPayment->getBalanceDocumentsFolder();
745 $folderName = !empty($documentFolder) ? $documentFolder->getName() : null;
746 if (!empty($folderName)) {
747 $balanceDocuments[$folderName] = true;
748 }
749 }
750
751 $yearSequences = [];
752 foreach (array_keys($balanceDocuments) as $folderName) {
753 $sequence = substr($folderName, -3);
754 $year = substr($folderName, -8, 4);
755 $yearSequences[$year] = $yearSequences[$year] ?? [];
756 $yearSequences[$year][] = $sequence;
757 }
758 ksort($yearSequences);
759
760 $balanceSequences = [];
761 if (count($yearSequences) == 1) {
762 $projectYear = array_keys($yearSequences)[0];
763 $projectName .= $yearSeparator . ($projectYear % 100);
764 $sequences = $yearSequences[$projectYear];
765 sort($sequences, SORT_NUMERIC);
766 $len = (int)log10(end($sequences)) + 1;
767 foreach ($sequences as $sequence) {
768 $balanceSequences[] = sprintf('%0' . $len . 'd', $sequence);
769 }
770 } else {
771 foreach ($yearSequences as $year => $sequences) {
772 sort($sequences, SORT_NUMERIC);
773 $len = (int)log10(end($sequences)) + 1;
774 foreach ($sequences as $sequence) {
775 $balanceSequences[] = ($year % 100) . '-' . sprintf('%0' . $len . 'd', $sequence);
776 }
777 }
778 }
779
780 $subjectPrefix = $projectName;
781 if (count($balanceSequences) >= 1) {
782 $subjectPrefix .= '-' . implode(trim(self::SUBJECT_PREFIX_SEPARATOR), $balanceSequences);
783 }
784
785 $subjects = $this->projectPayments
786 ->map(fn(ProjectPayment $payment) => $transliterate($payment->getSubject()))
787 ->toArray();
788 natsort($subjects);
789
790 // try to compactify the composite subject by attempting to group similar subjects
791 $oldPrefix = false;
792 $postfix = [];
793 $purpose = '';
794 foreach ($subjects as $subject) {
795 $parts = array_map(
796 function(string $part) {
797 // This should remain human-readable, therefore "camel-casing"
798 // should only be performed on "real words".
799 $part = Util::normalizeSpaces($part);
800 $words = explode(' ', $part);
801 $result = '';
802 $prevIsWord = true;
803 foreach ($words as $word) {
804 $word = preg_replace('/(?<![0-9])[0-9]{2}([0-9]{2})(?![0-9])/', '\1', $word);
805 if (preg_match('/^[a-zA-Z-_]+$/', $word)) {
806 $word = Util::dashesToCamelCase($word, capitalizeFirstCharacter: true, dashes: '-_');
807 if (!$prevIsWord) {
808 $result .= ' ';
809 }
810 $prevIsWord = true;
811 } else {
812 $result .= ' ';
813 $prevIsWord = false;
814 }
815 $result .= $word;
816 }
817 return $result;
818 },
819 Util::explode(trim(self::SUBJECT_OPTION_SEPARATOR), $subject, Util::TRIM|Util::OMIT_EMPTY_FIELDS|Util::ESCAPED),
820 );
821 $prefix = $parts[0];
822 if (count($parts) < 2 || $oldPrefix != $prefix) {
823 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
824 if (strlen($purpose) > 0) {
825 $purpose .= self::SUBJECT_GROUP_SEPARATOR;
826 }
827 $purpose .= $prefix;
828 if (count($parts) >= 2) {
829 $purpose .= self::SUBJECT_OPTION_SEPARATOR;
830 $oldPrefix = $prefix;
831 } else {
832 $oldPrefix = false;
833 }
834 $postfix = [];
835 }
836 if (count($parts) >= 2) {
837 $postfix = array_merge($postfix, array_splice($parts, 1));
838 }
839 }
840 if (!empty($postfix)) {
841 $purpose .= implode(self::SUBJECT_ITEM_SEPARATOR, $postfix);
842 }
843 $purpose = $transliterate(Util::unescapeDelimiter($purpose, trim(self::SUBJECT_OPTION_SEPARATOR)));
844
845 $subject = $subjectPrefix . self::SUBJECT_PREFIX_SEPARATOR . $purpose;
846
847 if (strlen($subject) > self::SUBJECT_FIELD_LENGTH) {
848 $subject = substr(
849 Util::shortenCamelCaseString($subject, self::SUBJECT_FIELD_LENGTH, minLen: 4),
850 0,
851 self::SUBJECT_FIELD_LENGTH,
852 );
853 }
854
855 return $subject;
856 }
