295 public function addProjectInstrument(MusicianInstrument $musicianInstrument, int $voice = ProjectInstrument::UNVOICED):ProjectParticipant
296 {
297 $musician = $musicianInstrument->getMusician();
298 $instrument = $musicianInstrument->getInstrument();
299 if ($this->projectInstruments->exists(fn(mixed $key, ProjectInstrument $projectInstrument)
300 =>
301 $projectInstrument->getMusician() == $musician
302 && $projectInstrument->getInstrument() == $instrument
303 && $projectInstrument->getVoice() == $voice)) {
304 return $this;
305 }
306 $projectInstrument = new ProjectInstrument($this, $musicianInstrument, $voice);
307 $instrumentationNumbers = $this->project->getInstrumentationNumbers()->matching(DBUtil::criteriaWhere([
308 'project' => $this->project,
309 'instrument' => $instrument,
310 'voice' => $voice,
311 ]));
312 if ($instrumentationNumbers->isEmpty()) {
313 $instrumentationNumber = new ProjectInstrumentationNumber(
314 project: $this->project,
315 instrument: $instrument,
316 voice: $voice,
317 );
318 } else {
319 $instrumentationNumber = $instrumentationNumbers->first();
320 }
321 $instrumentationNumber->getProjectInstruments()->set($musician->getId(), $projectInstrument);
322 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
323
324 return $this;
325 }

537 public function usage(string|ParticipationContext $participationContext = ParticipationContext::UNRESTRICTED):int
538 {
