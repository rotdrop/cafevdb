125 public function floatValue(string $value, ?string $locale = null)
126 {
127 $amount = preg_replace('/\s+/u', '', $value);
128 empty($locale) && $locale = $this->getLocale();
129 $locales = [ $locale, 'en_US_POSIX' ];
130 $parsed = false;
131 foreach ($locales as $locale) {
132 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
133
134 $decimalSeparator = $fmt->getSymbol(\NumberFormatter::DECIMAL_SEPARATOR_SYMBOL);
135 $groupingSeparator = $fmt->getSymbol(\NumberFormatter::GROUPING_SEPARATOR_SYMBOL);
136
137 $decPos = strpos($amount, $decimalSeparator);
138 $grpPos = strpos($amount, $groupingSeparator);
139
140 if ($grpPos !== false && $decPos === false) {
141 // unlikely: 1,000, we assume 1,000.00 would be used
142 continue;
143 } elseif ($decPos < $grpPos) {
144 // unlikely: 1.000,00 in en_US
145 continue;
146 }
147
148 $parsed = $fmt->parse($amount);
149 if ($parsed !== false) {
150 $percent = $fmt->getSymbol(\NumberFormatter::PERCENT_SYMBOL);
151 if (preg_match('/'.$percent.'/u', $amount)) {
152 $parsed /= 100.0;
153 }
154 break;
155 }
156 }
157 return $parsed !== false ? (float)$parsed : $parsed;
158 }

195 public function storageValue(?string $value, ?string $locale = null)
196 {
197 if ($value === null || $value === '') {
198 return null;
199 }
200 if ($value === '0') {
201 return (int)0;
202 }
203 $factor = [
204 'b' => 1,
205 'kb' => 1000,
206 'kib' => (1 << 10),
207 'mb' => (1000 * 1000),
208 'mib' => (1 << 20),
209 'gb' => (1000 * 1000 * 1000),
210 'gib' => (1 << 30),
211 'tb' => (1000 * 1000 * 1000 * 1000),
212 'tib' => (1 << 40),
213 'pb' => (1000 * 1000 * 1000 * 1000 * 1000),
214 'pib' => (1 << 50),
215 'eb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000),
216 'eib' => (1 << 60),
217 'zb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000),
218 'zib' => (1 << 70),
219 'yb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000),
220 'yib' => (1 << 80),
221 ];
222 $value = preg_replace('/\s+/u', '', $value);
223 $value = strtolower(
224 str_ireplace(
225 [ 'bytes', 'kilo', 'kibi', 'mega', 'mebi', 'giga', 'gibi', 'tera', 'tibi', 'peta', 'pebi', 'exa', 'exbi', 'zetta', 'zebi', 'yotta', 'yobi', ],
226 [ 'b', 'k', 'ki', 'm', 'mi', 'g', 'gi', 't', 'ti', 'p', 'pi', 'e', 'ei', 'z', 'zi', 'y', 'yi', ],
227 $value));
228
229 if (preg_match('/([-0-9,.]+)([kmgtpezy]?i?b?)?$/', $value, $matches)) {
230 $this->logInfo('MATCHES ' . print_r($matches, true));
231 $value = $this->floatValue($matches[1], $locale);
232 if (empty($value)) {
233 return null;
234 }
235 if (!empty($matches[2])) {
236 if (empty($factor[$matches[2]])) {
237 return null;
238 }
239 $value *= $factor[$matches[2]];
240 if ($value <= PHP_INT_MAX) {
241 $value = (int)$value;
242 }
243 }
244 return $value;
245 } else {
246 return null;
247 }
248 }

263 protected function formatStorageValue(int $bytes, string $format = 'binary', int $digits = 2, ?string $locale = null):?string
264 {
265 $units = [
266 'decimal' => [
267 '', 'k', 'M', 'G', 'T', 'P', 'E', 'Z', 'Y',
268 ],
269 'longDecimal' => [
270 'bytes', 'kilo', 'mega', 'giga', 'tera', 'peta', 'exa', 'zetta', 'yotta',
271 ],
272 'binary' => [
273 '', 'Ki', 'Mi', 'Gi', 'Ti', 'Pi', 'Ei', 'Zi', 'Yi',
274 ],
275 'longBinary' => [
276 '', 'kibi', 'mebi', 'gibi', 'tebi', 'pebi', 'exbi', 'zebi', 'yobi'
277 ],
278 ];
279 $bytesUnit = [
280 'decimal' => 'B',
281 'longDecimal' => ' ' . $this->l->t('bytes'),
282 'binary' => 'B',
283 'longBinary' => ' ' . $this->l->t('bytes'),
284 ];
285 $radix = [
286 'decimal' => 1000.0,
287 'longDecimal' => 1000.0,
288 'binary' => 1024.0,
289 'longBinary' => 1024.0,
290 ];
291
292 $units = $units[$format] ?? null;
293 $bytesUnit = $bytesUnit[$format] ?? null;
294 $radix = $radix[$format] ?? null;
295
296 if ($units === null || $bytesUnit === null || $radix === null) {
297 // maybe throw InvalidArgumentException
298 return null;
299 }
300
301 empty($locale) && $locale = $this->getLocale();
302
303 $floatVal = $bytes;
304 $exponent = 0;
305 $exponentLimit = count($units) - 1;
306 while ($floatVal > $radix && $exponent < $exponentLimit) {
307 $floatVal /= $radix;
308 ++$exponent;
309 }
310
311 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
312 $fmt->setAttribute(NumberFormatter::MIN_FRACTION_DIGITS, 0);
313 $fmt->setAttribute(NumberFormatter::MAX_FRACTION_DIGITS, $digits);
314 $stringVal = $fmt->format($floatVal);
315
316 return $stringVal . ' ' . $units[$exponent] . $bytesUnit;
317 }

349 protected function getCommonPath(array $paths, bool $leadingSlash = true):string
350 {
351 // $this->logInfo('PATHS ' . print_r($paths, true));
352 $lastOffset = (int)$leadingSlash;
353 $common = $leadingSlash ? '/' : '';
354 while (($index = strpos($paths[0], '/', $lastOffset)) !== false) {
355 $dirLen = $index - $lastOffset + 1; // include /
356 $dir = substr($paths[0], $lastOffset, $dirLen);
357 foreach ($paths as $path) {
358 if (substr($path, $lastOffset, $dirLen) != $dir) {
359 return $leadingSlash ? substr($common, 0, -1) : $common;
360 }
361 }
362 $common .= $dir;
363 $lastOffset = $index + 1;
364 }
365 return $leadingSlash ? substr($common, 0, -1) : $common;
366 }
