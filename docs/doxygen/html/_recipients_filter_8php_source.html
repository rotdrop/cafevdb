492 private function pushHistory(bool $fastMode = false):void
493 {
494 $filter = [];
495 foreach (self::HISTORY_KEYS as $key) {
496 $filter[$key] = $this->cgiValue($key, []);
497 }
498
499 if (!$fastMode) {
500 // exclude musicians deselected by the filter from the set of
501 // selected recipients before recording the history
502 $filter['selectedRecipients'] =
503 array_intersect(
504 $filter['selectedRecipients'],
505 array_keys($this->eMailsDpy)
506 );
507
508 // tweak: sort the selected recipients by key
509 }
510
511 sort($filter['selectedRecipients']);
512 $md5 = md5(serialize($filter));
513
514 /* Avoid pushing duplicate history entries. If the new
515 * filter-record matches the current one, then simply discard
516 * the new filter. This is in order to avoid bloating the
517 * history records by repeated user submits of the same filter
518 * or duplicated double-clicks.
519 */
520 $historyFilter = $this->filterHistory[$this->historyPosition];
521 if ($historyFilter['md5'] != $md5) {
522 // Pushing a new record removes the history up to the current
523 // position, i.e. redos are not longer possible then. This
524 // seems to be common behaviour as "re-doing" is no longer
525 // well defined in this case.
526 array_splice($this->filterHistory, 0, $this->historyPosition);
527 array_unshift($this->filterHistory, [ 'md5' => $md5, 'data' => $filter ]);
528 $this->historyPosition = 0;
529 $this->historySize = count($this->filterHistory);
530 while ($this->historySize > self::MAX_HISTORY_SIZE) {
531 array_pop($this->filterHistory);
532 --$this->historySize;
533 }
534 }
535 }

544 private function applyHistory(int $offset):void
545 {
546 $newPosition = $this->historyPosition + $offset;
547
548 // Check for valid position.
549 if ($newPosition >= $this->historySize || $newPosition < 0) {
550 if ($this->shouldDebug(ConfigService::DEBUG_EMAILFORM)) {
551 throw new OutOfBoundsException(
552 $this->l->t(
553 'Invalid history position %d request, history size is %d',
554 [ $newPosition, $this->historySize ])
555 );
556 }
557 return;
558 }
559
560 // Move to the new history position.
561 $this->historyPosition = $newPosition;
562
563 $filter = $this->filterHistory[$newPosition]['data'];
564 foreach (self::HISTORY_KEYS as $key) {
565 $this->cgiData[$key] = $filter[$key];
566 }

635 private function fetchMusicians(array $criteria):void
636 {
637 // add the instruments filter
638 if (!empty($this->instrumentsFilter)) {
639 $criteria[] = [ 'instruments.instrument' => $this->instrumentsFilter ];
640 if ($this->projectId > 0 && ($this->userBase & self::MUSICIANS_EXCEPT_PROJECT) == 0) {
641 $criteria[] = [ 'projectInstruments.instrument' => $this->instrumentsFilter ];
642 $criteria[] = [ 'projectInstruments.project' => $this->projectId ];
643 }
644 }
645 if ($this->frozen && $this->projectId > 0) {
646 $criteria[] = [ 'id' => $this->emailRecs ];
647 }
648 if ($this->projectId <= 0) {
649 $criteria[] = [ '!defaultParticipationStatus' => $this->participationStatusBlackList() ];
650 } elseif (($this->userBase & self::MUSICIANS_EXCEPT_PROJECT) == 0) {
651 $criteria[] = [ 'projectParticipation.project' => $this->projectId ];
652 $criteria[] = [ '!projectParticipation.participationStatus' => $this->participationStatusBlackList() ];
653 }
654
655 $musicians = $this->musiciansRepository->findBy($criteria, [ 'id' => 'INDEX' ]);
656
657 // use the mailer-class we are using later anyway in order to
658 // validate email addresses syntactically now. Add broken
659 // addresses to the "brokenEMails" list.
660 $mailer = new PHPMailer;
661
663 foreach ($musicians as $rec => $musician) {
664
665 $displayName = $musician->getPublicName(true);
666 if (!empty($musician->getEmail())) {
667 // We allow comma separated multiple addresses
668 $musMail = explode(',', $musician->getEmail());
669 foreach ($musMail as $emailVal) {
670 if (!$mailer->validateAddress($emailVal)) {
671 $bad = htmlspecialchars($displayName.' <'.$emailVal.'>');
672 if (isset($this->brokenEMails[$rec])) {
673 $this->brokenEMails[$rec]['label'] .= ', '.$bad;
674 } else {
675 $this->brokenEMails[$rec] = [
676 'participant' => $this->projectId > 0 && $musician->isMemberOf($this->projectId),
677 'label' => $bad,
678 ];
679
680 }
681 } else {
682 $participant = !empty($this->project) ? $musician->getProjectParticipantOf($this->project) : null;
683 $isParticipant = !empty($participant);
684 $isNonParticipant = $this->projectId > 0 && !$isParticipant;
685 $userBase = 0;
686 if ($isParticipant) {
687 $userBase |= self::MUSICIANS_FROM_PROJECT;
688 } elseif ($isNonParticipant) {
689 $userBase |= self::MUSICIANS_EXCEPT_PROJECT;
690 }
691 if ($isParticipant) {
692 $status = $participant->getParticipationStatus();
693 } else {
694 $status = $musician->getDefaultParticipationStatus();
695 }
696 $emailRecord = [
697 'email' => $emailVal,
698 'name' => $displayName,
699 'status' => $status,
700 'project' => $this->projectId ?? 0,
701 'participant' => $isParticipant,
702 'userBase' => $userBase,
703 'dbdata' => $musician,
704 ];
705 $this->eMails[$rec] = $emailRecord;
706 $this->eMailsDpy[$rec] = htmlspecialchars($displayName.' <'.$emailVal.'>');
707 }
708 }
709 } else {
710 $this->brokenEMails[$rec] = [
711 'participant' => $this->projectId > 0 && $musician->isMemberOf($this->projectId),
712 'label' => htmlspecialchars($displayName),
713 ];
714 }
715 }

723 private function getMusiciansFromDB():void
724 {
725 $this->brokenEMails = [];
726 $this->eMails = [];
727 $this->eMailsDpy = []; // display records
728
729 $userBase = $this->userBase & self::DATABASE_MUSICIANS;
730 $criteria = [];
731 if (!empty($this->project) && $userBase != self::ALL_MUSICIANS) {
732
733 // OK, perhaps one should switch to sub-queries ...
734 //
735 // The problem is here that we need HAVING clauses in order to filter by
736 // the project id. WHERE and HAVING are implicitly joined by an AND, so
737 // we cannot first restrict to confirmed participation in the current
738 // project by WHERE and later allow non-project-musicians by HAVING.
739
740 if (($userBase & self::MUSICIANS_EXCEPT_PROJECT) && ($userBase & self::MUSICIANS_FROM_PROJECT)) {
741 $criteria[] = [
742 "(|projectParticipation.project@GROUP_CONCAT(IF(IDENTITY(%s) = {$this->projectId}, 1, NULL))" => null
743 // "(|projectParticipation.project@GROUP_CONCAT(CASE WHEN IDENTITY(%s) = {$this->projectId} THEN 1 ELSE NULL END))" => null
744 ];
745 if ($userBase & self::MUSICIANS_FROM_PROJECT_CONFIRMED) {
746 $criteria[] = [
747 "projectParticipation.project@GROUP_CONCAT(IF(IDENTITY(%s) = {$this->projectId}, projectParticipation.registration, NULL))" => 1
748 ];
749 } else {
750 $criteria[] = [
751 "projectParticipation.project@GROUP_CONCAT(IF(IDENTITY(%s) = {$this->projectId}, NULLIF(projectParticipation.registration, 0), NULL))" => null
752 ];
753 }
754 $criteria[] = [ ')@' => true ];
755 } elseif ($userBase & self::MUSICIANS_EXCEPT_PROJECT) {
756 $criteria[] = [
757 "projectParticipation.project@GROUP_CONCAT(IF(IDENTITY(%s) = {$this->projectId}, 1, NULL))" => null
758 ];
759 } elseif ($userBase & self::MUSICIANS_FROM_PROJECT) {
760 $criteria[] = [ '(&projectParticipation.project' => $this->projectId ];
761 switch ($userBase & self::MUSICIANS_FROM_PROJECT) {
762 case self::MUSICIANS_FROM_PROJECT:
763 break; // just all participants
764 case self::MUSICIANS_FROM_PROJECT_CONFIRMED:
765 $criteria[] = [ 'projectParticipation.registration' => 1 ];
766 break;
767 case self::MUSICIANS_FROM_PROJECT_PRELIMINARY:
768 $criteria[] = [ '(|projectParticipation.registration' => 0 ];
769 $criteria[] = [ 'projectParticipation.registration' => null ];
770 $criteria[] = [ ')' => true ];
771 break;
772 }
773 $criteria[] = [ ')' => true ];
774 }
775 if (empty($criteria)) {
776 return;
777 }
778 }
779 $this->fetchMusicians($criteria);
780
781 // Finally sort the display array
782 asort($this->eMailsDpy);
783 }

1393 public function substituteProjectMailingList(array &$selectedRecipients):bool
1394 {
1395 if (empty($this->project)) {
1396 // this is not a project email
1397 $this->logInfo('NO PROJECT');
1398 return false;
1399 }
1400
1401 if (!($this->userBase & self::MUSICIANS_FROM_PROJECT_CONFIRMED)) {
1402 // intentionally not for the list which should contain all confirmed
1403 // participants + further people.
1404 $this->logInfo('NOT FOR THE LIST');
1405 return false;
1406 }
1407
1408 $defaultByParticipationStatus = $this->defaultByParticipationStatus();
1409 $participationStatusFilter = $this->participationStatusFilter ?? [];
1410 if (array_intersect($participationStatusFilter, $defaultByParticipationStatus) !== $defaultByParticipationStatus) {
1411 // intentionally not for the list as less than the default status type
1412 // were addressed
1413 $this->logInfo('MEMBER FILTER INCOMPATIBLE');
1414 return false;
1415 }
1416
1417 if (!empty($this->instrumentsFilter)) {
1418 // intentionally not for the list, which is also archived
1419 $this->logInfo('INSTRUMENTS FILTER NOT EMPTY');
1420 return false;
1421 }
1422
1423 $numberOfPossibleRecipients = count($this->emailRecipientsChoices());
1424 $numberOfSelectedRecipients = count($selectedRecipients);
1425 if ($numberOfPossibleRecipients != $numberOfSelectedRecipients) {
1426 // although the recipient options intentionally include the list members
1427 // there is a particular choice of selected recipients, so this should
1428 // not go to the list.
1429 $this->logInfo('SPECIFIC CHOICE OF RECIPIENTS');
1430 return false;
1431 }
1432
1433 // keep this check last as it implies a REST call which is time-consuming.
1434 $listInfo = $this->getMailingListInfo();
1435 if (empty($listInfo)) {
1436 // well, there is not list ...
1437 return false;
1438 }
1439
1440 $listId = $listInfo['list_id'];
1442 $listsService = $this->di(MailingListsService::class);
1443 $listMembers = $listsService->findMembers($listId, flat: true, criteria: [
1444 MailingListsService::MEMBER_DELIVERY_STATUS => MailingListsService::DELIVERY_STATUS_ENABLED,
1445 MailingListsService::MEMBER_DELIVERY_MODE => MailingListsService::DELIVERY_MODE_REGULAR,
1446 ]);
1447 $bulkEmailFromAddress = $this->getConfigValue('emailfromaddress');
1448 Util::unsetValue($listMembers, $bulkEmailFromAddress);
1449 $recipientsByEmail = [];
1450 foreach ($selectedRecipients as $recipient) {
1451 $recipientsByEmail[$recipient['email']] = $recipient;
1452 }
1453 $remainingRecipients = array_diff_key($recipientsByEmail, array_flip($listMembers));
1454 if (!empty($remainingRecipients)) {
1455 $this->logInfo('NOT ON THE LIST ' . print_r(array_keys($remainingRecipients), true));
1456 }
1457
1458 $remainingListMembers = array_diff($listMembers, array_keys($recipientsByEmail));
1459 if (!empty($remainingListMembers)) {
1460 $this->logInfo('EXCESS MEMBERS ' . print_r($remainingListMembers, true));
1461 // It is ok if there are more members on the list, as intentionally this
1462 // should be mailing list traffic. return false;
1463 }
1464
1465 // throw away all recipients which are also reached by posting to the list
1466 // and add the list address as additional recipient.
1467 $selectedRecipients = array_merge(
1468 $this->selectedRecipients(self::PROJECT_MAILING_LIST),
1469 array_values($remainingRecipients),
1470 );
1471 return true;
1472 }
