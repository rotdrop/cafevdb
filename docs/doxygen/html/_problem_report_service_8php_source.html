92 public function submit(
93 array $userData,
94 array $errorData,
95 ?string $errorHtml,
96 ?string $userComment,
97 ):?array {
98
99 $this->logInfo(
100 'Requested problem report: '
101 . 'USER "' . print_r($userData, true)
102 . '" DATA "' . (int)!empty($errorData)
103 . '" HTML "' . (int)!empty($errorHtml)
104 . '" COMMENT "' . $userComment . '".'
105 );
106
107 // slightly more complicated than neccessary in case we want to add
108 // further communication channels like Github or Gilab issues in the
109 // future.
110 $notifications = [];
111 $exceptions = [];
112 try {
113 $notifications[self::REPORT_METHOD_EMAIL] = $this->submitViaEmail($userData, $errorData, $errorHtml, $userComment);
114 } catch (Throwable $e) {
115 $exceptions[self::REPORT_METHOD_EMAIL] = $e;
116 }
117
118 if (!empty($exceptions)) {
119 if (count($exceptions) == 1) {
120 throw reset($exceptions);
121 }
122 throw new Exceptions\EnduserNotificationException(
123 $this->l->t('All possibilities (i.e. %s) to submit your problem report have failed. Please use other communication channels to submit it.', implode(', ', array_keys($exceptions))),
124 );
125 }
126
127 return array_values($notifications);
128 }
