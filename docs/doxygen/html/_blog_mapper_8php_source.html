311 );
312 $cursor = $qb->execute();
313 $userPendingNotifications = false;
314 while ($row = $cursor->fetch()) {
315 $regex = '/(^|[,])+'.$userId.'($|[,])+/';
316 if (preg_match($regex, $row['reader']) !== 1) {
317 if (!$userPendingNotifications) {
318 $this->logInfo('User "' . $userId . '" has pending notifications.');
319 }
320 $userPendingNotifications = true;
321 }
322 }
