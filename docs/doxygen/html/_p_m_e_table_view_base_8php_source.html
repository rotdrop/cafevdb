747 public function beforeAnythingTrimAnything(PHPMyEdit &$pme, string $op, string $step, array &$oldValues, ?array &$changed, ?array &$newValues):bool
748 {
749 foreach ($newValues as $key => &$value) {
750 if (!is_scalar($value)) { // don't trim arrays
751 continue;
752 }
753
754 // Convert unicode space to ordinary space and trim
755 $value = Util::normalizeSpaces($value);
756
757 $fdn = $pme->fdn[$key];
758 if ($pme->col_has_multiple($fdn)) {
759 $value = preg_replace('/\s*,\s*/', DataConstants::VALUES_SEP, $value);
760 }
761
762 // @todo what is this
763 if ($op !== 'insert' && ($pme->skipped($fdn) || $pme->readonly($fdn))) {
764 continue;
765 }
766
767 $chgIdx = array_search($key, $changed);
768 if ($chgIdx === false && $oldValues[$key] != $value) {
769 $changed[] = $key;
770 }
771 if ($op == 'insert' && (string)$value === '' && $chgIdx !== false) {
772 unset($changed[$chgIdx]);
773 $changed = array_values($changed);
774 unset($newValues[$key]);
775 }
776 }
777
778 return true;
779 }

847 public function beforeUpdateDoUpdateAll(PHPMyEdit &$pme, string $op, string $step, array &$oldValues, ?array &$changed, ?array &$newValues):bool
848 {
849 // leave time-stamps to the ORM "behaviors"
850 Util::unsetValue($changed, 'updated');
851
852 $this->changeSetSize = count($changed);
853
854 $this->debugPrintValues($oldValues, $changed, $newValues, null, 'before');
855
856 $changeSets = [];
857 foreach ($changed as $field) {
858 if (str_ends_with($field, self::MASTER_FIELD_SUFFIX)) {
859 Util::unsetValue($changed, $field);
860 --$this->changeSetSize;
861 continue;
862 }
863 $fieldInfo = $this->joinTableField($field);
864 $changeSets[$fieldInfo['table']][$fieldInfo['column']] = $field;
865 }
866 $this->debug('CHANGESETS: '.print_r($changeSets, true));
867
868 $masterTable = null;
869 $masterEntity = null; // cache for a reference to the master entity
870 foreach ((empty($changeSets) ? [] : $this->joinStructure) as $table => $joinInfo) {
871 $changeSet = $changeSets[$table]??[];
872 if (empty($changeSet)) {
873 $this->debug('TABLE ' . $table . ' HAS EMPTY CHANGESET');
874 continue;
875 }
876 if ($joinInfo['flags'] & self::JOIN_READONLY) {
877 foreach ($changeSet as $column => $field) {
878 if (Util::unsetValue($changed, $field) > 0) {
879 --$this->changeSetSize;
880 }
881 }
882 $this->debug('TABLE ' . $table . ' IS SET READONLY');
883 continue;
884 }
885 $this->debug('TABLE ' . $table . ' HAS CHANGESET ' . print_r($changeSet, true));
886 if ($joinInfo['flags'] & self::JOIN_MASTER) {
887 // fill the identifier keys as they are left out for
888 // convenience in $this->joinStructure
889 foreach (array_keys($this->pme->key) as $key) {
890 $joinInfo['identifier'][$key] = $key;
891 }
892 $masterTable = $table;
893 }
894 $this->debug('CHANGESET '.$table.' '.print_r($changeSet, true));
895 $entityClass = $joinInfo['entity'];
896 /* $repository = */$this->getDatabaseRepository($entityClass);
897 $meta = $this->classMetadata($entityClass);
898
899 $multiple = null;
900 $identifier = [];
901 $identifierColumns = $meta->getIdentifierColumnNames();
902 foreach ($identifierColumns as $key) {
903 $pivotColumn = $this->findJoinColumnPivot($joinInfo, $key);
904 if ($pivotColumn === false) {
905 if (!empty($multiple)) {
906 throw new RuntimeException($this->l->t('Table "%s": missing identifier for field "%s" and grouping field "%s" already set.', [ $table, $key, $multiple ]));
907 }
908 // assume that the 'column' component contains the keys.
909 $keyField = self::joinTableFieldName($joinInfo, $joinInfo['column']);
910 $identifier[$key] = [
911 'old' => Util::explode(DataConstants::VALUES_SEP, Util::removeSpaces($oldValues[$keyField])),
912 'new' => Util::explode(DataConstants::VALUES_SEP, Util::removeSpaces($newValues[$keyField])),
913 ];
914 // Handle "deleted" information if present. This is meant for
915 // disabled instruments and the like. This stems from split-inputs
916 // where one select hold the deleted items and another one the active items.
917 //
918 // Unfortunaley this conflicts with the usual KEY:VALUE convention
919 // in other tables where KEY would be the ID of the database entity
920 // and VALUE in this case the deleted timestamp.
921 //
922 // We try to copy with this problem by checking for the colon syntax
923 // and not doing this hack if we find a usual
924 // KEY1:VALUE1,KEY2:VALUE2,... field.
925 $deletedField = self::joinTableFieldName($joinInfo, 'deleted');
926 if (!empty($oldValues[$deletedField]) && !str_contains($oldValues[$deletedField], self::JOIN_KEY_SEP)) {
927 $deletedKeys = Util::explode(DataConstants::VALUES_SEP, $oldValues[$deletedField]);
928 foreach (array_intersect($deletedKeys, $identifier[$key]['new']) as $deletedKey) {
929 $identifier[$key]['old'][] = $deletedKey;
930 $changeSet['deleted'] = $deletedField;
931 }
932 $identifier[$key]['old'] = array_values(array_unique($identifier[$key]['old']));
933 $newValues[$deletedField] = implode(DataConstants::VALUES_SEP, array_diff($deletedKeys, $identifier[$key]['new']));
934 }
935
936 $identifier[$key]['del'] = array_diff($identifier[$key]['old'], $identifier[$key]['new']);
937 $identifier[$key]['add'] = array_diff($identifier[$key]['new'], $identifier[$key]['old']);
938 $identifier[$key]['rem'] = array_intersect($identifier[$key]['new'], $identifier[$key]['old']);
939
940 if (isset($changeSet[$joinInfo['column']])) {
941 Util::unsetValue($changed, $changeSet[$joinInfo['column']]);
942 unset($changeSet[$joinInfo['column']]);
943 }
944 $multiple = $key;
945 if ($joinInfo['flags'] & self::JOIN_SINGLE_VALUED) {
946 if (count($identifier[$key]['old']) > 1 || count($identifier[$key]['old']) > 1) {
947 throw new RuntimeException(
948 $this->l->t(
949 'Identifier column "%s" for single-valued join-table "%s" contains more than one key: "%s" / "%s".', [
950 $key,
951 $table,
952 implode(',', $identifier[$key]['old']),
953 implode(',', $identifier[$key]['new']),
954 ]));
955 }
956 }
957 } else {
958 if (!is_array($pivotColumn)) {
959 $identifier[$key] = $oldValues[$pivotColumn];
960 } elseif (isset($pivotColumn['value'])) {
961 $identifier[$key] = $pivotColumn['value'];
