830 public function beforeAnythingTrimAnything(PHPMyEdit &$pme, string $op, string $step, array &$oldValues, ?array &$changed, ?array &$newValues):bool
831 {
832 foreach ($newValues as $key => &$value) {
833 if (!is_scalar($value)) { // don't trim arrays
834 continue;
835 }
836
837 // Convert unicode space to ordinary space and trim
838 $value = Util::normalizeSpaces($value);
839
840 $fdn = $pme->fdn[$key];
841 if ($pme->col_has_multiple($fdn)) {
842 $value = preg_replace('/\s*,\s*/', self::VALUES_SEP, $value);
843 }
844
845 // @todo what is this
846 if ($op !== 'insert' && ($pme->skipped($fdn) || $pme->readonly($fdn))) {
847 continue;
848 }
849
850 $chgIdx = array_search($key, $changed);
851 if ($chgIdx === false && $oldValues[$key] != $value) {
852 $changed[] = $key;
853 }
854 if ($op == 'insert' && (string)$value === '' && $chgIdx !== false) {
855 unset($changed[$chgIdx]);
856 $changed = array_values($changed);
857 unset($newValues[$key]);
858 }
859 }
860
861 return true;
862 }

930 public function beforeUpdateDoUpdateAll(PHPMyEdit &$pme, string $op, string $step, array &$oldValues, ?array &$changed, ?array &$newValues):bool
931 {
932 // leave time-stamps to the ORM "behaviors"
933 Util::unsetValue($changed, 'updated');
934
935 $this->changeSetSize = count($changed);
936
937 $this->debugPrintValues($oldValues, $changed, $newValues, null, 'before');
938
939 $changeSets = [];
940 foreach ($changed as $field) {
941 if (str_ends_with($field, self::MASTER_FIELD_SUFFIX)) {
942 Util::unsetValue($changed, $field);
943 --$this->changeSetSize;
944 continue;
945 }
946 $fieldInfo = $this->joinTableField($field);
947 $changeSets[$fieldInfo['table']][$fieldInfo['column']] = $field;
948 }
949 $this->debug('CHANGESETS: '.print_r($changeSets, true));
950
951 $masterTable = null;
952 $masterEntity = null; // cache for a reference to the master entity
953 foreach ((empty($changeSets) ? [] : $this->joinStructure) as $table => $joinInfo) {
954 $changeSet = $changeSets[$table]??[];
955 if (empty($changeSet)) {
956 $this->debug('TABLE ' . $table . ' HAS EMPTY CHANGESET');
957 continue;
958 }
959 if ($joinInfo['flags'] & self::JOIN_READONLY) {
960 foreach ($changeSet as $column => $field) {
961 if (Util::unsetValue($changed, $field) > 0) {
962 --$this->changeSetSize;
963 }
964 }
965 $this->debug('TABLE ' . $table . ' IS SET READONLY');
966 continue;
967 }
968 $this->debug('TABLE ' . $table . ' HAS CHANGESET ' . print_r($changeSet, true));
969 if ($joinInfo['flags'] & self::JOIN_MASTER) {
970 // fill the identifier keys as they are left out for
971 // convenience in $this->joinStructure
972 foreach (array_keys($this->pme->key) as $key) {
973 $joinInfo['identifier'][$key] = $key;
974 }
975 $masterTable = $table;
976 }
977 $this->debug('CHANGESET '.$table.' '.print_r($changeSet, true));
978 $entityClass = $joinInfo['entity'];
979 /* $repository = */$this->getDatabaseRepository($entityClass);
980 $meta = $this->classMetadata($entityClass);
981
982 $multiple = null;
983 $identifier = [];
984 $identifierColumns = $meta->getIdentifierColumnNames();
985 foreach ($identifierColumns as $key) {
986 $pivotColumn = $this->findJoinColumnPivot($joinInfo, $key);
987 if ($pivotColumn === false) {
988 if (!empty($multiple)) {
989 throw new RuntimeException($this->l->t('Table "%s": missing identifier for field "%s" and grouping field "%s" already set.', [ $table, $key, $multiple ]));
990 }
991 // assume that the 'column' component contains the keys.
992 $keyField = $this->joinTableFieldName($joinInfo, $joinInfo['column']);
993 $identifier[$key] = [
994 'old' => Util::explode(self::VALUES_SEP, Util::removeSpaces($oldValues[$keyField])),
995 'new' => Util::explode(self::VALUES_SEP, Util::removeSpaces($newValues[$keyField])),
996 ];
997 // Handle "deleted" information if present. This is meant for
998 // disabled instruments and the like. This stems from split-inputs
999 // where one select hold the deleted items and another one the active items.
1000 //
1001 // Unfortunaley this conflicts with the usual KEY:VALUE convention
1002 // in other tables where KEY would be the ID of the database entity
1003 // and VALUE in this case the deleted timestamp.
1004 //
1005 // We try to copy with this problem by checking for the colon syntax
1006 // and not doing this hack if we find a usual
1007 // KEY1:VALUE1,KEY2:VALUE2,... field.
