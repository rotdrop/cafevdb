161 protected function findFiles(string $dirName, bool $rootIsMandatory = false):array
162 {
163 $dirName = self::normalizeDirectoryName($dirName);
164 if (!empty($this->files[$dirName])) {
165 return $this->files[$dirName];
166 }
167
168 $dirComponents = Util::explode(self::PATH_SEPARATOR, $dirName);
169
170 $filterState = $this->disableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
171
173 $folderDirEntry = $this->rootFolder;
174 if (!$rootIsMandatory && empty($dirName) && empty($folderDirEntry)) {
175 $rootFolder = $folderDirEntry ?? new EmptyDirectoryNode('.', new DateTimeImmutable('@1'), $this->getShortId());
176 $this->files[$dirName] = [ '.' => $rootFolder ];
177 if (empty($folderDirEntry)) {
178 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
179 $readMeNode = $this->getReadMeFactory()->generateReadMe($rootFolder, $dirName);
180 if ($readMeNode !== null) {
181 $this->files[$dirName][$readMeNode->getName()] = $readMeNode;
182 }
183 return $this->files[$dirName];
184 }
185 } elseif (empty($folderDirEntry)) {
186 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
187 throw new Exceptions\DatabaseEntityNotFoundException($this->l->t(
188 'Unable to find directory entry for folder "%s".', $dirName
189 ));
190 } else {
191 foreach ($dirComponents as $component) {
192 $folderDirEntry = $folderDirEntry->getFolderByName($component);
193 if (empty($folderDirEntry)) {
194 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
195 throw new Exceptions\DatabaseEntityNotFoundException($this->l->t(
196 'Unable to find directory entry for folder "%s", component "%s", storage "%s".', [
197 $dirName, $component, $this->getShortId(),
198 ],
199 ));
200 }
201 }
202 }
203
204 $this->files[$dirName] = [ '.' => $folderDirEntry ];
205
206 $dirEntries = $folderDirEntry->getDirectoryEntries();
207
208 $hasReadme = false;
209
211 foreach ($dirEntries as $dirEntry) {
212
213 $baseName = $dirEntry->getName();
214 $fileName = $this->buildPath($dirName . self::PATH_SEPARATOR . $baseName);
215 list('basename' => $baseName) = self::pathInfo($fileName);
216
217 if ($dirEntry instanceof Entities\DatabaseStorageFolder) {
219 // add a directory entry
220 $baseName .= self::PATH_SEPARATOR;
221 $this->files[$dirName][$baseName] = $dirEntry;
222 } else {
224 $this->files[$dirName][$baseName] = $dirEntry;
225 $hasReadme = $hasReadme || $this->getReadMeFactory()->isReadMe($baseName);
226 }
227 }
228 if (!$hasReadme) {
229 $readMeNode = $this->getReadMeFactory()->generateReadMe($folderDirEntry, $dirName);
230 if ($readMeNode !== null) {
231 $this->files[$dirName][$readMeNode->getName()] = $readMeNode;
232 }
233 }
234
235 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
236
237 return $this->files[$dirName];
238 }

260 protected function persistInMemoryFileNode(InMemoryFileNode $memoryNode, ?Entities\Musician $owner = null):Entities\DatabaseStorageFile
261 {
262 $this->entityManager->beginTransaction();
263 try {
264 if ($memoryNode->getParent() instanceof EmptyDirectoryNode) {
265 $memoryNode->setParent($this->getRootFolder(create: true));
266 }
267
268 $file = new Entities\EncryptedFile(
269 $memoryNode->getName(),
270 $memoryNode->getFileData()->getData(),
271 $memoryNode->getMimeType(),
272 owner: $owner,
273 );
274 $file->setCreated($memoryNode->getUpdated());
275 $file->setUpdated($memoryNode->getUpdated());
276 $this->persist($file);
277 $this->flush();
278 $fileNode = $memoryNode->getParent()->addDocument($file, $memoryNode->getName());
279
280 if ($fileNode === null) {
281 throw new Exceptions\DatabaseStorageException(
282 $this->l->t('Unable to convert the in-memory file "%s".', $memoryNode->getParent()->getPathName()),
283 );
284 }
285
286 $this->flush();
287
288 $this->entityManager->commit();
289
290 return $fileNode;
291 } catch (Throwable $t) {
292 if ($this->entityManager->isTransactionActive()) {
293 $this->entityManager->rollback();
294 }
295 if ($t instanceof Exceptions\DatabaseStorageException) {
296 throw $t;
297 } else {
298 throw new Exceptions\DatabaseStorageException(
299 $this->l->t('Unable to convert the in-memory file "%s".', $memoryNode->getParent()->getPathName()),
300 0,
301 $t,
302 );
303 }
304 }
305 return null;

728 public function writeStream(string $path, $stream, ?int $size = null): int
729 {
730 // $this->logInfo('WRITE STREAM ' . $path);
731
732 if (!$this->touch($path)) {
733 return false;
734 }
736 $file = $this->fileFromFileName($path);
737 if (empty($file)) {
738 return false;
739 }
740 if ($size === null) {
741 $stream = CountWrapper::wrap($stream, function ($writtenSize) use (&$size) {
742 $size = $writtenSize;
743 });
744 }
745
746 $fileData = stream_get_contents($stream);
747 fclose($stream);
748
749 $this->entityManager->beginTransaction();
750 try {
751 if ($file instanceof InMemoryFileNode) {
753 // Hack: if the contents actually has not changed (Text app is
754 // notoriously writing content when a conflict is resolved and the
755 // server version is to be kept)
756 if ($fileData == $file->getData()) {
757 $this->logError('Working around buggy text-app notoriously saving unchanged content.');
758 $this->entityManager->commit();
759 return $size;
760 } else {
761 $this->logInfo('SAME DATA? "' . $fileData . '" || "' . $file->getData() . '"');
762 }
763 // this needs first to be replaced by a real file-node
764 $file = $this->persistInMemoryFileNode($file);
765 }
767 $file->getFileData()->setData($fileData);
769 $mimeTypeDetector = $this->appContainer()->get(IMimeTypeDetector::class);
770 $file->setMimeType($mimeTypeDetector->detectString($fileData));
771 $file->setSize(strlen($fileData));
772
773 $this->flush();
774
775 $this->entityManager->commit();
776 } catch (Throwable $t) {
777 if ($this->entityManager->isOwnTransactionActive()) {
778 $this->entityManager->pushTransactionException($t);
779 $this->entityManager->rollback();
780 } else {
781 $this->logException($t, 'writeStream() failed');
782 }
783 return -1;
784 }
785
786 return $size;
787 }

790 public function readStream(string $path)
791 {
793 $file = $this->fileFromFileName($path);
794 if (empty($file)) {
795 return false;
796 }
797 // using data:// would also result in copying things around as something
798 // would have to be prepended to the file-data ... so we can as well just
799 // use fwrite(). Unfortunately there is not such a nice thing like
800 // memory-backed streams as there is in C++ -- where you just can use an
801 // existing buffer verbatim without the need to copy it around.
802 $stream = fopen('php://memory', 'w+');
803 $result = fwrite($stream, $file->getFileData()->getData());
804 rewind($stream);
805 if ($result === false) {
806 fclose($stream);
807 return false;
808 }
809 return $stream;
810 }
