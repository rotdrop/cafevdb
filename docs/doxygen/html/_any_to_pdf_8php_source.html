258 public function findConverters():array
259 {
260 $result = [];
261
262 if (!empty($this->universalConverter)) {
263 $executable = $this->executableFinder->find($this->universalConverter, force: true);
264 if (empty($executable)) {
265 $executable = $this->l->t('not found');
266 }
267 $result[self::UNIVERSAL] = [ [ $this->universalConverter => $executable ] ];
268 }
269 if ($this->builtinConvertersDisabled) {
270 return $result;
271 }
272 $result = [];
273 foreach (self::CONVERTERS as $mimeType => $chainAlternatives) {
274 foreach ($chainAlternatives as $converterChain) {
275 $probedChain = [];
276 foreach ($converterChain as $converters) {
277 $probedConverters = [];
278 foreach ($converters as $converter) {
279 if ($converter == self::PASS_THROUGH) {
280 // TRANSLATORS: This is actually just the name of the "converter"
281 // TRANSLATORS: which does "nothing", i.e. just copies the input
282 // TRANSLATORS: data unchanged to the output.
283 $probedConverters[$converter] = $this->l->t('pass through');
284 continue;
285 }
286 if ($converter == self::FALLBACK) {
287 $converter = $this->fallbackConverter;
288 }
289 list($programName,) = explode(':', $converter);
290 try {
291 $executable = $this->executableFinder->find($programName, force: true);
292 } catch (Exceptions\EnduserNotificationException $e) {
293 $this->logException($e);
294 $executable = null;
295 }
296 if (empty($executable)) {
297 $executable = $this->l->t('not found');
298 }
299 $probedConverters[$converter] = $executable;
300 }
301 $probedChain[] = $probedConverters;
302 }
303 $result[] = [
304 'mimeType' => $mimeType,
305 'chain' => $probedChain,
306 ];
307 }
308 }
309 try {
310 $executable = $this->executableFinder->find($this->fallbackConverter, force: true);
311 } catch (Exceptions\EnduserNotificationException $e) {
312 $this->logException($e);
313 }
314 if (empty($executable)) {
315 $executable = $this->l->t('not found');
316 }
317 $result[] = [
318 'mimeType' => self::FALLBACK,
319 'chain' => [ [ $this->fallbackConverter => $executable ], ],
320 ];
321 return $result;
322 }

336 public function convertData(string $data, ?string $mimeType = null):string
337 {
338 if (empty($mimeType) || $mimeType == 'application/octet-stream') {
339 $mimeType = $this->mimeTypeDetector->detectString($data);
340 }
341
342 if (!empty($this->universalConverter)) {
343 try {
344 $data = $this->genericConvert($data, $mimeType, $this->universalConverter);
345 } catch (Throwable $t) {
346 if ($this->builtinConvertersDisabled) {
347 throw new RuntimeException(
348 $this->l->t('Universal converter "%1$s" has failed trying to convert MIME type "%2$s"', [
349 $this->universalConverter, $mimeType,
350 ]),
351 0,
352 $t,
353 );
354 } else {
355 $this->logException($t, 'Ignoring failed universal converter ' . $this->universalConverter);
356 }
357 }
358 }
359
360 $chains = self::CONVERTERS[$mimeType] ?? [ [ [ self::FALLBACK, ], ], ];
361
362 $originalData = $data;
363 foreach ($chains as $chain) {
364 foreach ($chain as $subStep) {
365 $convertedData = null;
366 foreach ($subStep as $tryConverter) {
367 if ($tryConverter == self::FALLBACK) {
368 $tryConverter = $this->fallbackConverter;
369 } elseif ($tryConverter == self::PASS_THROUGH) {
370 $tryConverter = 'passThrough';
371 }
372 $tryConverter = lcfirst(implode(array_map(fn($part) => ucfirst($part), explode(':', $tryConverter))));
373 try {
374 $method = $tryConverter . 'Convert';
375 if (method_exists($this, $method)) {
376 $convertedData = $this->$method($data);
377 } else {
378 $convertedData = $this->genericConvert($data, $mimeType, $tryConverter);
379 }
380 break;
381 } catch (Throwable $t) {
382 $this->logException($t, 'Failed converter ' . $tryConverter);
383 $convertedData = null;
384 }
385 }
386 if (empty($convertedData)) {
387 $this->logError('Converter chain substep for ' . $mimeType . ' has failed: ' . print_r($subStep, true));
388 $data = $originalData;
389 break; // no chance to continue, but perhaps there is an alternative
390 }
391 $data = $convertedData;
392 }
393 if (!empty($convertedData)) {
394 break;
395 }
396 }
397 if (empty($convertedData)) {
398 throw new RuntimeException(
399 $this->l->t('Converter "%1$s" has failed trying to convert MIME type "%2$s"', [
400 print_r($chains, true), $mimeType,
401 ]),
402 );
403 }
404
405 return $data;
406 }

454 $converterName = 'unoconvert';

456 $retry = false;
457 $count = 0;
458 do {
459 $process = new Process([
460 $converter,
461 '--convert-to', 'pdf',
462 '--filter-options', 'ExportNotes=False',
463 '-', '-',
464 ]);
465 $process->setInput($data);
466 try {
467 $process->run();
468 $retry = false;

470 $this->logException($timedOutException, 'Unrecoverable exception');
471 $retry = false;

473 $this->logException($t, 'Retry after exception, trial number ' . ($count + 1));
474 $retry = true;
475 }
476 } while ($retry && $count++ < self::UNOCONV_RETRIES);

701 protected function img2pdfConvert(string $data):string
702 {
703 putenv('LC_ALL=C');
704 $converterName = 'img2pdf';
705 $converter = $this->findExecutable($converterName);
706
707 try {
708 // we actually want to have "--rotation=ifvalid", see issue #100 on https://gitlab.mister-muffin.de/josch/img2
709 $process = new Process([ $converter, '--version' ]);
710 $process->run();
711 $versionString = $process->getOutput();
712 // img2pdf 0.4.4
713 $matches = null;
714 if (preg_match('/' . $converterName . '\\s+(\\d+)\\.(\\d+)\\.(\\d+)/', $versionString, $matches)) {
715 $version = [
716 'major' => $matches[1],
717 'minor' => $matches[2],
718 'patch' => $matches[3],
719 ];
720 }
721 } catch (Throwable $t) {
722 throw new RuntimeException(
723 $this->l->t('Unable to determine the version of the helper program "%1$s", is it installed?', $converter), 0, $t,
724 );
725 }
726 $processAndArgs = [
727 $converter,
728 '-', // from stdin
729 ];
730 if ($version['minor'] >= 4 && $version['patch'] >= 4) {
731 $processAndArgs[] = '--rotation=ifvalid'; // ignore broken rotation settings in EXIF meta data
732 }
733 $process = new Process($processAndArgs);
734 $process->setInput($data)->run();
735 return $process->getOutput();
736 }
