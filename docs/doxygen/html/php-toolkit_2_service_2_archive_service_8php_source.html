259 public function canOpen(File $fileNode):bool
260 {
261 $this->setProcessEnvironment();
262
263 $localPath = self::getLocalPath($fileNode);
264
265 $format = ArchiveFormats::detectArchiveFormat($localPath);
266 $canOpen = $format !== null && ArchiveFormats::canOpen($format);
267
268 $this->restoreProcessEnvironment();
269
270 if ($format === null || $format === false) {
271 throw new Exceptions\ArchiveCannotOpenException(
272 $this->l->t('Unable to detect the archive format of "%1$s".', $fileNode->getName())
273 );
274 }
275 if (!$canOpen) {
276 $messages = [];
277 if (str_starts_with($format, 't') && !str_starts_with($format, 'tar')) {
278 $innerFormat = 'tar';
279 $compositeFormat = $format;
280 $format = substr($format, 1);
281 $this->setProcessEnvironment();
282 $canDecompress = $format !== null && ArchiveFormats::canOpen($format);
283 $canUntar = ArchiveFormats::canOpen($innerFormat);
284 $this->restoreProcessEnvironment();
285 if (!$canDecompress) {
286 $messages[] = $this->l->t('Archive format of "%1$s" detected as "%2$s", but there is no backend driver installed which can decompress ".%3$s" files.', [
287 $fileNode->getName(),
288 $compositeFormat,
289 $format,
290 ]);
291 }
292 if (!$canUntar) {
293 // this should never be the case ...
294 $messages[] = $this->l->t(
295 'Unable to deal with tar-files. Please check the installation of the app.'
296 );
297 }
298 } else {
299 $messages[] = $this->l->t('The archive format of "%1$s" has been detected as "%2$s", but there is no backend driver installed which can deal with this format.', [
300 $fileNode->getName(),
301 $format,
302 ]);
303 }
304 $formats = ArchiveFormats::getDeclaredDriverFormats();
305 foreach ($formats[$format] as $driverClass) {
306 $shortDriver = substr($driverClass, strrpos($driverClass, '\\') + 1);
307 if (!$driverClass::isInstalled()) {
308 $messages[] = $this->l->t('The "%1$s" driver could handle this format, but it is not installed.', $shortDriver);
309 $typeLabel = BasicDriver::$typeLabels[$driverClass::TYPE];
310 $instructions = $driverClass::getInstallationInstruction();
311 $messages[] = $this->l->t('Installation instructions:')
312 . ' '
313 . ucfirst($typeLabel)
314 . '. '
315 . ucfirst($instructions)
316 . '.';
317 continue;
318 }
319 $abilities = $driverClass::getFormatAbilities($format);
320 $requiredAbilities = DriverAbilities::OPEN|DriverAbilities::EXTRACT_CONTENT;
321 if (($abilities & $requiredAbilities) != $requiredAbilities) {
322 $messages[] = $this->l->t('The "%1$s" driver claims to handle this format, but cannot extract the archive content.', $shortDriver);
323 }
324 }
325 throw new Exceptions\ArchiveCannotOpenException(
326 implode(PHP_EOL, $messages)
327 );
328 }
329
330 return true;
331 }

366 public function open(File $fileNode, ?int $sizeLimit = null, #[SensitiveParameter] ?string $password = null):?ArchiveService
367 {
368 if (!$this->canOpen($fileNode)) {
369 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
370 $fileNode->getPath(), self::getLocalPath($fileNode),
371 ]));
372 }
373
374 $this->setProcessEnvironment();
375
376 $this->archiver = ArchiveBackend::open(self::getLocalPath($fileNode), password: $password);
377
378 $this->restoreProcessEnvironment();
379
380 if (empty($this->archiver)) {
381 throw new Exceptions\ArchiveCannotOpenException($this->t('Unable to open archive file %s (%s)', [
382 $fileNode->getPath(), self::getLocalPath($fileNode),
383 ]));
384 }
385 if ($sizeLimit === null) {
386 $sizeLimit = $this->sizeLimit;
387 }
388 $this->fileNode = $fileNode;
389 $archiveInfo = $this->getArchiveInfo();
390 $archiveSize = $archiveInfo->originalSize;
391 if ($sizeLimit !== null && $archiveSize > $sizeLimit) {
392 $this->archiver = null;
393 $this->fileNode = null;
394 throw new Exceptions\ArchiveTooLargeException(
395 $this->t('Uncompressed size of archive "%1$s" is too large: %2$s > %3$s', [
396 $fileNode->getInternalPath(), CloudUtil::humanFileSize($archiveSize), CloudUtil::humanFileSize($sizeLimit),
397 ]),
398 $sizeLimit,
399 $archiveSize,
400 );
401 }
402
403 // Build a per-member map from the NFC-normalized name to the raw name as
404 // stored in the archive. Nextcloud normalizes every path to NFC, so this
405 // is the key by which members will be requested again on read. Resolving
406 // each member on its own avoids choosing a single normalization for the
407 // whole archive, which breaks archives that mix NFC and NFD names.
408 $this->memberNameMap = [];
409 $this->collidingMembers = [];
410 foreach ($this->archiver->getFileNames() as $rawName) {
411 $key = Normalizer::normalize($rawName, Normalizer::NFC);
412 if ($key === false) {
413 // The name is not valid UTF-8 (e.g. a legacy DOS/codepage name stored
414 // without the UTF-8 flag). Keep the raw bytes as key so the member
415 // stays addressable instead of being dropped.
416 $key = $rawName;
417 }
418 if (array_key_exists($key, $this->memberNameMap) && $this->memberNameMap[$key] !== $rawName) {
419 $this->collidingMembers[$key] ??= [ $this->memberNameMap[$key] ];
420 $this->collidingMembers[$key][] = $rawName;
421 }
422 $this->memberNameMap[$key] = $rawName;
423 }
424
425 return $this;
426 }

478 public static function getArchiveFolderName(string $archiveFileName):?string
479 {
480 // double to account for "nested" archive types
481 $archiveFolderName = pathinfo($archiveFileName, PATHINFO_FILENAME);
482 $secondExtension = pathinfo($archiveFolderName, PATHINFO_EXTENSION);
483
484 // as a rule of thumb we only strip the second extension if it contains no
485 // spaces and is no longer that 4 characters.
486 if (strlen($secondExtension) <= 4 && str_replace(' ', '', $secondExtension) === $secondExtension) {
487 $archiveFolderName = pathinfo($archiveFolderName, PATHINFO_FILENAME);
488 }
489
490 return $archiveFolderName;
491 }

637 {
