104 public function decrypt(?string $data):?string
105 {
106 if (!SealService::isSealedData($data)) {
107 return $data;
108 }
109 $sealData = $this->sealService->parseSeal($data);
110 $candidates = array_intersect(array_keys($this->sealCryptors), array_keys($sealData['keys']));
111 $candidates = array_filter($candidates, fn($id) => $this->sealCryptors[$id]->canDecrypt());
112 if (empty($candidates)) {
113 throw new Exceptions\DecryptionFailedException('Unable to unseal, no valid candidates');
114 }
115 foreach ($candidates as $id) {
116 try {
117 return $this->sealService->unseal($data, $id, $this->sealCryptors[$id]);
118 } catch (Exceptions\EncryptionException $e) {
119 $this->logDebug('Decryption for id "' . $id . '" failed.');
120 }
121 }
122 throw new Exceptions\EncryptionFailedException('Unable to decrypt sealed data, all candidates have failed: ' . implode(', ', $candidates));
123 }
