655 public function findCalendarObjectByCategories(
656 array $categories,
657 ?string $userId = null,
658 array $calendarUris = [],
659 ?string $type = null,
660 ):array {
661 $principalUri = 'principals/users/' . ($userId ?? $this->userId());
662 $query = $this->calendarManager->newQuery($principalUri);
663 foreach ($calendarUris as $calendarUri) {
664 $query->addSearchCalendar($calendarUri);
665 }
666 $categories = array_values($categories);
667 if (!empty($categories)) {
668 $firstCategory = $categories[0];
669 $query->addSearchProperty(ICalendarQuery::SEARCH_PROPERTY_CATEGORIES);
670 $query->setSearchPattern($firstCategory);
671 }
672 if (!empty($type)) {
673 $query->addType($type);
674 }
675 $calendarObjects = $this->calendarManager->searchForPrincipal($query);
676 $result = [];
677 foreach ($calendarObjects as $objectInfo) {
678 if (!empty($categories)) {
679 $match = false;
680 foreach ($objectInfo['objects'] as $calendarObject) {
681 if (isset($calendarObject['CATEGORIES'])) {
682 // Format of the array is [PROP][PROP_IDX][ VALUE, PARAMETERS ... ]
683 foreach ($calendarObject['CATEGORIES'] as $property) {
684 $objectCategories = explode(',', $property[0]);
685 if (empty(array_diff($categories, $objectCategories))) {
686 $match = true;
687 break 2;
688 }
689 }
690 }
691 }
692 if (!$match) {
693 continue;
694 }
695 }
696 $result[] = $this->getCalendarObject($objectInfo['calendar-key'], $objectInfo['uri']);
697 }
698 return $result;
699 }

715 {
