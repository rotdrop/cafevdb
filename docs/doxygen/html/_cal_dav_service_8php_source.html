665 public function findCalendarObjectByCategories(
666 array $categories,
667 ?string $userId = null,
668 array $calendarUris = [],
669 ?string $type = null,
670 ):array {
671 $principalUri = 'principals/users/' . ($userId ?? $this->userId());
672 $query = $this->calendarManager->newQuery($principalUri);
673 foreach ($calendarUris as $calendarUri) {
674 $query->addSearchCalendar($calendarUri);
675 }
676 $categories = array_values($categories);
677 if (!empty($categories)) {
678 $firstCategory = $categories[0];
679 $query->addSearchProperty(ICalendarQuery::SEARCH_PROPERTY_CATEGORIES);
680 $query->setSearchPattern($firstCategory);
681 }
682 if (!empty($type)) {
683 $query->addType($type);
684 }
685 $calendarObjects = $this->calendarManager->searchForPrincipal($query);
686 $result = [];
687 if (!empty($categories)) {
688 foreach ($calendarObjects as $objectInfo) {
689 $match = false;
690 foreach ($objectInfo['objects'] as $calendarObject) {
691 if (isset($calendarObject['CATEGORIES'])) {
692 // Format of the array is [PROP][PROP_IDX][ VALUE, PARAMETERS ... ]
693 foreach ($calendarObject['CATEGORIES'] as $property) {
694 $objectCategories = explode(',', $property[0]);
695 if (empty(array_diff($categories, $objectCategories))) {
696 $match = true;
697 break 2;
698 }
699 }
700 }
701 }
702 if (!$match) {
703 continue;
704 }
705 $result[] = $this->getCalendarObject($objectInfo['calendar-key'], $objectInfo['uri']);
706 }
707 } else {
708 foreach ($calendarObjects as $objectInfo) {
709 $result[] = $this->getCalendarObject($objectInfo['calendar-key'], $objectInfo['uri']);
710 }
711 }
712 return $result;
713 }
