87 public function floatValue(string $value, ?string $locale = null)
88 {
89 $amount = preg_replace('/\s+/u', '', $value);
90 empty($locale) && $locale = $this->getLocale();
91 $locales = [ $locale, 'en_US_POSIX' ];
92 $parsed = false;
93 foreach ($locales as $locale) {
94 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
95
96 $decimalSeparator = $fmt->getSymbol(\NumberFormatter::DECIMAL_SEPARATOR_SYMBOL);
97 $groupingSeparator = $fmt->getSymbol(\NumberFormatter::GROUPING_SEPARATOR_SYMBOL);
98
99 $decPos = strpos($amount, $decimalSeparator);
100 $grpPos = strpos($amount, $groupingSeparator);
101
102 if ($grpPos !== false && $decPos === false) {
103 // unlikely: 1,000, we assume 1,000.00 would be used
104 continue;
105 } elseif ($decPos < $grpPos) {
106 // unlikely: 1.000,00 in en_US
107 continue;
108 }
109
110 $parsed = $fmt->parse($amount);
111 if ($parsed !== false) {
112 $percent = $fmt->getSymbol(\NumberFormatter::PERCENT_SYMBOL);
113 if (preg_match('/'.$percent.'/u', $amount)) {
114 $parsed /= 100.0;
115 }
116 break;
117 }
118 }
119 return $parsed !== false ? (float)$parsed : $parsed;
120 }

157 public function storageValue(?string $value, ?string $locale = null)
158 {
159 if ($value === null || $value === '') {
160 return null;
161 }
162 if ($value === '0') {
163 return (int)0;
164 }
165 $factor = [
166 'b' => 1,
167 'kb' => 1000,
168 'kib' => (1 << 10),
169 'mb' => (1000 * 1000),
170 'mib' => (1 << 20),
171 'gb' => (1000 * 1000 * 1000),
172 'gib' => (1 << 30),
173 'tb' => (1000 * 1000 * 1000 * 1000),
174 'tib' => (1 << 40),
175 'pb' => (1000 * 1000 * 1000 * 1000 * 1000),
176 'pib' => (1 << 50),
177 'eb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000),
178 'eib' => (1 << 60),
179 'zb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000),
180 'zib' => (1 << 70),
181 'yb' => (1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000 * 1000),
182 'yib' => (1 << 80),
183 ];
184 $value = preg_replace('/\s+/u', '', $value);
185 $value = strtolower(
186 str_ireplace(
187 [ 'bytes', 'kilo', 'kibi', 'mega', 'mebi', 'giga', 'gibi', 'tera', 'tibi', 'peta', 'pebi', 'exa', 'exbi', 'zetta', 'zebi', 'yotta', 'yobi', ],
188 [ 'b', 'k', 'ki', 'm', 'mi', 'g', 'gi', 't', 'ti', 'p', 'pi', 'e', 'ei', 'z', 'zi', 'y', 'yi', ],
189 $value));
190
191 if (preg_match('/([-0-9,.]+)([kmgtpezy]?i?b?)?$/', $value, $matches)) {
192 $this->logInfo('MATCHES ' . print_r($matches, true));
193 $value = $this->floatValue($matches[1], $locale);
194 if (empty($value)) {
195 return null;
196 }
197 if (!empty($matches[2])) {
198 if (empty($factor[$matches[2]])) {
199 return null;
200 }
201 $value *= $factor[$matches[2]];
202 if ($value <= PHP_INT_MAX) {
203 $value = (int)$value;
204 }
205 }
206 return $value;
207 } else {
208 return null;
209 }

225 protected function formatStorageValue(int $bytes, string $format = 'binary', int $digits = 2, ?string $locale = null):?string
226 {
227 $units = [
228 'decimal' => [
229 '', 'k', 'M', 'G', 'T', 'P', 'E', 'Z', 'Y',
230 ],
231 'longDecimal' => [
232 'bytes', 'kilo', 'mega', 'giga', 'tera', 'peta', 'exa', 'zetta', 'yotta',
233 ],
234 'binary' => [
235 '', 'Ki', 'Mi', 'Gi', 'Ti', 'Pi', 'Ei', 'Zi', 'Yi',
236 ],
237 'longBinary' => [
238 '', 'kibi', 'mebi', 'gibi', 'tebi', 'pebi', 'exbi', 'zebi', 'yobi'
239 ],
240 ];
241 $bytesUnit = [
242 'decimal' => 'B',
243 'longDecimal' => ' ' . $this->l->t('bytes'),
244 'binary' => 'B',
245 'longBinary' => ' ' . $this->l->t('bytes'),
246 ];
247 $radix = [
248 'decimal' => 1000.0,
249 'longDecimal' => 1000.0,
250 'binary' => 1024.0,
251 'longBinary' => 1024.0,
252 ];
253
254 $units = $units[$format] ?? null;
255 $bytesUnit = $bytesUnit[$format] ?? null;
256 $radix = $radix[$format] ?? null;
257
258 if ($units === null || $bytesUnit === null || $radix === null) {
259 // maybe throw InvalidArgumentException
260 return null;
261 }
262
263 empty($locale) && $locale = $this->getLocale();
264
265 $floatVal = $bytes;
266 $exponent = 0;
267 $exponentLimit = count($units) - 1;
268 while ($floatVal > $radix && $exponent < $exponentLimit) {
269 $floatVal /= $radix;
270 ++$exponent;
271 }
272
273 $fmt = new NumberFormatter($locale, \NumberFormatter::DECIMAL);
274 $fmt->setAttribute(NumberFormatter::MIN_FRACTION_DIGITS, 0);
275 $fmt->setAttribute(NumberFormatter::MAX_FRACTION_DIGITS, $digits);
276 $stringVal = $fmt->format($floatVal);
277
278 return $stringVal . ' ' . $units[$exponent] . $bytesUnit;
279 }

311 protected function getCommonPath(array $paths, bool $leadingSlash = true):string
312 {
313 // $this->logInfo('PATHS ' . print_r($paths, true));
314 $lastOffset = (int)$leadingSlash;
315 $common = $leadingSlash ? '/' : '';
316 while (($index = strpos($paths[0], '/', $lastOffset)) !== false) {
317 $dirLen = $index - $lastOffset + 1; // include /
318 $dir = substr($paths[0], $lastOffset, $dirLen);
319 foreach ($paths as $path) {
320 if (substr($path, $lastOffset, $dirLen) != $dir) {
321 return $leadingSlash ? substr($common, 0, -1) : $common;
322 }
323 }
324 $common .= $dir;
325 $lastOffset = $index + 1;
326 }
327 return $leadingSlash ? substr($common, 0, -1) : $common;
328 }
