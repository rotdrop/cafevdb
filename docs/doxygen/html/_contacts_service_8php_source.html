144 foreach ($this->contactsManager->getUserAddressBooks() as $addressBook) {

177 public function emailContacts():array
178 {
180 $musicianCardBackend = $this->appContainer->get(MusicianCardBackend::class);
181 $musiciansKey = $musicianCardBackend->getURI();
182 $result = [];
183 $addressBooks = $this->contactsManager->getUserAddressBooks();
185 foreach ($addressBooks as $addressBook) {
186 if ($addressBook->getKey() == $musiciansKey) {
187 continue;
188 }
189 $bookName = $addressBook->getDisplayName();
190 $contacts = $addressBook->search('', [ 'FN', 'EMAIL' ], [] /* options */);
191 foreach ($contacts as $contact) {
192 $uid = $contact['UID'];
193 $fullName = $contact['FN']??'';
194 $emails = $contact['EMAIL']??null;
195 if (empty($emails)) {
196 continue;
197 }
198 if (!is_array($emails)) {
199 $emails = [ $emails ];
200 }
201 $theseContacts = [];
202 foreach ($emails as $email) {
203 if (empty($email)) {
204 continue;
205 }
206 $theseContacts[] = [
207 'uid' => $uid,
208 'email' => $email,
209 'name' => $fullName,
210 'addressBook' => $bookName,
211 ];
212 }
213 usort($theseContacts, function($a, $b) {
214 $aName = $a['name'] != '' ? $a['name'] : $a['email'];
215 $bName = $b['name'] != '' ? $b['name'] : $b['email'];
216 return strcmp($aName, $bName);
217 });
218 $result = array_merge($result, $theseContacts);
219 }
220 }
221 return $result;

265 public function flattenVCard(?string $cardUri, VCard $vCard, bool $withTypes = true):array
266 {
267 $result = [
268 'URI' => $cardUri,
269 ];
270 foreach ($vCard->children() as $property) {
271 if ($property->name === 'PHOTO') {
272 if ($property->getValueType() === 'BINARY' && $this->getTypeFromProperty($property)) {
273 $uri = 'data:image/' . strtolower($property['TYPE']) . ';base64,' . $property->getRawMimeDirValue();
274 $result[$property->name] = 'VALUE=uri:' . $uri;
275 } elseif ($property->getValueType() === 'URI') {
276 $result[$property->name] = 'VALUE=uri:' . $property->getValue();
277 } else {
278 $result[$property->name] = $property->getValue();
279 }
280 } elseif (in_array($property->name, ['URL', 'GEO', 'CLOUD', 'ADR', 'EMAIL', 'IMPP', 'TEL', 'X-SOCIALPROFILE', 'RELATED', 'LANG', 'X-ADDRESSBOOKSERVER-MEMBER'])) {
281 if (!isset($result[$property->name])) {
282 $result[$property->name] = [];
283 }
284
285 $type = $this->getTypeFromProperty($property);
286 if ($withTypes) {
287 $result[$property->name][] = [
288 'type' => $type,
289 'value' => $property->getValue()
290 ];
291 } else {
292 $result[$property->name][] = $property->getValue();
293 }
294 } else {
295 $result[$property->name] = $property->getValue();
296 }
297 }
298 return $result;

676 $entity->getInstruments()->remove($musicianInstrument->getInstrument()->getId());

697 break;
698 }
699 }
700 }
701
702 if (!empty($entity->getOrganization())) {
703 $entity->setDefaultParticipationStatus(EnumParticipationStatus::ASSOCIATED);
704 }
705
706 // Ignore image data.
707
708 return $entity;

905 public function mergeMusician(
906 array|VCard $target,
907 Entities\Musician $musician,
908 ): array {
909 if ($target instanceof VCard) {
910 $target = $this->flattenVCard($target->URI || null, $target, withTypes: true);
911 }
912
913 $instrumentCategories = $this->getDatabaseRepository(Entities\Instrument::class)->findNames();
914 $projectCategories = $this->getDatabaseRepository(Entities\Project::class)->findNames();
915 // Remove all categories which are set by us
916 $targetCategories = array_diff(explode(',', $target['CATEGORIES']), $instrumentCategories, [$this->appName()]);
917 // Remove all categories starting whith a project name.
918 foreach ($projectCategories as $projectName) {
919 foreach ($targetCategories as $index => $category) {
920 if (str_starts_with($category, $projectName)) {
921 unset($targetCategories[$index]);
922 }

924 }

927 // unlink the contact and remove all app-categories
928 sort($targetCategories);
929 $target['CATEGORIES'] = implode(',', $targetCategories);
930 return $target;
931 }
932
933 $source = $this->flattenVCard(cardUri: null, vCard: $this->export($musician), withTypes: true);
934
935 $result = Util::arrayMergeRecursive([], $target, $source, [ 'URI' => $target['URI'] ?? null ]);
936 if (empty($result['URI'])) {
937 unset($result['URI']);
938 }
939
940 $sourceCategories = explode(',', $source['CATEGORIES']);
941 $categories = array_unique(array_merge($targetCategories, $sourceCategories));
942 sort($categories);
943 $result['CATEGORIES'] = implode(',', $categories);
944
945 // eliminate redundant or empty typed properties.
946 foreach (self::TYPED_PROPERTIES as $typedPropery) {
947 if (!empty($target[$typedPropery])) {
948 $result[$typedPropery] = $target[$typedPropery];
949 }
950 foreach (($result[$typedPropery] ?? []) as $key => $property) {
951 if (empty($property['value'])) {
952 unset($result[$typedPropery][$key]);
953 }
954 if ($typedPropery == 'TEL') {
955 $phoneNumberService = $this->phoneNumberService();
956 if ($phoneNumberService->validate($property['value'])) {
957 $result[$typedPropery][$key]['value'] = $phoneNumberService->format();
958 if ($phoneNumberService->isMobile()) {
959 $result[$typedPropery][$key]['type'] = str_replace('VOICE', 'CELL', $result[$typedPropery][$key]['type']);
960 } else {
961 $result[$typedPropery][$key]['type'] = str_replace('CELL', 'VOICE', $result[$typedPropery][$key]['type']);
962 }
963 } else {
964 $this->logError('PHONE NUMBER VALIDATION FAILED FOR ' . $property['value']);
965 // unset($result[$typedPropery][$key]); should we?
966 }
967 }
968 }
969 if (!empty($source[$typedPropery])) {
970 $sourceProperty = $source[$typedPropery][0];
971 foreach ($result[$typedPropery] as $key => $property) {
972 if ($typedPropery == 'ADR') {
973 // assume a match if street, city and PO-code match
974 $address = explode(';', Util::normalizeSpaces($property['value']));
975 $sourceAddress = explode(';', Util::normalizeSpaces($sourceProperty['value']));

977 && $address[3] == $sourceAddress[3]
978 && $address[5] == $sourceAddress[5]) {
979 unset($result[$typedPropery][$key]);
980 }
981 } elseif ($property['value'] == $sourceProperty['value']) {
982 unset($result[$typedPropery][$key]);
983 }
984 }
985 array_unshift($result[$typedPropery], $sourceProperty);
986 }
987 }

1009 Entities\Musician $musician,
1010 ?string $oldAddressBookUri = null,
1011 ): ?bool {
1012 if (!empty($this->contactSynchronizations[$musician->getId()])) {
1013 $this->logInfo('Contact sync already active for ' . $musician->getPublicName());
1014 return null;
1015 }
1016 if (!$this->entityManager->isTransactionActive()) {
1017 $this->logInfo('Called without active database transaction for ' . $musician->getPublicName());
1018 return false;
1019 }

1022 $listener = $this->appContainer->get(ContactsCardEventListener::class);
1023 $this->entityManager->registerPreCommitAction(
1024 new GenericUndoable(
1025 function() use ($musician, $oldAddressBookUri, $listener) {
1026 if ($listener->isActive()) {
1027 // avoid recursion, the listener may cause change events which
1028 // should not in turn cause modifications in the address books.
1029 unset($this->contactSynchronizations[$musician->getId()]);
1030 return null;
1031 }
1032 if ($musician->isExpired() && $this->entityManager->contains($musician)) {
1033 $musician->setAddressBookUri(null);
1034 $this->flush();
1035 }
1036 if ($musician->getAddressBookUri() === null) {
1037 // addressbook has been unlinked, do nothing.
1038 unset($this->contactSynchronizations[$musician->getId()]);
1039 return null;
1040 }
1041 $addressBookUri = $oldAddressBookUri ?? $musician->getAddressBookUri();
1042 if (empty($addressBookUri)) {
1043 unset($this->contactSynchronizations[$musician->getId()]);
1044 return null;
1045 }
1046 $oldAddressBook = $this->addressBookByUri($addressBookUri);
1047 $oldContact = $this->findMusicianContact($musician, $addressBookUri);
1048 if ($oldContact === null) {
1049 $this->logError(
1050 'Broken address-book link for musician "' . $musician->getPublicName(). '"'
1051 . ', addressbook "' . $musician->getAddressBookUri() . '"'
1052 . ', uuid "' . $musician->getUuid() . '".',
1053 );
1054 unset($this->contactSynchronizations[$musician->getId()]);
1055 return null;
1056 }
1057 $newContact = $this->mergeMusician($oldContact, $musician);
1058 if ($addressBookUri != $musician->getAddressBookUri()) {
1059 $addressBook = $this->addressBookByUri($musician->getAddressBookUri());
1060 } else {
1061 $addressBook = $oldAddressBook;
1062 }
1063 $oldState = $listener->setEnabled(false);
1064 $addressBook->createOrUpdate($newContact);
1065 $listener->setEnabled($oldState);
1066 if ($oldAddressBook !== $addressBook) {
1067 $oldAddressBook->delete($oldContact['id']);
1068 }
1069 unset($this->contactSynchronizations[$musician->getId()]);
1070 return [ $addressBook, $newContact, $oldAddressBook, $oldContact ];
1071 },
1072 function (?array $oldData) use ($musician, $listener) {
1073 if ($oldData !== null) {
1074 $oldState = $listener->setEnabled(false);
1075 list($newAddressBook, $newContact, $oldAddressBook, $oldContact) = $oldData;
1076 if ($newAddressBook !== $oldAddressBook) {
1077 $newAddressBook->delete($newContact['id']);
1078 }
1079 $oldAddressBook->createOrUpdate($oldContact);
1080 $listener->setEnabled($oldState);
1081 $this->contactSynchronizations[$musician->getId()] = $musician;
1082 // the rollback should restore the addressbook link.
1083 }
1084 }
1085 ),
1086 );
1087 return true;
1088 }
