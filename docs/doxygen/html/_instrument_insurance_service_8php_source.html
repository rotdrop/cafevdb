134 private function yearFraction(
135 DateTimeInterface $insuranceStart,
136 ?DateTimeInterface $insuranceEnd,
137 DateTimeInterface $dueDate,
138 ): RationalNumber {
139 $timeZone = new DateTimeZone('UTC'); // $this->getDateTimeZone();
140 $startDate = self::convertToTimezoneDate(self::convertToDateTime($insuranceStart), $timeZone);
141 $dueDate = self::convertToTimezoneDate(self::convertToDateTime($dueDate), $timeZone);
142
143 $startDistance = $startDate->diff($dueDate);
144
145 // $dueDate is before $insuranceStart
146 if ($startDistance->invert) {
147 return new RationalNumber(0, 0, 12);
148 }
149
150 // for our purpose everything > 0 days is a month, we only charge
151 // full months
152 $startDistance->d = 0;
153
154 $months = $startDistance->y > 0 ? 12 : $startDistance->m;
155
156 if (!empty($insuranceEnd)) {
157 // to get the diff right -- $insuranceEnd is the last day where the
158 // instrument was included by into the insurance, we have to add one
159 // day. E.g.: Start 01.07.YYYY, end 30.06.ZZZZ should yield one year and
160 // not 365 days.
161 $endDate = self::convertToTimezoneDate(self::convertToDateTime($insuranceEnd), $timeZone)->modify('+1 day');
162 $endDistance = $dueDate->diff($endDate);
163 if ($endDistance->invert) {
164 // due-date after end-date
165 if ($endDistance->y > 0) {
166 // ended longer than one year ago, so return 0
167 return new RationalNumber(0, 0, 12);
168 }
169 $endDistance->d = 0; // just include fractional months
170 $months -= $endDistance->m;
171 }
172 }
173
174 $fraction = new RationalNumber(0, $months, 12);
175
176 return $fraction;
177 }

249 public function insuranceFee(
250 mixed $musicianOrId,
251 null|string|Entities\InsuranceBroker $broker,
252 $date = null,
253 ?array &$dueInterval = null,
254 ): MonetaryNumberType {
255 $timeZone = $this->getDateTimeZone();
256 if (empty($date)) {
257 $date = new DateTime();
258 }
259 $date = self::convertToTimezoneDate($date, $timeZone);
260
261 $payables = $this->billableInsurances($musicianOrId, $broker);
262
263 $taxFactor = $this->getTaxRate()->add(1);
264
265 $fee = MonetaryNumberType::zero();
268 $minDueDate = $maxDueDate = null;
270 foreach ($payables as $insurance) {
271 $insuranceStart = self::convertToTimezoneDate($insurance->getStartOfInsurance(), $timeZone);
272 $insuranceEnd = $insurance->getDeleted();
273 if (!empty($insuranceEnd)) {
274 $insuranceEnd = self::convertToTimezoneDate($insuranceEnd, $timeZone);
275 }
276
278 $rate = $insurance->getInsuranceRate();
279
280 // end of insurance period
281 $dueDate = $this->dueDate($rate->getDueDate(), $date);
282 if (!empty($insuranceEnd) && $dueDate->modify('-1 year') > $insuranceEnd) {
283 continue;
284 }
285 $minDueDate = empty($minDueDate) ? $dueDate : min($dueDate, $minDueDate);
286 $maxDueDate = empty($maxDueDate) ? $dueDate : max($dueDate, $maxDueDate);
287
288 $annualFee = MonetaryNumberType::create($insurance->getInsuranceAmount())->mul($rate->getRate());
289 $annualFee->mulEq($this->yearFraction($insuranceStart, $insuranceEnd, $dueDate));
290
291 $fee->addEq($annualFee->mul($taxFactor));
292 }
293 $dueInterval = [ 'min' => $minDueDate, 'max' => $maxDueDate ];
294
295 return $fee->round(2);
296 }

324 public function getRates(bool $translate = false, bool $nested = false):array

425
426 $payableInsurances = $billToParty->getPayableInsurances();
427 if ($broker !== null) {

434 }
435
436 $insuranceOverview = [
437 'billTo' => EntityArrayAdapter::create($billToParty, depth: 0),
438 'taxRate' => $this->getTaxRate(),
439 'broker' => $broker,
440 'musicians' => [],
441 'date' => $date,
442 ];
443
445 foreach ($payableInsurances as $insurance) {
446 $insuranceStart = self::convertToTimezoneDate($insurance->getStartOfInsurance(), $timeZone);
447 $insuranceEnd = $insurance->getDeleted();
448 if (!empty($insuranceEnd)) {
449 $insuranceEnd = self::convertToTimezoneDate($insuranceEnd, $timeZone);
450 }
451
453 $rate = $insurance->getInsuranceRate();
454
455 // end of the insurance year
456 $dueDate = $this->dueDate($rate->getDueDate(), $date);
457 $endDate = $dueDate->modify('-1 day'); // last day of insurance year
458
459 // start of insurance year
460 $lastDueDate = $dueDate->modify('-1 year');
461
462 if (!empty($insuranceEnd) && $lastDueDate > $insuranceEnd) {
463 // exclude instruments which are no longer insured
464 continue;
465 }
466
467 if ($dueDate <= $insuranceStart) {
468 // exclude instruments which were not yet insured in that year
469 continue;
470 }
471
472 $amount = MonetaryNumberType::create($insurance->getInsuranceAmount());
473 $fraction = $this->yearFraction($insuranceStart, $insuranceEnd, $dueDate);
474 $annualFee = $amount->mul($rate->getRate());
475
476 $instrumentHolder = $insurance->getInstrumentHolder();
477 $instrumentHolderId = $instrumentHolder->getId();
478 if (empty($insuranceOverview['musicians'][$instrumentHolderId])) {
479 $insuranceOverview['musicians'][$instrumentHolderId] = [
480 'name' => $instrumentHolder->getPublicName(true),
481 'subTotals' => MonetaryNumberType::create(0),
482 'items' => [],
483 ];
484 }
485
486 $itemInfo = [
487 'broker' => $insurance->getBroker()->getShortName(),
488 'scope' => $insurance->getGeographicalScope(),
489 'object' => $insurance->getObject(),
490 'manufacturer' => $insurance->getManufacturer(),
491 'amount' => $amount,
492 'rate' => $rate->getRate(),
493 'lastDue' => $lastDueDate,
494 'due' => empty($insuranceEnd) ? $endDate : $insuranceEnd,
495 'start' => $insuranceStart,
496 'fullFee' => $annualFee,
497 'fraction' => $fraction,
498 'fee' => $annualFee->mul($fraction),
499 ];
500
501 $insuranceOverview['musicians'][$instrumentHolderId]['items'][] = $itemInfo;
502 }
503
504 $annual = MonetaryNumberType::zero();
505 foreach ($insuranceOverview['musicians'] as $id => $info) {
506 // ordinary annular fees
507 $subTotals = MonetaryNumberType::zero();
508 foreach ($info['items'] as $itemInfo) {
509 $subTotals = $subTotals->add($itemInfo['fee']);
510 }
511 $insuranceOverview['musicians'][$id]['subTotals'] = $subTotals;
512 // $this->logInfo('SUBTOTALS '.$subTotals);
513 $annual = $annual->add($subTotals);
514 }
515 $insuranceOverview['annual'] = $annual;
516
517 return $insuranceOverview;
