132 private function yearFraction(
133 DateTimeInterface $insuranceStart,
134 ?DateTimeInterface $insuranceEnd,
135 DateTimeInterface $dueDate,
136 ):RationalNumber
137 {
138 $timeZone = new DateTimeZone('UTC'); // $this->getDateTimeZone();
139 $startDate = self::convertToTimezoneDate(self::convertToDateTime($insuranceStart), $timeZone);
140 $dueDate = self::convertToTimezoneDate(self::convertToDateTime($dueDate), $timeZone);
141
142 $startDistance = $startDate->diff($dueDate);
143
144 // $dueDate is before $insuranceStart
145 if ($startDistance->invert) {
146 return new RationalNumber(0, 0, 12);
147 }
148
149 // for our purpose everything > 0 days is a month, we only charge
150 // full months
151 $startDistance->d = 0;
152
153 $months = $startDistance->y > 0 ? 12 : $startDistance->m;
154
155 if (!empty($insuranceEnd)) {
156 // to get the diff right -- $insuranceEnd is the last day where the
157 // instrument was included by into the insurance, we have to add one
158 // day. E.g.: Start 01.07.YYYY, end 30.06.ZZZZ should yield one year and
159 // not 365 days.
160 $endDate = self::convertToTimezoneDate(self::convertToDateTime($insuranceEnd), $timeZone)->modify('+1 day');
161 $endDistance = $dueDate->diff($endDate);
162 if ($endDistance->invert) {
163 // due-date after end-date
164 if ($endDistance->y > 0) {
165 // ended longer than one year ago, so return 0
166 return new RationalNumber(0, 0, 12);
167 }
168 $endDistance->d = 0; // just include fractional months
169 $months -= $endDistance->m;
170 }
171 }
172
173 $fraction = new RationalNumber(0, $months, 12);
174
175 return $fraction;
176 }

248 public function insuranceFee(
249 mixed $musicianOrId,
250 null|string|Entities\InsuranceBroker $broker,
251 $date = null,
252 ?array &$dueInterval = null,
253 ):RationalNumber {
254 $timeZone = $this->getDateTimeZone();
255 if (empty($date)) {
256 $date = new DateTime();
257 }
258 $date = self::convertToTimezoneDate($date, $timeZone);
259
260 $payables = $this->billableInsurances($musicianOrId, $broker);
261
262 $taxFactor = $this->getTaxRate()->add(1);
263
264 $fee = RationalNumber::zero();
267 $minDueDate = $maxDueDate = null;
269 foreach ($payables as $insurance) {
270 $insuranceStart = self::convertToTimezoneDate($insurance->getStartOfInsurance(), $timeZone);
271 $insuranceEnd = $insurance->getDeleted();
272 if (!empty($insuranceEnd)) {
273 $insuranceEnd = self::convertToTimezoneDate($insuranceEnd, $timeZone);
274 }
275
277 $rate = $insurance->getInsuranceRate();
278
279 // end of insurance period
280 $dueDate = $this->dueDate($rate->getDueDate(), $date);
281 if (!empty($insuranceEnd) && $dueDate->modify('-1 year') > $insuranceEnd) {
282 continue;
283 }
284 $minDueDate = empty($minDueDate) ? $dueDate : min($dueDate, $minDueDate);
285 $maxDueDate = empty($maxDueDate) ? $dueDate : max($dueDate, $maxDueDate);
286
287 $annualFee = $rate->getRate()->mul($insurance->getInsuranceAmount());
288 $annualFee->mulEq($this->yearFraction($insuranceStart, $insuranceEnd, $dueDate));
289
290 $fee->addEq($annualFee->mul($taxFactor));
291 }
292 $dueInterval = [ 'min' => $minDueDate, 'max' => $maxDueDate ];
293
294 return $fee->round(2);
295 }

438

441 'taxRate' => $this->getTaxRate(),

443 'musicians' => [],
444 'date' => $date,
445 ];
446
448 foreach ($payableInsurances as $insurance) {
449 $insuranceStart = self::convertToTimezoneDate($insurance->getStartOfInsurance(), $timeZone);
450 $insuranceEnd = $insurance->getDeleted();
451 if (!empty($insuranceEnd)) {
452 $insuranceEnd = self::convertToTimezoneDate($insuranceEnd, $timeZone);
453 }
454
456 $rate = $insurance->getInsuranceRate();
457
458 // end of the insurance year
459 $dueDate = $this->dueDate($rate->getDueDate(), $date);
460 $endDate = $dueDate->modify('-1 day'); // last day of insurance year
461
462 // start of insurance year
463 $lastDueDate = $dueDate->modify('-1 year');
464
465 if (!empty($insuranceEnd) && $lastDueDate > $insuranceEnd) {
466 // exclude instruments which are no longer insured
467 continue;
468 }
469
470 if ($dueDate <= $insuranceStart) {
471 // exclude instruments which were not yet insured in that year
472 continue;
473 }
474
475 $amount = $insurance->getInsuranceAmount();
476 $fraction = $this->yearFraction($insuranceStart, $insuranceEnd, $dueDate);
477 $annualFee = $rate->getRate()->mul($amount);
478
479 $instrumentHolder = $insurance->getInstrumentHolder();
480 $instrumentHolderId = $instrumentHolder->getId();
481 if (empty($insuranceOverview['musicians'][$instrumentHolderId])) {
482 $insuranceOverview['musicians'][$instrumentHolderId] = [
483 'name' => $instrumentHolder->getPublicName(true),
484 'subTotals' => RationalNumber::create(0),
485 'items' => [],
486 ];
487 }
488
489 $itemInfo = [
490 'broker' => $insurance->getBroker()->getShortName(),
491 'scope' => $insurance->getGeographicalScope(),
492 'object' => $insurance->getObject(),
493 'manufacturer' => $insurance->getManufacturer(),
494 'amount' => $amount,
495 'rate' => $rate->getRate(),
496 'lastDue' => $lastDueDate,
497 'due' => empty($insuranceEnd) ? $endDate : $insuranceEnd,
498 'start' => $insuranceStart,
499 'fullFee' => $annualFee,
500 'fraction' => $fraction,
501 'fee' => $annualFee->mul($fraction),
502 ];
503
504 $insuranceOverview['musicians'][$instrumentHolderId]['items'][] = $itemInfo;
505 }
506
507 $annual = RationalNumber::zero();
508 foreach ($insuranceOverview['musicians'] as $id => $info) {
509 // ordinary annular fees
510 $subTotals = RationalNumber::zero();
511 foreach ($info['items'] as $itemInfo) {
512 $subTotals = $subTotals->add($itemInfo['fee']);
513 }
514 $insuranceOverview['musicians'][$id]['subTotals'] = $subTotals;
515 // $this->logInfo('SUBTOTALS '.$subTotals);
516 $annual = $annual->add($subTotals);
517 }
518 $insuranceOverview['annual'] = $annual;
519
520 return $insuranceOverview;
