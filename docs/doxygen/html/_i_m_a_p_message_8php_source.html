281 public function getFullMessage(int $id): array {
282 $mailBody = $this->plainMessage;
283 $data = $this->jsonSerialize();
284 if ($this->hasHtmlMessage) {
285 $data['hasHtmlBody'] = true;
286 $data['body'] = $this->getHtmlBody($id);
287 $data['attachments'] = $this->attachments;
288 } else {
289 $mailBody = $this->htmlService->convertLinks($mailBody);
290 [$mailBody, $signature] = $this->htmlService->parseMailBody($mailBody);
291 $data['body'] = $mailBody;
292 $data['signature'] = $signature;
293 $data['attachments'] = array_merge($this->attachments, $this->inlineAttachments);
294 }
295
296 return $data;
297 }
