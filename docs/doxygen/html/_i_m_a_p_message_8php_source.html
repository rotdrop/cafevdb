283 public function getFullMessage(int $id): array {
284 $mailBody = $this->plainMessage;
285 $data = $this->jsonSerialize();
286 if ($this->hasHtmlMessage) {
287 $data['hasHtmlBody'] = true;
288 $data['body'] = $this->getHtmlBody($id);
289 $data['attachments'] = $this->attachments;
290 } else {
291 $mailBody = $this->htmlService->convertLinks($mailBody);
292 [$mailBody, $signature] = $this->htmlService->parseMailBody($mailBody);
293 $data['body'] = $mailBody;
294 $data['signature'] = $signature;
295 $data['attachments'] = array_merge($this->attachments, $this->inlineAttachments);
296 }
297
298 return $data;
