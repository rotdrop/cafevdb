279 public function getFullMessage(int $id): array {
280 $mailBody = $this->plainMessage;
281 $data = $this->jsonSerialize();
282 if ($this->hasHtmlMessage) {
283 $data['hasHtmlBody'] = true;
284 $data['body'] = $this->getHtmlBody($id);
285 $data['attachments'] = $this->attachments;
286 } else {
287 $mailBody = $this->htmlService->convertLinks($mailBody);
288 [$mailBody, $signature] = $this->htmlService->parseMailBody($mailBody);
289 $data['body'] = $mailBody;
290 $data['signature'] = $signature;
291 $data['attachments'] = array_merge($this->attachments, $this->inlineAttachments);
292 }
293
294 return $data;
