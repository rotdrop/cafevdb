304 public function addProjectInstrument(MusicianInstrument $musicianInstrument, int $voice = ProjectInstrument::UNVOICED):ProjectParticipant
305 {
306 $musician = $musicianInstrument->getMusician();
307 $instrument = $musicianInstrument->getInstrument();
308 if ($this->projectInstruments->exists(fn(mixed $key, ProjectInstrument $projectInstrument)
309 =>
310 $projectInstrument->getMusician() == $musician
311 && $projectInstrument->getInstrument() == $instrument
312 && $projectInstrument->getVoice() == $voice)) {
313 return $this;
314 }
315 $projectInstrument = new ProjectInstrument($this, $musicianInstrument, $voice);
316 $instrumentationNumbers = $this->project->getInstrumentationNumbers()->matching(DBUtil::criteriaWhere([
317 'project' => $this->project,
318 'instrument' => $instrument,
319 'voice' => $voice,
320 ]));
321 if ($instrumentationNumbers->isEmpty()) {
322 $instrumentationNumber = new ProjectInstrumentationNumber(
323 project: $this->project,
324 instrument: $instrument,
325 voice: $voice,
326 );
327 } else {
328 $instrumentationNumber = $instrumentationNumbers->first();
329 }
330 $instrumentationNumber->getProjectInstruments()->set($musician->getId(), $projectInstrument);
331 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
332
333 return $this;
334 }
