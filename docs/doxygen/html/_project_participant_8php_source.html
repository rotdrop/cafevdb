305 public function addProjectInstrument(MusicianInstrument $musicianInstrument, int $voice = ProjectInstrument::UNVOICED):ProjectParticipant
306 {
307 $musician = $musicianInstrument->getMusician();
308 $instrument = $musicianInstrument->getInstrument();
309 if ($this->projectInstruments->exists(fn(mixed $key, ProjectInstrument $projectInstrument)
310 =>
311 $projectInstrument->getMusician() == $musician
312 && $projectInstrument->getInstrument() == $instrument
313 && $projectInstrument->getVoice() == $voice)) {
314 return $this;
315 }
316 $projectInstrument = new ProjectInstrument($this, $musicianInstrument, $voice);
317 $instrumentationNumbers = $this->project->getInstrumentationNumbers()->matching(DBUtil::criteriaWhere([
318 'project' => $this->project,
319 'instrument' => $instrument,
320 'voice' => $voice,
321 ]));
322 if ($instrumentationNumbers->isEmpty()) {
323 $instrumentationNumber = new ProjectInstrumentationNumber(
324 project: $this->project,
325 instrument: $instrument,
326 voice: $voice,
327 );
328 } else {
329 $instrumentationNumber = $instrumentationNumbers->first();
330 }
331 $instrumentationNumber->getProjectInstruments()->set($musician->getId(), $projectInstrument);
332 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
