303 public function addProjectInstrument(MusicianInstrument $musicianInstrument, int $voice = ProjectInstrument::UNVOICED):ProjectParticipant
304 {
305 $musician = $musicianInstrument->getMusician();
306 $instrument = $musicianInstrument->getInstrument();
307 if ($this->projectInstruments->exists(fn(mixed $key, ProjectInstrument $projectInstrument)
308 =>
309 $projectInstrument->getMusician() == $musician
310 && $projectInstrument->getInstrument() == $instrument
311 && $projectInstrument->getVoice() == $voice)) {
312 return $this;
313 }
314 $projectInstrument = new ProjectInstrument($this, $musicianInstrument, $voice);
315 $instrumentationNumbers = $this->project->getInstrumentationNumbers()->matching(DBUtil::criteriaWhere([
316 'project' => $this->project,
317 'instrument' => $instrument,
318 'voice' => $voice,
319 ]));
320 if ($instrumentationNumbers->isEmpty()) {
321 $instrumentationNumber = new ProjectInstrumentationNumber(
322 project: $this->project,
323 instrument: $instrument,
324 voice: $voice,
325 );
326 } else {
327 $instrumentationNumber = $instrumentationNumbers->first();
328 }
329 $instrumentationNumber->getProjectInstruments()->set($musician->getId(), $projectInstrument);
330 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
331
332 return $this;
