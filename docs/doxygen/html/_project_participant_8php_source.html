290 public function addProjectInstrument(MusicianInstrument $musicianInstrument, int $voice = ProjectInstrument::UNVOICED):ProjectParticipant
291 {
292 $musician = $musicianInstrument->getMusician();
293 $instrument = $musicianInstrument->getInstrument();
294 if ($this->projectInstruments->exists(fn(mixed $key, ProjectInstrument $projectInstrument)
295 =>
296 $projectInstrument->getMusician() == $musician
297 && $projectInstrument->getInstrument() == $instrument
298 && $projectInstrument->getVoice() == $voice)) {
299 return $this;
300 }
301 $projectInstrument = new ProjectInstrument($this, $musicianInstrument, $voice);
302 $instrumentationNumbers = $this->project->getInstrumentationNumbers()->matching(DBUtil::criteriaWhere([
303 'project' => $this->project,
304 'instrument' => $instrument,
305 'voice' => $voice,
306 ]));
307 if ($instrumentationNumbers->isEmpty()) {
308 $instrumentationNumber = new ProjectInstrumentationNumber(
309 project: $this->project,
310 instrument: $instrument,
311 voice: $voice,
312 );
313 } else {
314 $instrumentationNumber = $instrumentationNumbers->first();
315 }
316 $instrumentationNumber->getProjectInstruments()->set($musician->getId(), $projectInstrument);
317 $projectInstrument->setInstrumentationNumber($instrumentationNumber);
318
319 return $this;
320 }
