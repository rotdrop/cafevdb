103 {
104 $operation = EnumProgressStatusOperation::get($operation);

108 $target = $this->request->getParam('target', null);

111 $progress = $this->progressStatusService->create(
112 start: $current,
113 stop: $target,
114 data: $data,

116 return self::progressResponse($progress);

118 throw new Exceptions\EnduserNotificationException(
119 message: $this->l->t('Caught an exception.'),
120 previous: $t,
121 context: compact('current', 'target', 'data'),
122 );
123 }
124 break;
125 case EnumProgressStatusOperation::UPDATE:
126 $id = $this->request->getParam('id');
127 try {
128 $progress = $this->progressStatusService->get($id);
129 } catch (Throwable $t) {
130 throw new Exceptions\EnduserNotificationException(
131 message: $this->l->t('Caught an exception.'),
132 previous: $t,
133 context: compact('id'),
134 );
135 }
136 try {
137 $current = $this->request->getParam('current', null);
138 $target = $this->request->getParam('target', null);
139 $data = $this->request->getParam('data', null);
140 $progress->update($current, $target, $data);
141 return self::progressResponse($progress);
142 } catch (Throwable $t) {
143 throw new Exceptions\EnduserNotificationException(
144 message: $this->l->t('Caught an exception.'),
145 previous: $t,
146 context: compact('id', 'current', 'target', 'data'),
147 );
148 }
149 break;
150 case EnumProgressStatusOperation::DELETE:
151 $id = $this->request->getParam('id');

153 $progress = $this->progressStatusService->get($id);
154 } catch (Throwable $t) {

157 previous: $t,
158 context: compact('id'),

160 }
