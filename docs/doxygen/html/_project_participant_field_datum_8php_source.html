429 public static function fromDefaultValue(ProjectParticipantField $field, ProjectParticipant $participant): ?self
430 {
431 $defaultValue = $field->getDefaultValue();
432 if ($defaultValue === null) {
433 return null;
434 }
435 $fieldContext = $field->getParticipationContext();
436 $participantContext = $participant->getParticipationContext();
437 if ($fieldContext != ParticipationContext::UNRESTRICTED
438 && $participantContext != ParticipationContext::UNRESTRICTED
439 && $fieldContext != $participantContext) {
440 return null;
441 }
442
443 $datum = (new ProjectParticipantFieldDatum)
444 ->setField($field)
445 ->setProjectParticipant($participant)
446 ->setDataOption($defaultValue);
447 switch ($field->getMultiplicity()) {
448 case FieldMultiplicity::RECURRING:
449 case FieldMultiplicity::SIMPLE:
450 // value is stored in the the data item
451 $datum->setOptionValue($defaultValue->getData());
452 $datum->setDeposit($defaultValue->getDeposit());
453 break;
454 default:
455 break;
456 }
457 // Perform some validation and check whether the resulting field value makes sense.
458 $effectiveValue = $datum->getEffectiveValue();
459 switch ($field->getDataType()) {
460 case FieldDataType::BOOLEAN:
461 $valid = null !== filter_var($effectiveValue, FILTER_VALIDATE_BOOLEAN, ['flags' => FILTER_NULL_ON_FAILURE]);
462 break;
463 case FieldDataType::INTEGER:
464 $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_INT);
465 break;
466 case FieldDataType::DB_FILE:
467 // database id must be >= 1
468 $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]]);
469 break;
470 case FieldDataType::DATE:
471 // fallthrough
472 case FieldDataType::DATETIME:
473 try {
474 new DateTimeImmutable($effectiveValue);
475 $valid = true;
476 } catch (Throwable) {
477 $valid = false;
478 }
479 break;
480 case FieldDataType::FLOAT:
481 $valid = false !== filter_var($effectiveValue, FILTER_VALIDATE_FLOAT);
482 break;
483 // string types
484 case FieldDataType::CLOUD_FILE:
485 case FieldDataType::CLOUD_FOLDER:
486 case FieldDataType::TEXT:
487 case FieldDataType::HTML:
488 $valid = is_string($effectiveValue);
489 break;
490 case FieldDataType::LIABILITIES:
491 case FieldDataType::RECEIVABLES:
492 $valid = null !== MonetaryNumberType::create($effectiveValue);
493 break;
494 default:
495 // not reached, but a safeguard for future changes
496 return null;
497 }
498 if (!$valid) {
499 return null;
500 }
501
502 $field->getFieldData()->add($datum);
503 $defaultValue->getFieldData()->set($participant->getMusician()->getId(), $datum);
504
505 return $datum;
506 }

519 public function amountPayable(): MonetaryNumberType
520 {
521 switch ($this->field->getMultiplicity()) {
522 case FieldMultiplicity::SINGLE:
523 case FieldMultiplicity::MULTIPLE:
524 case FieldMultiplicity::PARALLEL:
525 case FieldMultiplicity::GROUPSOFPEOPLE:
526 $storedValue = $this->dataOption->getData();
527 try {
528 $value = MonetaryNumberType::fromDecimal($storedValue);
529 } catch (Throwable $t) {
530 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
531 }
532 break;
533 case FieldMultiplicity::GROUPOFPEOPLE:
534 // value in management option of $field
535 $managementOption = $this->field->getManagementOption();
536 if (empty($managementOption)) {
537 throw new RuntimeException('Unable to access management option for obtaining the field value.');
538 }
539 $storedValue = $managementOption->getData();
540 try {
541 $value = MonetaryNumberType::fromDecimal($storedValue);
542 } catch (Throwable $t) {
543 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
544 }
545 break;
546 case FieldMultiplicity::SIMPLE:
547 case FieldMultiplicity::RECURRING:
548 if (empty($this->optionValue)) {
549 $value = MonetaryNumberType::zero();
550 } else {
551 $storedValue = $this->optionValue;
552 try {
553 $value = MonetaryNumberType::fromDecimal($storedValue);
554 } catch (Throwable $t) {
555 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
556 }
557 }
558 break;
559 default:
560 throw new RuntimeException('Unhandled multiplicity tag: "' . (string)$this->field->getMultiplicity() . '".');
561 }
562 if ($this->field->getDataType() == FieldDataType::LIABILITIES) {
563 $value = $value->neg();
564 }
565 return $value;
566 }
