432 public function amountPayable():float
433 {
434 $value = 0.0;
435 switch ($this->field->getMultiplicity()) {
436 case Multiplicity::SINGLE():
437 case Multiplicity::MULTIPLE():
438 case Multiplicity::PARALLEL():
439 case Multiplicity::GROUPSOFPEOPLE():
440 $value = filter_var($this->dataOption->getData(), FILTER_VALIDATE_FLOAT);
441 if ($value === false) {
442 throw new RuntimeException('Stored value cannot be converted to float.');
443 }
444 break;
445 case Multiplicity::GROUPOFPEOPLE():
446 // value in management option of $field
447 $managementOption = $this->field->getManagementOption();
448 if (empty($managementOption)) {
449 throw new RuntimeException('Unable to access management option for obtaining the field value.');
450 }
451 $value = filter_var($managementOption->getData(), FILTER_VALIDATE_FLOAT);
452 if ($value === false) {
453 throw new RuntimeException('Stored value cannot be converted to float.');
454 }
455 break;
456 case Multiplicity::SIMPLE():
457 case Multiplicity::RECURRING():
458 if (!empty($this->optionValue)) {
459 $value = filter_var($this->optionValue, FILTER_VALIDATE_FLOAT);
460 if ($value === false) {
461 throw new RuntimeException('Stored value cannot be converted to float: ' . (string)$this->optionValue);
462 }
463 }
464 break;
465 default:
466 throw new RuntimeException('Unhandled multiplicity tag: '.(string)$this->field->getMultiplicity());
467 }
468 if ($this->field->getDataType() == DataType::LIABILITIES) {
469 $value = -$value;
470 }
471 return $value;
472 }
