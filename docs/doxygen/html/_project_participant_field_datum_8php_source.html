420 public static function fromDefaultValue(ProjectParticipantField $field, ProjectParticipant $participant):?ProjectParticipantFieldDatum
421 {
422 $defaultValue = $field->getDefaultValue();
423 if ($defaultValue === null) {
424 return null;
425 }
426 $fieldContext = $field->getParticipationContext();
427 $participantContext = $participant->getParticipationContext();
428 if ($fieldContext != ParticipationContext::UNRESTRICTED
429 && $participantContext != ParticipationContext::UNRESTRICTED
430 && $fieldContext != $participantContext) {
431 return null;
432 }
433
434 $datum = (new ProjectParticipantFieldDatum)
435 ->setField($field)
436 ->setProjectParticipant($participant)
437 ->setDataOption($defaultValue);
438 switch ($field->getMultiplicity()) {
439 case Multiplicity::SIMPLE:
440 // value is stored in the the data item
441 $datum->setOptionValue($defaultValue->getData());
442 $datum->setDeposit($defaultValue->getDeposit());
443 break;
444 default:
445 break;
446 }
447 $field->getFieldData()->add($datum);
448 $defaultValue->getFieldData()->set($participant->getMusician()->getId(), $datum);
449
450 return $datum;
451 }

464 public function amountPayable():RationalNumber
465 {
466 switch ($this->field->getMultiplicity()) {
467 case Multiplicity::SINGLE():
468 case Multiplicity::MULTIPLE():
469 case Multiplicity::PARALLEL():
470 case Multiplicity::GROUPSOFPEOPLE():
471 $storedValue = $this->dataOption->getData();
472 try {
473 $value = RationalNumber::fromDecimal($storedValue);
474 } catch (Throwable $t) {
475 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
476 }
477 break;
478 case Multiplicity::GROUPOFPEOPLE():
479 // value in management option of $field
480 $managementOption = $this->field->getManagementOption();
481 if (empty($managementOption)) {
482 throw new RuntimeException('Unable to access management option for obtaining the field value.');
483 }
484 $storedValue = $managementOption->getData();
485 try {
486 $value = RationalNumber::fromDecimal($storedValue);
487 } catch (Throwable $t) {
488 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
489 }
490 break;
491 case Multiplicity::SIMPLE():
492 case Multiplicity::RECURRING():
493 if (empty($this->optionValue)) {
494 $value = RationalNumber::zero();
495 } else {
496 $storedValue = $this->optionValue;
497 try {
498 $value = RationalNumber::fromDecimal($storedValue);
499 } catch (Throwable $t) {
500 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
501 }
502 }
503 break;
504 default:
505 throw new RuntimeException('Unhandled multiplicity tag: "' . (string)$this->field->getMultiplicity() . '".');
506 }
507 if ($this->field->getDataType() == DataType::LIABILITIES) {
508 $value = $value->neg();
509 }
510 return $value;
511 }
