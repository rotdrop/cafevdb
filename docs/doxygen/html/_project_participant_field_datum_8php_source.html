445 public static function fromDefaultValue(ProjectParticipantField $field, ProjectParticipant $participant):?ProjectParticipantFieldDatum
446 {
447 $defaultValue = $field->getDefaultValue();
448 if ($defaultValue === null) {
449 return null;
450 }
451 $fieldContext = $field->getParticipationContext();
452 $participantContext = $participant->getParticipationContext();
453 if ($fieldContext != ParticipationContext::UNRESTRICTED
454 && $participantContext != ParticipationContext::UNRESTRICTED
455 && $fieldContext != $participantContext) {
456 return null;
457 }
458
459 $datum = (new ProjectParticipantFieldDatum)
460 ->setField($field)
461 ->setProjectParticipant($participant)
462 ->setDataOption($defaultValue);
463 switch ($field->getMultiplicity()) {
464 case Multiplicity::SIMPLE:
465 // value is stored in the the data item
466 $datum->setOptionValue($defaultValue->getData());
467 $datum->setDeposit($defaultValue->getDeposit());
468 break;
469 default:
470 break;
471 }
472 $field->getFieldData()->add($datum);
473 $defaultValue->getFieldData()->set($participant->getMusician()->getId(), $datum);
474
475 return $datum;
476 }

489 public function amountPayable():RationalNumber
490 {
491 switch ($this->field->getMultiplicity()) {
492 case Multiplicity::SINGLE():
493 case Multiplicity::MULTIPLE():
494 case Multiplicity::PARALLEL():
495 case Multiplicity::GROUPSOFPEOPLE():
496 $storedValue = $this->dataOption->getData();
497 try {
498 $value = RationalNumber::fromDecimal($storedValue);
499 } catch (Throwable $t) {
500 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
501 }
502 break;
503 case Multiplicity::GROUPOFPEOPLE():
504 // value in management option of $field
505 $managementOption = $this->field->getManagementOption();
506 if (empty($managementOption)) {
507 throw new RuntimeException('Unable to access management option for obtaining the field value.');
508 }
509 $storedValue = $managementOption->getData();
510 try {
511 $value = RationalNumber::fromDecimal($storedValue);
512 } catch (Throwable $t) {
513 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
514 }
515 break;
516 case Multiplicity::SIMPLE():
517 case Multiplicity::RECURRING():
518 if (empty($this->optionValue)) {
519 $value = RationalNumber::zero();
520 } else {
521 $storedValue = $this->optionValue;
522 try {
523 $value = RationalNumber::fromDecimal($storedValue);
524 } catch (Throwable $t) {
525 throw new RuntimeException('Stored value cannot be converted to decimal: "' . $storedValue . '".');
526 }
527 }
528 break;
529 default:
530 throw new RuntimeException('Unhandled multiplicity tag: "' . (string)$this->field->getMultiplicity() . '".');
531 }
532 if ($this->field->getDataType() == DataType::LIABILITIES) {
533 $value = $value->neg();
534 }
535 return $value;
536 }
