334 public function handlePreNotification(
335 Entities\SepaDebitNote $debitNote,
336 Entities\CompositePayment $payment,
337 ): void {
338
339 $notifiedCount = $debitNote->getPayments()->filter(
340 fn(Entities\CompositePayment $payment) => !empty($payment->getNotificationMessageId())
341 )->count();
342 $totalCount = $debitNote->getPayments()->count();
343
344 if ($notifiedCount == $totalCount) {
345 $percentage = 100;
346 } else {
347 $percentage = (int)round((float)$notifiedCount * 100.0 / (float)$totalCount);
348 }
349
350 try {
351 $this->logInfo('TWEAK PRE NOTIFICATION TASK');
352 $preNotificationTaskUri = $debitNote->getPreNotificationTaskUri();
353 $preNotificationTask = $this->financeService->findFinanceCalendarEntry($preNotificationTaskUri);
354 if (!empty($preNotificationTask)) {
355 $this->eventsService->setCalendarTaskStatus($preNotificationTask, percentComplete: $percentage);
356 }
357 } catch (Throwable $t) {
358 $this->logException($t, 'Unable to tweak pre-notification task ' . $preNotificationTaskUri);
359 }
360
361 try {
362 $this->logInfo('TWEAK PRE NOTIFICATION EVENT');
363 $preNotificationEventUri = $debitNote->getPreNotificationEventUri();
364 $preNotificationEvent = $this->financeService->findFinanceCalendarEntry($preNotificationEventUri);
365 if (!empty($preNotificationEvent)) {
366 $vCalendar = VCalendarService::getVCalendar($preNotificationEvent);
367 $description = VCalendarService::getDescription($vCalendar);
368 $musician = $payment->getMusician();
369 $description .= "\n----\n"
370 . $this->l->t('PRENOTIFICATION HAS BEEN SENT:')
371 . "\n"
372 . $this->l->t('Person: %s', $musician->getPublicName(true) . ' <' . $musician->getEmail() . '>')
373 . "\n"
374 . $this->l->t('Date: %s', $this->dateTimeFormatter->formatDateTime($this->timeFactory->now()))
375 . "\n"
376 . $this->l->t('MessageId: %s', $payment->getNotificationMessageId());
377 $updateData = [
378 'description' => $description,
379 ];
380 if ($percentage == 100) {
381 $updateData['alarm'] = 0;
382 }
383 $this->eventsService->updateCalendarEntry($preNotificationEvent, $updateData);
384 }
385 } catch (Throwable $t) {
386 $this->logException($t, 'Unable to tweak pre-notification event ' . $preNotificationEventUri);
387 }
388 }

475 public function getBulkTransactionSlug(Entities\SepaBulkTransaction $transaction):string
476 {
477 if ($transaction instanceof Entities\SepaBankTransfer) {
478 return self::TRANSACTION_TYPE_BANK_TRANSFER;
479 }
480 $slugParts = [ self::TRANSACTION_TYPE_DEBIT_NOTE => true ];
481
482 $filterState = $this->disableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
483
485 foreach ($transaction->getPayments() as $compositePayment) {
487 foreach ($compositePayment->getProjectPayments() as $projectPayment) {
489 $field = $projectPayment->getReceivable()->getField();
490
491 if ($field->getMultiplicity() == FieldMultiplicity::RECURRING) {
492 $generator = $field->getManagementOption()->getData();
493 $optionSlug = method_exists($generator, 'slug')
494 ? $generator::slug() : $field->getName();
495 } else {
496 $optionSlug = $field->getName();
497 }
498 $slugParts[$optionSlug] = true;
499 }
500 }
501
502 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
503
504 $slugParts = array_keys($slugParts);
505 if (count($slugParts) > 2) {
506 return $slugParts[0];
507 } else {
508 return implode('-', $slugParts);
509 }

531 public function generateProjectPayments(
532 Entities\ProjectParticipant $participant,
533 array $receivableOptions,
534 ?\DateTimeInterface $transactionDueDate = null,
535 ): Entities\CompositePayment {
536 $payments = new ArrayCollection();
537 $totalAmount = MonetaryNumberType::zero();
538 $project = $participant->getProject();
539 $musician = $participant->getMusician();
540
542 $compositePayment = (new Entities\CompositePayment)
543 ->setProjectParticipant($participant)
544 ->setProjectPayments($payments)
545 ->setAmount($totalAmount)
546 ->setSubject('');
547
548 if (empty($receivableOptions)) {
549 return $compositePayment;
550 }
551
553 foreach ($receivableOptions as $receivableOption) {
554 if ($project != $receivableOption->getField()->getProject()) {
555 throw new RuntimeException(
556 $this->l->t('Refusing to generate payments for mismatching projects, current is "%s", but the receivable belongs to project "%s".', [
557 $project->getName(),
558 $receivableOption->getField()->getProject()->getName(),
559 ]));
560 }
561 $receivableDueDate = $receivableOption->getField()->getDueDate();
562 $depositDueDate = $receivableOption->getField()->getDepositDueDate();
564 foreach ($receivableOption->getMusicianFieldData($musician) as $receivable) {
565 $paidAmount = $receivable->amountPaid();
566 $payableAmount = $receivable->amountPayable();
567 $depositAmount = $receivable->depositAmount() ?? MonetaryNumberType::zero();
568 if ($payableAmount->sign() * $depositAmount->sign() < 0) {
569 throw new RuntimeException(
570 $this->l->t(
571 'Payable amount "%1$s" and deposit amount "%2$s" should have the compatible signs.', [
572 $payableAmount->toDecimal(2), $depositAmount->toDecimal(2),
573 ]));
574 }
575 if (!empty($transactionDueDate) && !empty($receivableDueDate)) {
576 if ($payableAmount->gt(0)) {
577 // debit note
578 if ($receivableDueDate <= $transactionDueDate) {
579 // past due-date, just keep billing the entire amount
580 } elseif ($depositDueDate <= $transactionDueDate) {
581 // past deposit-due date, charge the deposit
582 $payableAmount = $depositAmount;
583 } else {
584 // too early, just don't charge anything
585 $payableAmount->assign(0);
586 $this->logInfo('NOT YET DUE; DOING NOTHING');
587 }
588 } else {
589 // bank transfer
590 if ($transactionDueDate <= $depositDueDate) {
591 // early payment, only up to deposit
592 $payableAmount = $depositAmount;
593 } else {
594 // just transfer anything, i.e. keep the amount as is
595 }
596 }
597 }
598 $debitAmount = $payableAmount->sub($paidAmount)->round(2);
599 if ($debitAmount->eq(0)) {
600 // No need to debit empty amounts
601 // @todo Perhaps empty amounts should also be recorded.
602 continue;
603 }
604 $this->logInfo('RECEIVABLE FIELD ID ' . $receivable->getField()->getId());
605 $this->logInfo('RECEIVABLE FIELD MULTIPLICITY ' . $receivable->getField()->getMultiplicity()->value);
606 $this->refreshEntity($receivable->getField());
607 $this->logInfo('RECEIVABLE FIELD NAME ' . $receivable->getField()->getName());
609 $payment = (new Entities\ProjectPayment)
610 ->setProject($project)
611 ->setMusician($musician)
612 ->setReceivable($receivable)
613 ->setReceivableOption($receivableOption)
614 ->setAmount($debitAmount)
615 ->setCompositePayment($compositePayment)
616 ->setSubject($receivable->paymentReference())
617 ->setIsDonation(false)
618 ;
619
620 $payments->add($payment);
621 $totalAmount->addEq($debitAmount);
622 }
623 }
624
625 $compositePayment
626 ->setAmount($totalAmount)
627 ->updateSubject(fn($x) => $this->financeService->sepaTranslit($x));
628
629 return $compositePayment;
630 }

708 {
709 $calendarIds = $this->getCalendarObjects($bulkTransaction);
710
711 // $this->logInfo('CALIDS ' . print_r($calendarIds, true));
712
713 if ($only === null) {
714 $only = array_keys($calendarIds);

725 if (!empty($calendarObject)) {

763 {
764 if (!$force && $bulkTransaction->getSubmitDate() !== null) {
765 throw new Exceptions\DatabaseReadonlyException(

770
771 $this->entityManager->beginTransaction();
772
773 $this->entityManager->registerPreCommitAction(

789 },
790 );

793 // transaction data is handled in a separate listener.
794 $this->remove($bulkTransaction, flush: true);
795 $this->entityManager->commit();
796
797 } catch (Throwable $t) {
798 $this->entityManager->rollback();
799 throw new Exceptions\DatabaseException(
800 $this->l->t('Failed to remove bulk-transaction with id %d', $bulkTransaction->getId()),

855 public function generateTransactionData(
856 Entities\SepaBulkTransaction $bulkTransaction,
857 ?Entities\Project $project = null,
858 string $format = self::EXPORT_AQBANKING,
859 ): ?Entities\DatabaseStorageFile {
860 // as a safe-guard regenerate the subject in order to catch changes in
861 // linked supporting documents.
862 $this->updateBulkTransaction($bulkTransaction, flush: true);
863
864 $exportDocument = null;
865 $transcationData = $bulkTransaction->getSepaTransactionData();
867 foreach ($transcationData as $exportDocument) {
868 if (strpos($exportDocument->getName(), $format) !== false) {
869 break;
870 }
871 $exportDocument = null;
872 }
873
874 if (true || empty($exportDocument) || $bulkTransaction->getUpdated() > $exportDocument->getUpdated()) {
875
877 $storage = $this->appContainer->get(BankTransactionsStorage::class);
878
880 $exporter = $this->getTransactionExporter($format);
881 if (empty($exporter)) {
882 throw new InvalidArgumentException($this->l->t('Unable to find exporter for format "%s".', $format));
883 }
884 if ($bulkTransaction instanceof Entities\SepaBankTransfer) {
885 $transactionType = self::TRANSACTION_TYPE_BANK_TRANSFER;
886 } elseif ($bulkTransaction instanceof Entities\SepaDebitNote) {
887 $transactionType = self::TRANSACTION_TYPE_DEBIT_NOTE;
888 }
889
