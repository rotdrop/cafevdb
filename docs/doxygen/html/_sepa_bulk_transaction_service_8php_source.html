312 'description' => $description,

316 }
317 $this->eventsService->updateCalendarEntry($preNotificationEvent, $updateData);
318 }

320 $this->logException($t, 'Unable to tweak pre-notification event ' . $preNotificationEventUri);

413 {
414 if ($transaction instanceof Entities\SepaBankTransfer) {

416 }
417 $slugParts = [ self::TRANSACTION_TYPE_DEBIT_NOTE => true ];
418
419 $filterState = $this->disableFilter(EntityManager::SOFT_DELETEABLE_FILTER);
420
422 foreach ($transaction->getPayments() as $compositePayment) {

439 $filterState && $this->enableFilter(EntityManager::SOFT_DELETEABLE_FILTER);

468 public function generateProjectPayments(Entities\ProjectParticipant $participant, array $receivableOptions, ?\DateTimeInterface $transactionDueDate = null):Entities\CompositePayment
469 {
470 $payments = new ArrayCollection();
471 $totalAmount = RationalNumber::zero();
472 $project = $participant->getProject();
473 $musician = $participant->getMusician();
474
476 $compositePayment = (new Entities\CompositePayment)
477 ->setProjectParticipant($participant)
478 ->setProjectPayments($payments)
479 ->setAmount($totalAmount)
480 ->setSubject('');
481
482 if (empty($receivableOptions)) {
483 return $compositePayment;
484 }
485
487 foreach ($receivableOptions as $receivableOption) {
488 if ($project != $receivableOption->getField()->getProject()) {
489 throw new RuntimeException(
490 $this->l->t('Refusing to generate payments for mismatching projects, current is "%s", but the receivable belongs to project "%s".', [
491 $project->getName(),
492 $receivableOption->getField()->getProject()->getName(),
493 ]));
494 }
495 $receivableDueDate = $receivableOption->getField()->getDueDate();
496 $depositDueDate = $receivableOption->getField()->getDepositDueDate();
498 foreach ($receivableOption->getMusicianFieldData($musician) as $receivable) {
499 $paidAmount = $receivable->amountPaid();
500 $payableAmount = $receivable->amountPayable();
501 $depositAmount = $receivable->depositAmount() ?? RationalNumber::zero();
502 if ($payableAmount->sign() * $depositAmount->sign() < 0) {
503 throw new RuntimeException(
504 $this->l->t(
505 'Payable amount "%1$s" and deposit amount "%2$s" should have the compatible signs.', [
506 $payableAmount->toDecimal(2), $depositAmount->toDecimal(2),
507 ]));
508 }
509 if (!empty($transactionDueDate) && !empty($receivableDueDate)) {
510 if ($payableAmount->gt(0)) {
511 // debit note
512 if ($receivableDueDate <= $transactionDueDate) {
513 // past due-date, just keep billing the entire amount
514 } elseif ($depositDueDate <= $transactionDueDate) {
515 // past deposit-due date, charge the deposit
516 $payableAmount = $depositAmount;
517 } else {
518 // too early, just don't charge anything
519 $payableAmount->assign(0);
520 $this->logInfo('NOT YET DUE; DOING NOTHING');
521 }
522 } else {
523 // bank transfer
524 if ($transactionDueDate <= $depositDueDate) {
525 // early payment, only up to deposit
526 $payableAmount = $depositAmount;
527 } else {
528 // just transfer anything, i.e. keep the amount as is
529 }
530 }
531 }
532 $debitAmount = $payableAmount->sub($paidAmount)->round(2);
533 if ($debitAmount->eq(0)) {
534 // No need to debit empty amounts
535 // @todo Perhaps empty amounts should also be recorded.
536 continue;
537 }
539 $payment = (new Entities\ProjectPayment)
540 ->setProject($project)
541 ->setMusician($musician)
542 ->setReceivable($receivable)
543 ->setReceivableOption($receivableOption)
544 ->setAmount($debitAmount)
545 ->setCompositePayment($compositePayment)
546 ->setSubject($receivable->paymentReference())
547 ->setIsDonation(false)
548 ;
549
550 $payments->add($payment);
551 $totalAmount->addEq($debitAmount);
552 }
553 }
554
555 $compositePayment
556 ->setAmount($totalAmount)
557 ->updateSubject(fn($x) => $this->financeService->sepaTranslit($x));
558
559 return $compositePayment;
560 }

637 public function stashCalendarObjects(Entities\SepaBulkTransaction $bulkTransaction, ?array $only = null):array
638 {
639 $calendarIds = $this->getCalendarObjects($bulkTransaction);
640
641 // $this->logInfo('CALIDS ' . print_r($calendarIds, true));
642
643 if ($only === null) {
644 $only = array_keys($calendarIds);
645 } else {
646 $only = array_intersect($only, array_keys($calendarIds));
647 }
648
649 $calendarObjects = [];

652 foreach ($eventIds as $objectType => $objectIds) {

655 if (!empty($calendarObject)) {

657 break;
658 }
659 }
660 }

785 public function generateTransactionData(
786 Entities\SepaBulkTransaction $bulkTransaction,
787 ?Entities\Project $project = null,
788 null|string $format = null,
789 ):?Entities\DatabaseStorageFile {
790 $format = $format ?? self::EXPORT_AQBANKING;
791
792 // as a safe-guard regenerate the subject in order to catch changes in
793 // linked supporting documents.
794 $this->updateBulkTransaction($bulkTransaction, flush: true);
795
796 $exportDocument = null;
797 $transcationData = $bulkTransaction->getSepaTransactionData();
799 foreach ($transcationData as $exportDocument) {
800 if (strpos($exportDocument->getName(), $format) !== false) {
801 break;
802 }
803 $exportDocument = null;
804 }
805
806 if (true || empty($exportDocument) || $bulkTransaction->getUpdated() > $exportDocument->getUpdated()) {
807
809 $storage = $this->appContainer->get(BankTransactionsStorage::class);
810
812 $exporter = $this->getTransactionExporter($format);
813 if (empty($exporter)) {
814 throw new InvalidArgumentException($this->l->t('Unable to find exporter for format "%s".', $format));
815 }
816 if ($bulkTransaction instanceof Entities\SepaBankTransfer) {
817 $transactionType = self::TRANSACTION_TYPE_BANK_TRANSFER;
818 } elseif ($bulkTransaction instanceof Entities\SepaDebitNote) {
819 $transactionType = self::TRANSACTION_TYPE_DEBIT_NOTE;
820 }
821
822 $timeStamp = $this->timeStamp();
823
825 foreach ($bulkTransaction->getPayments() as $payment) {
826 $paymentProject = $payment->getProject();
827 if (empty($project)) {
828 $project = $paymentProject;
829 } elseif ($project != $paymentProject) {
830 throw new UnexpectedValueException($this->l->t(
831 'Conflicting projects "%1$s" vs. "%2$s".', [
832 (string)$project, (string)$paymentProject
833 ]));
834 }
835 }
836
837 $fileName = implode('-', array_filter([
838 $timeStamp,
839 $transactionType,
840 !empty($project) ? $project->getName() : null,
841 $format,
842 ])) . '.' . $exporter->fileExtension($bulkTransaction);
843
844 $fileData = $exporter->fileData($bulkTransaction);
845
846 if (empty($exportDocument)) {
847 $exportFile = new Entities\EncryptedFile(
848 fileName: $fileName,
849 data: $fileData,
850 mimeType: $exporter->mimeType($bulkTransaction)
851 );
852 $this->persist($exportFile);
853 } else {
854 $exportDocument
855 ->setName($fileName);
856 $exportFile = $exportDocument->getFile();
857 $exportFile
858 ->setFileName($fileName)
859 ->setMimeType($exporter->mimeType($bulkTransaction))
860 ->setSize(strlen($fileData))
861 ->getFileData()->setData($fileData);
862
863 }
864
865 $this->entityManager->beginTransaction();
866 try {
867 $exportDocument = $storage->addDocument($bulkTransaction, $exportFile, flush: false);
868 $bulkTransaction->addTransactionData($exportDocument);
869 $this->flush();
870 $this->entityManager->commit();
871 } catch (Throwable $t) {
872 $this->entityManager->rollback();
873 throw new Exceptions\DatabaseException(
874 $this->l->t(
875 'Unable to generate export data for bulk-transaction id %1$d, format "%2$s".', [
876 $bulkTransaction->getId(), $format
877 ]
878 ),
879 $t->getCode(),
880 $t
881 );
882 }
883 }
884 return $exportDocument;
885 }
