246 public function initUserKeyPair(bool $forceNewKeyPair = false):void
247 {
248 $exception = null;
249 try {
250 $this->asymKeyService->initEncryptionKeyPair($this->userId, $this->userPassword, $forceNewKeyPair);
251 } catch (Exceptions\EncryptionException $exception) {
252 // Gracefully accept a broken key-pair if the app-encryption key is empty.
253 try {
254 $userKey = $this->getUserEncryptionKey();
255 $exception = null;
256 if (empty($userKey)) {
257 // after all, this means that all values are unencrypted, so be graceful here
258 $this->asymKeyService->initEncryptionKeyPair($this->userId, $this->userPassword, forceNewKeyPair: true);
259 }
260 } catch (Throwable $t) {
261 // give up
262 $this->logException($t, 'User\'s "' . $this->userId . '" asymmetric key pair is broken and encryption key appears to be non-empty.');
263 }
264 }
265 $this->userAsymmetricCryptor = $this->asymKeyService->getCryptor($this->userId);
266 if (!empty($exception)) {
267 $this->userAsymmetricCryptor
268 ->setPrivateKey(null)
269 ->setPublicKey(null);
270 throw $exception;
271 }

374 public function initAppEncryptionKey():bool
375 {
376 if (!$this->bound()) {
377 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize global encryption key without bound user credentials.'));
378 }
379
380 $userDatabaseKey = $this->getUserEncryptionKey();
381 if (empty($userDatabaseKey)) {
382 // No key -> unencrypted
383 $this->logDebug("No Encryption Key, setting to empty string in order to disable encryption.");
384 $this->appCryptor->setEncryptionKey(''); // not null, just empty
385 } else {
386 $this->appCryptor->setEncryptionKey($userDatabaseKey);
387 }
388
389 // compare the user-key with the stored encryption key hash
390 $sysDatabaseKeyHash = $this->getConfigValue(ConfigConstants::APP_ENCRYPTION_KEY_HASH_KEY);
391 if (!$this->verifyHash($userDatabaseKey, $sysDatabaseKeyHash)) {
392 $failCount = 0;
393 $numberOfTests = 1000;
394 for ($i = 0; $i < $numberOfTests; ++$i) {
395 $failCount += 1 - (int)$this->verifyHash($userDatabaseKey, $sysDatabaseKeyHash);
396 }
397 if ($failCount < 10) {
398 $msg = 'BOGUS HARDWARE OR SOFTWARE, VERIFY HASH FAILED ' . $failCount . ' TIMES OUT OF ' . $numberOfTests . ' TIMES.';
399 $this->logError(
400 $msg,
401 [ 'exception' => new Exceptions\EncryptionException($msg) ],
402 );
403 return true;
404 }
405 // Failed
406 $this->appCryptor->setEncryptionKey(null);
407 throw new Exceptions\EncryptionKeyException($this->l->t('Failed to validate user encryption key. ' . $failCount . ' "' . $sysDatabaseKeyHash) . '"');
408 $this->logError('Unable to validate HASH for encryption key.');
409 } else {
410 $this->logDebug('Encryption keys validated'.(empty($userDatabaseKey) ? ' (no encryption)' : '').'.');
411 // $this->logInfo($this->l->t('Validated user encryption key. ' . $sysDatabaseKeyHash));
412 }
413
414 return true;
415 }

474 public function encryptionKeyValid(?string $encryptionKey = null):bool
475 {
476 if (!$this->bound()) {
477 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot validate app encryption key without bound user credentials'));
478 }
479
480 if (empty($encryptionKey)) {
481 $encryptionKey = $this->appCryptor->getEncryptionKey();
482 }
483 if (empty($encryptionKey)) {
484 $this->logWarn('Provided encryption-key is empty, encryption is switched off.');
485 }
486
487 $sysDatabaseKeyHash = $this->getConfigValue(ConfigConstants::APP_ENCRYPTION_KEY_HASH_KEY);
488
489 if (empty($sysDatabaseKeyHash) !== empty($encryptionKey)) {
490 if (empty($sysDatabaseKeyHash)) {
491 $this->logError('Stored encryption key is empty while provided encryption key is not empty.');
492 } else {
493 $this->logError('Stored encryption key is not empty while provided encryption key is empty.');
494 }
495 return false;
496 }
497
498 $match = $this->verifyHash($encryptionKey, $sysDatabaseKeyHash);
499
500 if (!$match) {
501 $this->logError('Unable to validate HASH for encryption key.');
502 } else {
503 $this->logDebug('Validated HASH for encryption key.');
504 }
505
506 return $match;

522 public function getConfigValue(string $key, mixed $default = null, bool $ignoreLock = false)
523 {
524 if (!$ignoreLock && !empty($this->getAppValue(ConfigConstants::CONFIG_LOCK_KEY))) {
525 throw new Exceptions\ConfigLockedException('Configuration locked, not retrieving value for ' . $key);
526 }
527 $value = $this->getAppValue($key, $default);
528
529 if (in_array($key, self::NEVER_ENCRYPT) && $this->appCryptor->isEncrypted($value) === false) {
530 // short-cut, no need to bail out or complain. The "not enecrypted"
531 // check is necessary for the case that items are added to the
532 // NEVER_ENCRYPT list.
533 return $value;
534 }
535
536 if (!empty($value) && ($value !== $default)) {
537 // null is error or uninitialized, string '' means no encryption
538 if ($this->appCryptor->getEncryptionKey() === null) {
539 if ($this->appCryptor->isEncrypted($value) === false) {
540 return $value;
541 }
542 if (!empty($this->userId)) {
543 $message = $this->l->t('Decryption requested for user "%s", but not configured, empty encryption key.', $this->userId);
544 throw new Exceptions\EncryptionKeyException($message);
545 // $this->logError($message);
546 }
547 return false;
548 }
549 try {
550 $value = $this->appCryptor->decrypt($value);
551 } catch (Throwable $t) {
552 throw new Exceptions\DecryptionFailedException($this->l->t('Unable to decrypt value "%s" for "%s"', [$value, $key]), $t->getCode(), $t);
553 }
554 //$this->logInfo("Decrypted value for $key: ".$value);
555 }
556
557 return $value;
558 }
