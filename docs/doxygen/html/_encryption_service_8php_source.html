253 public function initUserKeyPair(bool $forceNewKeyPair = false):void
254 {
255 $exception = null;
256 try {
257 $this->asymKeyService->initEncryptionKeyPair($this->userId, $this->userPassword, $forceNewKeyPair);
258 } catch (Exceptions\EncryptionException $exception) {
259 // Gracefully accept a broken key-pair if the app-encryption key is empty.
260 try {
261 $userKey = $this->getUserEncryptionKey();
262 $exception = null;
263 if (empty($userKey)) {
264 // after all, this means that all values are unencrypted, so be graceful here
265 $this->asymKeyService->initEncryptionKeyPair($this->userId, $this->userPassword, forceNewKeyPair: true);
266 }
267 } catch (Throwable $t) {
268 // give up
269 $this->logException($t, 'User\'s "' . $this->userId . '" asymmetric key pair is broken and encryption key appears to be non-empty.');
270 }
271 }
272 $this->userAsymmetricCryptor = $this->asymKeyService->getCryptor($this->userId);
273 if (!empty($exception)) {
274 $this->userAsymmetricCryptor
275 ->setPrivateKey(null)
276 ->setPublicKey(null);
277 throw $exception;
278 }

378 public function initAppEncryptionKey():bool
379 {
380 if (!$this->bound()) {
381 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize global encryption key without bound user credentials.'));
382 }
383
384 $userDatabaseKey = $this->getUserEncryptionKey();
385 if (empty($userDatabaseKey)) {
386 // No key -> unencrypted
387 $this->logDebug("No Encryption Key, setting to empty string in order to disable encryption.");
388 $this->appCryptor->setEncryptionKey(''); // not null, just empty
389 } else {
390 $this->appCryptor->setEncryptionKey($userDatabaseKey);
391 }
392
393 // compare the user-key with the stored encryption key hash
394 $sysDatabaseKeyHash = $this->getConfigValue(self::APP_ENCRYPTION_KEY_HASH_KEY);
395 if (!$this->verifyHash($userDatabaseKey, $sysDatabaseKeyHash)) {
396 $failCount = 0;
397 $numberOfTests = 1000;
398 for ($i = 0; $i < $numberOfTests; ++$i) {
399 $failCount += 1 - (int)$this->verifyHash($userDatabaseKey, $sysDatabaseKeyHash);
400 }
401 if ($failCount < 10) {
402 $msg = 'BOGUS HARDWARE OR SOFTWARE, VERIFY HASH FAILED ' . $failCount . ' TIMES OUT OF ' . $numberOfTests . ' TIMES.';
403 $this->logError(
404 $msg,
405 [ 'exception' => new Exceptions\EncryptionException($msg) ],
406 );
407 return true;
408 }
409 // Failed
410 $this->appCryptor->setEncryptionKey(null);
411 throw new Exceptions\EncryptionKeyException($this->l->t('Failed to validate user encryption key. ' . $failCount . ' "' . $sysDatabaseKeyHash) . '"');
412 $this->logError('Unable to validate HASH for encryption key.');
413 } else {
414 $this->logDebug('Encryption keys validated'.(empty($userDatabaseKey) ? ' (no encryption)' : '').'.');
415 // $this->logInfo($this->l->t('Validated user encryption key. ' . $sysDatabaseKeyHash));
416 }
417
418 return true;
419 }

478 public function encryptionKeyValid(?string $encryptionKey = null):bool
479 {
480 if (!$this->bound()) {
481 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot validate app encryption key without bound user credentials'));
482 }
483
484 if (empty($encryptionKey)) {
485 $encryptionKey = $this->appCryptor->getEncryptionKey();
486 }
487 if (empty($encryptionKey)) {
488 $this->logWarn('Provided encryption-key is empty, encryption is switched off.');
489 }
490
491 $sysDatabaseKeyHash = $this->getConfigValue(self::APP_ENCRYPTION_KEY_HASH_KEY);
492
493 if (empty($sysDatabaseKeyHash) !== empty($encryptionKey)) {
494 if (empty($sysDatabaseKeyHash)) {
495 $this->logError('Stored encryption key is empty while provided encryption key is not empty.');
496 } else {
497 $this->logError('Stored encryption key is not empty while provided encryption key is empty.');
498 }
499 return false;
500 }
501
502 $match = $this->verifyHash($encryptionKey, $sysDatabaseKeyHash);
503
504 if (!$match) {
505 $this->logError('Unable to validate HASH for encryption key.');
506 } else {
507 $this->logDebug('Validated HASH for encryption key.');
508 }
509

594 public function getConfigValue(string $key, mixed $default = null, bool $ignoreLock = false)
595 {
596 if (!$ignoreLock && !empty($this->getAppValue(self::CONFIG_LOCK_KEY))) {
597 throw new Exceptions\ConfigLockedException('Configuration locked, not retrieving value for ' . $key);
598 }
599 $value = $this->getAppValue($key, $default);
600
601 if (in_array($key, self::NEVER_ENCRYPT) && $this->appCryptor->isEncrypted($value) === false) {
602 // short-cut, no need to bail out or complain. The "not enecrypted"
603 // check is necessary for the case that items are added to the
604 // NEVER_ENCRYPT list.
605 return $value;
606 }
607
608 if (!empty($value) && ($value !== $default)) {
609 // null is error or uninitialized, string '' means no encryption
610 if ($this->appCryptor->getEncryptionKey() === null) {
611 if ($this->appCryptor->isEncrypted($value) === false) {
612 return $value;
613 }
614 if (!empty($this->userId)) {
615 $message = $this->l->t('Decryption requested for user "%s", but not configured, empty encryption key.', $this->userId);
616 throw new Exceptions\EncryptionKeyException($message);
617 // $this->logError($message);
618 }
619 return false;
620 }
621 try {
622 $value = $this->appCryptor->decrypt($value);
623 } catch (Throwable $t) {
624 throw new Exceptions\DecryptionFailedException($this->l->t('Unable to decrypt value "%s" for "%s"', [$value, $key]), $t->getCode(), $t);
625 }
626 //$this->logInfo("Decrypted value for $key: ".$value);
627 }
628
629 return $value;
630 }
