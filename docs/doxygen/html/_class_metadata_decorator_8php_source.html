244 private function updateInverseSide(
245 object $entity,
246 array $association,
247 object $targetEntity,
248 ClassMetadataInterface $targetMeta,
249 ):void {
250 if (!empty($association['mappedBy'])) {
251 $this->debug('WE ARE THE INVERSE SIDE ' . print_r($association, true));
252 }
253 // try to maintain connectivity if it does not cause additional direct database access
254 if (!empty($association['inversedBy'])
255 && (!($targetEntity instanceof Proxy) || $targetEntity->__isInitialized())) {
256 $inversedBy = $association['inversedBy'];
257 $associationType = $association['type'];
258 switch ($associationType) {
259 case ClassMetadata::ONE_TO_ONE:
260 $targetMeta->setFieldValue($targetEntity, $inversedBy, $entity);
261 break;
262 case ClassMetadata::MANY_TO_ONE:
263 $inversedByValue = $targetMeta->getFieldValue($targetEntity, $inversedBy);
264 if (!($inversedByValue instanceof Collection)
265 || ($inversedByValue instanceof PersistentCollection) && !$inversedByValue->isInitialized()) {
266 // skip if the collection would have to be fetches or if
267 // it is not there already (the latter should not happen).
268 break;
269 }
270 $targetAssociation = $targetMeta->associationMappings[$inversedBy];
271 $indexBy = $targetAssociation['indexBy'] ?? null;
272 // $orderBy = $targetAssociation['orderBy']; complicated
273 if (!empty($indexBy)) {
274 $indexByValue = (string)$this->metaData->getFieldValue($entity, $indexBy);
275 $inversedByValue->set($indexByValue, $entity);
276 } elseif (!$inversedByValue->contains($entity)) {
277 $inversedByValue->add($entity);
278 }
279 break;
280 }
281 }

301 public function getIdentifierColumnValues(mixed $entity)
302 {
303 $columnValues = [];
304 foreach ($this->metaData->getIdentifierValues($entity) as $field => $value) {
305 if (isset($this->metaData->associationMappings[$field])) {
306 $association = $this->metaData->associationMappings[$field];
307 $targetEntity = $association['targetEntity'];
309 $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
310 if (count($association['joinColumns']) != 1) {
311 throw new Exceptions\DatabaseException($this->l->t('Foreign keys as principle keys cannot be composite'));
312 }
313 $joinInfo = $association['joinColumns'][0];
314 $columnName = $joinInfo->name;
315 $targetColumn = $joinInfo->referencedColumnName;
316 $targetField = $targetMeta->fieldNames[$targetColumn];
317 if ($value instanceof $targetEntity) {
318 $columnValues[$columnName] = $targetMeta->getFieldValue($value, $targetField);
319 } else {
320 // avoid references with empty identifiers
321 if (empty($value)) {
322 $value = null;
323 $reference = null;
324 } else {
325 // replace the value by a reference
326 $reference = $this->entityManager->getReference($targetEntity, [ $targetField => $value ]);
327
328 // try to maintain connectivity if it does not cause additional direct database access
329 $this->updateInverseSide($entity, $association, $reference, $targetMeta);
330 }
331 $this->doSetFieldValue($this->metaData, $entity, $field, $reference);
332 // assume this is the column value, not the entity of the foreign key
333 $columnValues[$columnName] = $value;
334 }
335 } else {
336 $columnName = $this->metaData->fieldMappings[$field]['columnName'];
337 $columnValues[$columnName] = $value;
338 }
339 }
340 return $columnValues;

368 public function extractKeyValues(array $columnValues, bool $ignoreMissing = false):array
369 {
370 $entityId = [];
371 foreach ($this->metaData->identifier as $field) {
372 $dbalType = null;
373 if (isset($this->metaData->associationMappings[$field])) {
374 if (count($this->metaData->associationMappings[$field]['joinColumns']) != 1) {
375 throw new Exceptions\DatabaseException($this->l->t('Foreign keys as principle keys cannot be composite'));
376 }
377 $columnName = $this->metaData->associationMappings[$field]['joinColumns'][0]->name;
378 } else {
379 $columnName = $this->metaData->fieldMappings[$field]->columnName;
380 if (!isset($columnValues[$columnName])) {
381 // possibly an attempt to extract from non-existing field.
382 if ($ignoreMissing || $this->metaData->usesIdGenerator()) {
383 continue;
384 }
385 throw new Exceptions\DatabaseException(
386 $this->l->t('Missing value and no generator for identifier field: %s::%s', [ $this->getName(), $field ]));
387 }
388 $dbalType = Type::getType($this->metaData->fieldMappings[$field]->type);
389 }
390 $value = $columnValues[$columnName];
391 if (!empty($dbalType) && is_string($value)) {
392 $value = $dbalType->convertToPHPValue($value, $this->entityManager->getPlatform());
393 }
394 $entityId[$field] = $value;
395 }
396 return $entityId;
397 }

432 private function doSetFieldValue(mixed $meta, mixed $entity, string $field, mixed $value):void
433 {
434 // try first the setter/getter of the entity
435 $method = 'set'.ucfirst($field);
436
437 // try to convert string to correct type if possible
438 try {
439 $dbalTypeName = $meta->getTypeOfField($field);
440 if (!empty($dbalTypeName)) {
441 $dbalType = Type::getType($dbalTypeName);
442 if (!empty($dbalType) && is_string($value)) {
443 $value = $dbalType->convertToPHPValue($value, $this->entityManager->getPlatform());
444 }
445 }
446 } catch (ConversionException $e) {
447 if (empty($value) && is_callable([ $entity, $method ])) {
448 $this->logException($e, 'Type conversion for field ' . $field . ' failed, continuing anyway.');
449 } else {
450 throw new Exceptions\DatabaseException($this->l->t('Type conversion for field "%s" failed.', $field), 0, $e);
451 }
452 }
453 if (is_callable([ $entity, $method ])) {
454 $entity->$method($value);
455 } else {
456 $this->logWarn('Probably missing method "'.$method.'" on an entity '.get_class($entity));
457 $meta->setFieldValue($entity, $field, $value);
458 }

470 public function setColumnValue(mixed $entity, string $column, mixed $value):void
471 {
472 $this->debug('Set column value for ' . $column . ' to ' . $value);
473
474 $meta = $this->metaData;
475 $numSetters = 0;
476 if (isset($meta->fieldNames[$column])) {
477 // simple case, just set the entity property
478 $this->doSetFieldValue($meta, $entity, $meta->fieldNames[$column], $value);
479 ++$numSetters;
480 }
481 $columnAssociations = $this->columnAssociations();
482 foreach (($columnAssociations[$column]??[]) as $field) {
483 // If the column is a join-column then try to create references to the
484 // respective entity. Multi-column associations are supported if the
485 // column is the only one which is not known.
486 if (empty($value)) {
487 // an empty join-column value must void the association as the
488 // join-columns are the identifier of the referenced entity and hence
489 // cannot be null unless the referenced entity is null
490 $this->debug('DO CLEAR FIELD VALUE FOR ' . $field);
491 $this->doSetFieldValue($meta, $entity, $field, null);
492 continue;
493 }
494 $associationMapping = $meta->associationMappings[$field];
495 $targetEntity = $associationMapping['targetEntity'];
496 $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
497 $referencedColumnId = [];
498 foreach ($associationMapping['joinColumns'] as $joinInfo) {
499 $joinColumnName = $joinInfo->name;
500 $targetColumn = $joinInfo->referencedColumnName;
501 if ($joinColumnName == $column) {
502 $fieldValue = $meta->getFieldValue($entity, $field);
503 if (!empty($fieldValue) &&
504 $value == $targetMeta->getColumnValue($fieldValue, $targetColumn)) {
505 // do not destroy already installed associated entities
506 continue 2;
507 }
508 $targetValue = $value;
509 } else {
510 $targetValue = $this->getColumnValue($entity, $joinColumnName);
511 }
512 if (empty($targetValue)) {
513 continue 2;
514 }
515 $referencedColumnId[$targetColumn] = $targetValue;
516 }
517 $referencedId = $targetMeta->extractKeyValues($referencedColumnId);
518 $fieldValue = $this->entityManager->getReference($targetEntity, $referencedId);
519 $this->doSetFieldValue($meta, $entity, $field, $fieldValue);
520 $this->updateInverseSide($entity, $associationMapping, $fieldValue, $targetMeta);
521 ++$numSetters;
522 }
523 if ($numSetters == 0) {
524 $this->debug('Remember value for column ' . $column . ': ' . $value);
525 $this->temporaryColumnStorage[$column] = $value;
526 } else {
527 $this->debug('Clear value for column ' . $column);
528 unset($this->temporaryColumnStorage[$column]);
529 }
530 }

541 public function getColumnValue(mixed $entity, string $column)
542 {
543 $meta = $this->metaData;
544 // simple case ...
545 if (isset($meta->fieldNames[$column])) {
546 return $meta->getFieldValue($entity, $meta->fieldNames[$column]);
547 }
548 // look into the association mappings
549 $columnAssociations = $this->columnAssociations();
550 foreach (($columnAssociations[$column]??[]) as $targetColumn => $field) {
551 $fieldValue = $meta->getFieldValue($entity, $field);
552 if (empty($fieldValue)) {
553 continue;
554 }
555 $associationMapping = $meta->associationMappings[$field];
556 $targetEntity = $associationMapping['targetEntity'];
557 if ($fieldValue instanceof $targetEntity) {
558 $targetMeta = $this->entityManager->getClassMetadata($targetEntity);
559 return $targetMeta->getColumnValue($fieldValue, $targetColumn);
560 } else {
561 // assume the field-value is the just the column value of the single join-column
562 if (count($associationMapping['joinColumns']) != 1) {
563 throw new Exceptions\DatabaseException(
564 $this->l->t(
565 'Association field must eiter be an entity or the value of the single join column: %s',
566 print_r($associationMapping['joinColumns'], true),
567 ),
568 );
569 }
570 return $fieldValue;
571 }
572 }
573 if (!empty($this->temporaryColumnStorage[$column])) {
574 $this->debug('USE TEMPORARY COLUMN STORAGE ' . $column . ' => ' . $this->temporaryColumnStorage[$column]);
575 return $this->temporaryColumnStorage[$column];
576 }
577 return null;
578 }
