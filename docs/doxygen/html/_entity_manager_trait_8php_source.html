158 protected function remove(
159 mixed $entity,
160 bool $flush = false,
161 bool $hard = false,
162 bool $soft = false,
163 bool $useTransaction = false,
164 ):void {
165 if (filter_var($entity, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]])) {
166 $entity = [ 'id' => $entity ];
167 }
168 if (is_array($entity)) {
169 $key = $entity;
170 // getReference() will not always work, and in particular not with
171 // foreign keys, so just fetch the entity first.
172 //
173 // $entity = $this->entityManager->getReference($this->entityClassName, $key);
174 $entity = $this->entityManager->find($this->entityClassName, $key);
175 if (empty($entity)) {
176 throw new Exceptions\DatabaseEntityNotFoundException($this->entityClassName);
177 }
178 }
179 if ($soft && !$hard && method_exists($entity, 'isDeleted') && $entity->isDeleted()) {
180 return;
181 }
182 try {
183 if ($useTransaction && $flush) {
184 $this->entityManager->beginTransaction();
185 }
186 if ($hard && method_exists($entity, 'setDeleted')) {
187 $entity->setDeleted('now');
188 }
189 $this->entityManager->remove($entity);
190 if ($hard && !method_exists($entity, 'setDeleted')) {
191 $this->flush();
192 $this->entityManager->remove($entity);
193 }
194 if ($flush) {
195 $this->flush();
196 if ($useTransaction) {
197 $this->entityManager->commit();
198 }
199 }
200 } catch (Throwable $t) {
201 if ($useTransaction) {
202 $this->entityManager->rollback();
203 }
204 throw $t;
205 }
206 }

265 {

277 $this->entityManager->rollback();
278 } else {
279 \OCP\Service::get(\Psr\Log\LoggerInterface::class)->error('Exception during flush()', [ 'exception' => $t ]);
