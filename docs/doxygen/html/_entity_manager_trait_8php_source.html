157 protected function remove(
158 mixed $entity,
159 bool $flush = false,
160 bool $hard = false,
161 bool $soft = false,
162 bool $useTransaction = false,
163 ):void {
164 if (filter_var($entity, FILTER_VALIDATE_INT, ['min_range' => 1])) {
165 $entity = [ 'id' => $entity ];
166 }
167 if (is_array($entity)) {
168 $key = $entity;
169 // getReference() will not always work, and in particular not with
170 // foreign keys, so just fetch the entity first.
171 //
172 // $entity = $this->entityManager->getReference($this->entityClassName, $key);
173 $entity = $this->entityManager->find($this->entityClassName, $key);
174 if (empty($entity)) {
175 throw new Exceptions\DatabaseEntityNotFoundException($this->entityClassName);
176 }
177 }
178 if ($soft && !$hard && method_exists($entity, 'isDeleted') && $entity->isDeleted()) {
179 return;
180 }
181 try {
182 if ($useTransaction && $flush) {
183 $this->entityManager->beginTransaction();
184 }
185 $this->entityManager->remove($entity);
186 if ($hard && (!method_exists($entity, 'isDeleted') || !$entity->isDeleted())) {
187 $this->flush();
188 $this->entityManager->remove($entity);
189 }
190
191 if ($flush) {
192 $this->flush();
193 if ($useTransaction) {
194 $this->entityManager->commit();
195 }
196 }
197 } catch (Throwable $t) {
198 if ($useTransaction) {
199 $this->entityManager->rollback();
200 }
201 throw $t;
202 }
203 }
