875 /**
876 * Guess the gender from the name.
877 *
878 * @param null|\OCP\IL10N $l
879 *
880 * @return array An array of guesses.
881 */
882 public function guessGender(?\OCP\IL10N $l = null):array
883 {
884 $detector = new GenderDetector\GenderDetector();
885 $country = $this->country;
886 if (!empty($this->language)) {
887 $country = substr($this->language, 0, 2);
888 }
889 $country = strtoupper($country);
890 $names = array_filter(array_merge(explode(' ', $this->firstName ?? ''), explode(' ', $this->nickName ?? '')));
891 $genderTypes = array_filter(array_map(fn(string $name) => $detector->getGender($name, $country), $names));
892 $genderTypes = array_unique(array_map(fn(GenderDetector\Gender $gender) => str_replace('mostly', '', strtolower($gender->name)), $genderTypes));
893 if ($l !== null) {
894 return array_combine($genderTypes, array_map(fn(string $name) => $this->l->t($name), $genderTypes));
895 } else {
896 return $genderTypes;
897 }
898 }
