189 protected static function response(mixed $message, int $status = Http::STATUS_OK):DataResponse
190 {
191 $responseData = [
192 'messages' => [],
193 'message' => null,
194 ];
195 if (is_string($message)) {
196 $responseData = [
197 'messages' => [ $message ],
198 'message' => $message,
199 ];
200 } elseif (is_array($message)) {
201 $responseData = [
202 'messages' => $message,
203 'message' => implode('; ', $message),
204 ];
205 }
206 return self::dataResponse($responseData, $status);
