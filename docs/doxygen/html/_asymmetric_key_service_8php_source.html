122 public function initEncryptionKeyPair(?string $ownerId = null, ?string $keyPassphrase = null, bool $forceNewKeyPair = false)
123 {
124 if (empty($ownerId)) {
125 $ownerId = $this->getSessionUserId();
126 }
127
128 if (empty($ownerId)) {
129 throw new Exceptions\EncryptionException($this->l->t('The owner-id was not specified and could not be determined automatically.'));
130 }
131
132 if (empty($keyPassphrase)) {
133 $keyPassphrase = $this->getLoginPassword($ownerId);
134 }
135
136 if (empty($ownerId) || empty($keyPassphrase)) {
137 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize SSL key-pair without user and password'));
138 }
139
140 if (!$forceNewKeyPair && !empty(self::$keyPairs[$ownerId])) {
141 return self::$keyPairs[$ownerId];
142 }
143
144 try {
145 $keyPair = $forceNewKeyPair ? null : $this->keyStorage->getKeyPair($ownerId, $keyPassphrase);
146 } catch (Throwable $t) {
147 $this->logException($t, $this->l->t('Unable to retrieve old key-pair for "%s".', [ $ownerId ]));
148 $keyPair = [];
149 // throw new Exceptions\EnduserNotificationException('DEBUG: IGNORE FAILED KEYPAIR REQUESTS');
150 }
151 if (empty($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG]) || empty($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG])) {
152
153 $oldKeyPair = self::$keyPairs[$ownerId] ?? null;
154 if (empty($oldKeyPair) && $ownerId == $this->getSessionUserId()) {
155 $loginPassword = $this->getLoginPassword($ownerId);
156 if (!empty($loginPassword)) {
157 try {
158 $oldKeyPair = $this->keyStorage->getKeyPair($ownerId, $loginPassword);
159 } catch (Throwable $t) {
160 $this->logException($t, 'Unable to fetch old encryption key pair for "' . $ownerId . '".');
161 }
162 }
163 }
164
165 // make sure the old key-pair cache is set
166 self::$keyPairs[$ownerId] = $oldKeyPair;
167
168 $this->eventDispatcher->dispatchTyped(new Events\BeforeEncryptionKeyPairChanged($ownerId, $oldKeyPair));
169
170 $this->keyStorage->backupKeyPair($ownerId); // make a backup-copy if old key exists
171 $keyPair = $this->keyStorage->generateKeyPair($ownerId, $keyPassphrase);
172
173 $this->eventDispatcher->dispatchTyped(new Events\AfterEncryptionKeyPairChanged($ownerId, $oldKeyPair, $keyPair));
174 }
175
176 self::$keyPairs[$ownerId] = $keyPair;
177
178 // ensure that the cached cryptor has the correct key
179 /* $cryptor = */$this->getCryptor($ownerId)
180 ->setPrivateKey($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null)
181 ->setPublicKey($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? null);
182
183 return $keyPair;
184 }

261 public function getCryptor(string $ownerId):AsymmetricCryptorInterface
262 {
264 $cryptor = self::$cryptors[$ownerId] ?? null;
265 if (empty($cryptor)) {
266 $keyPair = self::$keyPairs[$ownerId] ?? null;
267 if (empty($keyPair) && $ownerId == $this->getSessionUserId()) {
268 // effect of LazyGhosts: EncryptionService CTOR has not been run
269 $this->initEncryptionKeyPair($ownerId);
270 $keyPair = self::$keyPairs[$ownerId] ?? null;
271 }
272 $privKey = $keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null;
273 $pubKey = $keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? $this->keyStorage->getPublicKey($ownerId);
274 $cryptor = clone $this->cryptorPrototype;
275 $cryptor->setPrivateKey($privKey);
276 $cryptor->setPublicKey($pubKey);
277 self::$cryptors[$ownerId] = $cryptor;
278 }
279 return $cryptor;

295 public function setSharedPrivateValue(string $ownerId, string $key, mixed $value):void
296 {
297 $value = (string)$value;
298 $configKey = self::CONFIG_KEY_PREFIX . $key;
299 if (empty($value)) {
300 $this->deleteUserValue($configKey, userId: $ownerId);
301 return;
302 }
303 $cryptor = $this->getCryptor($ownerId);
304 if (!$cryptor->canEncrypt()) {
305 throw new Exceptions\CannotEncryptException($this->l->t('Cannot encrypt personal value "%1$s" for "%2$s".', [ $key, $ownerId ]));
306 }
307 $this->setUserValue($configKey, $cryptor->encrypt($value), userId: $ownerId);
308 }

323 public function getSharedPrivateValue(string $ownerId, string $key, mixed $default = null):?string
324 {
325 $configKey = self::CONFIG_KEY_PREFIX . $key;
326 $value = $this->getUserValue($configKey, $default, userId: $ownerId);
327 if (empty($value) || $value === $default) { // allow empty and default values
328 return $value === null ? null : (string)$value;
329 }
330 $cryptor = $this->getCryptor($ownerId);
331 if (!$cryptor->canDecrypt()) {
332 throw new Exceptions\CannotDecryptException($this->l->t('Cannot decrypt personal value "%1$s" for "%2$s".', [ $key, $ownerId ]));
333 }
334 return $cryptor->decrypt($value);
335 }

459 {
460 $requestData = $this->appContainer->get(ITimeFactory::class)->getTime();
461 $this->setUserValue(self::RECRYPTION_REQUEST_KEY, $requestData, userId: $ownerId);
462
464 $notificationManager = $this->appContainer->get(NotificationManager::class);
465 $notification = $notificationManager->createNotification();
466
467 $notification->setApp($this->appName)
468 ->setDateTime(new DateTime)
469 ->setObject('owner_id', $ownerId)
470 ->setSubject(Notifier::RECRYPT_USER_SUBJECT, [ 'timestamp' => $requestData ])
471 ->addAction($notification->createAction()
472 ->setLabel(Notifier::ACCEPT_ACTION)
473 ->setLink('https://user_recrypt_request', 'POST'))
474 ->addAction($notification->createAction()
475 ->setLabel(Notifier::DECLINE_ACTION)
476 ->setLink('https://user_recrypt_request', 'DELETE'));
477
479 $organizationalRoles = $this->appContainer->get(OrganizationalRolesService::class);
481 foreach ($organizationalRoles->getGroupAdmins() as $groupAdmin) {
482 $groupAdminUid = $groupAdmin->getUID();
483 if ($groupAdminUid == $ownerId) {
484 $this->logInfo('Omitting key-owner from recryption notification.');
485 continue;
486 }
487 $this->logInfo('Notifying ' . $groupAdminUid);
488 $notification->setUser($groupAdminUid);
489 $notificationManager->notify($notification);
490 }
