116 public function initEncryptionKeyPair(?string $ownerId = null, ?string $keyPassphrase = null, bool $forceNewKeyPair = false)
117 {
118 if (empty($ownerId)) {
119 $ownerId = $this->getSessionUserId();
120 }
121
122 if (empty($ownerId)) {
123 throw new Exceptions\EncryptionException($this->l->t('The owner-id was not specified and could not be determined automatically.'));
124 }
125
126 if (empty($keyPassphrase)) {
127 $keyPassphrase = $this->getLoginPassword($ownerId);
128 }
129
130 if (empty($ownerId) || empty($keyPassphrase)) {
131 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize SSL key-pair without user and password'));
132 }
133
134 if (!$forceNewKeyPair && !empty(self::$keyPairs[$ownerId])) {
135 return self::$keyPairs[$ownerId];
136 }
137
138 try {
139 $keyPair = $forceNewKeyPair ? null : $this->keyStorage->getKeyPair($ownerId, $keyPassphrase);
140 } catch (Throwable $t) {
141 $this->logException($t, $this->l->t('Unable to retrieve old key-pair for "%s".', [ $ownerId ]));
142 $keyPair = [];
143 throw new Exceptions\EnduserNotificationException('DEBUG: IGNORE FAILED KEYPAIR REQUESTS');
144 }
145 if (empty($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG]) || empty($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG])) {
146
147 $oldKeyPair = self::$keyPairs[$ownerId] ?? null;
148 if (empty($oldKeyPair) && $ownerId == $this->getSessionUserId()) {
149 $loginPassword = $this->getLoginPassword($ownerId);
150 if (!empty($loginPassword)) {
151 try {
152 $oldKeyPair = $this->keyStorage->getKeyPair($ownerId, $loginPassword);
153 } catch (Throwable $t) {
154 $this->logException($t, 'Unable to fetch old encryption key pair for "' . $ownerId . '".');
155 }
156 }
157 }
158
159 // make sure the old key-pair cache is set
160 self::$keyPairs[$ownerId] = $oldKeyPair;
161
162 $this->eventDispatcher->dispatchTyped(new Events\BeforeEncryptionKeyPairChanged($ownerId, $oldKeyPair));
163
164 $this->keyStorage->backupKeyPair($ownerId); // make a backup-copy if old key exists
165 $keyPair = $this->keyStorage->generateKeyPair($ownerId, $keyPassphrase);
166
167 $this->eventDispatcher->dispatchTyped(new Events\AfterEncryptionKeyPairChanged($ownerId, $oldKeyPair, $keyPair));
168 }
169
170 self::$keyPairs[$ownerId] = $keyPair;
171
172 // ensure that the cached cryptor has the correct key
173 /* $cryptor = */$this->getCryptor($ownerId)
174 ->setPrivateKey($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null)
175 ->setPublicKey($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? null);
176
177 return $keyPair;
178 }

255 public function getCryptor(string $ownerId):AsymmetricCryptorInterface
256 {
258 $cryptor = self::$cryptors[$ownerId] ?? null;
259 if (empty($cryptor)) {
260 $keyPair = self::$keyPairs[$ownerId] ?? null;
261 if (empty($keyPair) && $ownerId == $this->getSessionUserId()) {
262 // effect of LazyGhosts: EncryptionService CTOR has not been run
263 $this->initEncryptionKeyPair($ownerId);
264 $keyPair = self::$keyPairs[$ownerId] ?? null;
265 }
266 $privKey = $keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null;
267 $pubKey = $keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? $this->keyStorage->getPublicKey($ownerId);
268 $cryptor = clone $this->cryptorPrototype;
269 $cryptor->setPrivateKey($privKey);
270 $cryptor->setPublicKey($pubKey);
271 self::$cryptors[$ownerId] = $cryptor;
272 }
273 return $cryptor;

289 public function setSharedPrivateValue(string $ownerId, string $key, mixed $value):void
290 {
291 $value = (string)$value;
292 $configKey = self::CONFIG_KEY_PREFIX . $key;
293 if (empty($value)) {
294 $this->cloudConfig->deleteUserValue($ownerId, $this->appName, $configKey);
295 return;
296 }
297 $cryptor = $this->getCryptor($ownerId);
298 if (!$cryptor->canEncrypt()) {
299 throw new Exceptions\CannotEncryptException($this->l->t('Cannot encrypt personal value "%1$s" for "%2$s".', [ $key, $ownerId ]));
300 }
301 $this->cloudConfig->setUserValue($ownerId, $this->appName, $configKey, $cryptor->encrypt($value));
302 }
