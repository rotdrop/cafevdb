117 public function initEncryptionKeyPair(?string $ownerId = null, ?string $keyPassphrase = null, bool $forceNewKeyPair = false)
118 {
119 if (empty($ownerId)) {
120 $ownerId = $this->getSessionUserId();
121 }
122
123 if (empty($ownerId)) {
124 throw new Exceptions\EncryptionException($this->l->t('The owner-id was not specified and could not be determined automatically.'));
125 }
126
127 if (empty($keyPassphrase)) {
128 $keyPassphrase = $this->getLoginPassword($ownerId);
129 }
130
131 if (empty($ownerId) || empty($keyPassphrase)) {
132 throw new Exceptions\EncryptionKeyException($this->l->t('Cannot initialize SSL key-pair without user and password'));
133 }
134
135 if (!$forceNewKeyPair && !empty(self::$keyPairs[$ownerId])) {
136 return self::$keyPairs[$ownerId];
137 }
138
139 try {
140 $keyPair = $forceNewKeyPair ? null : $this->keyStorage->getKeyPair($ownerId, $keyPassphrase);
141 } catch (Throwable $t) {
142 $this->logException($t, $this->l->t('Unable to retrieve old key-pair for "%s".', [ $ownerId ]));
143 $keyPair = [];
144 throw new Exceptions\EnduserNotificationException('DEBUG: IGNORE FAILED KEYPAIR REQUESTS');
145 }
146 if (empty($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG]) || empty($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG])) {
147
148 $oldKeyPair = self::$keyPairs[$ownerId] ?? null;
149 if (empty($oldKeyPair) && $ownerId == $this->getSessionUserId()) {
150 $loginPassword = $this->getLoginPassword($ownerId);
151 if (!empty($loginPassword)) {
152 try {
153 $oldKeyPair = $this->keyStorage->getKeyPair($ownerId, $loginPassword);
154 } catch (Throwable $t) {
155 $this->logException($t, 'Unable to fetch old encryption key pair for "' . $ownerId . '".');
156 }
157 }
158 }
159
160 // make sure the old key-pair cache is set
161 self::$keyPairs[$ownerId] = $oldKeyPair;
162
163 $this->eventDispatcher->dispatchTyped(new Events\BeforeEncryptionKeyPairChanged($ownerId, $oldKeyPair));
164
165 $this->keyStorage->backupKeyPair($ownerId); // make a backup-copy if old key exists
166 $keyPair = $this->keyStorage->generateKeyPair($ownerId, $keyPassphrase);
167
168 $this->eventDispatcher->dispatchTyped(new Events\AfterEncryptionKeyPairChanged($ownerId, $oldKeyPair, $keyPair));
169 }
170
171 self::$keyPairs[$ownerId] = $keyPair;
172
173 // ensure that the cached cryptor has the correct key
174 /* $cryptor = */$this->getCryptor($ownerId)
175 ->setPrivateKey($keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null)
176 ->setPublicKey($keyPair[self::PUBLIC_ENCRYPTION_KEY_CONFIG] ?? null);
177
178 return $keyPair;
179 }

257 {

260 if (empty($cryptor)) {
261 $keyPair = self::$keyPairs[$ownerId] ?? null;
262 $privKey = $keyPair[self::PRIVATE_ENCRYPTION_KEY_CONFIG] ?? null;

264 $cryptor = clone $this->cryptorPrototype;
265 $cryptor->setPrivateKey($privKey);

267 self::$cryptors[$ownerId] = $cryptor;

285 public function setSharedPrivateValue(string $ownerId, string $key, mixed $value):void
286 {
287 $value = (string)$value;
288 $configKey = self::CONFIG_KEY_PREFIX . $key;
289 if (empty($value)) {
290 $this->cloudConfig->deleteUserValue($ownerId, $this->appName, $configKey);
291 return;
292 }
293 $cryptor = $this->getCryptor($ownerId);
294 if (!$cryptor->canEncrypt()) {
295 throw new Exceptions\CannotEncryptException($this->l->t('Cannot encrypt personal value "%1$s" for "%2$s".', [ $key, $ownerId ]));
296 }
297 $this->cloudConfig->setUserValue($ownerId, $this->appName, $configKey, $cryptor->encrypt($value));
298 }

313 public function getSharedPrivateValue(string $ownerId, string $key, mixed $default = null):?string
314 {

317 if (empty($value) || $value === $default) { // allow empty and default values

472 $groupAdminUid = $groupAdmin->getUID();

474 $this->logInfo('Omitting key-owner from recryption notification.');
475 continue;

477 $this->logInfo('Notifying ' . $groupAdminUid);
