150 public function generateAccountsAutocompleteData(): ?array
151 {
152 $accountsExport = $this->encryptionService->getAppValue(AdminSettings::GNU_CASH_ACCOUNTS_TREE_DATA_KEY);
153 if (empty($accountsExport)) {
154 $this->logError('ACCOUNTS EXPORT FILE IS NOT SET');
155 return null;
156 }
157 $accountsExportFile = $this->userStorage->getFile($accountsExport);
158 if (empty($accountsExportFile)) {
159 $this->logError('UNABLE TO OPEN ACCOUNTS EXPORTS FILE ' . $accountsExport);
160 return null;
161 }
162
163 $accountsAutocompleteFile = $this->appStorage->getFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE, throw: false);
164 if ($accountsAutocompleteFile !== null && $accountsExportFile->getMTime() <= $accountsAutocompleteFile->getMTime()) {
165 return json_decode($accountsAutocompleteFile->getContent(), true);
166 }
167
168 $permanentProjects = $this->entityManager->getRepository(Entities\Project::class)->findNames(onlyType: ProjectType::PERMANENT);
169
170 $leafAccountRe = '/:([^0-9]+[0-9]{4}|'
171 . implode('|', array_map(fn(string $name) => preg_quote($name), $permanentProjects))
172 . ')$/';
173
174 $autocompleteData = [
175 self::GNU_CASH_EXPENSE_KEY => [],
176 self::GNU_CASH_INCOME_KEY => [],
177 ];
178
179 $exportData = explode("\n", $accountsExportFile->getContent());
180
181 foreach ($exportData as $dataLine) {
182 $lineData = str_getcsv($dataLine, ';', escape: '');
183 if (empty($lineData) || $lineData[0] === null) {
184 break;
185 }
186 $type = strtolower($lineData[0]);
187 if ($type != self::GNU_CASH_EXPENSE_KEY && $type != self::GNU_CASH_INCOME_KEY) {
188 continue;
189 }
190 $account = preg_replace($leafAccountRe, '', $lineData[1]);
191 $autocompleteData[$type][] = $account;
192 }
193 foreach ($autocompleteData as &$accounts) {
194 $accounts = array_unique($accounts);
195 sort($accounts);
196 $accounts = array_values($accounts);
197
198 $count = count($accounts);
199 for ($i = 0; $i < $count - 1; ++$i) {
200 if (str_starts_with($accounts[$i + 1], $accounts[$i] . ':')) {
201 unset($accounts[$i]);
202 }
203 }
204 $accounts = array_values($accounts);
205 }
206
207 if (!$accountsAutocompleteFile) {
208 $accountsAutocompleteFile = $this->appStorage->ensureFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE);
209 }
210 $accountsAutocompleteFile->putContent(json_encode($autocompleteData));
211
212 return $autocompleteData;
213 }

290 public function exportCompositePaymentBalancingEntries(Entities\CompositePayment $compositePayment): array
291 {
292 $currencyCode = $this->getAppCurrencyCode();
293
294 $receivableAccounts = [];
295 $musician = $compositePayment->getProjectParticipant()->getMusician();
296 $description = $compositePayment->getSubject() . '; ' . $musician->getPublicName(false);
297
298 // it need not be the case that a composite payment result in the same
299 // balancing account for each splits, though in general this should be the
300 // case.
302 foreach ($compositePayment->getProjectPayments() as $projectPayment) {
303 $receivableAccount = $this->generateParticipantReceivablesAccount($projectPayment->getReceivable());
304 if ($receivableAccount === null) {
305 continue;
306 }
307 if (!isset($receivableAccounts[$receivableAccount])) {
308 $receivableAccounts[$receivableAccount] = [
309 'payments' => [ $projectPayment ],
310 ];
311 } else {
312 $receivableAccounts[$receivableAccount]['payments'][] = $projectPayment;
313 }
314 }
315 $data = [];
316 foreach ($receivableAccounts as $receivableAccount => $accountData) {
317 $dueDate = max(
318 array_map(
319 fn(Entities\ProjectPayment $payment) => $this->financeService->getDueDate($payment->getReceivable()),
320 $accountData['payments'],
321 ),
322 );
324 $subTotals = array_reduce(
325 $accountData['payments'],
326 fn(MonetaryNumberType $carry, Entities\ProjectPayment $payment) => $carry->add($payment->getAmount()),
327 MonetaryNumberType::zero(),
328 );
329 $transactionId = md5($receivableAccount);
330 $data[] = [
331 'transactionId' => $transactionId,
332 'date' => $dueDate->format('d-m-Y'),
333 'amount' => $subTotals->toDecimal(2), // + or minus?
334 'negativeAmount' => $subTotals->neg()->toDecimal(2), // + or minus?
335 'account' => $receivableAccount,
336 'description' => $description,
337 'currency' => $currencyCode,
338 'notes' => '',
339 'memo' => '',
340 ];
342 foreach ($accountData['payments'] as $projectPayment) {
343 $receivable = $projectPayment->getReceivable();
344 $balancingAccount = $receivable->getBalancingAccount();
345 if (empty($balancingAccount)) {
346 $field = $receivable->getField();
347 if ($field->getMultiplicity() == FieldMultiplicity::RECURRING) {
348 $generatorOption = $field->getManagementOption();
349 $class = $generatorOption->getData();
350 try {
351 $generator = $this->appContainer->get($class);
352 if (method_exists($generator, 'generateLegacyBalancingAccount')) {
353 $balancingAccount = $generator->generateLegacyBalancingAccount($receivable);
354 }
355 } catch (Throwable $t) {
356 // echo $t->getMessage() . PHP_EOL;
357 // ignore
358 $balancingAccount = '';
359 }
360 }
361 }
363 $data[] = [
364 'transactionId' => $transactionId,
365 'date' => '',
366 'amount' => $projectPayment->getAmount()->neg()->toDecimal(2),
367 'negativeAmount' => $projectPayment->getAmount()->toDecimal(2),
368 'account' => $balancingAccount,
369 'subject' => '',
370 'description' => '',
371 'notes' => '',
372 'memo' => $projectPayment->getSubject(),
373 ];
374 }
375 }
376
377 return $data;
378 }
