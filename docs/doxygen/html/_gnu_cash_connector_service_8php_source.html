285 public function generateAccountsAutocompleteData(): ?array
286 {
287 $accountsExport = $this->encryptionService->getAppValue(AdminSettings::GNU_CASH_ACCOUNTS_TREE_DATA_KEY);
288 if (empty($accountsExport)) {
289 $this->logError('ACCOUNTS EXPORT FILE IS NOT SET');
290 return null;
291 }
293 $userStorage = $this->appContainer->get(UserStorage::class);
294 $accountsExportFile = $userStorage->getFile($accountsExport);
295 if (empty($accountsExportFile)) {
296 $this->logError('UNABLE TO OPEN ACCOUNTS EXPORTS FILE ' . $accountsExport);
297 return null;
298 }
299
301 $appStorage = $this->appContainer->get(AppStorage::class);
302 $accountsAutocompleteFile = $appStorage->getFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE, throw: false);
303 if ($accountsAutocompleteFile !== null && $accountsExportFile->getMTime() <= $accountsExportFile->getMTime()) {
304 return json_decode($accountsAutocompleteFile->getContent(), true);
305 }
306
308 $entityManager = $this->appContainer->get(EntityManager::class);
309 $permanentProjects = $entityManager->getRepository(Entities\Project::class)->findNames(onlyType: ProjectType::PERMANENT);
310
311 $leafAccountRe = '/:([^0-9]+[0-9]{4}|'
312 . implode('|', array_map(fn(string $name) => preg_quote($name), $permanentProjects))
313 . ')$/';
314
315 $autocompleteData = [
316 self::GNU_CASH_EXPENSE_KEY => [],
317 self::GNU_CASH_INCOME_KEY => [],
318 ];
319
320 $exportData = explode("\n", $accountsExportFile->getContent());
321
322 foreach ($exportData as $dataLine) {
323 $lineData = str_getcsv($dataLine, ';');
324 if ($lineData === false) {
325 break;
326 }
327 $type = strtolower($lineData[0]);
328 if ($type != self::GNU_CASH_EXPENSE_KEY && $type != self::GNU_CASH_INCOME_KEY) {
329 continue;
330 }
331 $account = preg_replace($leafAccountRe, '', $lineData[1]);
332 $autocompleteData[$type][] = $account;
333 }
334 foreach ($autocompleteData as &$accounts) {
335 $accounts = array_unique($accounts);
336 sort($accounts);
337 $accounts = array_values($accounts);
338
339 $count = count($accounts);
340 for ($i = 0; $i < $count - 1; ++$i) {
341 if (str_starts_with($accounts[$i + 1], $accounts[$i] . ':')) {
342 unset($accounts[$i]);
343 }
344 }
345 $accounts = array_values($accounts);
346 }
347
348 if (!$accountsAutocompleteFile) {
349 $accountsAutocompleteFile = $appStorage->ensureFile(self::GNU_CASH_AUTOCOMPLETE_ACCOUNTS_APP_DATA_FILE);
350 }
351 $accountsAutocompleteFile->putContent(json_encode($autocompleteData));
352
353 return $autocompleteData;
354 }
