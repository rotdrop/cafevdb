333 public function testFetchFromDatabaseCollectionArchive(): void
334 {
335 $zipFileBase = 'äfolder';
336 $dbFiles = $this->generateDatabasFileNodes();
337 $result = $this->downloadsController->fetch(
338 section: DownloadsController::SECTION_DATABASE,
339 object: DownloadsController::OBJECT_COLLECTION,
340 items: array_map(fn($entity) => $entity->getId(), $dbFiles),
341 fileName: $zipFileBase,
342 );
343 $this->assertInstanceOf(DataDownloadResponse::class, $result);
345 $this->assertEquals(Http::STATUS_OK, $result->getStatus());
346 $headers = $result->getHeaders();
347 $this->assertArrayHasKey('Content-Type', $headers);
348 $this->assertEquals('application/zip', $headers['Content-Type']);
349 preg_match_all('/(?:filename="([^"]+)"|filename\\*=UTF-8\'\'([^ ]+)(?: |$))/', $headers['Content-Disposition'], $matches);
350 $transliteratedFileName = $matches[1][0] ?? null;
351 $encodedFileName = $matches[2][1] ?? null;
352 $this->assertNotNull($transliteratedFileName);
353 $this->assertNotNull($encodedFileName);
354 $zipFileName = $zipFileBase . '.zip';
355 $this->assertEquals('ae' . mb_substr($zipFileName, 1), $transliteratedFileName);
356 $this->assertEquals($zipFileName, urldecode($encodedFileName));
357 $data = $result->render();
358
359 $tempManager = \OCP\Server::get(\OCP\ITempManager::class);
360 $zipFile = $tempManager->getTemporaryFile('.zip');
361 file_put_contents($zipFile, $data);
362 $zip = new ZipArchive;
363 $this->assertTrue($zip->open($zipFile));
364 $fileCount = $zip->numFiles;
365 $this->assertEquals(count($dbFiles), $fileCount);
366 for ($i = 0; $i < $fileCount; $i++) {
367 $info = $zip->statIndex($i);
368 $this->assertEquals($zipFileBase . '/' . $dbFiles[$i]->getName(), $info['name']);
369 $this->assertEquals($dbFiles[$i]->getSize(), $info['size']);
370 $extracted = stream_get_contents($zip->getStreamIndex($i));
371 $this->assertEquals($dbFiles[$i]->getData(), $extracted);
372 }
