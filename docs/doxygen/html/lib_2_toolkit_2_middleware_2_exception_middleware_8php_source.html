132 protected function afterThrowable(Controller $controller, string $methodName, Throwable $exception, bool $wrap)
133 {
134 $reflectionMethod = new ReflectionMethod($controller, $methodName);
135 if ($this->hasAnnotationOrAttribute($reflectionMethod, Attributes\DoNotCatchExceptions::class)) {
136 throw $exception;
137 }
138 if (!($exception instanceof EnduserNotificationException)) {
139 if (!($this->catchAll || $wrap)) {
140 throw $exception;
141 }
142 try {
143 $folderPrefix = $this->appContainer->get(App::APP_ROOT_FOLDER);
144 } catch (Throwable) {
145 // ignore
146 $folderPrefix = \OC::$SERVERROOT;
147 }
148
149 $originalException = $exception;
150 $exceptionMessage = $this->l->t(
151 'Unable to serve request to "%1$s": %2$s',
152 [ $this->request->getPathInfo(), $originalException->getMessage() ],
153 );
154 $exceptionMessage = str_replace($folderPrefix, '...', $exceptionMessage);
155
156 $context = [];
157 $httpStatusCode = $this->httpStatusCodeMapping[get_class($originalException)]
158 ?? $this->httpStatusCodeMapping['default']
159 ?? Http::STATUS_INTERNAL_SERVER_ERROR;
160
161 $exception = new EnduserNotificationException(
162 $exceptionMessage, 0, $originalException,
163 httpStatusCode: $httpStatusCode,
164 context: $context,
165 );
166 }
167 $httpStatusCode = $exception->getHttpStatusCode();
168 $context = array_merge_recursive(
169 [ 'httpStatusCode' => $httpStatusCode ],
170 $exception->getContext() ?? [],
171 );
172 $logEntry = $this->logException(
173 $exception,
174 message: $exception->getMessage(),
175 context: $context,
176 returnLogEntry: true,
177 shift: PHP_INT_MIN, // do not decorate with prefix
178 );
179 if (is_array($logEntry)) {
180 array_walk_recursive($logEntry, fn(&$value) => $value = str_replace(\OC::$SERVERROOT, '', mb_scrub($value, 'UTF-8')));
181 } else {
182 $this->logError('Log entry is null');
183 }
184 $this->logDebug('LOG_ENTRY ' . print_r($logEntry, true));
185 // the frontend shows the "messages" array to the user
