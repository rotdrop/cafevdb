75 protected function mapLogLevels(mixed $level)
76 {
77 if (is_int($level) || is_numeric($level)) {
78 return $level;
79 }
80 switch ($level) {
81 case LogLevel::EMERGENCY:
82 return ILogger::FATAL;
83 case LogLevel::ALERT:
84 return ILogger::ERROR;
85 case LogLevel::CRITICAL:
86 return ILogger::ERROR;
87 case LogLevel::ERROR:
88 return ILogger::ERROR;
89 case LogLevel::WARNING:
90 return ILogger::WARN;
91 case LogLevel::NOTICE:
92 return ILogger::INFO;
93 case LogLevel::INFO:
94 return ILogger::INFO;
95 case LogLevel::DEBUG:
96 return ILogger::DEBUG;
97 default:
98 return ILogger::ERROR;
99 }
100 }

114 public function log(
115 mixed $level,
116 string $message,
117 array $context = [],
118 int $shift = 0,
119 bool $showTrace = false,
120 bool $returnLogEntry = false,
121 ):?array {
122 $level = $this->mapLogLevels($level);
123
124 if ($shift < 0) {
125 $prefix = '';
126 } else {
127 $trace = debug_backtrace();
128 $prefix = '';
129 $shift = min($shift, count($trace));
130
131 do {
132 $caller = $trace[$shift];
133 $file = $caller['file'] ?? 'unknown';
134 $line = $caller['line'] ?? 'unknown';
135 $caller = $trace[$shift + 1] ?? 'unknown';
136 $class = $caller['class'] ?? 'unknown';
137 $method = $caller['function'];
138
139 $prefix .= $file . ':' . $line . ': ' . $class . '::' . $method . '(): ';
140 } while ($showTrace && --$shift > 0);
141 }
142
143 $listener = null;
144 if ($returnLogEntry) {
146 $eventDispatcher = $this->appContainer()->get(IEventDispatcher::class);
148 $listener = $this->appContainer()->get(BeforeMessageLoggedEventListener::class);
149 $listener->clearLogEntry();
150 $eventDispatcher->addListener(
151 BeforeMessageLoggedEventListener::EVENT,
152 [$listener, 'handle'],
153 );
154 }
155 $this->logger()->log($level, $prefix . $message, $context);
156
157 return $listener?->getLogEntry();
158 }
