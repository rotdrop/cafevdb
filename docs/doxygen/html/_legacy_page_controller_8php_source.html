295 private function getTemplate(?string $template, string $renderAs): string
296 {
297 // Replace colons back to path separators. All our template parameters are
298 // alpha-numeric with the exception that they man contain path separators
299 // (i.e. '/') which optionally are replaced by colons (i.e. ':') in order
300 // to avoid url en-/decoding. Here we need to convert back to path
301 // separators.
302 if (!$this->configCheck['summary']) {
303 return PageRenderer\ConfigCheck::TEMPLATE;
304 }
305 if (empty($template)) {
306 $template = self::DEFAULT_TEMPLATE;
307 }
308
310 $blogRenderer = $this->appContainer->get(PageRenderer\Blog::class);
311 if ($blogRenderer->notificationsPending()) {
312 $template = PageRenderer\BLOG::TEMPLATE;
313 }
314 $template = str_replace(':', Constants::PATH_SEP, $template);
315
316 return $template;
317 }
