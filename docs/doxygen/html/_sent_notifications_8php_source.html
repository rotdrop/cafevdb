158 protected function execute(InputInterface $input, OutputInterface $output): int
159 {
160 $this->dry = $input->getOption(self::OPTION_DRY_RUN);
161
162 $imapUri = $input->getOption(self::OPTION_IMAP_URI);
163 if ($imapUri) {
164 $imapOptions = parse_url($imapUri);
165 }
166
167 $result = $this->authenticate($input, $output);
168 if ($result != 0) {
169 return $result;
170 }
171
172 if ($input->getOption(self::ACTION_LIST_MISSING) || $input->getOption(self::ACTION_ADD_MISSING)) {
173 $sentEmailsRepository = $this->getDatabaseRepository(Entities\SentEmail::class);
174 $payments = $this->getDatabaseRepository(Entities\CompositePayment::class)->findBy(
175 [
176 [ '!notificationMessageId' => null, ],
177 [ '!notificationMessageId' => '', ],
178 ],
179 orderBy: [
180 'dateOfReceipt' => 'ASC',
181 ],
182 );
183 $rows = [];
185 foreach ($payments as $payment) {
186 $notificationMessageId = $payment->getNotificationMessageId();
188 $sentEmail = $sentEmailsRepository->find([ 'messageId' => $notificationMessageId, ]);
189 if (!empty($sentEmail)) {
190 $this->sentEmails[$notificationMessageId] = $sentEmail; // remember for later
191 continue;
192 }
193 $this->payments[$notificationMessageId] = $payment; // remember in order to reconstruct the SentEmail link.
194 $rows[$notificationMessageId] = [
195 $notificationMessageId,
196 $payment->getDateOfReceipt()->format('Y-m-d'),
197 '',
198 ];
199 }
200 $messageIdOptions = $input->getOption(self::OPTION_MESSAGE_ID);
201 if (!empty($messageIdOptions)) {
202 $rows = array_filter($rows, fn(array $row) => in_array($row[0], $messageIdOptions));
203 }
204
205 if (count($rows) == 0) {
206 $output->writeln('<info>' . $this->l->t('All email notifications seem to be recorded in the database.') . '</info>');
207 return 0;
208 }
209
210 if (!empty($imapOptions)) {
211 if (isset($imapOptions['user']) && !isset($imapOptions['pass'])) {
212 $helper = $this->getHelper('question');
213 $question = (new Question($this->l->t('IMAP Password') . ': ', ''))->setHidden(true);
214 $imapOptions['pass'] = $helper->ask($input, $output, $question);
215 }
216 }
217
218 $this->imapService->setAccount(
219 host: $imapOptions['host'] ?? null,
220 port: $imapOptions['port'] ?? null,
221 security: $imapOptions['scheme'] ?? null,
222 user: $imapOptions['user'] ?? null,
223 password: $imapOptions['pass'] ?? null,
224 );
225
226 $messageIds = array_map(fn(array $row) => $row[0], $rows);
227 $imapMessages = $this->imapService->searchMessageId($messageIds);
229 foreach ($imapMessages as $key => $imapMessage) {
230 unset($imapMessages[$key]);
231 $imapMessages[$imapMessage->getMessageId()] = $imapMessage;
232 }
233 array_merge($this->imapMessages, $imapMessages);
234
235 $output->writeln(
236 '<info>'
237 . $this->l->t(
238 'Found %1$d of %2$d messages on the server.',
239 [
240 count($imapMessages),
241 count($messageIds),
242 ]
243 )
244 . '</info>');
245 }
246
247 if ($input->getOption(self::ACTION_LIST_MISSING)) {
249 foreach ($imapMessages as $messageId => $imapMessage) {
250 $subject = $imapMessage->getSubject();
251 if (mb_strlen($subject) > 20) {
252 $subject = mb_substr($subject, 0, 8) . '...' . mb_substr($subject, -9);
253 }
254 $rows[$messageId][2] = $subject;
255 $rows[$messageId][3] = $this->l->t('found');
256 }
257
258 $headers = [
259 $this->l->t('Missing in DB'),
260 $this->l->t('Date of Receipt'),
261 $this->l->t('Subject'),
262 $this->l->t('IMAP Server'),
263 ];
264 (new Table($output))
265 ->setHeaders($headers)
266 ->setRows($rows)
267 ->render();
268 }
269
270 if ($input->getOption(self::ACTION_ADD_MISSING)) {
271 foreach ($imapMessages as $messageId => $imapMessage) {
272 $sentEmail = $this->sentEmailsService->sentEmailFromMessage($imapMessage, persist: true, flush: false);
273 $this->entityManager->beginTransaction();
274 try {
276 $payment = $this->payments[$messageId];
277 $payment->setPreNotificationEmail($sentEmail);
278 $sentEmail->setCompositePayment($payment);
279
280 $sepaTransaction = $payment->getSepaTransaction();
281 if (!empty($sepaTransaction)) {
282 $sentEmail->setSepaBulkTransaction($sepaTransaction);
283 }
284
285 $this->persist($sentEmail);
286 $this->flush();
287
288 $this->entityManager->commit();
289 } catch (Throwable $t) {
290 $this->entityManager->rollback();
291 $output->writeln(
292 '<error>'
293 . $this->l->t(
294 'Reconstruction of "%1$s" failed: %2$s.',
295 [
296 $messageId,
297 $t->getMessage(),
298 ],
299 )
300 . '</error>');
301 throw $t;
302 }
303 }
304 }
305
306 return 0;
307 }
