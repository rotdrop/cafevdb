131 public function callInternalRoute(
132 string $route,
133 string $method = self::POST,
134 array $routeParams = [],
135 array $requestData = [],
136 string $postType = self::JSON,
137 ): array {
138
139 $url = $this->urlGenerator->linkToRouteAbsolute($route, $routeParams);
140
141 $urlParameters = [
142 'format' => self::JSON,
143 ];
144
145 // catch excess url parameters
146 $query = parse_url($url, PHP_URL_QUERY);
147 if (!empty($query)) {
148 $url = substr($url, 0, strrpos($url, '?'));
149 $query = parse_str($query, $query);
150 $urlParameters = array_merge($urlParameters, $query);
151 }
152
153 if ($method == self::GET) {
154 $urlParameters = array_merge($urlParameters, $requestData);
155 $requestData = null;
156 }
157
158 $requestOptions = [];
159
160 if (!empty($requestData)) {
161 switch ($postType) {
162 case self::JSON:
163 $requestOptions['json'] = $requestData;
164 break;
165 case self::URL_ENCODED:
166 $requestOptions['form_params'] = $requestData;
167 break;
168 default:
169 throw new InvalidArgumentException(
170 $this->l->t('Supported data formats are "%1$s" and "%2$s", specified was "%3$s".', [
171 self::JSON, self::URL_ENCODED, $postType
172 ]));
173 }
174 }
175
176 $requestOptions['headers'] = [
177 'requesttoken' => \OCP\Util::callRegister(),
178 'Accept' => 'application/json',
179 'OCS-APIRequest' => 'true',
180 ];
181 $serverHost = $this->request->getServerHost();
182 if (!empty($this->request->cookies)) {
183 $requestOptions['cookies'] = CookieJar::fromArray(
184 $this->request->cookies,
185 $serverHost,
186 );
187 }
188 if ($serverHost === 'localhost' && str_starts_with($url, 'https:')) {
189 $requestOptions['verify'] = false;
190 }
191
192 $this->logInfo('CALL OUT ' . $url . ' ' . print_r($requestOptions, true));
193
195 $response = $this->client->request($method, $url, $requestOptions);
196
197 $body = $response->getBody();
198 $responseData = json_decode($body, true);
199 if (!is_array($responseData)) {
200 throw new RuntimeException(
201 $this->l->t('Invalid response from API call: "%s"', print_r([
202 'body' => $body,
203 'statusCode' => $response->getStatusCode(),
204 'headers' => $response->getHeaders(),
205 ], true)),
206 );
207 }
208
209 $this->logDebug('RESPONSE DATA ' . print_r($responseData, true));
210
211 // Some apps still return HTTP_STATUS_OK and code errors and success in
212 // the old way instead of using HTTP-error-codes.
213 if (($responseData['status'] ?? null) != 'success' && isset($responseData['data'])) {
214 throw new RuntimeException(
215 $this->l->t('Error response from call to internal route "%1$s": %2$s', [
216 $route, $responseData['data']['message'] ?? print_r($responseData, true)
217 ]));
218 }
219
220 // parse OCS responses if format matches
221 if (isset($responseData['ocs'])) {
222 $meta = $responseData['ocs']['meta'] ?? null;
223 $data = $responseData['ocs']['data'] ?? null;
224 if ($meta === null || $data === null) {
225 throw new RuntimeException(
226 $this->l->t('Invalid OCS response from call to internal route "%1$s": %2$s', [
227 $route, $responseData,
228 ]));
229 }
230 if ($meta['statuscode'] !== 100) {
231 throw new RuntimeException(
232 $this->l->t('Error response from call to internal route "%1$s": %2$s -- %3$s', [
233 $route, $meta['status'] ?? 'unknwown status', $meta['message'] ?? print_r($meta, true)
234 ]));
235 }
236 return $data;
237 }
238
239 if (isset($responseData['data'])) {
240 return $responseData['data'];
241 } else {
242 return $responseData;
243 }
244 }
