513 public function updateFieldDatumDocument(Entities\ProjectParticipantFieldDatum $fieldDatum, bool $flush = true):bool
514 {
515 // fetch the current file information based on the existing directory structure
516 $currentFileInfo = $this->projectService->participantFileInfo($fieldDatum, newFile: null, includeDeleted: true);
517
518 // fetch the target file information pretending it would be a new file
519 $targetFileInfo = $this->projectService->participantFileInfo($fieldDatum, newFile: $currentFileInfo['file'], includeDeleted: true);
520
521 $currentComponents = Util::explode(self::PATH_SEPARATOR, $currentFileInfo['pathName']);
522 $targetComponents = Util::explode(self::PATH_SEPARATOR, $targetFileInfo['pathName']);
523
524 $needsFlush = false;
525 if ($flush) {
526 $this->entityManager->beginTransaction();
527 }
528 try {
529
530 if (count($currentComponents) != count($targetComponents)) {
531 // two possibilities: remove and add or throw
532 $this->logWarn('Current path "' . $currentFileInfo['pathName'] . '" and target path "' . $targetFileInfo['pathName'] . '" have different numbers of components.');
533 // Use add-remove in order to first increase the number of links. As
534 // the paths have different numbers of components source and target
535 // directory entries must reside in different directories.
536 $this->addFieldDatumDocument($fieldDatum, $currentFileInfo['file'], flush: false);
537 $this->removeFieldDatumDocument($fieldDatum, flush: false);
538 $needsFlush = true;
539 } else {
541 $dirEntry = $currentFileInfo['dirEntry'];
542 while (!$dirEntry->isRootFolder()) {
543 $targetName = array_pop($targetComponents);
544 if ($dirEntry->getName() != $targetName) {
545 $needsFlush = true;
546 $dirEntry->setName($targetName);
547 }
548 $dirEntry = $dirEntry->getParent();
549 }
550 }
551
552 if ($flush) {
553 $this->flush();
554 $this->entityManager->commit();
555 }
556 } catch (Throwable $t) {
