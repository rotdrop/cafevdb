511 public function updateFieldDatumDocument(Entities\ProjectParticipantFieldDatum $fieldDatum, bool $flush = true):bool
512 {
513 // fetch the current file information based on the existing directory structure
514 $currentFileInfo = $this->projectService->participantFileInfo($fieldDatum, newFile: null, includeDeleted: true);
515
516 // fetch the target file information pretending it would be a new file
517 $targetFileInfo = $this->projectService->participantFileInfo($fieldDatum, newFile: $currentFileInfo['file'], includeDeleted: true);
518
519 $currentComponents = Util::explode(self::PATH_SEPARATOR, $currentFileInfo['pathName']);
520 $targetComponents = Util::explode(self::PATH_SEPARATOR, $targetFileInfo['pathName']);
521
522 $needsFlush = false;
523 if ($flush) {
524 $this->entityManager->beginTransaction();
525 }
526 try {
527
528 if (count($currentComponents) != count($targetComponents)) {
529 // two possibilities: remove and add or throw
530 $this->logWarn('Current path "' . $currentFileInfo['pathName'] . '" and target path "' . $targetFileInfo['pathName'] . '" have different numbers of components.');
531 // Use add-remove in order to first increase the number of links. As
532 // the paths have different numbers of components source and target
533 // directory entries must reside in different directories.
534 $this->addFieldDatumDocument($fieldDatum, $currentFileInfo['file'], flush: false);
535 $this->removeFieldDatumDocument($fieldDatum, flush: false);
536 $needsFlush = true;
537 } else {
539 $dirEntry = $currentFileInfo['dirEntry'];
540 while (!$dirEntry->isRootFolder()) {
541 $targetName = array_pop($targetComponents);
542 if ($dirEntry->getName() != $targetName) {
543 $needsFlush = true;
544 $dirEntry->setName($targetName);
545 }
546 $dirEntry = $dirEntry->getParent();
547 }
548 }
549
550 if ($flush) {
551 $this->flush();
552 $this->entityManager->commit();
553 }
554 } catch (Throwable $t) {
555 if ($flush && $this->entityManager->isTransactionActive()) {
556 $this->entityManager->rollback();
557 }
558 throw new Exceptions\DatabaseException($this->l->t('Unable to update file-name for field-datum "%s".', [
559 (string)$fieldDatum->getProjectParticipant()
560 ]));
561 }
562
563 return $needsFlush;
564 }
