293 protected function parsePath(string $path):?array
294 {
295 $path = StorageUtil::uploadBasename($path);
296 $fileName = pathinfo($path, PATHINFO_FILENAME);
297 $parts = explode('-', $fileName);
298 if (count($parts) != 4) {
299 return null;
300 }
301 $l10nTaxType = $parts[1];
302 $assessmentPeriodStart = $parts[2];
303 $assessmentPeriodEnd = $parts[3];
304 $taxType = null;
305 foreach (array_values(EnumTaxType::values()) as $taxType) {
306 if ($this->appL10n()->t((string)$taxType) == $l10nTaxType) {
307 break;
308 }
309 $taxType = null;
310 }
311 if (empty($taxType)) {
312 return null;
313 }
314 if ($fileName != $this->getLegacyTaxExemptionNoticeFileName($taxType, $assessmentPeriodStart, $assessmentPeriodEnd)) {
315 return null;
316 }
317 return compact(['taxType', 'assessmentPeriodStart', 'assessmentPeriodEnd']);

326 public function rename(string $path1, string $path2): bool
327 {
329 $dirEntry = $this->fileFromFileName($path1);
330 if (empty($dirEntry)) {
331 throw new Exceptions\DatabaseStorageException(
332 $this->l->t('Unable to find database entity for path "%s".', $path1)
333 );
334 }
335 if ($dirEntry instanceof Entities\DatabaseStorageFolder) {
336 throw new Exceptions\DatabaseStorageException(
337 $this->l->t('Path "%s" is a directory.', $path1)
338 );
339 }
340 if ($dirEntry instanceof InMemoryFileNode) {
341 $dirEntry = $this->persistInMemoryFileNode($dirEntry);
342 }
343
344 if (!$this->isReadMe($path2)) {
345 $newValues = $this->parsePath($path2);
346 if (empty($newValues)) {
347 return false;
348 }
349
351 $repository = $this->entityManager->getRepository(Entity::class);
352 $entity = $repository->findOneBy([
353 'writtenNotice' => $dirEntry,
354 ]);
355 }
356
357 $this->entityManager->beginTransaction();
358 try {
359 if (!empty($entity)) {
360 foreach ($newValues as $key => $value) {
361 $entity[$key] = $value;
362 }
363
364 $extension = pathinfo($path2, PATHINFO_EXTENSION);
365 $newName = $this->getTaxExemptionNoticeFileName($entity, $extension);
366
367 if ($newName != basename($path2)) {
368 throw new UnexpectedValueException($this->l->t('Generated name "%1$s" differs from target name "%2$s" after updating entity "%3$s".', [
369 $newName, basename($path2), (string)$entity,
370 ]));
371 }
372 }
373
374 $dirEntry->setName(basename($path2));
375
376 $this->flush();
377 $this->entityManager->commit();
378
379 // local in-memory file cache
380 $this->setFileNameCache($path2, $dirEntry);
381 $this->unsetFileNameCache($path1);
382 } catch (Throwable $t) {
383 if ($this->entityManager->isTransactionActive()) {
384 $this->entityManager->rollback();
385 }
386 throw new Exceptions\DatabaseException($this->l->t('Unable to update "%$1s" from value-set "%2$s".', (string)$dirEntry, print_r($newValues, true)));
387 }
388
389 return true;

393 public function touch($path, $mtime = null)
394 {
395 if ($this->is_dir($path)) {
396 // $this->logInfo('IS DIR ' . $path);
397 return false;
398 }
399
400 $this->entityManager->beginTransaction();
401 try {
403 $dirEntry = $this->fileFromFileName($path);
404 if (empty($dirEntry)) {
405 if (!$this->isReadMe($path)) {
406 $newValues = $this->parsePath($path);
407 if (empty($newValues)) {
408 throw new UnexpectedValueException($this->l->t('Path "%s" seems to be invalid.', $path));
409 }
410
412 $repository = $this->entityManager->getRepository(Entity::class);
413 $entity = $repository->findOneBy($newValues);
414 if (empty($entity)) {
415 // The entity must exist before hand
416 throw new Exceptions\DatabaseEntityNotFoundException($this->l->t('Unable to find a matching entity for the given path "%s".', $path));
417 }
418 }
419 $file = new Entities\EncryptedFile(basename($path), '', '');
420 if ($mtime !== null) {
421 $file->setCreated($mtime);
422 }
423 $this->persist($file);
424 $this->flush();
425 // search for the folder
426 $rootFolder = $this->getRootFolder(create: true);
427 if (empty($rootFolder)) {
428 throw new UnexpectedValueException($this->l->t('Root-folder does not exist.'));
429 }
430 $dirEntry = $rootFolder->addDocument($file, basename($path));
431 if ($mtime !== null) {
432 $dirEntry->setCreated($mtime);
433 }
434 if (!empty($entity)) {
435 $entity->setWrittenNotice($dirEntry);
436 }
437 }
438 if ($dirEntry instanceof InMemoryFileNode) {
439 $dirEntry = $this->persistInMemoryFileNode($dirEntry);
440 }
441
442 if ($mtime !== null) {
443 $dirEntry->setUpdated($mtime);
444 $dirEntry->getFile()->setUpdated($mtime);
445 }
446 $this->flush();
447
448 $this->entityManager->commit();
449
450 $this->setFileNameCache($path, $dirEntry);
451
452 } catch (\Throwable $t) {
453 $this->logException($t);
454 if ($this->entityManager->isTransactionActive()) {
455 $this->entityManager->rollback();
456 }
457 return false;
