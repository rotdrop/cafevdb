234
236 $dirEntry = $this->fileFromFileName($path);
237 if (empty($dirEntry)) {

242 if ($dirEntry instanceof Entities\DatabaseStorageFolder) {

246 }
247
249 $repository = $this->entityManager->getRepository(Entity::class);
250 $entities = $repository->findBy([
251 'writtenNotice' => $dirEntry,
252 ]);

254 $this->entityManager->beginTransaction();
255 try {

259 $entity->setWrittenNotice(null);
260 }

262 $dirEntry->unlink();
263 $dirEntry->setFile(null);
264 $this->entityManager->remove($dirEntry);
265
266 $this->flush();
267 $this->entityManager->commit();
268
269 // update our local files cache
270 $this->unsetFileNameCache($path);
271 } catch (\Throwable $t) {
272 $this->logException($t);
